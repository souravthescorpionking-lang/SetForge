   140→function ProgramDayRow({
   141→  day,
   142→  index,
   143→  minutes,
   144→  handle,
   145→  onOpen,
   146→}: {
   147→  day: ProgramDetailDayDTO;
   148→  index: number;
   149→  minutes: number | null;
   150→  handle: ReactNode;
   151→  onOpen: (dayId: string) => void;
   152→}) {
   153→  const isRest = (day.dayType ?? "WORKOUT") === "REST";
   154→  return (
   155→    <div
   156→      data-row
   157→      className="flex h-14 w-full items-center gap-1 overflow-hidden whitespace-nowrap rounded-lg border bg-card pl-1 pr-3"
   158→    >
   159→      {handle}
   160→      <div
   161→        {...(isRest
   162→          ? { "aria-label": `Day ${index + 1} — rest day` }
   163→          : {
   164→              role: "button",
   165→              tabIndex: 0,
   166→              "aria-label": `Open ${day.name}`,
   167→              ...tourAttrs({ id: "programDetail.dayRow", label: "Day row", help: "Open the day to see its exercise groups.", order: 70 }),
   168→            })}
   169→        className={cn(
   170→          "flex h-14 min-w-0 flex-1 items-center gap-2",
   171→          !isRest && "cursor-pointer select-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none",
   172→        )}
   173→        onClick={isRest ? undefined : () => onOpen(day.id)}
   174→        onKeyDown={
   175→          isRest
   176→            ? undefined
   177→            : (e) => {
   178→                if (e.target !== e.currentTarget) return;
   179→                if (e.key === "Enter" || e.key === " ") {
   180→                  e.preventDefault();
   181→                  onOpen(day.id);
   182→                }
   183→              }
   184→        }
   185→      >
   186→        <span
   187→          className={cn(
   188→            "min-w-0 flex-1 truncate text-sm font-semibold leading-none",
   189→            isRest && "font-normal italic text-muted-foreground",
   190→          )}
   191→        >
   192→          {isRest ? "Rest day" : day.name}
   193→        </span>
   194→        <span className="flex-none text-xs font-semibold leading-none text-muted-foreground">Day {index + 1}</span>
   195→        {!isRest && minutes != null ? (
   196→          <span className="flex-none text-xs leading-none text-muted-foreground">{minutes} min</span>
   197→        ) : null}
   198→      </div>
   199→    </div>
   200→  );
   201→}

   280→  const phases = useMemo(() => resolvedVariant?.phases ?? [], [resolvedVariant]);
   281→  const phaseCount = phases.length;
   282→  const equipment = useMemo(
   283→    () => (resolvedVariant?.equipment ?? []).map((v) => EQUIPMENT_LABELS[v as Equipment] ?? v),
