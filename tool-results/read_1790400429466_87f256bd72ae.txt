     3→// ─────────────────────────────────────────────────────────────────────────────
     4→// SetRow — THE single source of truth set row (Part 3 §SET ROW).
     5→//
     6→// One 40px CSS-grid row for every mode (edit / read / preview / template).
     7→// The grid template is COMPUTED, never styled ad hoc:
     8→//   • spec track widths:  #24 · type32 · f1:72 · f2:56 · rpe:52 · tempo:64 ·
     9→//                          rest:56 · done:40 · more:32
    10→//   • missing modality fields or settings-hidden columns REMOVE tracks from
    11→//     the template (they are never "hidden" cells).
    12→//   • <468px viewport: rpe + tempo tracks are REMOVED — their values/editors
    13→//     move into the `more` ⋯ popover. (PLATFORM ADAPTATION: the spec's bottom
    14→//     sheet is forbidden here; an anchored popover — shadcn Popover on
    15→//     Floating UI with collision flip — is the allowed replacement.)
    16→//   • <360px viewport: f1/f2 drop their floors to spec−16 (minmax) so the row
    17→//     still fits a 320px screen. All small tracks stay exact px.
    18→//   • f1/f2 carry `minmax(spec, 1fr)` so value columns fill leftover card width
    19→//     at every viewport — the row can never overflow its card and never leaves
    20→//     a dead right margin.
    21→//
    22→// Grid math (weight+reps modality, all columns on):
    23→//   wide fixed tracks 24+32+72+56+52+64+56+40+32 = 428px
    24→//   mobile (<468, rpe+tempo dropped)             = 312px ≤ 328px (360−32 law)
    25→//   320px viewport floors 24+32+56+40+56+40+32   = 280px ≤ 282px available
    26→//   (The spec contract names 460 as the wide threshold; the corrected value is
    27→//   468 — 428px of tracks + 32px page gutters + 6px card chrome (border×2 +
    28→//   4px colour bar) need ≥466px, so the wide layout only engages at 468.)
    29→// ─────────────────────────────────────────────────────────────────────────────

    66→/** Spec track widths in px — the single source of truth for the row grid. */
    67→const TRACK = { index: 24, setType: 32, f1: 72, f2: 56, rpe: 52, tempo: 64, rest: 56, done: 40, more: 32 } as const;
    68→
    69→/** Viewport at which rpe/tempo tracks return to the grid (see file header). */
    70→const WIDE_VIEWPORT = 468;
    71→
    72→/** Tap→cycle order for the set-type chip. */
    73→const TYPE_CYCLE: SetType[] = ["NORMAL", "WARMUP", "DROP", "FAILURE", "AMRAP"];
    74→
    75→/** Rest presets (sec) — spec list. */
    76→const REST_PRESETS = [30, 60, 90, 120, 150, 180];
    77→
    78→/** RPE chip scale: 1–10 in half steps (spec). */
    79→const RPE_CHIPS: number[] = [];
    80→for (let v = 1; v <= 10; v += 0.5) RPE_CHIPS.push(v);

    82→type TrackKey = "index" | "setType" | "f1" | "f2" | "rpe" | "tempo" | "rest" | "done" | "more";
    83→
    84→type GridSpec = { keys: TrackKey[]; template: string };
    85→
    86→function buildGrid(
    87→  fields: SetField[],
    88→  visible: CardVisibleColumns,
    89→  mode: CardMode,
    90→  vw: number,
    91→  moreTrack: boolean,
    92→): GridSpec {
    93→  const keys: TrackKey[] = [];
    94→  const parts: string[] = [];
    95→  const push = (key: TrackKey, css: string) => {
    96→    keys.push(key);
    97→    parts.push(css);
    98→  };
    99→
   100→  push("index", `${TRACK.index}px`);
   101→  if (visible.setType) push("setType", `${TRACK.setType}px`);
   102→
   103→  // Value tracks: spec floor (−16 below 360px — documented adaptation so the
   104→  // fixed small tracks still fit a 320px screen) + 1fr growth to fill the card.
   105→  const narrow = vw < 360;
   106→  if (fields[0]) push("f1", `minmax(${TRACK.f1 - (narrow ? 16 : 0)}px, 1fr)`);
   107→  if (fields[1]) push("f2", `minmax(${TRACK.f2 - (narrow ? 16 : 0)}px, 1fr)`);
   108→
   109→  if (visible.rpe && vw >= WIDE_VIEWPORT) push("rpe", `${TRACK.rpe}px`);
   110→  if (visible.tempo && vw >= WIDE_VIEWPORT) push("tempo", `${TRACK.tempo}px`);
   111→  if (visible.rest) push("rest", `${TRACK.rest}px`);
   112→
   113→  // `done` never applies to routine templates → track removed.
   114→  if (mode !== "template") push("done", `${TRACK.done}px`);
   115→  if (mode === "edit" || mode === "template") push("more", `${TRACK.more}px`);
   116→  else if (moreTrack) push("more", `${TRACK.more}px`);
   117→
   118→  return { keys, template: parts.join(" ") };
   119→}

   173→function TempoEditor({ value, onChange }: { value: string | null; onChange: (v: string | null) => void }) {

   178→    initial[2] ?? "",

   180→  ]);
   181→  const labels = ["Ecc", "Pause", "Con", "Pause"];
   182→  return (

   190→            <span className="mb-1 block text-center text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">

   194→              aria-label={`Tempo ${labels[i]} seconds`}

   310→function TypeCell({
   311→  set,
   312→  mode,
   313→  editable,
   314→  onAction,
   315→}: {
   316→  set: CardSet;
   317→  mode: CardMode;
   318→  editable: boolean;
   319→  onAction?: (a: CardAction) => void;
   320→}) {
   321→  const [open, setOpen] = useState(false);
   322→  const holdRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   323→  const startRef = useRef<{ x: number; y: number } | null>(null);
   324→  const longPressed = useRef(false);
   325→
   326→  const endHold = () => {
   327→    if (holdRef.current) clearTimeout(holdRef.current);
   328→    holdRef.current = null;
   329→    startRef.current = null;
   330→  };
   331→
   332→  useEffect(() => () => endHold(), []);
   333→
   334→  const value = set.setType ?? null;
   335→  const meta = value ? SET_TYPE_META[value as SetType] : null;
   336→  const templateBlank = mode === "template" && value == null;
   337→
   338→  // read / preview / read-only fallback → static glyph
   339→  if (!editable || !onAction) {
   340→    if (meta) {
   341→      return (
   342→        <span
   343→          className={cn("flex h-6 w-6 items-center justify-center rounded-md text-[11px] font-bold", meta.className)}
   344→          title={`${meta.label} — ${meta.description}`}
   345→        >
   346→          {meta.letter}
   347→        </span>
   348→      );
   349→    }
   350→    return (
   351→      <span className="text-xs text-muted-foreground/40" title="Set type not set">
   352→        –
   353→      </span>
   354→    );
   355→  }
   356→
   357→  const cycle = () => {
   358→    const current = (value ?? "NORMAL") as SetType;
   359→    const next = TYPE_CYCLE[(TYPE_CYCLE.indexOf(current) + 1) % TYPE_CYCLE.length];
   360→    onAction({ type: "update-set", setId: set.id, patch: { setType: next } });
   361→  };
   362→
   363→  return (
   364→    <Popover open={open} onOpenChange={setOpen}>
   365→      <PopoverTrigger asChild>
   366→        <button
   367→          type="button"
   368→          aria-label={
   369→            meta
   370→              ? `Set type ${meta.label}. Tap to cycle, hold for picker`
   371→              : "Set type not set. Tap to choose"
   372→          }
   373→          title={meta ? `${meta.label} — ${meta.description}` : "Choose set type"}
   374→          onClick={(e) => {
   375→            // Fully controlled popover: preventDefault suppresses Radix's own
   376→            // click-toggle (composeEventHandlers skips when defaultPrevented),
   377→            // so tap = cycle · hold / right-click / Enter = picker.
   378→            e.preventDefault();
   379→            if (longPressed.current) {
   380→              longPressed.current = false;
   381→              return;
   382→            }
   383→            if (templateBlank) {
   384→              setOpen(true);
   385→              return;
   386→            }
   387→            cycle();
   388→          }}
   389→          onPointerDown={(e) => {
   390→            startRef.current = { x: e.clientX, y: e.clientY };
   391→            longPressed.current = false;
   392→            holdRef.current = setTimeout(() => {
   393→              longPressed.current = true;
   394→              setOpen(true);
   395→            }, 420);
   396→          }}
   397→          onPointerMove={(e) => {
   398→            const s = startRef.current;
   399→            if (s && (Math.abs(e.clientX - s.x) > 8 || Math.abs(e.clientY - s.y) > 8)) endHold();
   400→          }}
   401→          onPointerUp={endHold}
   402→          onPointerCancel={endHold}
   403→          onPointerLeave={endHold}
   404→          onContextMenu={(e) => {
   405→            e.preventDefault();
   406→            endHold();
   407→            setOpen(true);
   408→          }}
   409→          className={cn(
   410→            "flex h-7 w-7 items-center justify-center rounded-md text-[11px] font-bold transition-transform active:scale-95",
   411→            templateBlank ? "border border-dashed border-border text-muted-foreground" : meta?.className,
   412→          )}
   413→        >
   414→          {templateBlank ? "–" : (meta?.letter ?? "–")}
   415→        </button>
   416→      </PopoverTrigger>
   417→      <PopoverContent align="center" className="w-56 p-2">
   418→        <p className="px-1 pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Set type</p>
   419→        <div className="flex flex-col gap-1">
   420→          {SET_TYPES.map((t) => (
   421→            <button
   422→              key={t}
   423→              type="button"
   424→              onClick={() => {
   425→                onAction({ type: "update-set", setId: set.id, patch: { setType: t } });
   426→                setOpen(false);
   427→              }}
   428→              className="flex w-full items-center gap-2 rounded-lg px-2 py-1 text-left transition-colors hover:bg-accent"
   429→            >
   430→              <span
   431→                className={cn(
   432→                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[11px] font-bold",
   433→                  SET_TYPE_META[t].className,
   434→                )}
   435→              >
   436→                {SET_TYPE_META[t].letter}
   437→              </span>
   438→              <span className="min-w-0 flex-1">
   439→                <span className="block text-sm font-semibold leading-tight">{SET_TYPE_META[t].label}</span>
   440→                <span className="block text-[11px] leading-tight text-muted-foreground">
   441→                  {SET_TYPE_META[t].description}
   442→                </span>
   443→              </span>
   444→            </button>
   445→          ))}
   446→        </div>
   447→      </PopoverContent>
   448→    </Popover>
   449→  );
   450→}

   452→function ValueCell({
   453→  field,
   454→  set,
   455→  exercise,
   456→  mode,
   457→  editable,
   458→  onAction,
   459→}: {
   460→  field: SetField;
   461→  set: CardSet;
   462→  exercise: CardExercise;
   463→  mode: CardMode;
   464→  editable: boolean;
   465→  onAction?: (a: CardAction) => void;
   466→}) {
   467→  const key = FIELD_TO_KEY[field];
   468→  const value = (set[key] as number | null | undefined) ?? null;
   469→  const [draft, setDraft] = useState<string | null>(null);
   470→  const [focused, setFocused] = useState(false);
   471→  const step = stepForField(field, exercise);
   472→  const templateBlank = mode === "template" && value == null;
   473→  const label = FIELD_LABEL[field];
   474→
   475→  if (!editable || !onAction) {
   476→    return (
   477→      <span
   478→        className="w-full truncate px-1 text-right text-sm tabular-nums text-foreground"
   479→        title={formatFieldValue(field, value, exercise.unit)}
   480→      >
   481→        {formatFieldValue(field, value, exercise.unit)}
   482→      </span>
   483→    );
   484→  }
   485→
   486→  const display = draft ?? (value == null ? "" : trimNum(value));
   487→
   488→  const commit = () => {
   489→    if (draft != null) {
   490→      const parsed = parseFieldValue(field, draft);
   491→      if (parsed !== undefined && parsed !== value) {
   492→        onAction({ type: "update-set", setId: set.id, patch: { [key]: parsed } as Partial<CardSet> });
   493→      }
   494→    }
   495→    setDraft(null);
   496→  };
   497→
   498→  const stepBy = (dir: 1 | -1) => {
   499→    const parsedDraft = draft != null ? parseFieldValue(field, draft) : undefined;
   500→    const base = (parsedDraft != null ? parsedDraft : value) ?? 0;
   501→    const next = Math.max(0, Math.round((base + dir * step) * 100) / 100);
   502→    onAction({ type: "update-set", setId: set.id, patch: { [key]: next } as Partial<CardSet> });
   503→    setDraft(null);
   504→  };
   505→
   506→  const stepperButton = (dir: 1 | -1) => (
   507→    <button
   508→      type="button"
   509→      aria-label={dir < 0 ? `Decrease ${label}` : `Increase ${label}`}
   510→      // Keep focus on the input so the steppers don't unmount before click.
   511→      onMouseDown={(e) => e.preventDefault()}
   512→      onClick={() => stepBy(dir)}
   513→      className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   514→    >
   515→      {dir < 0 ? <Minus className="h-3 w-3" aria-hidden /> : <Plus className="h-3 w-3" aria-hidden />}
   516→    </button>
   517→  );
   518→
   519→  return (
   520→    <div className="flex h-full min-w-0 items-center">
   521→      {/* template blank: ↺ = copy-last (typed value replaces it) */}
   522→      {templateBlank && !focused ? (
   523→        <button
   524→          type="button"
   525→          aria-label={`Copy previous set ${label.toLowerCase()}`}
   526→          title="Copy previous set"
   527→          onClick={() => onAction({ type: "copy-last", setId: set.id })}
   528→          className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   529→        >
   530→          <RotateCcw className="h-3 w-3" aria-hidden />
   531→        </button>
   532→      ) : null}
   533→      {/* focused: inline steppers inside the cell ([− 20px | input | + 20px]) */}

   694→function MoreCell({
   695→  set,
   696→  editable,
   697→  showRpeEditor,
   698→  showTempoEditor,
   699→  onAction,
   700→}: {
   701→  set: CardSet;
   702→  editable: boolean;
   703→  showRpeEditor: boolean;
   704→  showTempoEditor: boolean;
   705→  onAction?: (a: CardAction) => void;
   706→}) {
   707→  const [open, setOpen] = useState(false);
   708→  const [noteDraft, setNoteDraft] = useState<string | null>(null);
   709→
   710→  // read / preview: PR/note glyph (inline — the preferred PR badge spot)
   711→  if (!editable || !onAction) {
   712→    if (set.isNewPr) return <Trophy className="h-3.5 w-3.5 text-amber-400" role="img" aria-label="New personal record" />;
   713→    if (set.note) return <StickyNote className="h-3.5 w-3.5 text-muted-foreground" role="img" aria-label="Has note" />;
   714→    return null;
   715→  }
   716→
   717→  const commitNote = () => {
   718→    if (noteDraft != null && noteDraft !== (set.note ?? "")) {
   719→      onAction({ type: "update-set", setId: set.id, patch: { note: noteDraft } });
   720→    }
   721→  };
   722→
   723→  const handleOpenChange = (next: boolean) => {
   724→    if (!next) commitNote();
   725→    setOpen(next);
   726→    if (next) setNoteDraft(null);
   727→  };
   728→
   729→  const dropped = showRpeEditor || showTempoEditor;
   730→
   731→  return (
   732→    <Popover open={open} onOpenChange={handleOpenChange}>
   733→      <PopoverTrigger asChild>
   734→        <button
   735→          type="button"
   736→          aria-label={`More options for set ${set.index}`}
   737→          className="flex h-10 w-8 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   738→        >
   739→          <MoreHorizontal className="h-4 w-4" aria-hidden />
   740→        </button>
   741→      </PopoverTrigger>
   742→      <PopoverContent align="end" className="w-64 p-3">
   743→        {dropped ? (
   744→          <div className="flex flex-col gap-3">
   745→            {showRpeEditor ? (
   746→              <RpeEditor
   747→                value={set.rpe ?? null}
   748→                onChange={(v) => onAction({ type: "update-set", setId: set.id, patch: { rpe: v } })}
   749→              />
   750→            ) : null}
   751→            {showTempoEditor ? (
   752→              <div className="border-t border-border pt-3">
   753→                <TempoEditor
   754→                  value={set.tempo ?? null}
   755→                  onChange={(v) => onAction({ type: "update-set", setId: set.id, patch: { tempo: v } })}
   756→                />
   757→              </div>
   758→            ) : null}
   759→          </div>
   760→        ) : null}
   761→        <div className={cn(dropped && "border-t border-border pt-3")}>
   762→          <p className="pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Note</p>
   763→          <textarea
   764→            aria-label="Set note"
   765→            rows={2}
   766→            value={noteDraft ?? set.note ?? ""}
   767→            onChange={(e) => setNoteDraft(e.target.value)}
   768→            onBlur={commitNote}
   769→            placeholder="Set note…"
   770→            className="w-full resize-none rounded-md border border-input bg-background px-2 py-1 text-xs outline-none focus:ring-2 focus:ring-ring/40"
   771→          />
   772→          <div className="mt-2 flex gap-1">
   773→            <Button
   774→              type="button"
   775→              variant="outline"
   776→              size="sm"
   777→              className="h-8 flex-1 rounded-lg text-xs"
   778→              onClick={() => onAction({ type: "copy-last", setId: set.id })}
   779→            >
   780→              <RotateCcw className="h-3.5 w-3.5" aria-hidden /> Copy last
   781→            </Button>
   782→            <Button
   783→              type="button"
   784→              variant="outline"
   785→              size="sm"
   786→              className="h-8 flex-1 rounded-lg text-xs"
   787→              onClick={() => {
   788→                onAction({ type: "rest-timer", setId: set.id });
   789→                handleOpenChange(false);
   790→              }}
   791→            >
   792→              <Timer className="h-3.5 w-3.5" aria-hidden /> Rest
   793→            </Button>
   794→          </div>
   795→        </div>
   796→      </PopoverContent>
   797→    </Popover>
   798→  );
   799→}

   803→export interface SetRowProps {
   804→  /** Render mode — see CardMode. */
   805→  mode: CardMode;
   806→  exercise: CardExercise;
   807→  set: CardSet;
   808→  visibleColumns: CardVisibleColumns;
   809→  /**
   810→   * read/preview: whether the card shows the more track (ExerciseCard computes
   811→   * "any set has a PR/note"). Standalone default: this set has a PR/note.
   812→   */
   813→  moreTrack?: boolean;
   814→  /** Omit → the row renders fully read-only (no inputs/menus, even in edit mode). */
   815→  onAction?: (action: CardAction) => void;
   816→  className?: string;
   817→}
   818→
   819→export function SetRow({ mode, exercise, set, visibleColumns, moreTrack, onAction, className }: SetRowProps) {
   820→  const vw = useViewportWidth();
   821→  const fields = useMemo(() => fieldsForType(exercise.modality), [exercise.modality]);
   822→  const editable = onAction != null && (mode === "edit" || mode === "template" || mode === "preview");
   823→  const effectiveMoreTrack =
   824→    moreTrack ?? (mode === "read" || mode === "preview" ? !!(set.isNewPr || set.note) : true);
   825→  const grid = useMemo(
   826→    () => buildGrid(fields, visibleColumns, mode, vw, effectiveMoreTrack),
   827→    [fields, visibleColumns, mode, vw, effectiveMoreTrack],
   828→  );
   829→
   830→  // rpe/tempo editors move into the more ⋯ popover below the wide threshold
   831→  const rpeDropped = visibleColumns.rpe && vw < WIDE_VIEWPORT;
   832→  const tempoDropped = visibleColumns.tempo && vw < WIDE_VIEWPORT;
   833→  const typeEditable = editable && (mode === "edit" || mode === "template");
   834→  const moreEditable = editable && (mode === "edit" || mode === "template");
   835→
   836→  const cells: Record<TrackKey, ReactNode> = {
   837→    index: <IndexCell set={set} mode={mode} onAction={editable ? onAction : undefined} />,
   838→    setType: <TypeCell set={set} mode={mode} editable={typeEditable} onAction={typeEditable ? onAction : undefined} />,
   839→    f1: fields[0] ? (
   840→      <ValueCell
   841→        field={fields[0]}
   842→        set={set}
   843→        exercise={exercise}
   844→        mode={mode}
   845→        editable={editable}
   846→        onAction={editable ? onAction : undefined}
   847→      />
   848→    ) : null,
   849→    f2: fields[1] ? (
   850→      <ValueCell
   851→        field={fields[1]}
   852→        set={set}
   853→        exercise={exercise}
   854→        mode={mode}
   855→        editable={editable}
   856→        onAction={editable ? onAction : undefined}
   857→      />
   858→    ) : null,
   859→    rpe: <RpeCell set={set} editable={editable} onAction={editable ? onAction : undefined} />,
   860→    tempo: <TempoCell set={set} editable={editable} onAction={editable ? onAction : undefined} />,
   861→    rest: <RestCell set={set} editable={editable} onAction={editable ? onAction : undefined} />,
   862→    done: <DoneCell set={set} mode={mode} onAction={onAction} />,
   863→    more: (
   864→      <MoreCell
   865→        set={set}
   866→        editable={moreEditable}
   867→        showRpeEditor={rpeDropped}
   868→        showTempoEditor={tempoDropped}
   869→        onAction={moreEditable ? onAction : undefined}
   870→      />
   871→    ),
   872→  };
   873→
   874→  return (
   875→    <div
   876→      data-row
   877→      role="group"
   878→      aria-label={`Set ${set.index}`}
   879→      className={cn(rowGrid, rowTint(set), className)}
   880→      style={{ gridTemplateColumns: grid.template }}
   881→    >
   882→      {grid.keys.map((key) => (
   883→        <div key={key} className={cn(CELL, key !== "f1" && key !== "f2" && "justify-center")}>
   884→          {cells[key]}
   885→        </div>
   886→      ))}
   887→    </div>
   888→  );
   889→}
