    98→    98→/** Map a CardSet patch (SetRow/ExerciseCard contract) onto the API's SetInput. */
    99→    99→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   100→   100→  const out: SetInput = {};
   101→   101→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   102→   102→  if ("reps" in patch) out.reps = patch.reps ?? null;
   103→   103→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   104→   104→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   105→   105→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   106→   106→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   107→   107→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   108→   108→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   109→   109→  if ("note" in patch) out.comment = patch.note ?? null;
   110→   110→  if ("done" in patch) out.isComplete = !!patch.done;
   111→   111→  return out;
   112→   112→}

   114→   114→/** §4.11 showMaxWeightBar — "Best 110×5" from the exercise's record table
   115→   115→ *  (max-weight entry; ties break to the higher rep count). */
   116→   116→function bestSetLine(records: RecordsDTO | undefined): string | null {
   117→   117→  if (!records) return null;
   118→   118→  const usable = records.actual.filter((r) => r.weight != null && r.weight > 0 && r.reps != null);
   119→   119→  const rows = usable.some((r) => !r.superseded) ? usable.filter((r) => !r.superseded) : usable;
   120→   120→  if (rows.length === 0) return null;
   121→   121→  let best = rows[0];
   122→   122→  for (const r of rows) {
   123→   123→    if (r.weight! > best.weight! || (r.weight === best.weight && r.reps! > best.reps!)) best = r;
   124→   124→  }
   125→   125→  return `Best ${trimNum(best.weight!)}×${best.reps!}`;
   126→   126→}

   859→   859→            {exercises.map((we) => {
   860→   860→              const group = we.groupId ? groupById.get(we.groupId) ?? null : null;
   861→   861→              const isCurrentGuidedCard = guidedCardWeId === we.id;
   862→   862→              // §4.11 guided video panel — only the CURRENT guided card, only
   863→   863→              // when a video exists (MediaBlock itself collapses to 0px otherwise)
   864→   864→              const showGuidedVideo =
   865→   865→                isCurrentGuidedCard && !!settings?.showVideoPanel && !!we.exercise.videoUrl;
   866→   866→              const showTipRow = tipsOpen && !!we.exercise.trainerTip;
   867→   867→              const underHeader =
   868→   868→                showGuidedVideo || showTipRow ? (
   869→   869→                  <>
   870→   870→                    {showGuidedVideo ? (
   871→   871→                      // Part 7 — exerciseCard.mediaBlock anchor (the block itself
   872→   872→                      // is consumer-supplied via the underHeader slot).
   873→   873→                      <div
   874→   874→                        {...tourAttrs({ id: "exerciseCard.mediaBlock", label: "Form video", help: "Guided demo video for the current exercise.", order: 140, when: ["guided"] })}
   875→   875→                      >
   876→   876→                        <GuidedVideoPanel weId={we.id} videoUrl={we.exercise.videoUrl} height={180} />
   877→   877→                      </div>
   878→   878→                    ) : null}
   879→   879→                    {showTipRow ? (
   880→   880→                      // Part 7 — exerciseCard.tipRow anchor (same slot contract).
   881→   881→                      <div
   882→   882→                        {...tourAttrs({ id: "exerciseCard.tipRow", label: "Trainer tip", help: "Coaching cue for this exercise; tap to expand.", order: 150 })}
   883→   883→                      >
   884→   884→                        <TrainerTipRow tip={we.exercise.trainerTip!} />
   885→   885→                      </div>
   886→   886→                    ) : null}
   887→   887→                  </>
   888→   888→                ) : undefined;
   889→   889→              return (
   890→   890→                <div
   891→   891→                  key={we.id}
   892→   892→                  ref={(el) => {
   893→   893→                    if (el) cardRefs.current.set(we.id, el);
   894→   894→                    else cardRefs.current.delete(we.id);
   895→   895→                  }}
   896→   896→                  className="relative flex-none"
   897→   897→                >
   898→   898→                  <SoloCard
   899→   899→                    mode="edit"
   900→   900→                    exercise={toCardExercise(we)}
   901→   901→                    sets={cardSets(we)}
   902→   902→                    collapsed={collapsedIds.has(we.id)}
   903→   903→                    visibleColumns={visibleColumns}
   904→   904→                    groupColour={group?.colour}
   905→   905→                    groupName={group?.name}
   906→   906→                    groupCode={groupCodes.codes.get(we.id)}
   907→   907→                    metaExtra={bestByWeId.get(we.id)}
   908→   908→                    underHeader={underHeader}
   909→   909→                    className={isCurrentGuidedCard ? "ring-2 ring-primary/40" : undefined}
   910→   910→                    onAction={handleCardAction(we)}
   911→   911→                  />
   912→   912→                  {notesWeId === we.id && (
   913→   913→                    <ExerciseNotesPopover
   914→   914→                      exercise={we.exercise}
   915→   915→                      open
   916→   916→                      onClose={() => setNotesWeId(null)}
   917→   917→                    />
   918→   918→                  )}
   919→   919→                  {groupWeId === we.id && (
   920→   920→                    <ExerciseGroupPopover
   921→   921→                      workout={workout}
   922→   922→                      we={we}
   923→   923→                      open
   924→   924→                      onClose={() => setGroupWeId(null)}
   925→   925→                    />
   926→   926→                  )}
   927→   927→                </div>
   928→   928→              );
   929→   929→            })}
   930→   930→
   931→   931→            {/* Part 7 — today.summary anchor (SummaryRow is a display-only row;
   932→   932→                the wrapper carries the declaration without touching it). */}
   933→   933→            <div
   934→   934→              {...tourAttrs({ id: "today.summary", label: "Day summary", help: "Sets, volume and PRs logged for the day.", order: 70, when: ["populated"] })}
   935→   935→            >
   936→   936→              <SummaryRow workout={workout} unit={defaultUnitFor(settings)} />
   937→   937→            </div>
   938→   938→
   939→   939→            {/* bottom breathing spacer — bars are flex siblings, nothing to clear */}
   940→   940→            <div className="h-4 flex-none" aria-hidden />
   941→   941→
   942→   942→            <ConfirmRemoveExercise
   943→   943→              exerciseName={removeWe?.exercise.name ?? ""}
   944→   944→              open={removeWeId != null}
   945→   945→              onClose={() => setRemoveWeId(null)}
   946→   946→              onConfirm={async () => {
   947→   947→                const id = removeWeId;
   948→   948→                setRemoveWeId(null);
   949→   949→                if (id) await removeExercise(id);
   950→   950→              }}
   951→   951→            />
   952→   952→
   953→   953→            {/* §4.11 finishBehaviour=ASK — Save/Discard dialogs (mounted only
   954→   954→                while the ASK stage is open; ALWAYS_SAVE never reaches it) */}
   955→   955→            {settings?.finishBehaviour === "ASK" ? (
   956→   956→              <FinishAskFlow
   957→   957→                workout={workout}
   958→   958→                open={finishAskOpen}
   959→   959→                onClose={() => setFinishAskOpen(false)}
   960→   960→                onSave={() => void finishWorkout()}
   961→   961→                dateKey={dateKey}
   962→   962→                onDiscarded={() => {
   963→   963→                  // a discarded day can't keep a rest countdown running
   964→   964→                  rest.skip();
   965→   965→                  setFinishAskOpen(false);
   966→   966→                }}
   967→   967→              />
   968→   968→            ) : null}
   969→   969→
   970→   970→            {sessionFromWorkout.dialog}
   971→   971→          </>
   972→   972→        )}
   973→   973→      </ScrollBody>
   974→   974→    </Screen>
   975→   975→  );
   976→   976→}
