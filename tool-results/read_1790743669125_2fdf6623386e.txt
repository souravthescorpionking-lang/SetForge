   328→function RecordsTab({ exercise }: { exercise: ExerciseDTO }) {
   329→  const settings = useApp((s) => s.settings);
   330→  const unit = exerciseUnit(exercise, settings);
   331→  const fields = useMemo(() => new Set(fieldsForType(exercise.type)), [exercise.type]);
   332→  const tracksWeightReps = fields.has("weight") && fields.has("reps");
   333→
   334→  const recordsQuery = useQuery({
   335→    queryKey: qk.exerciseRecords(exercise.id),
   336→    queryFn: () => exercisesApi.records(exercise.id),
   337→    staleTime: 60_000,
   338→  });
   339→  const historyQuery = useQuery({
   340→    queryKey: qk.exerciseHistory(exercise.id),
   341→    queryFn: () => exercisesApi.history(exercise.id),
   342→    staleTime: 60_000,
   343→  });
   344→
   345→  const rows = useMemo<{ out: RecordRow[]; actual: RecordRow[]; estimated: RecordRow[] }>(() => {
   346→    const out: RecordRow[] = [];
   347→    const entries = historyQuery.data ?? [];
   348→    const records: RecordsDTO | undefined = recordsQuery.data;
   349→
   350→    // ---- bests from the logged sets (legacy records-tab computations) ----
   351→    let bestWeight: { w: number; day: string } | null = null;
   352→    let bestVolume: { v: number; day: string } | null = null;
   353→    let bestE1rm: { v: number; day: string } | null = null;
   354→    let bestDistance: { v: number; day: string } | null = null;
   355→    let bestTime: { v: number; day: string } | null = null;
   356→    const repLimit = settings?.estOneRmRepLimit ?? 10;
   357→    const method = settings?.e1rmMethod ?? "BRZYCKI";
   358→
   359→    for (const en of entries) {
   360→      const day = dayKeyOf(en.date);
   361→      for (const s of en.sets) {
   362→        if (s.weight != null && (bestWeight == null || s.weight > bestWeight.w)) {
   363→          bestWeight = { w: s.weight, day };
   364→        }
   365→        if (
   366→          s.weight != null &&
   367→          s.reps != null &&
   368→          s.weight > 0 &&
   369→          s.reps > 0 &&
   370→          (bestVolume == null || s.weight * s.reps > bestVolume.v)
   371→        ) {
   372→          bestVolume = { v: s.weight * s.reps, day };
   373→        }
   374→        // same guard as the server: failure sets are not e1RM inputs
   375→        if (
   376→          s.weight != null &&
   377→          s.reps != null &&
   378→          s.reps >= 1 &&
   379→          s.reps <= repLimit &&
   380→          s.setType !== "FAILURE"
   381→        ) {
   382→          const e = estOneRmByMethod(s.weight, s.reps, method, s.rpe);
   383→          if (e > 0 && (bestE1rm == null || e > bestE1rm.v)) bestE1rm = { v: e, day };
   384→        }
   385→        if (s.distance != null && s.distance > 0 && (bestDistance == null || s.distance > bestDistance.v)) {
   386→          bestDistance = { v: s.distance, day };
   387→        }
   388→        if (s.timeSec != null && s.timeSec > 0 && (bestTime == null || s.timeSec > bestTime.v)) {
   389→          bestTime = { v: s.timeSec, day };
   390→        }
   391→      }
   392→    }
   393→
   394→    if (tracksWeightReps) {
   395→      if (bestWeight) {
   396→        out.push({ lift: "Best weight", value: weightLabel(bestWeight.w, unit), date: formatDayShort(bestWeight.day) });
   397→      }
   398→      if (bestVolume) {
   399→        out.push({ lift: "Best volume set", value: `${round1(bestVolume.v)} ${unit}`, date: formatDayShort(bestVolume.day) });
   400→      }
   401→      if (bestE1rm) {
   402→        // value prefers the server computation when loaded (authoritative)
   403→        const v = records && records.estimatedOneRm > 0 ? records.estimatedOneRm : bestE1rm.v;
   404→        out.push({ lift: "Estimated 1RM", value: `${round1(v)} ${unit}`, date: formatDayShort(bestE1rm.day) });
   405→      }
   406→    }
   407→    if (bestDistance) out.push({ lift: "Best distance", value: `${round2(bestDistance.v)} km`, date: formatDayShort(bestDistance.day) });
   408→    if (bestTime) out.push({ lift: "Best time", value: formatDuration(bestTime.v), date: formatDayShort(bestTime.day) });
   409→
   410→    // ---- per-reps actual PRs (records API, superseded dimmed) ----
   411→    const actual: RecordRow[] = (records?.actual ?? []).map((r) => ({
   412→      lift: `Best set at ${r.reps} ${r.reps === 1 ? "rep" : "reps"}`,
   413→      value: weightLabel(r.weight, unit),
   414→      date: formatDayShort(dayKeyOf(r.date)),
   415→      dim: r.superseded,
   416→    }));
   417→
   418→    // ---- estimated rep maxes (nRM, legacy grid → rows; 1RM row above) ----
   419→    const estimated: RecordRow[] = (records?.estimated ?? [])
   420→      .filter((r) => r.reps > 1)
   421→      .map((r) => ({ lift: `Estimated ${r.reps}RM`, value: `${round1(r.weight)} ${unit}`, date: null }));
   422→
   423→    return { out, actual, estimated };
   424→  }, [historyQuery.data, recordsQuery.data, settings, unit, tracksWeightReps]);
   425→
   426→  if (historyQuery.isLoading || recordsQuery.isLoading) {
   427→    return (
   428→      <div className="flex flex-col gap-2" aria-busy="true" aria-label="Loading records">
   429→        {[0, 1, 2, 3, 4, 5, 6].map((i) => (
   430→          <Skeleton key={i} className="h-10 rounded-lg" />
   431→        ))}
   432→      </div>
   433→    );
   434→  }
   435→
   436→  const nothing = rows.out.length === 0 && rows.actual.length === 0 && rows.estimated.length === 0;
   437→
   438→  return (
   439→    <div className="flex flex-col gap-3">
   440→      {nothing ? (
   441→        <div
   442→          data-row
   443→          className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"
   444→        >
   445→          No records yet — log a set to start tracking.
   446→        </div>
   447→      ) : null}
   448→
   449→      {rows.out.length > 0 ? (
   450→        <section className="flex flex-col gap-1" aria-label="Bests">
   451→          <SectionHeader label="Bests" />
   452→          {rows.out.map((r) => (
   453→            <RecordRowLine key={r.lift} row={r} />
   454→          ))}
   455→        </section>
   456→      ) : null}
   457→
   458→      {!tracksWeightReps && (rows.out.length > 0 || nothing) ? (
   459→        <p className="px-1 text-xs text-muted-foreground">
   460→          Weight × reps records need a weighted exercise — showing bests only.
   461→        </p>
   462→      ) : null}
   463→
   464→      {rows.actual.length > 0 ? (
   465→        <section className="flex flex-col gap-1" aria-label="Rep records">
   466→          <SectionHeader label="Rep records" />
   467→          {rows.actual.map((r) => (
   468→            <RecordRowLine key={r.lift} row={r} />
   469→          ))}
   470→        </section>
   471→      ) : null}
   472→
   473→      {rows.estimated.length > 0 ? (
   474→        <section className="flex flex-col gap-1" aria-label="Estimated maxes">
   475→          <SectionHeader label="Estimated maxes" />
   476→          {rows.estimated.map((r) => (
   477→            <RecordRowLine key={r.lift} row={r} />
   478→          ))}
   479→        </section>
   480→      ) : null}
   481→
   482→      {rows.actual.length > 0 ? (
   483→        <p className="px-1 text-xs text-muted-foreground">
   484→          A PR is superseded when a heavier set at the same-or-higher rep count was logged later.
   485→        </p>
   486→      ) : null}
   487→    </div>
   488→  );
   489→}

   600→/** 96px inline goal creator (NOT a data-row): type select + target input. */
   601→function GoalCreator({
   602→  exercise,
   603→  unit,
   604→  onDone,
   605→}: {
   606→  exercise: ExerciseDTO;
   607→  unit: WeightUnit;
   608→  onDone: () => void;
   609→}) {
   610→  const run = useOfflineRun();
   611→  const inv = useInvalidate();

   618→    field === "targetWeight"

   621→        ? "Target reps"

   669→            ))}
   670→          </SelectContent>
   671→        </Select>
   672→      </div>
   673→      <div className="flex min-h-0 flex-1 gap-2">

   675→          type="number"

   678→          min="0"
   679→          className="h-full flex-1 rounded-lg tabular-nums"

   682→          placeholder={placeholder}
   683→          aria-label={placeholder}

   686→        <Button
   687→          type="button"
   688→          className="h-full flex-1 rounded-lg font-semibold"

   690→          tour={{ id: "exerciseOverview.goalCreate", label: "Create goal", help: "Save the goal for this exercise.", order: 110 }}
   691→          onClick={() => void save()}
   692→        >
   693→          {saving ? "Saving…" : "Add goal"}
   694→        </Button>
   695→      </div>
   696→    </div>
   697→  );

   700→/** 96px inline expansion (NOT a data-row): progress + target editor + delete. */
   701→function GoalExpansion({
   702→  goal,
   703→  unit,
   704→  onDone,
   705→}: {
   706→  goal: GoalDTO;
   707→  unit: WeightUnit;
   708→  onDone: () => void;
   709→}) {
   710→  const run = useOfflineRun();
   711→  const inv = useInvalidate();
   712→  const field = goalTargetField(goal.type);
   713→  const initial =
   714→    field === "targetWeight"
   715→      ? goal.targetWeight
   716→      : field === "targetReps"
   717→        ? goal.targetReps
   718→        : field === "targetDistance"
   719→          ? goal.targetDistance
   720→          : goal.targetTimeSec;
   721→
   722→  const [target, setTarget] = useState<string>(initial != null ? String(initial) : "");
   723→  const [saving, setSaving] = useState(false);
   724→  const [armed, setArmed] = useState(false);
   725→
   726→  // auto-disarm the two-tap delete after a pause
   727→  useEffect(() => {
   728→    if (!armed) return;
   729→    const t = window.setTimeout(() => setArmed(false), 4000);
   730→    return () => window.clearTimeout(t);
   731→  }, [armed]);
   732→
   733→  const save = async () => {
   734→    const n = Number(target);
   735→    if (target.trim() === "" || !Number.isFinite(n) || n <= 0) {
   736→      toast.error("Set a target value above zero");
   737→      return;
   738→    }
   739→    setSaving(true);
   740→    const payload: GoalInput = {
   741→      type: goal.type,
   742→      targetWeight: field === "targetWeight" ? n : null,
   743→      targetReps: field === "targetReps" ? n : null,
   744→      targetDistance: field === "targetDistance" ? n : null,
   745→      targetTimeSec: field === "targetTimeSec" ? n : null,
   746→    };
   747→    const ok = await run({
   748→      label: "Goal update",
   749→      path: `/api/goals/${goal.id}`,
   750→      method: "PATCH",
   751→      body: payload,
   752→      run: () => goalsApi.update(goal.id, payload),
   753→      successMsg: "Goal updated",
   754→      onDone: () => inv.goals(),
   755→    });
   756→    setSaving(false);
   757→    if (ok) onDone();
   758→  };
   759→
   760→  const remove = async () => {
   761→    if (!armed) {
   762→      setArmed(true);
   763→      return;
   764→    }
   765→    const ok = await run({
   766→      label: "Goal delete",
   767→      path: `/api/goals/${goal.id}`,
   768→      method: "DELETE",
   769→      run: () => goalsApi.remove(goal.id),
   770→      successMsg: "Goal deleted",
   771→      onDone: () => inv.goals(),
   772→    });
   773→    if (ok) onDone();
   774→  };
   775→
   776→  return (
   777→    <div className="flex h-24 flex-none flex-col gap-1 rounded-lg border bg-card p-1.5">
   778→      <div className="flex min-h-0 flex-1 items-center gap-2 overflow-hidden whitespace-nowrap">
   779→        <Progress
   780→          value={Math.min(100, goal.pct)}
   781→          className={cn("h-2 flex-1", goal.achieved && "*:data-[slot=progress-indicator]:bg-emerald-500")}
   782→          aria-label={`${goal.pct}% of goal`}
   783→        />
   784→        <span className="flex-none truncate text-xs font-semibold tabular-nums text-muted-foreground">
   785→          {goalValueLabel(goal.type, goal.current, unit)} / {goalValueLabel(goal.type, goal.target, unit)}
   786→        </span>
   787→      </div>
   788→      <div className="flex min-h-0 flex-1 gap-2">
   789→        <Input
   790→          type="number"
   791→          inputMode="decimal"
   792→          step="any"
   793→          min="0"
   794→          className="h-full flex-1 rounded-lg tabular-nums"
   795→          value={target}
   796→          onChange={(e) => setTarget(e.target.value)}
   797→          aria-label="Goal target"
   798→          {...tourAttrs({ id: "exerciseOverview.goalSaveTarget", label: "New target", help: "Type the goal's new target value.", order: 120 })}
   799→        />
   800→        <Button
   801→          type="button"
   802→          className="h-full flex-1 rounded-lg font-semibold"
   803→          disabled={saving}
   804→          tour={{ id: "exerciseOverview.goalSave", label: "Save target", help: "Update the goal's target value.", order: 120 }}
   805→          onClick={() => void save()}
   806→        >
   807→          {saving ? "Saving…" : "Save target"}
   808→        </Button>
   809→        <Button
   810→          type="button"
   811→          variant="outline"
   812→          className={cn(
   813→            "h-full flex-none gap-1 rounded-lg",
   814→            armed ? "border-destructive/50 text-destructive" : "text-muted-foreground",
   815→          )}
   816→          tour={{ id: "exerciseOverview.goalDelete", label: "Delete goal", help: "Tap twice to delete the goal for good.", order: 130 }}
   817→          onClick={() => void remove()}
   818→        >
   819→          <Trash2 className="h-3.5 w-3.5" aria-hidden />
   820→          {armed ? "Sure?" : "Delete"}
   821→        </Button>
   822→      </div>
   823→    </div>
   824→  );
   825→}

   831→function HistoryTab({
   832→  exercise,

   836→  visibleColumns: CardVisibleColumns;
   837→}) {

   842→    queryKey: qk.exerciseHistory(exercise.id),
   843→    queryFn: () => exercisesApi.history(exercise.id),
   844→    staleTime: 60_000,
   845→  });
   846→  // records → date-level PR badges (legacy history-tab pattern)

   848→    queryKey: qk.exerciseRecords(exercise.id),
   849→    queryFn: () => exercisesApi.records(exercise.id),

   860→      const list = map.get(key);
   861→      if (list) list.push(en);
   862→      else map.set(key, [en]);
   863→    }

   866→

   868→    () => new Set((recordsQuery.data?.actual ?? []).map((r) => dayKeyOf(r.date))),
   869→    [recordsQuery.data],

   876→          <Skeleton key={i} className="h-14 rounded-lg" />
   877→        ))}
   878→      </div>
   879→    );

   883→    return (
   884→      <div
   885→        data-row
   886→        className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"

   938→              const collapsed = collapsedIds.has(entry.workoutExerciseId);

   980→                        }
   981→                      }}

   985→                    <div className="relative flex-none">
   986→                      <ExerciseNotesPopover
   987→                        exercise={exercise}
   988→                        open
   989→                        onClose={() => setNotesEntryId(null)}
   990→                      />
   991→                    </div>
   992→                  ) : null}
   993→                </div>
   994→              );
   995→            })}
   996→          </section>
   997→        );
   998→      })}
   999→      {(historyQuery.data?.length ?? 0) >= 100 ? (
  1000→        <p className="px-1 text-xs text-muted-foreground">Showing the most recent 100 sessions.</p>
  1001→      ) : null}
  1002→    </div>
  1003→  );
  1004→}
