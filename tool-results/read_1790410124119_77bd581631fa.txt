    74→// ---------- audio ----------
    75→function useBeep(enabled: boolean) {
    76→  const ctxRef = useRef<AudioContext | null>(null);
    77→
    78→  const ctx = useCallback(() => {
    79→    if (typeof window === "undefined") return null;
    80→    if (!ctxRef.current) {
    81→      const AC = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    82→      if (!AC) return null;
    83→      ctxRef.current = new AC();
    84→    }
    85→    if (ctxRef.current.state === "suspended") void ctxRef.current.resume();
    86→    return ctxRef.current;
    87→  }, []);
    88→
    89→  const beep = useCallback(
    90→    (freq: number, dur: number, when = 0, gain = 0.12) => {
    91→      if (!enabled) return;
    92→      const ac = ctx();
    93→      if (!ac) return;
    94→      const osc = ac.createOscillator();
    95→      const g = ac.createGain();
    96→      osc.type = "sine";
    97→      osc.frequency.value = freq;
    98→      const t = ac.currentTime + when;
    99→      g.gain.setValueAtTime(gain, t);
   100→      g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
   101→      osc.connect(g).connect(ac.destination);
   102→      osc.start(t);
   103→      osc.stop(t + dur);
   104→    },
   105→    [ctx, enabled],
   106→  );
   107→
   108→  return { beep, warmupAudio: ctx };
   109→}

   111→function useVibrate(enabled: boolean) {

   113→    (pattern: number | number[]) => {

   121→/** Speaks short phase cues via the local SpeechSynthesis engine (no network). */
   122→function useSpeech(enabled: boolean) {
   123→  return useCallback(
   124→    (text: string) => {
   125→      if (!enabled || typeof window === "undefined" || !window.speechSynthesis) return;
   126→      try {
   127→        window.speechSynthesis.cancel(); // cut off any still-speaking cue
   128→        const u = new SpeechSynthesisUtterance(text);
   129→        u.rate = 1.15;
   130→        u.volume = 0.9;
   131→        u.pitch = 1;
   132→        window.speechSynthesis.speak(u);
   133→      } catch {
   134→        /* speech unavailable — silent fallback */
   135→      }
   136→    },
   137→    [enabled],
   138→  );
   139→}

   141→/** Keeps the screen awake while active (re-acquires on tab visibility). */
   142→function useWakeLock(active: boolean) {
   143→  useEffect(() => {
   144→    if (!active || typeof navigator === "undefined" || !("wakeLock" in navigator)) return;
   145→    let lock: { release: () => Promise<void> } | null = null;
   146→    let cancelled = false;
   147→    const request = async () => {
   148→      try {
   149→        if (cancelled) return;
   150→        lock = await navigator.wakeLock.request("screen");
   151→      } catch {
   152→        /* wake lock is best-effort */
   153→      }
   154→    };
   155→    const onVisible = () => {
   156→      if (document.visibilityState === "visible") void request();
   157→    };
   158→    void request();
   159→    document.addEventListener("visibilitychange", onVisible);
   160→    return () => {
   161→      cancelled = true;
   162→      document.removeEventListener("visibilitychange", onVisible);
   163→      void lock?.release().catch(() => {});
   164→      lock = null;
   165→    };
   166→  }, [active]);
   167→}

   169→/** Exposes the timer on the OS media session: lock-screen / hardware-key
   170→ *  controls (play/pause, next = skip phase) plus live phase metadata and
   171→ *  position. Best-effort — silently inert where unsupported (headless). */
   172→function useMediaSession(opts: {
   173→  active: boolean;
   174→  playing: boolean;
   175→  title: string;
   176→  album: string;
   177→  positionSec: number;
   178→  durationSec: number;
   179→  onPlay: () => void;
   180→  onPause: () => void;
   181→  onNext: () => void;
   182→  onStop: () => void;
   183→}) {
   184→  const { active, playing, title, album, positionSec, durationSec } = opts;
   185→  const handlers = useRef(opts);
   186→  useEffect(() => {
   187→    handlers.current = opts;
   188→  });
   189→
   190→  const ms = typeof navigator !== "undefined" && "mediaSession" in navigator ? navigator.mediaSession : null;
   191→
   192→  // register action handlers once per session
   193→  useEffect(() => {
   194→    if (!ms) return;
   195→    const trySet = (action: string, fn: (() => void) | null) => {
   196→      try {
   197→        ms.setActionHandler(action as MediaSessionAction, fn as never);
   198→      } catch {
   199→        /* action unsupported on this platform */
   200→      }
   201→    };
   202→    trySet("play", () => handlers.current.onPlay());
   203→    trySet("pause", () => handlers.current.onPause());
   204→    trySet("nexttrack", () => handlers.current.onNext());
   205→    trySet("stop", () => handlers.current.onStop());
   206→    trySet("previoustrack", () => handlers.current.onStop());
   207→    return () => {
   208→      trySet("play", null);
   209→      trySet("pause", null);
   210→      trySet("nexttrack", null);
   211→      trySet("stop", null);
   212→      trySet("previoustrack", null);
   213→    };
   214→  }, [ms]);
   215→
   216→  // metadata while active; cleared when idle/complete
   217→  useEffect(() => {
   218→    if (!ms) return;
   219→    if (!active) {
   220→      try {
   221→        ms.metadata = null;
   222→        ms.playbackState = "none";
   223→      } catch {
   224→        /* best-effort */
   225→      }
   226→      return;
   227→    }
   228→    try {
   229→      if (typeof MediaMetadata !== "undefined") {
   230→        ms.metadata = new MediaMetadata({
   231→          title,
   232→          artist: "SetForge",
   233→          album,
   234→        });
   235→      }
   236→      ms.playbackState = playing ? "playing" : "paused";
   237→    } catch {
   238→      /* best-effort */
   239→    }
   240→  }, [ms, active, playing, title, album]);
   241→
   242→  // position state (scrubber on the lock screen) — only while playing
   243→  useEffect(() => {
   244→    if (!ms || !active || !playing) return;
   245→    try {
   246→      if (typeof ms.setPositionState === "function" && Number.isFinite(durationSec) && durationSec > 0) {
   247→        ms.setPositionState({
   248→          duration: durationSec,
   249→          playbackRate: 1,
   250→          position: Math.min(Math.max(0, positionSec), durationSec),
   251→        });
   252→      }
   253→    } catch {
   254→      /* best-effort */
   255→    }
   256→  }, [ms, active, playing, positionSec, durationSec]);
   257→}

   267→export function IntervalTimer() {
   268→  const [presetId, setPresetId] = useState("tabata");
   269→  const [config, setConfig] = useState<Config>(PRESETS[0].config);
   270→  const [status, setStatus] = useState<Status>("idle");
   271→  const [phase, setPhase] = useState<Phase>("prepare");
   272→  const [round, setRound] = useState(0);
   273→  const [remaining, setRemaining] = useState(PRESETS[0].config.prepareSec);
   274→  const [phaseTotalSec, setPhaseTotalSec] = useState(PRESETS[0].config.prepareSec);
   275→  const [elapsedTotal, setElapsedTotal] = useState(0);
   276→  const [sound, setSound] = useState(true);
   277→  const [haptics, setHaptics] = useState(true);
   278→  const [voice, setVoice] = useState(false);
   279→  const [saveOpen, setSaveOpen] = useState(false);
   280→  const [presetName, setPresetName] = useState("");
   281→  // rename dialog state
   282→  const [renameOpen, setRenameOpen] = useState(false);
   283→  const [renameTarget, setRenameTarget] = useState<TimerPresetDTO | null>(null);
   284→  const [renameName, setRenameName] = useState("");
   285→
   286→  // server-synced user presets
   287→  const { data: savedPresets } = useTimerPresets();
   288→  const invalidate = useInvalidate();
   289→
   290→  const savePreset = useMutation({
   291→    mutationFn: (data: { name: string; config: Config }) =>
   292→      timerPresetsApi.create({ name: data.name, ...data.config }),
   293→    onSuccess: (p) => {
   294→      invalidate.timerPresets();
   295→      setSaveOpen(false);
   296→      setPresetName("");
   297→      setPresetId(p.id); // highlight the freshly saved preset
   298→      toast.success(`Preset “${p.name}” saved`);
   299→    },
   300→    onError: (e) => toast.error(e instanceof Error ? e.message : "Could not save preset"),
   301→  });
   302→
   303→  const deletePreset = useMutation({
   304→    mutationFn: (id: string) => timerPresetsApi.remove(id),
   305→    onSuccess: () => {
   306→      invalidate.timerPresets();
   307→      toast.success("Preset deleted");
   308→    },
   309→    onError: (e) => toast.error(e instanceof Error ? e.message : "Could not delete preset"),
   310→  });
   311→
   312→  const renamePreset = useMutation({
   313→    mutationFn: (data: { id: string; name: string }) => timerPresetsApi.update(data.id, { name: data.name }),
   314→    onSuccess: (p) => {
   315→      invalidate.timerPresets();
   316→      setRenameOpen(false);
   317→      toast.success(`Renamed to “${p.name}”`);
   318→    },
   319→    onError: (e) => toast.error(e instanceof Error ? e.message : "Could not rename preset"),
   320→  });
   321→
   322→  const submitRename = () => {
   323→    const name = renameName.trim();
   324→    if (!name) {
   325→      toast.error("Give the preset a name");
   326→      return;
   327→    }
   328→    if (renameTarget) renamePreset.mutate({ id: renameTarget.id, name });
   329→  };
   330→
   331→  const engineRef = useRef<Engine | null>(null);
   332→  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null);
   333→  const beepedRef = useRef<Set<number>>(new Set());
   334→  const elapsedAtRef = useRef(0); // total elapsed ms when engine last ran
   335→  const engineStartRef = useRef(0); // wall clock when engine (re)started — for elapsed
   336→
   337→  const { beep, warmupAudio } = useBeep(sound);
   338→  const vibrate = useVibrate(haptics);
   339→  const speak = useSpeech(voice);
   340→  useWakeLock(status === "running");
   341→
   342→  const totalSec = config.prepareSec + config.rounds * config.workSec + Math.max(0, config.rounds - 1) * config.restSec;
   343→  const totalMs = totalSec * 1000;
   344→
   345→  const stopTick = () => {
   346→    if (tickRef.current) {
   347→      clearInterval(tickRef.current);
   348→      tickRef.current = null;
   349→    }
   350→  };
   351→
   352→  const enterPhase = useCallback(
   353→    (next: Phase, nextRound: number, now: number) => {
   354→      const durMs =
   355→        next === "prepare" ? config.prepareSec * 1000 : next === "work" ? config.workSec * 1000 : next === "rest" ? config.restSec * 1000 : 0;
   356→      engineRef.current = { phase: next, round: nextRound, phaseEndsAt: now + durMs, phaseTotal: durMs };
   357→      beepedRef.current = new Set();
   358→      setPhase(next);
   359→      setRound(nextRound);
   360→      setRemaining(Math.ceil(durMs / 1000));
   361→      setPhaseTotalSec(Math.max(1, Math.round(durMs / 1000)));
   362→
   363→      if (next === "work") {
   364→        beep(880, 0.35, 0, 0.16);
   365→        vibrate(200);
   366→        speak(nextRound === config.rounds && config.rounds > 1 ? "Last round — work!" : "Work!");
   367→      } else if (next === "rest") {
   368→        beep(440, 0.3, 0, 0.14);
   369→        vibrate([120, 60, 120]);
   370→        speak("Rest");
   371→      } else if (next === "prepare") {
   372→        beep(660, 0.2, 0, 0.12);
   373→        speak("Get ready");
   374→      }
   375→    },
   376→    [config, beep, vibrate, speak],
   377→  );
   378→
   379→  const tick = useCallback(() => {
   380→    const eng = engineRef.current;
   381→    if (!eng) return;
   382→    const now = Date.now();
   383→    const remainMs = Math.max(0, eng.phaseEndsAt - now);
   384→    const remainSec = Math.ceil(remainMs / 1000);
   385→    setRemaining(remainSec);
   386→    setElapsedTotal(elapsedAtRef.current + (now - engineStartRef.current));
   387→
   388→    // countdown cues at 3/2/1
   389→    if (remainSec <= 3 && remainSec > 0 && !beepedRef.current.has(remainSec)) {
   390→      beepedRef.current.add(remainSec);
   391→      beep(520, 0.08, 0, 0.1);
   392→    }
   393→
   394→    if (remainMs === 0) {
   395→      // advance
   396→      if (eng.phase === "prepare") {
   397→        enterPhase("work", 1, now);
   398→      } else if (eng.phase === "work") {
   399→        if (eng.round >= config.rounds) {
   400→          // finished
   401→          engineRef.current = null;
   402→          setPhase("done");
   403→          setRemaining(0);
   404→          setStatus("idle");
   405→          stopTick();
   406→          elapsedAtRef.current = 0;
   407→          beep(880, 0.25, 0, 0.15);
   408→          beep(1100, 0.35, 0.3, 0.15);
   409→          beep(1320, 0.5, 0.7, 0.16);
   410→          vibrate([300, 100, 300]);
   411→          speak("Complete! Great job!");
   412→          return;
   413→        }
   414→        enterPhase(config.restSec > 0 ? "rest" : "work", eng.round + (config.restSec > 0 ? 0 : 1), now);
   415→      } else if (eng.phase === "rest") {
   416→        enterPhase("work", eng.round + 1, now);
   417→      }
   418→    }
   419→  }, [config, enterPhase, beep, vibrate, speak]);
   420→
   421→  const start = () => {
   422→    warmupAudio(); // user gesture → unlock audio
   423→    const now = Date.now();
   424→    engineStartRef.current = now;
   425→    elapsedAtRef.current = 0;
   426→    enterPhase(config.prepareSec > 0 ? "prepare" : "work", config.prepareSec > 0 ? 0 : 1, now);
   427→    setStatus("running");
   428→    stopTick();
   429→    tickRef.current = setInterval(tick, 100);
   430→  };
   431→
   432→  const pause = () => {
   433→    const eng = engineRef.current;
   434→    if (!eng) return;
   435→    stopTick();
   436→    const remainMs = Math.max(0, eng.phaseEndsAt - Date.now());
   437→    elapsedAtRef.current += Date.now() - engineStartRef.current;
   438→    eng.phaseEndsAt = -1; // paused marker
   439→    (eng as Engine & { pausedRemainMs?: number }).pausedRemainMs = remainMs;
   440→    setStatus("paused");
   441→  };
   442→
   443→  const resume = () => {
   444→    const eng = engineRef.current;
   445→    if (!eng) return;
   446→    const remainMs = (eng as Engine & { pausedRemainMs?: number }).pausedRemainMs ?? 0;
   447→    const now = Date.now();
   448→    engineStartRef.current = now;
   449→    eng.phaseEndsAt = now + remainMs;
   450→    setStatus("running");
   451→    stopTick();
   452→    tickRef.current = setInterval(tick, 100);
   453→  };
   454→
   455→  const skipPhase = () => {
   456→    const eng = engineRef.current;
   457→    if (!eng) return;
   458→    const now = Date.now();
   459→    if (eng.phase === "prepare") {
   460→      enterPhase("work", 1, now);
   461→    } else if (eng.phase === "work") {
   462→      if (eng.round >= config.rounds) {
   463→        engineRef.current = null;
   464→        setPhase("done");
   465→        setStatus("idle");
   466→        stopTick();
   467→        return;
   468→      }
   469→      if (config.restSec > 0) enterPhase("rest", eng.round, now);
   470→      else enterPhase("work", eng.round + 1, now);
   471→    } else {
   472→      enterPhase("work", eng.round + 1, now);
   473→    }
   474→  };
   475→
   476→  const reset = () => {
   477→    stopTick();
   478→    engineRef.current = null;
   479→    elapsedAtRef.current = 0;
   480→    setStatus("idle");
   481→    setPhase("prepare");
   482→    setRound(0);
   483→    setRemaining(config.prepareSec);
   484→    setPhaseTotalSec(Math.max(1, config.prepareSec));
   485→    setElapsedTotal(0);
   486→  };
   487→
   488→  useEffect(() => stopTick, []);
   489→
   490→  // preset selection (resets the timer)
   491→  const applyPreset = (p: Preset) => {
   492→    reset();
   493→    setPresetId(p.id);
   494→    setConfig(p.config);
   495→    setRemaining(p.config.prepareSec);
   496→    setPhaseTotalSec(Math.max(1, p.config.prepareSec));
   497→  };
   498→
   499→  const applySavedPreset = (p: TimerPresetDTO) => {
   500→    reset();
   501→    setPresetId(p.id);
   502→    setConfig({ prepareSec: p.prepareSec, workSec: p.workSec, restSec: p.restSec, rounds: p.rounds });
   503→    setRemaining(p.prepareSec);
   504→    setPhaseTotalSec(Math.max(1, p.prepareSec));
   505→  };
   506→
   507→  const submitSavePreset = () => {
   508→    const name = presetName.trim();
   509→    if (!name) {
   510→      toast.error("Give the preset a name");
   511→      return;
   512→    }
   513→    savePreset.mutate({ name, config });
   514→  };
   515→
   516→  // ring geometry
   517→  const R = 120;
   518→  const C = 2 * Math.PI * R;
   519→  const phaseTotalMs = phaseTotalSec * 1000;
   520→  const phaseFrac =
   521→    status === "idle" && phase === "prepare" ? 1 : phase === "done" ? 1 : Math.min(1, Math.max(0, (remaining * 1000) / phaseTotalMs));
   522→  const dash = C * (1 - phaseFrac);
   523→
   524→  const mmss = (sec: number) => {
   525→    const m = Math.floor(sec / 60);
   526→    const s = sec % 60;
   527→    return `${m}:${String(s).padStart(2, "0")}`;
   528→  };
   529→
   530→  const style = PHASE_STYLE[phase];
   531→  const running = status === "running";
   532→  const currentRound = phase === "done" ? config.rounds : phase === "rest" ? round + 1 : Math.max(1, round || 1);
   533→
   534→  // expose the timer on the OS media session (lock-screen controls)
   535→  useMediaSession({
   536→    active: status === "running" || status === "paused",
   537→    playing: running,
   538→    title: phase === "done" ? "Complete!" : `${style.label} · ${remaining}s left`,
   539→    album: `Round ${Math.max(currentRound, 1)}/${config.rounds} · ${config.workSec}s/${config.restSec}s`,
   540→    positionSec: Math.max(0, phaseTotalSec - remaining),
   541→    durationSec: phaseTotalSec,
   542→    onPlay: resume,
   543→    onPause: pause,
   544→    onNext: skipPhase,
   545→    onStop: reset,
