    56→    56→// ── local unit conversion helpers (metric is canonical on the wire) ──────────
    57→    57→
    58→    58→const CM_PER_IN = 2.54;
    59→    59→const KG_PER_LB = 0.45359237;
    60→    60→
    61→    61→const roundTo1 = (n: number): number => Math.round(n * 10) / 10;
    62→    62→
    63→    63→/** 177.8 cm → "5 ft 10 in" (imperial height display). */
    64→    64→function cmToFtIn(cm: number): string {
    65→    65→  const totalIn = cm / CM_PER_IN;
    66→    66→  let ft = Math.floor(totalIn / 12);
    67→    67→  let inch = Math.round(totalIn - ft * 12);
    68→    68→  if (inch >= 12) {
    69→    69→    ft += 1;
    70→    70→    inch -= 12;
    71→    71→  }
    72→    72→  return `${ft} ft ${inch} in`;
    73→    73→}
    74→    74→
    75→    75→const cmToIn = (cm: number): number => cm / CM_PER_IN;
    76→    76→const inToCm = (inches: number): number => inches * CM_PER_IN;
    77→    77→const kgToLb = (kg: number): number => kg / KG_PER_LB;
    78→    78→const lbToKg = (lb: number): number => lb * KG_PER_LB;
    79→    79→const identity = (n: number): number => n;

   762→   762→function MeasureEditor({
   763→   763→  title,
   764→   764→  unitLabel,
   765→   765→  placeholder,
   766→   766→  initialMetric,
   767→   767→  imperial,
   768→   768→  toDisplay,
   769→   769→  toMetric,
   770→   770→  min,
   771→   771→  max,
   772→   772→  canonicalUnit,
   773→   773→  onSave,
   774→   774→  onCancel,
   775→   775→}: {
   776→   776→  title: string;
   777→   777→  unitLabel: string;
   778→   778→  placeholder: string;
   779→   779→  initialMetric: number | null;
   780→   780→  imperial: boolean;
   781→   781→  toDisplay: (metric: number) => number;
   782→   782→  toMetric: (display: number) => number;
   783→   783→  min: number;
   784→   784→  max: number;
   785→   785→  canonicalUnit: string;
   786→   786→  onSave: (metric: number) => Promise<void>;
   787→   787→  onCancel: () => void;
   788→   788→}) {
   789→   789→  const [raw, setRaw] = useState(initialMetric != null ? String(roundTo1(toDisplay(initialMetric))) : "");
   790→   790→  const [saving, setSaving] = useState(false);
   791→   791→
   792→   792→  const parsed = raw.trim() === "" ? null : Number(raw);
   793→   793→  const metric = parsed == null || !Number.isFinite(parsed) ? null : toMetric(parsed);
   794→   794→  const error =
   795→   795→    parsed == null ? null
   796→   796→    : !Number.isFinite(parsed) || parsed < min || parsed > max
   797→   797→      ? `Enter ${min}–${max} ${unitLabel}`
   798→   798→      : null;
   799→   799→
   800→   800→  const save = async () => {
   801→   801→    if (metric == null || error) return;
   802→   802→    setSaving(true);
   803→   803→    try {
   804→   804→      await onSave(roundTo1(metric));
   805→   805→    } finally {
   806→   806→      setSaving(false);
   807→   807→    }
   808→   808→  };
   809→   809→
   810→   810→  return (
   811→   811→    <EditorShell
   812→   812→      title={`${title} · ${unitLabel}`}
   813→   813→      footer={
   814→   814→        <SaveCancelButtons
   815→   815→          onSave={() => void save()}
   816→   816→          onCancel={onCancel}
   817→   817→          saveDisabled={metric == null || error != null}
   818→   818→          saving={saving}
   819→   819→        />
   820→   820→      }
   821→   821→    >
   822→   822→      <div className="flex flex-col gap-1.5">
   823→   823→        <Input
   824→   824→          type="number"
   825→   825→          inputMode="decimal"
   826→   826→          step="any"
   827→   827→          min="0"
   828→   828→          className={cn("h-12 rounded-lg text-base tabular-nums", error && "border-destructive")}
   829→   829→          value={raw}
   830→   830→          onChange={(e) => setRaw(e.target.value)}
   831→   831→          placeholder={placeholder ?? ""}
   832→   832→          aria-label={`${title} in ${unitLabel}`}
   833→   833→          aria-invalid={error != null}
   834→   834→        />
   835→   835→        {error ? (
   836→   836→          <p className="px-1 text-xs text-destructive" role="alert">
   837→   837→            {error}
   838→   838→          </p>
   839→   839→        ) : imperial && metric != null ? (
   840→   840→          <p className="px-1 text-xs tabular-nums text-muted-foreground">
   841→   841→            ≈ {roundTo1(metric)} {canonicalUnit}
   842→   842→          </p>
   843→   843→        ) : (
   844→   844→          <p className="px-1 text-xs text-muted-foreground">Stored as {canonicalUnit}.</p>
   845→   845→        )}
   846→   846→      </div>
   847→   847→    </EditorShell>
   848→   848→  );
   849→   849→}
