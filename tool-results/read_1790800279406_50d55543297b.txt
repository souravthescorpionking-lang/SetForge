    71→/** §4.2 Estimate: Σ per set (tempo total + planned rest) rounded to 5 —
    72→ *  unset tempo falls back to the repo's avg-set constant, unset rest to the
    73→ *  default rest (0 when restNone). */
    74→export function estimateMinutes(exercises: ExerciseLike[]): number {
    75→  let sec = 0;
    76→  for (const ex of exercises) {
    77→    for (const s of ex.sets) {
    78→      sec += tempoTotalSec(s.tempo ?? "") ?? EST_MINUTES_PER_SET_SEC;
    79→      sec += s.restPlannedSec ?? (ex.restNone ? 0 : EST_MINUTES_DEFAULT_REST_SEC);
    80→    }
    81→  }
    82→  return Math.min(300, Math.max(5, Math.round(sec / 60 / 5) * 5));
    83→}

   134→function invalidSetIdsOf(view: EditorExerciseView): Set<string> {
   135→  const out = new Set<string>();
   136→  for (const s of view.sets) {

   444→      action: { label: "Undo", onClick: () => void restoreServerExercise(re, flatIndex) },

   448→  /** Undo path for a persisted removal: re-add + re-create sets + group + order. */
   449→  const restoreServerExercise = async (re: RoutineExerciseDTO, flatIndex: number) => {
   450→    if (!day) return;
   451→    try {
   452→      await routinesApi.addExercise(routineId, day.id, re.exerciseId);
   453→      const fresh = await routinesApi.get(routineId);
   454→      const freshDay = fresh.days.find((d) => d.id === day.id);
   455→      const ordered = [...(freshDay?.exercises ?? [])].sort((a, b) => a.sortOrder - b.sortOrder);
   456→      const newRe = ordered[ordered.length - 1];
   457→      if (!newRe) return;
   458→      for (const s of [...re.sets].sort((a, b) => a.sortOrder - b.sortOrder)) {
   459→        const input: PredefinedSetInput = {
   460→          reps: s.reps ?? null,
   461→          setType: s.setType ?? null,
   462→          tempo: s.tempo ?? null,
   463→          restPlannedSec: s.restPlannedSec ?? null,
   464→        };
   465→        await routinesApi.addSet(routineId, day.id, newRe.id, input);
   466→      }
   467→      if (re.groupId != null || re.tip || re.restNone) {
   468→        await routinesApi.updateExercise(routineId, day.id, newRe.id, {
   469→          ...(re.groupId != null ? { groupId: re.groupId } : {}),
   470→          ...(re.tip ? { tip: re.tip } : {}),
   471→          ...(re.restNone != null ? { restNone: re.restNone } : {}),
   472→        });
   473→      }
   474→      // Restore the flat position (the re-add landed at the end).
   475→      const ids = ordered.map((e) => e.id).filter((id) => id !== newRe.id);
   476→      ids.splice(Math.max(0, Math.min(flatIndex, ids.length)), 0, newRe.id);
   477→      await routinesApi.reorderExercises(routineId, day.id, ids);
   478→      invalidate.routines();
   479→      invalidate.customWorkouts();
   480→      toast.success(`${re.exercise.name} restored`);

   576→          body: input,
   577→          label: "Set added",

   583→          path: `/api/routines/${routineId}/days/${day.id}/exercises/${view.id}/sets/${setId}`,

   585→          label: "Set removed",

   593→          body: patch,
   594→          label: "Exercise updated",

   659→        await savePersisted();

   664→      setSaving(false);

   668→  /** §4.9 create: Routine(kind→SESSION, source=CUSTOM) + day + series + sets
   669→   *  through the EXISTING routine-service paths, in the editor's own order. */
   670→  const saveDraft = async () => {
   671→    if (!draft) return;
   672→    // create as ROUTINE (the day-create endpoint refuses SESSION kinds — the
   673→    // hub's create-then-edit dance), flip to SESSION after the day exists.
   674→    let current = await routinesApi.create({
   675→      name: draft.name.trim(),
   676→      kind: "ROUTINE",
   677→      source: "CUSTOM",
   678→      difficulty: draft.difficulty,

   681→    const equipment = [...new Set(draft.exercises.flatMap((e) => e.exercise.equipment ?? []))];
   682→    const minutesToStore = draft.minutes ?? estimateMinutes(draft.exercises);
   683→    current = await routinesApi.addDay(current.id, "Workout", "WORKOUT", {
   684→      minutes: minutesToStore,
   685→      muscles,
   686→      equipment,

   695→      const ordered = [...(dayNow?.exercises ?? [])].sort((a, b) => a.sortOrder - b.sortOrder);
   696→      const newRe = ordered[ordered.length - 1];
   697→      if (newRe) reIds.push(newRe.id);
   698→    }

   700→    const groupIds = [...new Set(draft.exercises.map((e) => e.groupId).filter((g): g is string => g != null))];
   701→    for (const gid of groupIds) {
   702→      const members = draft.exercises.filter((e) => e.groupId === gid);
   703→      const firstReId = reIds[draft.exercises.indexOf(members[0])];
   704→      if (!firstReId) continue;

   706→      for (const m of members.slice(1)) {
   707→        const reId = reIds[draft.exercises.indexOf(m)];
   708→        if (reId) await routinesApi.updateExercise(current.id, dayId, reId, { groupId: res.groupId });
   709→      }
   710→    }
   711→    // sets + exercise meta.
   712→    for (let i = 0; i < draft.exercises.length; i++) {
   713→      const e = draft.exercises[i];
   714→      const reId = reIds[i];
   715→      if (!reId) continue;
   716→      for (const s of e.sets) {
   717→        await routinesApi.addSet(current.id, dayId, reId, {
   718→          reps: s.reps,
   719→          setType: s.setType,
   720→          tempo: s.tempo,
   721→          restPlannedSec: s.restPlannedSec,
   722→        });
   723→      }
   724→      if (e.tip || e.restNone) {
   725→        await routinesApi.updateExercise(current.id, dayId, reId, {
   726→          ...(e.tip ? { tip: e.tip } : {}),
   727→          ...(e.restNone ? { restNone: e.restNone } : {}),
   728→        });
   729→      }
   730→    }
   731→    // flip to SESSION (validates exactly one workout day server-side).
   732→    await routinesApi.update(current.id, { kind: "SESSION" });
   733→    invalidate.routines();
   734→    invalidate.customWorkouts();
   735→    draftStore.reset();
   736→    toast.success("Saved to your workouts");
   737→    navigate(`/days/${dayId}`);

   754→    setDifficultyDraft(null);

   821→            <Skeleton className="h-32 w-full rounded-lg" />

   824→          <>
   825→            {/* ---------- name row 56 ---------- */}
   826→            <div

   828→              data-row
   829→              className={cn(
   830→                "flex h-14 w-full flex-none items-center gap-2 overflow-hidden whitespace-nowrap rounded-lg border border-border bg-card px-3",
   831→                showErrors && nameError && "border-destructive/60",
   832→              )}

   834→            >
   835→              <Input
   836→                value={name}
   837→                maxLength={80}
   838→                placeholder="Workout name"
   839→                aria-label="Workout name"
   840→                aria-invalid={showErrors && nameError}
