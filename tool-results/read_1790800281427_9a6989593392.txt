    71→    71→/** §4.2 Estimate: Σ per set (tempo total + planned rest) rounded to 5 —
    72→    72→ *  unset tempo falls back to the repo's avg-set constant, unset rest to the
    73→    73→ *  default rest (0 when restNone). */
    74→    74→export function estimateMinutes(exercises: ExerciseLike[]): number {
    75→    75→  let sec = 0;
    76→    76→  for (const ex of exercises) {
    77→    77→    for (const s of ex.sets) {
    78→    78→      sec += tempoTotalSec(s.tempo ?? "") ?? EST_MINUTES_PER_SET_SEC;
    79→    79→      sec += s.restPlannedSec ?? (ex.restNone ? 0 : EST_MINUTES_DEFAULT_REST_SEC);
    80→    80→    }
    81→    81→  }
    82→    82→  return Math.min(300, Math.max(5, Math.round(sec / 60 / 5) * 5));
    83→    83→}
