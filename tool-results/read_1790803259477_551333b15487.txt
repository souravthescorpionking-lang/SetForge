     4→// GroupCard — THE single source of truth group card (Part 8 §2). One card per
     5→// GROUP (a superset group, or an ungrouped exercise as its own group of 1):
     6→//
     7→//   |▌ 4px colour bar full height (group colour; category colour on singletons)
     8→//   | A1                Superset | 32  code (12px bold muted) · label right
     9→//   | Deadlift                   | 40  name 16px
    10→//   | Reps:  5  5  5     💡 …    | 40  per-set reps tabular · tip · options
    11→//   | Tempo: 4/0/1/0              | 32  hidden when no tempo anywhere
    12→//   | ⏱ ›                        | 40  collapsed rest (toggle remembered)
    13→//   |    3× · Rest sec: 90 90 90 | 40  expanded rest row

    16→// Modes (§2.3):
    17→//   view — Program detail / Session detail: Reps / Tempo / ⏱ rows.
    18→//   read — Log detail: SetRow list read-only ("100 · 5 🏆").
    19→//   edit — Builder editors: view layout, fields tappable, ≡ drag replaces 💡.
    20→//   log  — Logging screen: Tempo row + SetRow list + "+ set"; collapsed
    21→//          exercises show "0/3" right.
    22→//
    23→// Laws honoured: ONE border per GROUP; colour = the 4px bar ONLY (Law 7);
    24→// rows 56/48/40/32, single line, nowrap + ellipsis (Law 4); card never
    25→// scrolls and never contains inner cards; width = container width.
    26→// Divider BETWEEN groups (16px gap + 1px 30%-muted line) is rendered by
    27→// <GroupCardStack> — never inside the card, never after the last group.

   202→function overviewRestText(set: CardSet): string {
   203→  const planned = set.restPlannedSec ?? null;
   204→  const actual = set.restActualSec ?? null;
   205→  if (planned == null || planned <= 0) return "–";
   206→  if (actual != null) return `${planned}→${actual}`; // §3.3 “60→75” once performed
   207→  return formatRestSec(planned);
   208→}

   287→function repsCellText(set: CardSet, fields: Array<"weight" | "reps" | "distance" | "timeSec">): string {
   288→  // Part 9 §5: AMRAP prescribed sets spell out "AMRAP" (tap → Term definition).
   289→  // Part 9 §8: performed AMRAP sets carry their actual reps → "AMRAP→{n}".
   290→  const reps =
   291→    set.setType === "AMRAP"
   292→      ? set.reps == null
   293→        ? "AMRAP"
   294→        : `AMRAP→${set.reps}`
   295→      : set.reps == null
   296→        ? "∞"
   297→        : String(set.reps);
   298→  if (!fields.includes("weight")) {
   299→    if (fields.includes("distance") && set.distanceM != null) return `${trimNum(set.distanceM)} m · ${reps}`;
   300→    if (fields.includes("timeSec") && set.timeSec != null) return `${set.timeSec}s · ${reps}`;
   301→    return reps;
   302→  }
   303→  const weight =
   304→    set.weightKind === "COPY_LAST"
   305→      ? "↺"
   306→      : set.weightKind === "PERCENT_1RM" && set.pct != null
   307→        ? `${trimNum(set.pct)}%`
   308→        : set.weightKg != null
   309→          ? trimNum(set.weightKg)
   310→          : "↺";
   311→  return `${weight}·${reps}`;
   312→}

   317→const NAME_ROW = "flex h-10 items-center gap-1 px-3";

   323→export function GroupCard({
   324→  mode,
   325→  group,
   326→  entries,
   327→  onAction,
   328→  visibleColumns,
   329→  restExpanded = false,
   330→  onToggleRest,
   331→  menuItems,
   332→  overview,
   333→  className,
   334→}: GroupCardProps) {
   335→  const [tipOpen, setTipOpen] = useState<string | null>(null);
   336→  const [noteOpen, setNoteOpen] = useState<string | null>(null);
   337→  const hasActions = onAction != null;
   338→  const barColour = group.colour ?? entries[0]?.exercise.categoryColour ?? "#f97316";
   339→  const menu = hasActions ? (menuItems ?? menuForMode(mode)) : [];
   340→  // Legacy Part 3 modes (template/preview/summary/edit-legacy) render through
   341→  // their own SetRow list below — the Part 8 layout branches do not run.
   342→  const isLegacyRows =
   343→    mode === "template" || mode === "preview" || mode === "summary" || mode === "edit-legacy";
   344→  const layoutMode: "view" | "read" | "edit" | "log" =
   345→    mode === "view" || mode === "read" || mode === "edit" || mode === "log" ? mode : "read";
   346→
   347→  // Rest row hidden entirely when no entry carries rest data (§5: restNone
   348→  // exercises always show their "Rest: none" row).
   349→  const anyRest = entries.some((e) => e.restNone || e.sets.some((s) => s.restPlannedSec != null));
   350→  const anyTempo = entries.some((e) => e.sets.some((s) => s.tempo != null && s.tempo !== ""));
   351→
   352→  const dispatch = (action: CardAction, entryIndex: number, exerciseId?: string) => {
   353→    if (!onAction) return;
   354→    // exerciseId-bearing actions get the concrete id injected here.
   355→    const a =
   356→      exerciseId && ("exerciseId" in action && (action as { exerciseId?: string }).exerciseId === "")
   357→        ? ({ ...action, exerciseId } as CardAction)
   358→        : action;
   359→    onAction(a, entryIndex);
   360→  };
   361→
   362→  return (
   363→    <article
   364→      className={cn("flex overflow-hidden rounded-lg border bg-card", className)}
   365→      aria-label={group.label ? `${group.label} ${group.code}` : (entries[0]?.exercise.name ?? "Group")}
   366→    >
   367→      <div aria-hidden className="w-1 flex-none" style={{ backgroundColor: barColour }} />
   368→      <div className="min-w-0 flex-1 py-1">
   369→        {entries.map((entry, i) => {
   370→          const fields = fieldsForType(entry.exercise.modality) as Array<"weight" | "reps" | "distance" | "timeSec">;
   371→          const e = entry.exercise;
   372→          const collapsed = layoutMode === "log" && entry.collapsed;
   373→          const doneCount = entry.sets.filter((s) => s.done).length;
   374→          const menuForEntry = menu.map((item) => ({
   375→            ...item,
   376→            action:
   377→              "exerciseId" in item.action && (item.action as { exerciseId?: string }).exerciseId === ""
   378→                ? ({ ...item.action, exerciseId: e.id } as CardAction)
   379→                : item.action,
   380→          }));
   381→          return (
   382→            <div key={e.id} className={cn("min-w-0", i > 0 && "mt-3")}>
   383→              {/* 32px code row */}
   384→              <div {...tourAttrs({ id: "groupCard.code", label: "Series code", help: "Group letter and position, e.g. A1.", order: 100 })} className={CODE_ROW}>
   385→                <span className="text-xs font-bold tabular-nums leading-none text-muted-foreground">{entry.code || group.code}</span>
   386→                {group.label ? (
   387→                  <span className="ml-auto text-xs leading-none text-muted-foreground">
   388→                    <Term id={group.label as "Superset" | "Triset" | "Giant set"}>{group.label}</Term>
   389→                  </span>
   390→                ) : null}
   391→              </div>
   392→
   393→              {/* 40px name row — §3.3 overview: tappable (§3.4 jump-to-exercise:
   394→                  focus jumps to this exercise's first unlogged set); the ⋯ menu
   395→                  span stops propagation so it stays its own target. */}
   396→              <div
   397→                {...tourAttrs({ id: "groupCard.name", label: "Exercise name", help: "Exercise name; long-press options via the menu.", order: 110 })}
   398→                className={cn(
   399→                  NAME_ROW,
   400→                  overview?.onJumpEntry && "cursor-pointer transition-colors hover:bg-accent/40 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-ring",
   401→                )}
   402→                role={overview?.onJumpEntry ? "button" : undefined}
   403→                tabIndex={overview?.onJumpEntry ? 0 : undefined}
   404→                aria-label={overview?.onJumpEntry ? `Focus ${e.name} — jump to its next set` : undefined}
   405→                onClick={overview?.onJumpEntry ? () => overview.onJumpEntry?.(i) : undefined}
   406→                onKeyDown={
   407→                  overview?.onJumpEntry
   408→                    ? (ev) => {
   409→                        if (ev.key === "Enter" || ev.key === " ") {
   410→                          ev.preventDefault();
   411→                          overview.onJumpEntry?.(i);
   412→                        }
   413→                      }
   414→                    : undefined
   415→                }
   416→              >
   417→                <h3 className="min-w-0 flex-1 truncate text-base font-semibold leading-none">{e.name}</h3>
   418→                {entry.note != null ? (
   419→                  <span className="flex flex-none items-center gap-1 text-xs leading-none text-muted-foreground" aria-label="Has a note">
   420→                    <StickyNote className="h-3.5 w-3.5" aria-hidden />
   421→                  </span>
   422→                ) : null}
   423→                {e.progressionDelta != null && e.progressionDelta !== 0 ? (
   424→                  <span className="flex-none text-xs font-medium leading-none text-primary">
   425→                    {e.progressionDeload ? "↓ deload" : `↑ +${trimNum(e.progressionDelta)} next`}
   426→                  </span>
   427→                ) : null}
   428→                {collapsed || overview ? (
   429→                  <span className="flex-none text-xs tabular-nums leading-none text-muted-foreground">{doneCount}/{entry.sets.length}</span>
   430→                ) : null}
   431→                {hasActions && menuForEntry.length > 0 ? (
   432→                  <span className="flex flex-none" onClick={(ev) => ev.stopPropagation()}>
   433→                    <DropdownMenu>
   434→                      <DropdownMenuTrigger asChild>
   435→                        <Button
   436→                          type="button"
   437→                          variant="ghost"
   438→                          tour={{ id: "groupCard.menu", label: "Options", help: "Detail, history, graph, records, notes, replace or remove.", order: 120 }}
   439→                          className="h-11 w-11 p-0"
   440→                          aria-label={`Options for ${e.name}`}
   441→                        >
   442→                          <MoreVertical className="h-5 w-5" aria-hidden />
   443→                        </Button>
   444→                      </DropdownMenuTrigger>
   445→                      <DropdownMenuContent align="end" className="w-44">
   446→                        {menuForEntry.map((item) => (
   447→                          <DropdownMenuItem
   448→                            key={item.label}
   449→                            onClick={() => dispatch(item.action, i)}
   450→                            className={item.destructive ? "text-destructive focus:text-destructive" : undefined}
   451→                          >
   452→                            <item.icon className="h-4 w-4" aria-hidden />
   453→                            {item.label}
   454→                          </DropdownMenuItem>
   455→                        ))}
   456→                      </DropdownMenuContent>
   457→                    </DropdownMenu>
   458→                  </span>
   459→                ) : null}
   460→              </div>
   461→
   462→              {/* 📝 note row (§5.4) — 32px single line under the header; tap
   463→                  expands the full text (prose block, exempt from nowrap). */}
   464→              {entry.note ? (
   465→                <>
   466→                  <button
   467→                    type="button"
   468→                    {...tourAttrs({ id: "groupCard.note", label: "Note", help: "Your coaching note for this exercise; tap to expand.", order: 120 })}
   469→                    aria-expanded={noteOpen === e.id}
   470→                    aria-label={`Note for ${e.name}`}
   471→                    onClick={() => setNoteOpen((cur) => (cur === e.id ? null : e.id))}
   472→                    className={cn(NOTE_ROW, "w-full gap-2 text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground")}
   473→                  >
   474→                    <StickyNote className="h-3.5 w-3.5 flex-none text-primary" aria-hidden />
   475→                    <span className="truncate">{noteOpen === e.id ? "Hide note" : "Note"}</span>

   477→                    <ChevronDown className={cn("h-3.5 w-3.5 flex-none text-muted-foreground/60 transition-transform", noteOpen !== e.id && "-rotate-90")} aria-hidden />
   478→                  </button>
   479→                  {noteOpen === e.id ? (
   480→                    <p className="mx-3 mb-1 max-h-24 overflow-y-auto whitespace-pre-wrap rounded-md bg-muted/60 px-3 py-2 text-xs leading-relaxed text-muted-foreground">
   481→                      {entry.note}
   482→                    </p>
   483→                  ) : null}
   484→                </>
   485→              ) : null}
   486→
   487→              {/* 💡 trainer tip toggle (edit mode: drag handle replaces it) */}
   488→              {mode === "edit" && entry.dragHandle != null ? (
   489→                <div className="flex h-8 items-center px-3">{entry.dragHandle}</div>
   490→              ) : entry.tip ? (
   491→                <>
   492→                  <button
   493→                    type="button"
   494→                    {...tourAttrs({ id: "groupCard.tip", label: "Trainer tip", help: "Expand this exercise's trainer tip.", order: 130 })}
   495→                    aria-expanded={tipOpen === e.id}
   496→                    aria-label={`Trainer tip for ${e.name}`}
   497→                    onClick={() => setTipOpen((cur) => (cur === e.id ? null : e.id))}
   498→                    className={cn(TEMPO_ROW, "w-full gap-2 text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground")}
   499→                  >
   500→                    <Lightbulb className="h-4 w-4 flex-none text-amber-500" aria-hidden />
   501→                    <span className="truncate">{tipOpen === e.id ? "Hide tip" : "Trainer tip"}</span>
   502→                  </button>
   503→                  {tipOpen === e.id ? (
   504→                    <p className="mx-3 mb-1 line-clamp-3 rounded-md bg-muted/60 px-3 py-2 text-xs leading-relaxed text-muted-foreground transition-all duration-150">
   505→                      {entry.tip}
   506→                    </p>
   507→                  ) : null}
   508→                </>
   509→              ) : null}
   510→
   511→              {/* mode bodies (§4.5: a consumer editor node replaces the
   512→                  default reps/tempo/rest block — the builder's editor rows) */}
   513→              {(mode === "view" || mode === "edit") && !collapsed ? (
   514→                entry.editor != null ? (
   515→                  entry.editor
   516→                ) : (
   517→                <>
   518→                  <div {...tourAttrs({ id: "groupCard.reps", label: "Reps row", help: "Planned reps per set; ↺ copies last time, % is 1RM-based.", order: 140 })} className={REPS_ROW}>

   521→                      {entry.sets.map((s) => (
   522→                        <span key={s.id} className="flex-none whitespace-nowrap">
   523→                          {s.setType === "AMRAP" ? (
   524→                            <Term id="AMRAP" className="tabular-nums">{repsCellText(s, fields)}</Term>
   525→                          ) : (
   526→                            repsCellText(s, fields)
   527→                          )}
   528→                        </span>
   529→                      ))}
   530→                      {entry.sets.length === 0 ? <span className="text-muted-foreground">–</span> : null}
   531→                    </span>
   532→                  </div>
   533→                  {anyTempo ? (
   534→                    <div {...tourAttrs({ id: "groupCard.tempo", label: "Tempo row", help: "Planned tempo, e.g. 4/0/1/0 — eccentric, pause, concentric, pause.", order: 150 })} className={TEMPO_ROW}>

   536→                      <span className="truncate text-sm tabular-nums leading-none text-muted-foreground">
   537→                        {entry.sets.find((s) => s.tempo)?.tempo?.replace(/-/g, "/") ?? "–"}
   538→                      </span>
   539→                    </div>
   540→                  ) : null}
   541→                  {anyRest ? (
   542→                    <button
   543→                      type="button"
   544→                      {...tourAttrs({ id: "groupCard.rest", label: "Rest row", help: "Planned rest between sets; tap to expand per-set seconds.", order: 160 })}
   545→                      onClick={() => onToggleRest?.()}
   546→                      aria-expanded={restExpanded}
   547→                      className={cn(REST_ROW, "w-full gap-2 text-left")}
   548→                    >
   549→                      <Timer className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   550→                      {entry.restNone ? (
   551→                        <span className="truncate text-sm leading-none text-muted-foreground">Rest: none</span>
   552→                      ) : restExpanded ? (
   553→                        <span className="truncate text-sm tabular-nums leading-none text-muted-foreground">
   554→                          {entry.sets.length}× · Rest sec: {entry.sets.map((s) => s.restPlannedSec ?? "–").join(" ")}
   555→                        </span>
   556→                      ) : (
   557→                        <span className="text-sm leading-none text-muted-foreground">Rest {entry.sets[0]?.restPlannedSec ?? "–"}s</span>
   558→                      )}
   559→                      <ChevronDown className={cn("ml-auto h-4 w-4 flex-none text-muted-foreground transition-transform", !restExpanded && "-rotate-90")} aria-hidden />
   560→                    </button>
   561→                  ) : null}
   562→                </>
   563→                )
   564→              ) : null}
   565→
   566→              {(mode === "read" || isLegacyRows) && mode !== "summary" ? (
   567→                <div className="px-1 pb-1">
   568→                  {entry.sets.map((s) => (
   569→                    <SetRow key={s.id} mode={mode} exercise={e} set={s} visibleColumns={effectiveCols(e, visibleColumns)} />
   570→                  ))}
   571→                </div>
   572→              ) : null}
   573→
   574→              {(mode === "log" || mode === "template" || mode === "edit-legacy") && !collapsed ? (
   575→                <div className="px-1">
   576→                  {overview ? (
   577→                    /* §3.3 Overview: the compact Set|Type|Reps|Weight|Rest table
   578→                       replaces the SetRow inputs + add-set (jump taps only). */
   579→                    <OverviewSetTable
   580→                      entry={entry}
   581→                      unit={e.unit}
   582→                      focusSetId={overview.focusSetId}
   583→                      onJumpSet={overview.onJumpSet}
   584→                    />
   585→                  ) : (
   586→                    <>
   587→                      {anyTempo ? (
   588→                        <div className={cn(TEMPO_ROW, "px-2")}>
   589→                          <span className="flex-none text-xs font-medium text-muted-foreground">Tempo:</span>
   590→                          <span className="truncate text-sm tabular-nums leading-none text-muted-foreground">
   591→                            {entry.sets.find((s) => s.tempo)?.tempo?.replace(/-/g, "/") ?? "–"}
   592→                          </span>
   593→                        </div>
   594→                      ) : null}
   595→                      {entry.sets.map((s) => (
   596→                        <SetRow
   597→                          key={s.id}
   598→                          mode={mode === "log" ? "log" : mode}
   599→                          exercise={e}
   600→                          set={s}
   601→                          visibleColumns={effectiveCols(e, visibleColumns)}
   602→                          current={entry.currentSetIndex === s.index - 1}
   603→                          onAction={onAction != null ? (a) => dispatch(a, i) : undefined}
   604→                        />
   605→                      ))}
   606→                      {hasActions ? (
   607→                        <button
   608→                          type="button"
   609→                          data-row
   610→                          {...tourAttrs({ id: "groupCard.addSet", label: "Add set", help: "Append another set to this exercise.", order: 170 })}
   611→                          aria-label={`Add set to ${e.name}`}
   612→                          className={cn(rowBase, "w-full gap-2 px-2 text-sm font-medium text-muted-foreground transition-colors hover:bg-accent/50")}
   613→                          onClick={() => dispatch({ type: "add-set" }, i)}
   614→                        >
   615→                          <Plus className="h-4 w-4" aria-hidden />
   616→                          Add set
   617→                        </button>
   618→                      ) : null}
   619→                    </>
   620→                  )}
   621→                </div>
   622→              ) : null}
   623→
   624→              {/* log mode collapsed tap target */}
   625→              {layoutMode === "log" && collapsed && hasActions ? (
   626→                <button
   627→                  type="button"
   628→                  {...tourAttrs({ id: "groupCard.expand", label: "Expand", help: "Show this exercise's sets.", order: 180 })}
   629→                  aria-label={`Expand ${e.name}`}
   630→                  className={cn(REST_ROW, "w-full px-3 text-left text-sm font-medium text-muted-foreground")}
   631→                  onClick={() => dispatch({ type: "toggle-collapse" }, i)}
   632→                >
   633→                  <ChevronDown className="h-4 w-4 -rotate-90" aria-hidden />
   634→                  Expand sets
   635→                </button>
   636→              ) : null}
   637→
   638→              {/* §8 per-entry footer — consumer-owned rows (performed data) */}
   639→              {entry.footer != null ? <div className="mt-1">{entry.footer}</div> : null}
   640→            </div>
   641→          );
   642→        })}
   643→      </div>
   644→    </article>
   645→  );
   646→}

   650→function effectiveCols(exercise: CardExercise, global?: CardVisibleColumns): CardVisibleColumns {
   651→  const base: CardVisibleColumns = global ?? { setType: true, rpe: true, tempo: true, rest: true };
   652→  return {
   653→    setType: base.setType,
   654→    rpe: exercise.showRpe == null ? base.rpe : exercise.showRpe,
   655→    tempo: exercise.showTempo == null ? base.tempo : exercise.showTempo,
   656→    rest: exercise.showRest == null ? base.rest : exercise.showRest,
   657→  };
   658→}

   660→// ---------- GroupCardStack: groups + the between-groups divider (§2.2) ----------
   661→
   662→function divideChildren(children: ReactNode): ReactNode[] {
   663→  const items = Children.toArray(children).filter(isValidElement);
   664→  return items.map((child, i) => (
   665→    <div key={child.key ?? String(i)} className="flex flex-col gap-4">
   666→      {i > 0 ? <div aria-hidden className="h-px flex-none bg-muted/30" /> : null}
   667→      {child}
   668→    </div>
   669→  ));
   670→}

   672→/** 16px gap + 1px faded (30% muted) divider BETWEEN groups; none after last. */
   673→export function GroupCardStack({ children, className }: { children: ReactNode; className?: string }) {
   674→  return <div className={cn("flex flex-col gap-4", className)}>{divideChildren(children)}</div>;
   675→}
