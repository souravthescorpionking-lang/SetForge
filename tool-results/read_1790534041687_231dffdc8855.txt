    99→/** Map a CardSet patch (SetRow/ExerciseCard contract) onto the API's SetInput. */
   100→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   101→  const out: SetInput = {};
   102→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   103→  if ("reps" in patch) out.reps = patch.reps ?? null;
   104→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   105→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   106→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   107→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   108→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   109→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   110→  if ("note" in patch) out.comment = patch.note ?? null;
   111→  if ("done" in patch) out.isComplete = !!patch.done;
   112→  return out;
   113→}

   115→/** §4.11 showMaxWeightBar — "Best 110×5" from the exercise's record table
   116→ *  (max-weight entry; ties break to the higher rep count). */
   117→function bestSetLine(records: RecordsDTO | undefined): string | null {
   118→  if (!records) return null;
   119→  const usable = records.actual.filter((r) => r.weight != null && r.weight > 0 && r.reps != null);
   120→  const rows = usable.some((r) => !r.superseded) ? usable.filter((r) => !r.superseded) : usable;
   121→  if (rows.length === 0) return null;
   122→  let best = rows[0];
   123→  for (const r of rows) {
   124→    if (r.weight! > best.weight! || (r.weight === best.weight && r.reps! > best.reps!)) best = r;
   125→  }
   126→  return `Best ${trimNum(best.weight!)}×${best.reps!}`;
   127→}

   491→    if (!we || !set || set.isComplete) return;
   492→    await toggleSetDone(we, set);
   493→    advanceGuidedPast(guidedIdx + 1);
   494→  };
   495→
   496→  // card element registry — the guided pointer scrolls the current card into view
   497→  const cardRefs = useRef<Map<string, HTMLDivElement>>(new Map());
   498→  useEffect(() => {
   499→    if (!guidedActive) return;
   500→    const p = pointerOrder[guidedIdx];
   501→    if (!p) return;
   502→    const el = cardRefs.current.get(p.exerciseId);
   503→    if (!el) return;
   504→    const reduced =
   505→      typeof window.matchMedia === "function" &&
   506→      window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   507→    el.scrollIntoView({ block: "nearest", behavior: reduced ? "auto" : "smooth" });
   508→  }, [guidedActive, guidedIdx, pointerOrder]);
   509→
   510→  // ---------- §4.11 derived card data ----------
   511→
   512→  /** §4.10 group codes for today's cards (A1/A2… on superset members). */
   513→  const groupCodes = useMemo(() => computeGroupCodes(exercises), [exercises]);
   514→
   515→  const guidedCardWeId = guidedIdx < pointerOrder.length ? pointerOrder[guidedIdx].exerciseId : null;
   516→  const guidedAllDone = guidedActive && guidedIdx >= pointerOrder.length;
   517→  const guidedLabel = (() => {
   518→    if (!guidedActive) return null;
   519→    if (guidedAllDone) return "All sets done";
   520→    const p = pointerOrder[guidedIdx];
   521→    const code = groupCodes.codes.get(p.exerciseId);
   522→    const total = sortedSetsByWe.get(p.exerciseId)?.length ?? 0;
   523→    return code ? `${code} · set ${p.setIndex + 1}/${total}` : `set ${p.setIndex + 1}/${total}`;
   524→  })();
   525→
   526→  /** §4.11 sets progress — non-warmup performed sets / total sets. */
   527→  const setsProgress = useMemo(() => {
   528→    if (!workout) return null;
   529→    let total = 0;
   530→    let completed = 0;
   531→    for (const we of workout.exercises) {
   532→      for (const s of we.sets) {
   533→        total += 1;
   534→        if (s.isComplete && !s.isWarmup && (s.setType ?? "NORMAL") !== "WARMUP") completed += 1;
   535→      }
   536→    }
   537→    return total > 0 ? { completed, total } : null;
   538→  }, [workout]);
   539→
   540→  /** §4.11 showMaxWeightBar — "Best W×R" per exercise from its records. */
   541→  const showBestLine = !!settings?.showMaxWeightBar;
   542→  const recordsQueries = useQueries({
   543→    queries: exercises.map((we) => ({
   544→      queryKey: qk.exerciseRecords(we.exerciseId),
   545→      queryFn: () => exercisesApi.records(we.exerciseId),
   546→      enabled: showBestLine,
   547→      staleTime: 60_000,
   548→    })),
   549→  });
   550→  const bestByWeId = useMemo(() => {
   551→    const m = new Map<string, string>();
   552→    if (!showBestLine) return m;
   553→    exercises.forEach((we, i) => {
   554→      const line = bestSetLine(recordsQueries[i]?.data);
   555→      if (line) m.set(we.id, line);
   556→    });
   557→    return m;
   558→  }, [showBestLine, exercises, recordsQueries]);
   559→
   560→  /** Any card has a trainer tip → the MetaRow "Tip" chip renders. */
   561→  const anyTips = exercises.some((we) => !!we.exercise.trainerTip);
   562→
   563→  // ---------- Part 5: finish / undo-finish ----------
   564→  const finishWorkout = async () => {
   565→    if (!workout || !online) {
   566→      if (!online) toast.info("Finishing needs a connection");
   567→      return;
   568→    }
   569→    try {
   570→      const res = await workoutsApi.finish(workout.id);
   571→      invalidate.workout();
   572→      invalidate.dashboard();
   573→      invalidate.schedule();
   574→      invalidate.programs();
   575→      // "Day X up next" — index comes from the routine days (fallback: name)
   576→      let nextLabel = "Done";
   577→      if (res.nextDay) {
   578→        const routine = routinesQuery.data?.routines.find(
   579→          (r) => r.id === workout.sourceRoutineId,
   580→        );
   581→        const idx = routine?.days.findIndex((d) => d.id === res.nextDay?.id) ?? -1;
   582→        nextLabel = idx >= 0 ? `Day ${idx + 1} up next` : `${res.nextDay.name} up next`;
   583→      }
   584→      toast.success(`Workout finished · ${nextLabel}`, {
   585→        duration: 10_000,
   586→        action: {
   587→          label: "Undo",
   588→          onClick: () => void undoFinish(workout.id),
   589→        },
   590→      });
   591→    } catch (e) {
   592→      toast.error(e instanceof Error ? e.message : "Could not finish workout");
   593→    }
   594→  };
   595→
   596→  const undoFinish = async (workoutId: string) => {
   597→    try {
   598→      await workoutsApi.undoFinish(workoutId);
   599→      invalidate.workout();
   600→      invalidate.dashboard();
   601→      invalidate.schedule();
   602→      invalidate.programs();
   603→      toast.success("Finish undone — keep logging");
   604→    } catch (e) {
   605→      if (e instanceof ApiError && e.status === 404) {
   606→        toast.info("Too late to undo (past the 10s window)");
   607→      } else {
   608→        toast.error(e instanceof Error ? e.message : "Could not undo");
   609→      }
   610→    }
   611→  };
   612→
   613→  // §4.11 finishBehaviour — ALWAYS_SAVE (default) finishes directly; ASK opens
   614→  // the Save/Discard dialogs (finish-ask.tsx).
   615→  const handleFinishPressed = () => {
   616→    if (settings?.finishBehaviour === "ASK") setFinishAskOpen(true);
   617→    else void finishWorkout();
   618→  };
   619→
   620→  // ---------- card mapping ----------
   621→  const visibleColumns: CardVisibleColumns = {
   622→    setType: settings?.showSetType ?? true,
   623→    rpe: settings?.showRpe ?? true,
   624→    tempo: settings?.showTempo ?? true,
   625→    rest: settings?.showRest ?? true,
   626→  };
   627→
   628→  const toCardExercise = (we: WorkoutExerciseDTO): CardExercise => ({
   629→    id: we.id,
   630→    name: we.exercise.name,
   631→    categoryLabel: we.exercise.category?.name ?? "Exercise",
   632→    categoryColour: we.exercise.category?.colour ?? "#71717a",
   633→    modality: we.exercise.type,
   634→    unit: exerciseUnit(we.exercise, settings),
   635→    weightIncrement: we.exercise.weightIncrement ?? settings?.defaultWeightIncrement ?? 2.5,
   636→  });
   637→
   638→  const cardSets = (we: WorkoutExerciseDTO): CardSet[] =>
   639→    [...we.sets].sort((a, b) => a.sortOrder - b.sortOrder).map((s, i) => toCardSet(s, i + 1));
   640→
   641→  const handleCardAction =
   642→    (we: WorkoutExerciseDTO) =>
   643→    (action: CardAction): void => {
   644→      switch (action.type) {
   645→        case "toggle-collapse":
   646→          setCollapsedIds((prev) => {
   647→            const next = new Set(prev);
   648→            if (next.has(we.id)) next.delete(we.id);
   649→            else next.add(we.id);
   650→            return next;
   651→          });
   652→          break;
   653→        case "add-set":
   654→          void addSetToCard(we);
   655→          break;
   656→        case "update-set":
   657→          void patchSet(we, action.setId, cardPatchToSetInput(action.patch));
   658→          break;
   659→        case "toggle-done": {
   660→          const set = we.sets.find((s) => s.id === action.setId);
   661→          if (set) void toggleSetDone(we, set);
   662→          break;
   663→        }
   664→        case "copy-last":
   665→          void copyLast(we, action.setId);
   666→          break;
   667→        case "notes":
   668→          setNotesWeId(we.id);
   669→          break;
   670→        case "rest-timer":
   671→          rest.start(we.exercise.restSec ?? undefined, action.setId ?? null);
   672→          break;
   673→        case "move-up":
   674→          void moveExercise(we, -1);
   675→          break;
   676→        case "move-down":
   677→          void moveExercise(we, 1);
   678→          break;
   679→        case "add-to-group":
   680→          setGroupWeId(we.id);
   681→          break;
   682→        case "replace":
   683→          // p3-4: the picker opens in replace mode with the day context
   684→          navigate(`/exercises?replace=${we.id}&date=${dateKey}`);
   685→          break;
   686→        case "remove":
   687→          setRemoveWeId(we.id);
   688→          break;
   689→        case "select":
   690→          // multi-select omitted in the p3-3 Today rebuild (see report)
   691→          toast.info("Multi-select arrives with the next build");
   692→          break;
   693→        case "open":
   694→          // p3-4 "Focus view" — carry the day context into the training screen
   695→          navigate(`/today/${we.id}?date=${dateKey}`);
   696→          break;
   697→        case "toggle-select":
   698→          break; // preview-mode only; edit rows never emit it
   699→      }
   700→    };
   701→
   702→  const restActive = rest.remainingSec != null;
   703→  // §4.11 RING display — in-flow block while resting (RestBar stays too)
   704→  const ringRestActive = restActive && settings?.restDisplay === "RING";
   705→
   706→  // ---------- render ----------
   707→  return (
   708→    <Screen
   709→      topBar={
   710→        <TopBar
   711→          title="SetForge"
   712→          actions={
   713→            <>
   714→              <Button
   715→                type="button"
   716→                variant="ghost"
   717→                size="icon"
   718→                className="h-11 w-11 flex-none"
   719→                aria-label="Calendar"
   720→                onClick={() => navigate("/calendar")}
   721→              >
   722→                <CalendarDays className="h-5 w-5" aria-hidden />
   723→              </Button>
   724→              <DropdownMenu>
   725→                <DropdownMenuTrigger asChild>
   726→                  <Button
   727→                    type="button"
   728→                    variant="ghost"
   729→                    size="icon"
   730→                    className="h-11 w-11 flex-none"
   731→                    aria-label="More actions"
   732→                  >
   733→                    <MoreVertical className="h-5 w-5" aria-hidden />
   734→                  </Button>
   735→                </DropdownMenuTrigger>
   736→                <DropdownMenuContent align="end" className="w-52">
   737→                  {workout && workout.exercises.length > 0 ? (
   738→                    <>
   739→                      <DropdownMenuItem
   740→                        onClick={() => sessionFromWorkout.openFor(workout.id, dateKey)}
   741→                      >
   742→                        <Dumbbell className="h-4 w-4" aria-hidden /> Save as session
   743→                      </DropdownMenuItem>
   744→                      {/* §4.10c — today's arrange screen (6-d) */}
   745→                      <DropdownMenuItem
   746→                        onClick={() =>
   747→                          navigate(
   748→                            dateKey === todayKey() ? "/today/arrange" : `/today/arrange?date=${dateKey}`,
   749→                          )
   750→                        }
   751→                      >
   752→                        <ArrowDownUp className="h-4 w-4" aria-hidden /> Arrange exercises
   753→                      </DropdownMenuItem>
   754→                    </>
   755→                  ) : null}
   756→                  <DropdownMenuItem onClick={() => navigate("/history")}>
   757→                    <History className="h-4 w-4" aria-hidden /> History
   758→                  </DropdownMenuItem>
   759→                  <DropdownMenuItem onClick={() => navigate("/exercises")}>
   760→                    <Dumbbell className="h-4 w-4" aria-hidden /> Exercises
   761→                  </DropdownMenuItem>
   762→                  <DropdownMenuItem onClick={() => navigate("/tools")}>
   763→                    <Wrench className="h-4 w-4" aria-hidden /> Tools
   764→                  </DropdownMenuItem>
   765→                  <DropdownMenuItem onClick={() => navigate("/settings")}>
   766→                    <Settings className="h-4 w-4" aria-hidden /> Settings
   767→                  </DropdownMenuItem>
   768→                </DropdownMenuContent>
   769→              </DropdownMenu>
   770→            </>
   771→          }
   772→        />
   773→      }
   774→      subBar={
   775→        <DateStrip
   776→          dateKey={dateKey}
   777→          onChange={goTo}
   778→          workoutExists={!!workout}
   779→          progress={settings?.showSetsProgressBar !== false ? setsProgress : null}
   780→        />
   781→      }
   782→      bottomBar={
   783→        <BottomBar>
   784→          {restActive ? (
   785→            <RestBar
   786→              display={formatDuration(rest.remainingSec ?? 0)}
   787→              onAdjust={rest.adjust}
   788→              onSkip={rest.skip}
   789→            />
   790→          ) : workout?.finishedAt ? (
   791→            // finished — single disabled confirmation row
   792→            <FinishedBar />
   793→          ) : guidedActive ? (
   794→            // §4.11 guided mode — Prev / Log set / Next replaces Add/Finish
   795→            <GuidedBar
   796→              onPrev={() => moveGuided(-1)}
   797→              onNext={() => moveGuided(1)}
   798→              onLog={() => void logGuidedSet()}
   799→              onFinish={handleFinishPressed}
   800→              canLog={guidedIdx < pointerOrder.length && !isPointerComplete(guidedIdx)}
   801→              allDone={guidedAllDone}
   802→              atStart={guidedIdx <= 0}
   803→            />
   804→          ) : workout && workout.exercises.length > 0 ? (
   805→            // logging in progress — two equal actions
   806→            <FinishBar
   807→              onAdd={() => navigate(`/exercises?date=${dateKey}`)}
   808→              onFinish={handleFinishPressed}
   809→            />
   810→          ) : (
   811→            <AddExerciseBar onAdd={() => navigate(`/exercises?date=${dateKey}`)} />
   812→          )}
   813→        </BottomBar>
   814→      }
   815→    >
   816→      <ScrollBody>
   817→        {isLoading ? (
   818→          <DaySkeleton />
   819→        ) : !workout ? (
   820→          <TodayEmpty
   821→            hasPrevious={hasPrevious}
   822→            starting={creating}
   823→            onStartNew={() => void startNewWorkout()}
   824→            onCopyPrevious={() => void copyPreviousWorkout()}
   825→          />
   826→        ) : (
   827→          <>
   828→            {/* §4.11 RING rest display — FIRST child, pushes content down,
   829→                removed when rest ends (no overlay; RestBar stays in the BottomBar) */}
   830→            {ringRestActive ? (
   831→              <RestRingBlock
   832→                remainingSec={rest.remainingSec ?? 0}
   833→                totalSec={rest.totalSec}
   834→                onAdjust={rest.adjust}
   835→                onSkip={rest.skip}
   836→              />
   837→            ) : null}
   838→
   839→            <MetaRow
   840→              workout={workout}
   841→              source={source}
   842→              restRemainingSec={rest.remainingSec}
   843→              onToggleTimer={() => void toggleTimer()}
   844→              setsProgress={settings?.showSetsProgressBar !== false ? setsProgress : null}
   845→              guidedPointer={guidedLabel}
   846→              tipToggle={
   847→                anyTips ? { open: tipsOpen, onToggle: () => setTipsOpen((v) => !v) } : null
   848→              }
   849→            />
   850→
   851→            {exercises.map((we) => {
   852→              const group = we.groupId ? groupById.get(we.groupId) ?? null : null;
   853→              const isCurrentGuidedCard = guidedCardWeId === we.id;
   854→              // §4.11 guided video panel — only the CURRENT guided card, only
   855→              // when a video exists (MediaBlock itself collapses to 0px otherwise)
   856→              const showGuidedVideo =
   857→                isCurrentGuidedCard &&
   858→                !!settings?.showVideoPanel &&
   859→                !!we.exercise.videoUrl;
   860→              const showTipRow = tipsOpen && !!we.exercise.trainerTip;
   861→              const underHeader =
   862→                showGuidedVideo || showTipRow ? (
   863→                  <>
   864→                    {showGuidedVideo ? (
   865→                      <div className="flex-none border-b border-border p-2">
   866→                        <MediaBlock videoUrl={we.exercise.videoUrl} height={180} />
   867→                      </div>
   868→                    ) : null}
   869→                    {showTipRow ? <TrainerTipRow tip={we.exercise.trainerTip!} /> : null}
   870→                  </>
   871→                ) : undefined;
   872→              return (
   873→                <div
   874→                  key={we.id}
   875→                  ref={(el) => {
   876→                    if (el) cardRefs.current.set(we.id, el);
   877→                    else cardRefs.current.delete(we.id);
   878→                  }}
   879→                  className="relative flex-none"
   880→                >
   881→                  <ExerciseCard
   882→                    mode="edit"
   883→                    exercise={toCardExercise(we)}
   884→                    sets={cardSets(we)}
   885→                    collapsed={collapsedIds.has(we.id)}
   886→                    visibleColumns={visibleColumns}
   887→                    groupColour={group?.colour}
   888→                    groupName={group?.name}
   889→                    groupCode={groupCodes.codes.get(we.id)}
   890→                    metaExtra={bestByWeId.get(we.id)}
   891→                    underHeader={underHeader}
   892→                    className={isCurrentGuidedCard ? "ring-2 ring-primary/40" : undefined}
   893→                    onAction={handleCardAction(we)}
   894→                  />
   895→                  {notesWeId === we.id && (
   896→                    <ExerciseNotesPopover
   897→                      exercise={we.exercise}
   898→                      open
   899→                      onClose={() => setNotesWeId(null)}
   900→                    />
   901→                  )}
   902→                  {groupWeId === we.id && (
   903→                    <ExerciseGroupPopover
   904→                      workout={workout}
   905→                      we={we}
   906→                      open
   907→                      onClose={() => setGroupWeId(null)}
   908→                    />
   909→                  )}
   910→                </div>
   911→              );
   912→            })}
   913→
   914→            <SummaryRow workout={workout} unit={defaultUnitFor(settings)} />
   915→
   916→            {/* bottom breathing spacer — bars are flex siblings, nothing to clear */}
   917→            <div className="h-4 flex-none" aria-hidden />
   918→
   919→            <ConfirmRemoveExercise
   920→              exerciseName={removeWe?.exercise.name ?? ""}
   921→              open={removeWeId != null}
   922→              onClose={() => setRemoveWeId(null)}
   923→              onConfirm={async () => {
   924→                const id = removeWeId;
   925→                setRemoveWeId(null);
   926→                if (id) await removeExercise(id);
   927→              }}
   928→            />
   929→
   930→            {/* §4.11 finishBehaviour=ASK — Save/Discard dialogs (mounted only
   931→                while the ASK stage is open; ALWAYS_SAVE never reaches it) */}
   932→            {settings?.finishBehaviour === "ASK" ? (
   933→              <FinishAskFlow
   934→                workout={workout}
   935→                open={finishAskOpen}
   936→                onClose={() => setFinishAskOpen(false)}
   937→                onSave={() => void finishWorkout()}
   938→                dateKey={dateKey}
   939→                onDiscarded={() => {
   940→                  // a discarded day can't keep a rest countdown running
   941→                  rest.skip();
   942→                  setFinishAskOpen(false);
   943→                }}
   944→              />
   945→            ) : null}
   946→
   947→            {sessionFromWorkout.dialog}
   948→          </>
   949→        )}
   950→      </ScrollBody>
   951→    </Screen>
   952→  );
   953→}
   954→
   955→function DaySkeleton() {
   956→  return (
   957→    <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading day">
   958→      <div className="h-12 animate-pulse rounded-lg bg-muted/60" />
   959→      {[0, 1, 2].map((i) => (
   960→        <div key={i} className="h-14 animate-pulse rounded-lg bg-muted/40" />
   961→      ))}
   962→    </div>
   963→  );
   964→}
