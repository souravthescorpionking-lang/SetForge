   104→/** Map a CardSet patch (SetRow/GroupCard contract) onto the API's SetInput. */
   105→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   106→  const out: SetInput = {};
   107→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   108→  if ("reps" in patch) out.reps = patch.reps ?? null;
   109→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   110→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   111→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   112→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   113→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   114→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   115→  if ("note" in patch) out.comment = patch.note ?? null;
   116→  if ("done" in patch) out.isComplete = !!patch.done;
   117→  return out;
   118→}

   122→  const [, force] = useState(0);
   123→  useEffect(() => {
   124→    if (!active) return;

   126→    return () => clearInterval(id);
   127→  }, [active]);

   130→/** §4.11 sets progress — non-warmup performed sets / total sets. */
   131→function setsProgressOf(workout: WorkoutDTO): { completed: number; total: number } | null {
   132→  let total = 0;
   133→  let completed = 0;
   134→  for (const we of workout.exercises) {
   135→    for (const s of we.sets) {
   136→      total += 1;
   137→      if (s.isComplete && !s.isWarmup && (s.setType ?? "NORMAL") !== "WARMUP") completed += 1;
   138→    }
   139→  }
   140→  return total > 0 ? { completed, total } : null;
   141→}
