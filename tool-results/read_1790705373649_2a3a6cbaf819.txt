    25→/** Full PR recompute for one exercise (delete + rebuild from all sets). */
    26→export async function recomputePRs(
    27→  tx: Prisma.TransactionClient,
    28→  userId: string,
    29→  exerciseId: string,
    30→): Promise<void> {
    31→  const sets = await tx.trainingSet.findMany({
    32→    where: {
    33→      userId,
    34→      workoutExercise: {
    35→        exerciseId,
    36→        workout: { deletedAt: null, discardedAt: null }, // Part 6: soft lifecycle excluded from PRs
    37→      },
    38→      weight: { not: null },
    39→      reps: { not: null },
    40→      isComplete: true, // only performed sets hold records
    41→      isWarmup: false, // warm-up ramps are preparation, not performance
    42→    },
    43→    include: { workoutExercise: { include: { workout: true } } },
    44→    orderBy: { createdAt: "asc" },
    45→  });
    46→  const best = new Map<number, { setId: string; weight: number; date: Date }>();
    47→  for (const s of sets) {
    48→    const reps = s.reps!;
    49→    const weight = s.weight!;
    50→    const date = s.workoutExercise.workout.date;
    51→    const cur = best.get(reps);
    52→    if (!cur || weight > cur.weight || (weight === cur.weight && date < cur.date)) {
    53→      best.set(reps, { setId: s.id, weight, date });
    54→    }
    55→  }
    56→  await tx.personalRecord.deleteMany({ where: { exerciseId } });
    57→  for (const [reps, v] of best) {
    58→    await tx.personalRecord.create({
    59→      data: { id: uuid7(), userId, exerciseId, reps, weight: v.weight, setId: v.setId, date: v.date },
    60→    });
    61→  }
    62→}

    93→export async function listWorkouts(
    94→  userId: string,
    95→  opts: { from?: string; to?: string; limit?: number; search?: string },
    96→) {
    97→  const where: Prisma.WorkoutWhereInput = { userId, deletedAt: null, discardedAt: null };
    98→  if (opts.from || opts.to) {
    99→    where.date = {
   100→      ...(opts.from ? { gte: toDayUtc(opts.from) } : {}),
   101→      ...(opts.to ? { lte: toDayUtc(opts.to) } : {}),
   102→    };
   103→  }
   104→  const rows = await db.workout.findMany({
   105→    where,
   106→    orderBy: { date: "desc" },
   107→    take: opts.limit ?? 500,
   108→    include: workoutInclude,
   109→  });
   110→  // Case-insensitive search across comment, exercise names and category names.
   111→  // Done in JS (not Prisma `contains`) so behaviour is identical on SQLite and
   112→  // Postgres (Prisma has no `mode: insensitive` on SQLite).
   113→  if (opts.search && opts.search.trim()) {
   114→    const q = opts.search.trim().toLowerCase();
   115→    return rows
   116→      .filter(
   117→        (w) =>
   118→          w.comment?.toLowerCase().includes(q) ||
   119→          w.exercises.some(
   120→            (we) =>
   121→              we.exercise.name.toLowerCase().includes(q) ||
   122→              we.exercise.category?.name.toLowerCase().includes(q),
   123→          ),
   124→      )
   125→      .map(mapWorkoutSummary);
   126→  }
   127→  return rows.map(mapWorkoutSummary);
   128→}

   142→export async function createOrGetWorkout(userId: string, dateInput: string | Date) {
   143→  const date = toDayUtc(dateInput);
   144→  // Part 6: a soft-deleted/discarded workout may still occupy (userId, date) —
   145→  // a fresh start on that date purges it (the discard/delete intent already won).
   146→  const occupant = await db.workout.findUnique({ where: { userId_date: { userId, date } } });
   147→  if (occupant && (occupant.deletedAt != null || occupant.discardedAt != null)) {
   148→    const exerciseIds = await db.workoutExercise.findMany({ where: { workoutId: occupant.id }, select: { exerciseId: true } });
   149→    await db.workout.delete({ where: { id: occupant.id } });
   150→    for (const { exerciseId } of exerciseIds) {
   151→      await db.$transaction(async (tx) => recomputePRs(tx, userId, exerciseId));
   152→    }
   153→  }
   154→  const existing = await db.workout.findFirst({ where: { userId, date, deletedAt: null, discardedAt: null }, include: workoutInclude });
   155→  if (existing) return mapWorkout(existing);
   156→  const created = await db.workout.create({
   157→    data: { id: uuid7(), userId, date },
   158→    include: workoutInclude,
   159→  });
   160→  return mapWorkout(created);
   161→}

   163→export async function updateWorkout(
   164→  userId: string,
   165→  id: string,
   166→  patch: { date?: string; comment?: string | null; startAt?: string | null; endAt?: string | null },
   167→) {
   168→  const w = await db.workout.findFirst({ where: { id, userId } });
   169→  if (!w) throw notFound("Workout not found");
   170→
   171→  let date = w.date;
   172→  if (patch.date !== undefined) {
   173→    const newDate = toDayUtc(patch.date);
   174→    if (newDate.getTime() !== w.date.getTime()) {
   175→      const clash = await db.workout.findFirst({ where: { userId, date: newDate, NOT: { id } } });
   176→      if (clash) throw conflict("A workout already exists on that date");
   177→      date = newDate;
   178→    }
   179→  }
   180→  const updated = await db.workout.update({
   181→    where: { id },
   182→    data: {
   183→      date,
   184→      ...(patch.comment !== undefined ? { comment: patch.comment } : {}),
   185→      ...(patch.startAt !== undefined ? { startAt: patch.startAt ? new Date(patch.startAt) : null } : {}),
   186→      ...(patch.endAt !== undefined ? { endAt: patch.endAt ? new Date(patch.endAt) : null } : {}),
   187→    },
   188→    include: workoutInclude,
   189→  });
   190→  return mapWorkout(updated);
   191→}

   194→export async function deleteWorkout(userId: string, id: string) {
   195→  const w = await db.workout.findFirst({ where: { id, userId }, include: { exercises: { select: { exerciseId: true } } } });

   207→/**
   208→ * POST /api/workouts/:id/discard — session discard: sets discardedAt, reverts
   209→ * cursor/schedule effects best-effort, recomputes PRs. Hidden everywhere until
   210→ * restored (10s client Undo window → restore).
   211→ */
   212→export async function discardWorkout(userId: string, id: string) {
   213→  const w = await db.workout.findFirst({ where: { id, userId }, include: { exercises: { select: { exerciseId: true } } } });
   214→  if (!w) throw notFound("Workout not found");
   215→  if (w.discardedAt) return { ok: true, discardedAt: w.discardedAt.toISOString() };
   216→
   217→  const exerciseIds = [...new Set(w.exercises.map((e) => e.exerciseId))];
   218→  const now = new Date();
   219→
   220→  // Cursor revert: when this workout came from the active program's day and the
   221→  // cursor has moved to (or past) the next day, pull it back to the source day.
   222→  if (w.sourceRoutineId && w.sourceDayId) {
   223→    const active = await db.activeRoutine.findFirst({ where: { userId, routineId: w.sourceRoutineId } });
   224→    if (active) {
   225→      const days = await db.routineDay.findMany({ where: { routineId: w.sourceRoutineId }, orderBy: { sortOrder: "asc" }, select: { id: true } });
   226→      const sourceIndex = days.findIndex((d) => d.id === w.sourceDayId);
   227→      if (sourceIndex >= 0 && active.cursorDayIndex !== sourceIndex) {
   228→        await db.activeRoutine.update({
   229→          where: { id: active.id },
   230→          data: { cursorDayIndex: sourceIndex, lastAdvancedAt: null, lastAdvancedForDate: null },
   231→        });
   232→      }
   233→    }
   234→  }
   235→
   236→  // Schedule revert: linked DONE entries — past dates flip to MISSED (truthful),
   237→  // today/future flip back to PLANNED — and every entry unlinks the workout.
   238→  // Two collision cases need care (partial unique index: only ONE live PLANNED
   239→  // entry per userId+date):
   240→  //   • workout.scheduledStart === false → the DONE entry was auto-created by
   241→  //     the start flow (nothing was planned); revert = soft-delete, not PLANNED.
   242→  //   • another live PLANNED entry already occupies that date (the slot was
   243→  //     re-taken while the workout was live) → revert = soft-delete.
   244→  const linked = await db.scheduleEntry.findMany({ where: { userId, workoutId: id } });
   245→  const todayUtc = toDayUtc(new Date());
   246→  for (const entry of linked) {
   247→    if (entry.status !== "DONE") continue;
   248→    if (entry.date.getTime() < todayUtc.getTime()) {
   249→      await db.scheduleEntry
   250→        .update({
   251→          where: { id: entry.id },
   252→          data: { status: "MISSED", workoutId: null, missedAt: entry.missedAt ?? now },
   253→        })
   254→        .catch(() => undefined);
   255→      continue;
   256→    }
   257→    const slotTaken =
   258→      w.scheduledStart === false ||
   259→      !!(await db.scheduleEntry.findFirst({
   260→        where: { userId, date: entry.date, status: "PLANNED", deletedAt: null, NOT: { id: entry.id } },
   261→        select: { id: true },
   262→      }));
   263→    if (slotTaken) {
   264→      await db.scheduleEntry
   265→        .update({ where: { id: entry.id }, data: { workoutId: null, deletedAt: now } })
   266→        .catch(() => undefined);
   267→    } else {
   268→      await db.scheduleEntry
   269→        .update({ where: { id: entry.id }, data: { status: "PLANNED", workoutId: null } })
   270→        .catch(() => undefined);
   271→    }
   272→  }
   273→
   274→  await db.$transaction(async (tx) => {
   275→    await tx.workout.update({ where: { id }, data: { discardedAt: now } });
   276→    for (const exerciseId of exerciseIds) await recomputePRs(tx, userId, exerciseId);
   277→  });
   278→  return { ok: true, discardedAt: now.toISOString() };
   279→}

   281→/** POST /api/workouts/:id/restore — clears discardedAt/deletedAt (409 on date clash) + PR recompute. */
   282→export async function restoreWorkout(userId: string, id: string) {
   283→  const w = await db.workout.findFirst({ where: { id, userId }, include: { exercises: { select: { exerciseId: true } } } });
   284→  if (!w) throw notFound("Workout not found");
   285→  if (!w.discardedAt && !w.deletedAt) return { ok: true };
   286→
   287→  const clash = await db.workout.findFirst({
   288→    where: { userId, date: w.date, deletedAt: null, discardedAt: null, NOT: { id } },
   289→    select: { id: true },
   290→  });
   291→  if (clash) throw conflict("Another workout already occupies that date — restore not possible");
   292→
   293→  const exerciseIds = [...new Set(w.exercises.map((e) => e.exerciseId))];
   294→  await db.$transaction(async (tx) => {
   295→    await tx.workout.update({ where: { id }, data: { discardedAt: null, deletedAt: null } });
   296→    for (const exerciseId of exerciseIds) await recomputePRs(tx, userId, exerciseId);
   297→  });
   298→  return { ok: true };
   299→}

   412→export type SetFieldsInput = {
   413→  weight?: number | null;
   414→  reps?: number | null;

   422→  rpe?: number | null;

   429→/** setType ⟺ isWarmup single source of truth: WARMUP type mirrors the legacy flag. */
   430→function syncWarmup(input: SetFieldsInput): { setType?: string; isWarmup?: boolean } {
   431→  if (input.setType !== undefined) {
   432→    return { setType: input.setType, isWarmup: input.setType === "WARMUP" };
   433→  }
   434→  if (input.isWarmup !== undefined) {
   435→    return { isWarmup: input.isWarmup, ...(input.isWarmup ? { setType: "WARMUP" } : {}) };
   436→  }
   437→  return {};
   438→}

   440→/** When a set turns complete: stamp completedAt and derive restActualSec
   441→ *  from the most recent earlier completed set in the same exercise. */
   442→function completionTimestamps(
   443→  now: Date,
   444→  input: SetFieldsInput,
   445→  prevCompletedAt: Date | null,
   446→): { completedAt?: Date | null; restActualSec?: number | null } {
   447→  if (input.isComplete === true) {
   448→    const completedAt = input.completedAt ? new Date(input.completedAt) : now;
   449→    const restActualSec =
   450→      input.restActualSec != null
   451→        ? input.restActualSec
   452→        : prevCompletedAt
   453→          ? Math.max(0, Math.round((completedAt.getTime() - prevCompletedAt.getTime()) / 1000))
   454→          : null;
   455→    return { completedAt, ...(restActualSec != null ? { restActualSec } : {}) };
   456→  }
   457→  if (input.isComplete === false) return { completedAt: null, restActualSec: null };
   458→  return {};
   459→}

   461→export async function createSet(
   462→  userId: string,
   463→  workoutId: string,
   464→  weId: string,
   465→  input: SetFieldsInput,
   466→) {
   467→  const we = await getWorkoutExerciseOwned(userId, workoutId, weId);
   468→  const settings = await db.userSettings.findUnique({ where: { userId } });
   469→  const trackPR = settings?.trackPR ?? true;
   470→
   471→  const count = await db.trainingSet.count({ where: { workoutExerciseId: weId } });
   472→  const warmupSync = syncWarmup(input);
   473→  const effectiveType = warmupSync.setType ?? "NORMAL";
   474→  const effectiveWarmup = warmupSync.isWarmup ?? effectiveType === "WARMUP";
   475→  const newPr =
   476→    trackPR &&
   477→    !effectiveWarmup &&
   478→    input.weight != null &&
   479→    input.reps != null &&
   480→    (await isPRForReps(userId, we.exerciseId, input.reps, input.weight));
   481→
   482→  // previous completed set in this exercise (for restActualSec)
   483→  const prevCompleted = await db.trainingSet.findFirst({
   484→    where: { workoutExerciseId: weId, isComplete: true, completedAt: { not: null } },
   485→    orderBy: { completedAt: "desc" },
   486→    select: { completedAt: true },
   487→  });
   488→  const stamps = completionTimestamps(new Date(), input, prevCompleted?.completedAt ?? null);
   489→
   490→  // Timeout headroom: concurrent set PATCHes (e.g. the §4.10d apply-to-all
   491→  // fan-out) serialize on the SQLite write lock — the 5s default expires while
   492→  // queued. 15s budget / 5s acquisition keeps every request green under burst.
   493→  const created = await db.$transaction(
   494→    async (tx) => {
   495→      const set = await tx.trainingSet.create({
   496→        data: {
   497→          id: uuid7(),
   498→          userId,
   499→          workoutExerciseId: weId,
   500→          weight: input.weight ?? null,
   501→          reps: input.reps ?? null,
   502→          distance: input.distance ?? null,
   503→          timeSec: input.timeSec ?? null,
   504→          comment: input.comment ?? null,
   505→          isComplete: input.isComplete ?? false,
   506→          isWarmup: effectiveWarmup,
   507→          setType: effectiveType,
   508→          rpe: input.rpe ?? null,
   509→          tempo: input.tempo ?? null,
   510→          restPlannedSec: input.restPlannedSec ?? null,
   511→          ...(stamps.completedAt !== undefined ? { completedAt: stamps.completedAt } : {}),
   512→          ...(stamps.restActualSec !== undefined ? { restActualSec: stamps.restActualSec } : {}),
   513→          sortOrder: count,
   514→        },
   515→      });
   516→      await recomputePRs(tx, userId, we.exerciseId);
   517→      return set;
   518→    },
   519→    TX_OPTIONS,
   520→  );
   521→  return mapSet(created, newPr);
   522→}

   524→export async function updateSet(
   525→  userId: string,
   526→  workoutId: string,
   527→  weId: string,
   528→  setId: string,
   529→  input: SetFieldsInput,
   530→) {
   531→  const we = await getWorkoutExerciseOwned(userId, workoutId, weId);
   532→  const existing = await db.trainingSet.findFirst({ where: { id: setId, workoutExerciseId: weId } });
   533→  if (!existing) throw notFound("Set not found");
   534→
   535→  const warmupSync = syncWarmup(input);
   536→  // previous completed set BEFORE this one (sortOrder-based) for restActualSec
   537→  let prevCompletedAt: Date | null = null;
   538→  if (input.isComplete === true && !existing.completedAt) {
   539→    const prev = await db.trainingSet.findFirst({
   540→      where: { workoutExerciseId: weId, isComplete: true, completedAt: { not: null }, id: { not: setId } },
   541→      orderBy: { completedAt: "desc" },
   542→      select: { completedAt: true },
   543→    });
   544→    prevCompletedAt = prev?.completedAt ?? null;
   545→  }
   546→  const stamps = completionTimestamps(new Date(), input, prevCompletedAt);
   547→
   548→  const updated = await db.$transaction(
   549→    async (tx) => {
   550→      const set = await tx.trainingSet.update({
   551→        where: { id: setId },
   552→        data: {
   553→          ...(input.weight !== undefined ? { weight: input.weight } : {}),
   554→          ...(input.reps !== undefined ? { reps: input.reps } : {}),
   555→          ...(input.distance !== undefined ? { distance: input.distance } : {}),
   556→          ...(input.timeSec !== undefined ? { timeSec: input.timeSec } : {}),
   557→          ...(input.comment !== undefined ? { comment: input.comment } : {}),
   558→          ...(input.isComplete !== undefined ? { isComplete: input.isComplete } : {}),
   559→          ...(input.isWarmup !== undefined ? { isWarmup: input.isWarmup } : {}),
   560→          // ---- Part 2 ----
   561→          ...(warmupSync.setType !== undefined ? { setType: warmupSync.setType } : {}),
   562→          ...(warmupSync.isWarmup !== undefined ? { isWarmup: warmupSync.isWarmup } : {}),
   563→          ...(input.rpe !== undefined ? { rpe: input.rpe } : {}),
   564→          ...(input.tempo !== undefined ? { tempo: input.tempo } : {}),
   565→          ...(input.restPlannedSec !== undefined ? { restPlannedSec: input.restPlannedSec } : {}),
   566→          ...(input.restActualSec !== undefined ? { restActualSec: input.restActualSec } : {}),
   567→          ...(stamps.completedAt !== undefined ? { completedAt: stamps.completedAt } : {}),
   568→          ...(stamps.restActualSec !== undefined ? { restActualSec: stamps.restActualSec } : {}),
   569→        },
   570→      });
   571→      await recomputePRs(tx, userId, we.exerciseId);
   572→      return set;
   573→    },
   574→    TX_OPTIONS,
   575→  );
   576→  return mapSet(updated);
   577→}

   588→      for (let i = 0; i < rest.length; i++) {
   589→        if (rest[i].sortOrder !== i) await tx.trainingSet.update({ where: { id: rest[i].id }, data: { sortOrder: i } });
   590→      }
   591→    },
   592→    TX_OPTIONS,
   593→  );
   594→  return { ok: true };

   648→/** Copy sets from another workout (default: the most recent before the target date). */
   649→export async function copyWorkout(
   650→  userId: string,
   651→  targetWorkoutId: string,
   652→  input: { fromDate?: string; setIds?: string[] },
   653→) {
   654→  const target = await db.workout.findFirst({ where: { id: targetWorkoutId, userId }, include: workoutInclude });
   655→  if (!target) throw notFound("Workout not found");
   656→
   657→  let source: WorkoutFull | null = null;
   658→  if (input.fromDate) {
   659→    source = await db.workout.findUnique({
   660→      where: { userId_date: { userId, date: toDayUtc(input.fromDate) } },
   661→      include: workoutInclude,
   662→    });
   663→  } else {
   664→    source = await db.workout.findFirst({
   665→      where: { userId, date: { lt: target.date } },
   666→      orderBy: { date: "desc" },
   667→      include: workoutInclude,
   668→    });
   669→  }
   670→  if (!source) throw notFound("No source workout found to copy");
   671→
   672→  const setFilter = input.setIds ? new Set(input.setIds) : null;
   673→  const affected = new Set<string>();
   674→
   675→  // map source group name → target group
   676→  const groupMap = new Map<string, string>();
   677→  for (const g of source.groups) {
   678→    const existing = target.groups.find((tg) => tg.name === g.name);
   679→    if (existing) {
   680→      groupMap.set(g.id, existing.id);
   681→    } else {
   682→      const created = await db.workoutGroup.create({
   683→        data: { id: uuid7(), userId, workoutId: target.id, name: g.name, colour: g.colour },
   684→      });
   685→      groupMap.set(g.id, created.id);
   686→    }
   687→  }
   688→
   689→  let sortOrder = await db.workoutExercise.count({ where: { workoutId: target.id } });
   690→  for (const swe of source.exercises) {
   691→    const sets = setFilter ? swe.sets.filter((s) => setFilter.has(s.id)) : swe.sets;
   692→    if (sets.length === 0) continue;
   693→    affected.add(swe.exerciseId);
   694→    // find-or-create target workout exercise
   695→    let twe = await db.workoutExercise.findFirst({ where: { workoutId: target.id, exerciseId: swe.exerciseId } });
   696→    if (!twe) {
   697→      twe = await db.workoutExercise.create({
   698→        data: {
   699→          id: uuid7(),
   700→          userId,
   701→          workoutId: target.id,
   702→          exerciseId: swe.exerciseId,
   703→          sortOrder: sortOrder++,
   704→          groupId: swe.groupId ? groupMap.get(swe.groupId) ?? null : null,
   705→        },
   706→      });
   707→    } else if (swe.groupId && !twe.groupId) {
   708→      await db.workoutExercise.update({ where: { id: twe.id }, data: { groupId: groupMap.get(swe.groupId) ?? null } });
   709→    }
   710→    let setOrder = await db.trainingSet.count({ where: { workoutExerciseId: twe.id } });
   711→    for (const s of sets) {
   712→      await db.trainingSet.create({
   713→        data: {
   714→          id: uuid7(),
   715→          userId,
   716→          workoutExerciseId: twe.id,
   717→          weight: s.weight,
   718→          reps: s.reps,
   719→          distance: s.distance,
   720→          timeSec: s.timeSec,
   721→          comment: s.comment,
   722→          isComplete: false,
   723→          isWarmup: s.isWarmup,
   724→          sortOrder: setOrder++,
   725→        },
   726→      });
   727→    }
   728→  }
   729→  await db.$transaction(async (tx) => {
   730→    for (const exerciseId of affected) await recomputePRs(tx, userId, exerciseId);
   731→  });
   732→  return getWorkout(userId, target.id);
   733→}

   735→/** Move the whole workout (or selected exercises) to another date. */
   736→export async function moveWorkout(
   737→  userId: string,
   738→  workoutId: string,
   739→  input: { toDate: string; workoutExerciseIds?: string[] },
   740→) {
   741→  const source = await db.workout.findFirst({ where: { id: workoutId, userId }, include: workoutInclude });
   742→  if (!source) throw notFound("Workout not found");
   743→  const newDate = toDayUtc(input.toDate);
   744→  if (newDate.getTime() === source.date.getTime() && !input.workoutExerciseIds?.length) {
   745→    return getWorkout(userId, source.id); // no-op
   746→  }
   747→
   748→  const movingAll = !input.workoutExerciseIds?.length;
   749→  const movingIds = new Set(input.workoutExerciseIds ?? []);
   750→
   751→  // find-or-create target workout at newDate
   752→  let target = await db.workout.findUnique({ where: { userId_date: { userId, date: newDate } }, include: workoutInclude });
   753→  if (!target) {
   754→    target = await db.workout.create({ data: { id: uuid7(), userId, date: newDate }, include: workoutInclude });
   755→  } else if (target.id === source.id) {
   756→    return updateWorkout(userId, source.id, { date: input.toDate });
   757→  }
   758→
   759→  const toMove = source.exercises.filter((we) => movingAll || movingIds.has(we.id));
   760→  let sortOrder = target.exercises.length;
   761→  for (const swe of toMove) {
   762→    await db.workoutExercise.update({ where: { id: swe.id }, data: { workoutId: target.id, sortOrder: sortOrder++, groupId: null } });
   763→  }
   764→  // copy groups of moved exercises into target
   765→  const groupMap = new Map<string, string>();
   766→  for (const g of source.groups) {
   767→    const hasMoved = toMove.some((we) => we.groupId === g.id);
   768→    if (!hasMoved) continue;
   769→    const existing = target.groups.find((tg) => tg.name === g.name);
   770→    const g2 = existing ?? (await db.workoutGroup.create({ data: { id: uuid7(), userId, workoutId: target.id, name: g.name, colour: g.colour } }));
   771→    groupMap.set(g.id, g2.id);
   772→  }
   773→  for (const swe of toMove) {
   774→    if (swe.groupId && groupMap.has(swe.groupId)) {
   775→      await db.workoutExercise.update({ where: { id: swe.id }, data: { groupId: groupMap.get(swe.groupId)! } });
   776→    }
   777→  }
   778→
   779→  const remaining = await db.workoutExercise.count({ where: { workoutId: source.id } });
   780→  if (remaining === 0) {
   781→    await db.workout.delete({ where: { id: source.id } });
   782→  } else {
   783→    const rest = await db.workoutExercise.findMany({ where: { workoutId: source.id }, orderBy: { sortOrder: "asc" } });
   784→    for (let i = 0; i < rest.length; i++) {
   785→      if (rest[i].sortOrder !== i) await db.workoutExercise.update({ where: { id: rest[i].id }, data: { sortOrder: i } });
   786→    }
   787→  }
   788→  return getWorkout(userId, target.id);
   789→}

   793→export async function previousWorkoutDate(userId: string, beforeDate?: string | Date): Promise<string | null> {
   794→  const d = beforeDate ? toDayUtc(beforeDate) : todayDayUtc();
   795→  const prev = await db.workout.findFirst({ where: { userId, date: { lt: d } }, orderBy: { date: "desc" } });
   796→  return prev ? dayKeyIso(prev.date) : null;

   799→function dayKeyIso(d: Date): string {
   800→  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}-${String(d.getUTCDate()).padStart(2, "0")}`;
   801→}
