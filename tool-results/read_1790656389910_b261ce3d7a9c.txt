   131→/**
   132→ * True when an event originates from an interactive control INSIDE the row —
   133→ * or from portal content that React bubbles through the row (Radix menu items
   134→ * live in a body portal but still propagate synthetic events up the React tree
   135→ * to the row, so [role=menuitem]/[role=menu] must be guarded too — without it

   139→function isInteractiveTarget(target: EventTarget | null): boolean {
   140→  const el = target instanceof HTMLElement ? target : null;
   141→  if (!el) return false;
   142→  return !!el.closest("button, input, a, [role=menuitem], [role=menu], [data-radix-popper-content-wrapper]");
   143→}

   145→/** `12 · 3d` — total sets logged · days since last performed. */
   146→function metaLabel(ex: ExerciseDTO, setCount: number | undefined): string {
   147→  const sets = setCount ?? 0;
   148→  if (!ex.lastPerformed) return `${sets} · –`;
   149→  const days = Math.floor((Date.now() - new Date(ex.lastPerformed).getTime()) / 86_400_000);
   150→  let since: string;
   151→  if (days <= 0) since = "today";
   152→  else if (days < 30) since = `${days}d`;
   153→  else if (days < 365) since = `${Math.floor(days / 30)}mo`;
   154→  else since = `${Math.floor(days / 365)}y`;
   155→  return `${sets} · ${since}`;
   156→}
