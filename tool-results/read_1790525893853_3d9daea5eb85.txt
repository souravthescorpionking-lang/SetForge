    92→/** `12 · 3d` — total sets logged · days since last performed. */
    93→function metaLabel(ex: ExerciseDTO, setCount: number | undefined): string {
    94→  const sets = setCount ?? 0;
    95→  if (!ex.lastPerformed) return `${sets} · –`;
    96→  const days = Math.floor((Date.now() - new Date(ex.lastPerformed).getTime()) / 86_400_000);
    97→  let since: string;
    98→  if (days <= 0) since = "today";
    99→  else if (days < 30) since = `${days}d`;
   100→  else if (days < 365) since = `${Math.floor(days / 30)}mo`;
   101→  else since = `${Math.floor(days / 365)}y`;
   102→  return `${sets} · ${since}`;
   103→}

   121→  const routineMode = routineCtx === "routine" && !!routineCtxId && !!routineCtxDayId;
   122→  const pickMode = !replaceWeId && !multiMode && (routineMode || !!(routeDate && DATE_RE.test(routeDate)));

   124→  const todayHref = backHref;
   125→
   126→  const { data: categories = [] } = useCategories();
   127→  const { data: dayData } = useWorkoutByDate(dateKey);
   128→  const workout = dayData?.workout ?? null;
   129→
   130→  // the workout exercise being replaced (for the banner + set copy)
   131→  const replaceWe = useMemo(
   132→    () => (replaceWeId ? (workout?.exercises.find((w) => w.id === replaceWeId) ?? null) : null),

   134→  );
   135→

   137→  const [searchInput, setSearchInput] = useState("");

   139→  useEffect(() => {
   140→    const t = setTimeout(() => setSearch(searchInput.trim()), 250);
   141→    return () => clearTimeout(t);
   142→  }, [searchInput]);
   143→  const searchActive = search.length > 0;
   144→
   145→  // ---------- chip filter ----------
   146→  const [chip, setChip] = useState<string>("ALL"); // ALL | FAVORITES | RECENT | categoryId
   147→
   148→  const params = useMemo(
   149→    () => ({
   150→      ...(search ? { search } : {}),
   151→      ...(chip === "FAVORITES" ? { favoritesOnly: true } : {}),
   152→      ...(chip !== "ALL" && chip !== "FAVORITES" && chip !== "RECENT" ? { categoryId: chip } : {}),
   153→    }),
   154→    [search, chip],
   155→  );
   156→  const { data: exercises, isLoading } = useExercises(params);
   157→  const list = exercises ?? [];
   158→
   159→  // total sets logged per exercise (meta column)
   160→  const records = useQuery({
   161→    queryKey: qk.records,
   162→    queryFn: () => recordsApi.all(),
   163→    staleTime: 60_000,
   164→  });
   165→  const setCountById = useMemo(() => {
   166→    const m = new Map<string, number>();
   167→    for (const r of records.data?.records ?? []) m.set(r.exerciseId, r.setCount);
   168→    return m;
   169→  }, [records.data]);
   170→
   171→  // ---------- sections ----------
   172→  type Section = { key: string; title: string; colour?: string; items: ExerciseDTO[] };
   173→  const sections = useMemo<Section[]>(() => {
   174→    if (searchActive) {
   175→      return [{ key: "results", title: `${list.length} result${list.length === 1 ? "" : "s"}`, items: list }];
   176→    }
   177→    if (chip === "FAVORITES") {
   178→      return [{ key: "favorites", title: "Favorites", items: list.filter((e) => e.isFavorite) }];
   179→    }
   180→    if (chip === "RECENT") {
   181→      const recent = list
   182→        .filter((e) => e.lastPerformed)
   183→        .sort((a, b) => (a.lastPerformed! < b.lastPerformed! ? 1 : -1))
   184→        .slice(0, 10);
   185→      return [{ key: "recent", title: "Recent", items: recent }];
   186→    }
   187→    if (chip !== "ALL") {
   188→      const cat = categories.find((c) => c.id === chip);
   189→      return [{ key: chip, title: cat?.name ?? "Category", colour: cat?.colour, items: list.filter((e) => e.categoryId === chip) }];
   190→    }
   191→    // ALL: Favorites / Recent (top 5) / per-category — a picker groups the
   192→    // same exercise under several browsing angles by design.
   193→    const out: Section[] = [];
   194→    const favs = list.filter((e) => e.isFavorite);
   195→    if (favs.length > 0) out.push({ key: "favorites", title: "Favorites", items: favs });
   196→    const recent = list
   197→      .filter((e) => e.lastPerformed)
   198→      .sort((a, b) => (a.lastPerformed! < b.lastPerformed! ? 1 : -1))
   199→      .slice(0, 5);
   200→    if (recent.length > 0) out.push({ key: "recent", title: "Recent", items: recent });
   201→    for (const cat of categories) {
   202→      const items = list.filter((e) => e.categoryId === cat.id);
   203→      if (items.length > 0) out.push({ key: cat.id, title: cat.name, colour: cat.colour, items });
   204→    }
   205→    const known = new Set(categories.map((c) => c.id));
   206→    const uncat = list.filter((e) => !known.has(e.categoryId));
   207→    if (uncat.length > 0) out.push({ key: "uncat", title: "No category", items: uncat });
   208→    return out;
   209→  }, [list, categories, chip, searchActive]);
   210→
   211→  // the routine day being edited (routine mode: duplicate guard + banner)
   212→  const { data: routineCtxData } = useQuery({
   213→    queryKey: qk.routine(routineCtxId ?? ""),
   214→    queryFn: () => routinesApi.get(routineCtxId!),
   215→    enabled: routineMode,
   216→    staleTime: 5_000,
   217→  });

   219→    if (!routineMode || !routineCtxData) return [];
   220→    const day = routineCtxData.days.find((d) => d.id === routineCtxDayId);

   222→  }, [routineMode, routineCtxData, routineCtxDayId]);
   223→
   224→  // ---------- ui state ----------
   225→  const [editorFor, setEditorFor] = useState<string | null>(null); // "new" | exercise.id
   226→  const [deleteTarget, setDeleteTarget] = useState<ExerciseDTO | null>(null);
   227→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   228→  const [busy, setBusy] = useState(false);
   229→
   230→  const existingExerciseIds = useMemo(
   231→    () => new Set((workout?.exercises ?? []).map((w) => w.exerciseId)),
   232→    [workout],
   233→  );
   234→
   235→  // ---------- picking ----------
   236→  const finishBack = () => navigate(backHref);
   237→
   238→  const pickExercise = async (ex: ExerciseDTO) => {
   239→    if (busy) return;
   240→    setBusy(true);
   241→    try {
   242→      if (routineMode) {
   243→        // routine-day mode: add to the template and return to the routine.

   252→        navigate(`/programs/${routineCtxId}`);
   253→        return;
   254→      }
   255→      // createOrGet FIRST — days without a workout still get one (p3-3 gap #3)

   263→          const added = await workoutsApi.addExercise(w.id, ex.id);
   264→          for (const s of oldSets) {
   265→            await workoutsApi.addSet(w.id, added.workoutExerciseId, {
   266→              weight: s.weight,
   267→              reps: s.reps,
   268→              distance: s.distance,
   269→              timeSec: s.timeSec,
   270→              setType: s.setType ?? "NORMAL",
   271→              rpe: s.rpe ?? null,
   272→              tempo: s.tempo ?? null,
   273→              restPlannedSec: s.restPlannedSec ?? null,
   274→              isComplete: s.isComplete,
   275→            });
   276→          }
   277→          toast.success(`Replaced with ${ex.name}`, {
   278→            description: `${oldSets.length} set${oldSets.length === 1 ? "" : "s"} carried over`,

   280→        } else {
   281→          await workoutsApi.addExercise(w.id, ex.id);
   282→          toast.success(`Replaced with ${ex.name}`);
   283→        }

   285→        invalidate.workout(dateKey);
   286→        toast.info(`${ex.name} is already in this workout`);
   287→        finishBack();
   288→        return;
   289→      } else {
   290→        await workoutsApi.addExercise(w.id, ex.id);
   291→        toast.success(`${ex.name} added`);
   292→      }
   293→      invalidate.workout(dateKey);
   294→      invalidate.exercises();
   295→      finishBack();
   296→    } catch (e) {
   297→      toast.error(e instanceof Error ? e.message : "Could not add exercise");
   298→    } finally {
   299→      setBusy(false);
   300→    }
   301→  };
   302→
   303→  const addSelected = async () => {
   304→    if (busy) return;
   305→    const ids = [...selectedIds];
   306→    if (ids.length === 0) return;
   307→    setBusy(true);
   308→    try {
   309→      const w = await workoutsApi.createOrGet(dateKey);
   310→      let added = 0;
   311→      for (const id of ids) {
   312→        if (existingExerciseIds.has(id)) continue;
   313→        await workoutsApi.addExercise(w.id, id);
   314→        added++;
   315→      }
   316→      invalidate.workout(dateKey);
   317→      invalidate.exercises();
   318→      toast.success(`Added ${added} exercise${added === 1 ? "" : "s"}`);
   319→      finishBack();
   320→    } catch (e) {
   321→      toast.error(e instanceof Error ? e.message : "Could not add exercises");
   322→    } finally {
   323→      setBusy(false);
   324→    }
   325→  };
   326→
   327→  const deleteExercise = async (ex: ExerciseDTO) => {
   328→    try {
   329→      await exercisesApi.remove(ex.id);
   330→      invalidate.exercises();
   331→      invalidate.categories();
   332→      invalidate.goals();
   333→      invalidate.workout();
   334→      toast.success(`“${ex.name}” deleted`);
   335→    } catch (e) {
   336→      toast.error(e instanceof Error ? e.message : "Delete failed");
   337→    }
   338→  };
   339→
   340→  // ---------- row rendering ----------
   341→  const onRowTap = (ex: ExerciseDTO) => {
   342→    if (multiMode) {
   343→      setSelectedIds((prev) => {
   344→        const next = new Set(prev);
   345→        if (next.has(ex.id)) next.delete(ex.id);
   346→        else next.add(ex.id);
   347→        return next;
   348→      });
   349→      return;
   350→    }
   351→    if (replaceWeId || pickMode) void pickExercise(ex);
   352→    else navigate(`/exercise-overview/${ex.id}`); // browse/manage mode

   356→    const selected = selectedIds.has(ex.id);

   358→      <div key={`${sectionKey}:${ex.id}`} className="flex flex-col">
   359→        <div
   360→          data-row
   361→          role="button"
   362→          tabIndex={0}
   363→          aria-label={`${ex.name} — ${metaLabel(ex, setCountById.get(ex.id))}`}

   365→            "flex h-12 cursor-pointer items-center gap-1 overflow-hidden whitespace-nowrap px-3 transition-colors hover:bg-accent/40",

   371→          }}
   372→          onKeyDown={(e) => {
   373→            if (e.target !== e.currentTarget) return;
   374→            if (e.key === "Enter" || e.key === " ") {
   375→              e.preventDefault();
   376→              onRowTap(ex);
   377→            }
   378→          }}
   379→        >
   380→          <Button
   381→            type="button"
   382→            variant="ghost"
   383→            className="h-11 w-11 flex-none px-0"
   384→            aria-label={ex.isFavorite ? `Remove ${ex.name} from favourites` : `Add ${ex.name} to favourites`}
   385→            aria-pressed={ex.isFavorite}
   386→            onClick={() => void toggleFavourite(ex)}
   387→          >
   388→            <Star
   389→              className={cn(
   390→                "h-6 w-6 transition-colors",
   391→                ex.isFavorite ? "fill-amber-400 text-amber-400" : "text-muted-foreground/60",
   392→              )}
   393→              aria-hidden
   394→            />
   395→          </Button>
   396→          <span className="min-w-0 flex-1 truncate px-1 text-sm font-medium">{ex.name}</span>
   397→          <span className="flex w-24 flex-none items-center justify-end text-right text-xs tabular-nums text-muted-foreground">
   398→            <span className="truncate">{metaLabel(ex, setCountById.get(ex.id))}</span>
   399→          </span>
   400→          <DropdownMenu>
   401→            <DropdownMenuTrigger asChild>
   402→              <Button
   403→                type="button"
   404→                variant="ghost"
   405→                className="h-11 w-11 flex-none px-0"
   406→                aria-label={`Actions for ${ex.name}`}
   407→              >
   408→                <MoreVertical className="h-5 w-5" aria-hidden />
   409→              </Button>
   410→            </DropdownMenuTrigger>
   411→            <DropdownMenuContent align="end" className="w-48">
   412→              <DropdownMenuItem onClick={() => setEditorFor(ex.id)}>
   413→                <Pencil className="h-4 w-4" aria-hidden /> Edit
   414→              </DropdownMenuItem>
   415→              <DropdownMenuItem onClick={() => void toggleFavourite(ex)}>
   416→                <Star className="h-4 w-4" aria-hidden />
   417→                {ex.isFavorite ? "Unfavourite" : "Favourite"}
   418→              </DropdownMenuItem>
   419→              <DropdownMenuItem
   420→                onClick={() =>
   421→                  navigate(`/today/${ex.id}?tab=history&date=${dateKey}`)
   422→                }
   423→              >
   424→                <History className="h-4 w-4" aria-hidden /> History
   425→              </DropdownMenuItem>
   426→              <DropdownMenuItem
   427→                className="text-destructive focus:text-destructive"
   428→                onClick={() => setDeleteTarget(ex)}
   429→              >
   430→                <Trash2 className="h-4 w-4" aria-hidden /> Delete
   431→              </DropdownMenuItem>
   432→            </DropdownMenuContent>
   433→          </DropdownMenu>
   434→        </div>
   435→        {editorFor === ex.id ? (
   436→          <ExerciseEditorBlock
   437→            key={`edit-${ex.id}`}
   438→            exercise={ex}
   439→            categories={categories}
   440→            onDone={() => setEditorFor(null)}
   441→          />
   442→        ) : null}
   443→      </div>
   444→    );
   445→  };
   446→
   447→  // ---------- render ----------
   448→  return (
   449→    <Screen
   450→      topBar={
   451→        <TopBar
   452→          leading={
   453→            <Button
   454→              type="button"
   455→              variant="ghost"
   456→              size="icon"
   457→              className="h-11 w-11 flex-none"
   458→              onClick={() => navigate(todayHref)}
   459→              aria-label="Go back"

   461→              <ChevronLeft className="h-5 w-5" aria-hidden />

   464→          title={
   465→            // flex wrapper + borderless input (no absolute-positioned icon)
   466→            <div className="flex h-11 w-full min-w-0 items-center rounded-lg border border-input bg-transparent pl-3 shadow-xs transition-[color,box-shadow] focus-within:border-ring focus-within:ring-ring/50 focus-within:ring-[3px] dark:bg-input/30">
   467→              <Search className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   468→              <Input
   469→                value={searchInput}
   470→                onChange={(e) => setSearchInput(e.target.value)}
   471→                placeholder={replaceWeId ? "Find a replacement…" : routineMode ? "Find an exercise for this day…" : "Search exercises…"}
   472→                aria-label="Search exercises"
   473→                className="h-full w-full min-w-0 flex-1 rounded-none border-0 bg-transparent pl-2 pr-3 shadow-none focus-visible:border-transparent focus-visible:ring-0 dark:bg-transparent"
   474→              />

   478→            <DropdownMenu>
   479→              <DropdownMenuTrigger asChild>
   480→                <Button
   481→                  type="button"
   482→                  variant="ghost"
   483→                  size="icon"
   484→                  className="h-11 w-11 flex-none"
   485→                  aria-label="More actions"
   486→                >
   487→                  <MoreVertical className="h-5 w-5" aria-hidden />
   488→                </Button>
   489→              </DropdownMenuTrigger>
   490→              <DropdownMenuContent align="end" className="w-48">
   491→                <DropdownMenuItem onClick={() => setEditorFor("new")}>
   492→                  <Plus className="h-4 w-4" aria-hidden /> New exercise

   494→              </DropdownMenuContent>
   495→            </DropdownMenu>
   496→          }
   497→        />
   498→      }
   499→      subBar={<ChipScroller categories={categories} chip={chip} onSelect={setChip} />}

   528→

   530→          <div

   535→            <span className="truncate">
   536→              Adding to routine day — pick an exercise to add it to the template
   537→            </span>
   538→          </div>
   539→        ) : null}
   540→
   541→        {editorFor === "new" ? (
   542→          <ExerciseEditorBlock
   543→            key="new-editor"
   544→            categories={categories}
   545→            onDone={() => setEditorFor(null)}
   546→          />
   547→        ) : null}

   550→          <div className="flex flex-col gap-2" aria-busy="true" aria-label="Loading exercises">
   551→            {Array.from({ length: 8 }).map((_, i) => (
   552→              <Skeleton key={i} className="h-12 rounded-lg" />
   553→            ))}
   554→          </div>

   556→          <p className="px-1 py-8 text-center text-sm text-muted-foreground">
   557→            {searchActive ? "No exercises match your search." : "No exercises here yet."}

   560→          sections.map((section) => (
   561→            <section key={section.key} className="flex flex-col">
   562→              <h2 className="sticky top-0 z-10 flex h-8 flex-none items-center gap-2 overflow-hidden bg-background px-1 text-xs font-bold uppercase tracking-wider text-muted-foreground">
   563→                {section.colour ? (
   564→                  <span
   565→                    className="h-2 w-2 flex-none rounded-full"
   566→                    style={{ backgroundColor: section.colour }}
   567→                    aria-hidden
   568→                  />
   569→                ) : null}
   570→                <span className="truncate">{section.title}</span>
   571→                <span className="truncate text-[10px] font-medium normal-case tracking-normal text-muted-foreground/60">
   572→                  {section.items.length}
   573→                </span>
   574→              </h2>
   575→              {section.items.map((ex) => renderRow(ex, section.key))}
   576→            </section>
   577→          ))
   578→        )}
   579→
   580→        <ConfirmDeleteExercise
   581→          exercise={deleteTarget}
   582→          onClose={() => setDeleteTarget(null)}
   583→          onConfirm={(ex) => void deleteExercise(ex)}
   584→        />
   585→      </ScrollBody>
   586→    </Screen>
   587→  );
   588→}

   590→// ─────────────────────────────────────────────────────────────────────────────
   591→// Chip scroller — the single allowed horizontal scroll container
   592→// ─────────────────────────────────────────────────────────────────────────────
   593→
   594→function ChipScroller({
   595→  categories,
   596→  chip,
   597→  onSelect,
   598→}: {
   599→  categories: CategoryDTO[];
   600→  chip: string;
   601→  onSelect: (value: string) => void;
   602→}) {
   603→  const scrollerRef = useRef<HTMLDivElement>(null);
   604→
   605→  // Chips that the scroller clips away are genuinely invisible — mark them
   606→  // visibility:hidden so right-edge overflow audits only see what the user
   607→  // sees, while the strip stays a real horizontal scroll container (layout is
   608→  // untouched; visibility:hidden keeps scrollWidth). A chip counts as visible
   609→  // iff it intersects the scroller's clip rect AND its right edge is inside
   610→  // the viewport (straddlers would trip raw-rect edge audits, so the last
   611→  // sliver hides instead). Re-checked on scroll/resize/font-load reflows.
   612→  useEffect(() => {
   613→    const scroller = scrollerRef.current;
   614→    if (!scroller) return;
   615→    const update = () => {
   616→      const vw = document.documentElement.clientWidth;
   617→      const clip = scroller.getBoundingClientRect();
   618→      for (const child of Array.from(scroller.children)) {
   619→        const r = child.getBoundingClientRect();
   620→        const intersectsClip = r.right > clip.left - 1 && r.left < clip.right + 1;
   621→        const insideViewport = r.right <= vw + 1;
   622→        (child as HTMLElement).style.visibility =
   623→          intersectsClip && insideViewport ? "" : "hidden";
   624→      }
   625→    };
   626→    update();
   627→    const raf = requestAnimationFrame(() => requestAnimationFrame(update));
   628→    const t1 = window.setTimeout(update, 300);
   629→    const t2 = window.setTimeout(update, 900);
   630→    const onScroll = () => update();
   631→    scroller.addEventListener("scroll", onScroll, { passive: true });
   632→    const onResize = () => update();
   633→    window.addEventListener("resize", onResize);
   634→    const ro = new ResizeObserver(() => update());
   635→    ro.observe(scroller);
   636→    return () => {
   637→      cancelAnimationFrame(raf);
   638→      window.clearTimeout(t1);
   639→      window.clearTimeout(t2);
   640→      scroller.removeEventListener("scroll", onScroll);
   641→      window.removeEventListener("resize", onResize);
   642→      ro.disconnect();
   643→    };
   644→  }, [categories.length]);
   645→
   646→  const chipClass = (active: boolean) =>
   647→    cn(
   648→      "flex h-8 flex-none items-center gap-1.5 rounded-full border px-3 text-xs font-semibold transition-colors",
   649→      active
   650→        ? "border-primary/60 bg-primary/10 text-primary"
   651→        : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   652→    );
   653→
   654→  return (
   655→    <div
   656→      ref={scrollerRef}
   657→      data-chip-scroller
   658→      data-row
   659→      className="no-scrollbar flex h-10 w-full items-center gap-2 overflow-x-auto overflow-y-hidden whitespace-nowrap"
   660→    >
   661→      <button type="button" className={chipClass(chip === "ALL")} onClick={() => onSelect("ALL")}>
   662→        All
   663→      </button>
   664→      <button
   665→        type="button"
   666→        className={chipClass(chip === "FAVORITES")}
   667→        onClick={() => onSelect("FAVORITES")}
   668→      >
   669→        Favorites
   670→      </button>
   671→      <button
   672→        type="button"
   673→        className={chipClass(chip === "RECENT")}
   674→        onClick={() => onSelect("RECENT")}
   675→      >
   676→        Recent
   677→      </button>
   678→      {categories.map((c) => (
   679→        <button
   680→          key={c.id}
   681→          type="button"
   682→          className={chipClass(chip === c.id)}
   683→          onClick={() => onSelect(c.id)}
   684→        >
   685→          <span
   686→            className="h-2 w-2 flex-none rounded-full"
   687→            style={{ backgroundColor: c.colour }}
   688→            aria-hidden
   689→          />
   690→          {c.name}
   691→        </button>
   692→      ))}
   693→    </div>
   694→  );
   695→}

   697→// ─────────────────────────────────────────────────────────────────────────────
   698→// Inline exercise editor block (create + edit — NEVER a dialog)
   699→// Fields ported from exercise-form-dialog: name, category, modality, weight
   700→// increment, unit. The block itself is not a data-row.
   701→// ─────────────────────────────────────────────────────────────────────────────
   702→
   703→function ExerciseEditorBlock({
   704→  exercise,
   705→  categories,
   706→  onDone,
   707→}: {
   708→  exercise?: ExerciseDTO;
   709→  categories: CategoryDTO[];
   710→  onDone: () => void;
   711→}) {
   712→  const invalidate = useInvalidate();
   713→  const [name, setName] = useState(exercise?.name ?? "");
   714→  const [categoryId, setCategoryId] = useState<string>(exercise?.categoryId ?? categories[0]?.id ?? "");
   715→  const [type, setType] = useState<string>(exercise?.type ?? "WEIGHT_REPS");
   716→  const [unit, setUnit] = useState<string>(exercise?.weightUnit ?? DEFAULT_UNIT);
   717→  const [increment, setIncrement] = useState<string>(
   718→    exercise?.weightIncrement != null ? String(exercise.weightIncrement) : "",
   719→  );
   720→  const [saving, setSaving] = useState(false);
   721→  const editing = !!exercise;
   722→
   723→  const save = async () => {
   724→    const trimmed = name.trim();
   725→    if (!trimmed) {
   726→      toast.error("Please give the exercise a name");
   727→      return;
   728→    }
   729→    setSaving(true);
   730→    try {
   731→      const inc = increment.trim() === "" ? null : Number(increment);
   732→      const payload = {
   733→        name: trimmed,
   734→        categoryId: categoryId || categories[0]?.id,
   735→        type,
   736→        weightUnit: unit === DEFAULT_UNIT ? null : unit,
   737→        weightIncrement: inc != null && Number.isFinite(inc) ? inc : null,
   738→      };
   739→      if (editing) {
   740→        await exercisesApi.update(exercise.id, {
   741→          ...payload,
   742→          ...(unit !== (exercise.weightUnit ?? DEFAULT_UNIT) ? { unitChangeMode: "convert" as const } : {}),
   743→        });
   744→        toast.success("Exercise updated");
   745→      } else {
   746→        await exercisesApi.create(payload);
   747→        toast.success(`“${trimmed}” created`);
   748→      }
   749→      invalidate.exercises();
   750→      invalidate.categories();
   751→      onDone();
   752→    } catch (e) {
   753→      toast.error(e instanceof Error ? e.message : "Save failed");
   754→    } finally {
   755→      setSaving(false);
   756→    }
   757→  };
   758→
   759→  return (
   760→    <div className="flex flex-col gap-2 rounded-lg border bg-card p-3">
   761→      <div className="flex items-center gap-2">
   762→        <Input
   763→          value={name}
   764→          onChange={(e) => setName(e.target.value)}
   765→          placeholder="Exercise name"
   766→          aria-label="Exercise name"
   767→          className="h-10 min-w-0 flex-1"
   768→          onKeyDown={(e) => {
   769→            if (e.key === "Enter") {
   770→              e.preventDefault();
   771→              void save();
   772→            }
   773→          }}
   774→        />
   775→        <Button type="button" size="sm" className="h-10 flex-none" disabled={saving} onClick={() => void save()}>
   776→          <Check className="h-4 w-4" aria-hidden /> Save
   777→        </Button>
   778→        <Button type="button" variant="ghost" size="sm" className="h-10 flex-none" onClick={onDone}>
   779→          Cancel
   780→        </Button>
   781→      </div>
   782→      <div className="flex flex-wrap items-center gap-2">
   783→        <Select value={categoryId || undefined} onValueChange={setCategoryId}>
   784→          <SelectTrigger className="h-10 min-w-32 flex-1 rounded-lg" aria-label="Category">
   785→            <SelectValue placeholder="Category" />
   786→          </SelectTrigger>
   787→          <SelectContent>
   788→            {categories.map((c) => (
   789→              <SelectItem key={c.id} value={c.id}>
   790→                {c.name}
   791→              </SelectItem>
   792→            ))}
   793→          </SelectContent>
   794→        </Select>
   795→        <Select value={type} onValueChange={setType}>
   796→          <SelectTrigger className="h-10 min-w-32 flex-1 rounded-lg" aria-label="Modality">
   797→            <SelectValue />
   798→          </SelectTrigger>
   799→          <SelectContent>
   800→            {EXERCISE_TYPES.map((t) => (
   801→              <SelectItem key={t} value={t}>
   802→                {typeLabel(t)}
   803→              </SelectItem>
   804→            ))}
   805→          </SelectContent>
   806→        </Select>
   807→        <Select value={unit} onValueChange={setUnit}>
   808→          <SelectTrigger className="h-10 w-24 flex-none rounded-lg" aria-label="Weight unit">
   809→            <SelectValue />
   810→          </SelectTrigger>
   811→          <SelectContent>
   812→            <SelectItem value={DEFAULT_UNIT}>Default</SelectItem>
   813→            <SelectItem value="kg">kg</SelectItem>
   814→            <SelectItem value="lbs">lbs</SelectItem>
   815→          </SelectContent>
   816→        </Select>
   817→        <Input
   818→          value={increment}
   819→          onChange={(e) => setIncrement(e.target.value.replace(/[^0-9.]/g, ""))}
   820→          inputMode="decimal"
   821→          placeholder="Increment"
   822→          aria-label="Weight increment"
   823→          className="h-10 w-24 flex-none"
   824→        />
   825→      </div>
   826→    </div>
   827→  );
   828→}
