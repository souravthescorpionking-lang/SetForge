   103→   103→function Stepper({
   104→   104→  value,
   105→   105→  min,
   106→   106→  max,
   107→   107→  formatValue,
   108→   108→  ariaLabel,
   109→   109→  onDecrement,
   110→   110→  onIncrement,
   111→   111→}: {
   112→   112→  value: number | null;
   113→   113→  min: number;
   114→   114→  max: number;
   115→   115→  formatValue?: (v: number | null) => string;
   116→   116→  ariaLabel: string;
   117→   117→  onDecrement: () => void;
   118→   118→  onIncrement: () => void;
   119→   119→}) {
   120→   120→  const text = formatValue ? formatValue(value) : value == null ? "–" : String(value);
   121→   121→  return (
   122→   122→    <span className="flex flex-none items-center gap-1">
   123→   123→      <Button
   124→   124→        type="button"
   125→   125→        variant="outline"
   126→   126→        className="h-10 w-10 flex-none p-0"
   127→   127→        disabled={value != null && value <= min}
   128→   128→        aria-label={`Decrease ${ariaLabel}`}
   129→   129→        onClick={() => {
   130→   130→          hapticTap();
   131→   131→          onDecrement();
   132→   132→        }}
   133→   133→      >
   134→   134→        <Minus className="h-4 w-4" aria-hidden />
   135→   135→      </Button>
   136→   136→      <span
   137→   137→        className="flex h-10 w-12 flex-none items-center justify-center text-sm font-bold tabular-nums leading-none"
   138→   138→        aria-live="polite"
   139→   139→        aria-label={`${ariaLabel}: ${text}`}
   140→   140→      >
   141→   141→        {text}
   142→   142→      </span>
   143→   143→      <Button
   144→   144→        type="button"
   145→   145→        variant="outline"
   146→   146→        className="h-10 w-10 flex-none p-0"
   147→   147→        disabled={value != null && value >= max}
   148→   148→        aria-label={`Increase ${ariaLabel}`}
   149→   149→        onClick={() => {
   150→   150→          hapticTap();
   151→   151→          onIncrement();
   152→   152→        }}
   153→   153→      >
   154→   154→        <Plus className="h-4 w-4" aria-hidden />
   155→   155→      </Button>
   156→   156→    </span>
   157→   157→  );
   158→   158→}

   163→   163→  const navigate = useApp((s) => s.navigate);
   164→   164→  const online = useOnline();

   166→   166→
   167→   167→  // ---------- draft state (local only — nothing persists before Create) ----------
   168→   168→  // #/programs/new/builder?name=… deep-links can pre-fill the name.
   169→   169→  const [name, setName] = useState(() => {

   176→   176→  const [estMinutes, setEstMinutes] = useState<number | null>(null);

   179→   179→  const [labels, setLabels] = useState<string[]>([]);
   180→   180→  const [addingLabel, setAddingLabel] = useState(false);

   183→   183→  const [pickerFor, setPickerFor] = useState<string | null>(null);
   184→   184→  const [pickerSearch, setPickerSearch] = useState("");

   186→   186→  const searchFocusRef = useRef<HTMLInputElement | null>(null);
   187→   187→
   188→   188→  // ---------- derived ----------
   189→   189→  const workoutDayCount = weekly.filter((w) => w.type === "WORKOUT").length;
   190→   190→  const daysPerWeek = daysManual ?? workoutDayCount;
   191→   191→  const templateNames = useMemo(() => {
   192→   192→    const seen: string[] = [];
   193→   193→    for (const w of weekly) {
   194→   194→      if (w.type !== "WORKOUT") continue;
   195→   195→      const n = w.name.trim();
   196→   196→      if (n && !seen.includes(n)) seen.push(n);
   197→   197→    }
   198→   198→    return seen;
   199→   199→  }, [weekly]);
   200→   200→  const unnamedWorkoutDays = weekly.some((w) => w.type === "WORKOUT" && !w.name.trim());
   201→   201→  const totalWeeks = phases.reduce((n, p) => n + p.weeks, 0);
   202→   202→  const totalDays = totalWeeks * 7;
   203→   203→  const canCreate =
   204→   204→    name.trim().length > 0 && templateNames.length > 0 && !unnamedWorkoutDays && !creating;
   205→   205→

   217→   217→    hapticTap();

   219→   219→  };
   220→   220→
   221→   221→  const setDayName = (index: number, dayName: string) => {
   222→   222→    setWeekly((prev) => prev.map((w, i) => (i === index ? { ...w, name: dayName.slice(0, 40) } : w)));
   223→   223→  };

   225→   225→  const addPhase = () => {
   226→   226→    hapticTap();
   227→   227→    setPhases((prev) =>
   228→   228→      prev.length >= 12 ? prev : [...prev, { name: `Phase ${prev.length + 1}`, weeks: 1 }],
   229→   229→    );
   230→   230→  };
   231→   231→
   232→   232→  const removePhase = (index: number) => {
   233→   233→    setPhases((prev) => (prev.length <= 1 ? prev : prev.filter((_, i) => i !== index)));

   237→   237→    hapticTap();

   239→   239→    setPickerFor((prev) => (prev === templateName ? null : templateName));
   240→   240→    if (pickerFor !== templateName) {
   241→   241→      requestAnimationFrame(() => searchFocusRef.current?.focus());
   242→   242→    }
   243→   243→  };
   244→   244→
   245→   245→  const addExerciseTo = (templateName: string, exerciseId: string, exerciseName: string) => {

   247→   247→    setExercises((prev) => {
   248→   248→      const list = prev[templateName] ?? [];
   249→   249→      if (list.some((e) => e.exerciseId === exerciseId)) return prev;

   251→   251→    });
   252→   252→  };
   253→   253→

   256→   256→      const list = (prev[templateName] ?? []).filter((e) => e.exerciseId !== exerciseId);
   257→   257→      const next = { ...prev, [templateName]: list };
   258→   258→      return next;
   259→   259→    });
   260→   260→  };
   261→   261→
   262→   262→  const addLabel = (label: string) => {
   263→   263→    const clean = label.trim().slice(0, 30);
   264→   264→    if (!clean) return;

   270→   270→      toast.info("Labels are limited to 20");
   271→   271→      return;
   272→   272→    }
   273→   273→    setLabels((prev) => [...prev, clean]);
   274→   274→  };
   275→   275→

   278→   278→    if (!online) {
   279→   279→      toast.info("Creating a program needs a connection");
   280→   280→      return;
   281→   281→    }
   282→   282→    setCreating(true);

   285→   285→      for (const templateName of templateNames) {
   286→   286→        const list = exercises[templateName] ?? [];
   287→   287→        if (list.length === 0) continue;
   288→   288→        exercisesPayload[templateName] = list.map((e) => ({
   289→   289→          exerciseId: e.exerciseId,
   290→   290→          sets: BLANK_BUILDER_SETS,
   291→   291→        }));
   292→   292→      }
   293→   293→      const input: BuilderProgramInput = {
   294→   294→        name: name.trim(),
   295→   295→        difficulty,
   296→   296→        ...(daysManual != null || workoutDayCount > 0 ? { daysPerWeek: Math.max(1, Math.min(7, daysPerWeek)) } : {}),
   297→   297→        ...(estMinutes != null ? { estMinutes } : {}),
   298→   298→        ...(labels.length > 0 ? { labels } : {}),
   299→   299→        phases: phases.map((p, i) => ({ name: p.name.trim() || `Phase ${i + 1}`, weeks: p.weeks })),
   300→   300→        weekly: weekly.map((w, i) => ({
   301→   301→          weekday: i,
   302→   302→          type: w.type,
   303→   303→          ...(w.type === "WORKOUT" ? { name: w.name.trim() } : { name: null }),
   304→   304→        })),
   305→   305→        ...(exercisesPayload && Object.keys(exercisesPayload).length > 0 ? { exercises: exercisesPayload } : {}),

   309→   309→      invalidate.routines();
   310→   310→      hapticSuccess();
   311→   311→      toast.success(`Program created · ${res.dayCount} days`);
   312→   312→      replaceHash(`#/programs/${res.id}`);
   313→   313→    } catch (e) {
   314→   314→      toast.error(errorMessage(e));

   317→   317→    }

   343→   343→            Rest day
   344→   344→          </span>
   345→   345→        )}
   346→   346→        <DropdownMenu>
   347→   347→          <DropdownMenuTrigger asChild>
