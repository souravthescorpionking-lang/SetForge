   316→function RecordsTab({ exercise }: { exercise: ExerciseDTO }) {
   317→  const settings = useApp((s) => s.settings);
   318→  const unit = exerciseUnit(exercise, settings);
   319→  const fields = useMemo(() => new Set(fieldsForType(exercise.type)), [exercise.type]);
   320→  const tracksWeightReps = fields.has("weight") && fields.has("reps");
   321→
   322→  const recordsQuery = useQuery({
   323→    queryKey: qk.exerciseRecords(exercise.id),
   324→    queryFn: () => exercisesApi.records(exercise.id),
   325→    staleTime: 60_000,
   326→  });
   327→  const historyQuery = useQuery({
   328→    queryKey: qk.exerciseHistory(exercise.id),
   329→    queryFn: () => exercisesApi.history(exercise.id),
   330→    staleTime: 60_000,
   331→  });
   332→
   333→  const rows = useMemo<{ out: RecordRow[]; actual: RecordRow[]; estimated: RecordRow[] }>(() => {
   334→    const out: RecordRow[] = [];
   335→    const entries = historyQuery.data ?? [];
   336→    const records: RecordsDTO | undefined = recordsQuery.data;
   337→
   338→    // ---- bests from the logged sets (legacy records-tab computations) ----
   339→    let bestWeight: { w: number; day: string } | null = null;
   340→    let bestVolume: { v: number; day: string } | null = null;
   341→    let bestE1rm: { v: number; day: string } | null = null;
   342→    let bestDistance: { v: number; day: string } | null = null;
   343→    let bestTime: { v: number; day: string } | null = null;
   344→    const repLimit = settings?.estOneRmRepLimit ?? 10;
   345→    const method = settings?.e1rmMethod ?? "BRZYCKI";
   346→
   347→    for (const en of entries) {
   348→      const day = dayKeyOf(en.date);
   349→      for (const s of en.sets) {
   350→        if (s.weight != null && (bestWeight == null || s.weight > bestWeight.w)) {
   351→          bestWeight = { w: s.weight, day };
   352→        }
   353→        if (
   354→          s.weight != null &&
   355→          s.reps != null &&
   356→          s.weight > 0 &&
   357→          s.reps > 0 &&
   358→          (bestVolume == null || s.weight * s.reps > bestVolume.v)
   359→        ) {
   360→          bestVolume = { v: s.weight * s.reps, day };
   361→        }
   362→        // same guard as the server: failure sets are not e1RM inputs
   363→        if (
   364→          s.weight != null &&
   365→          s.reps != null &&
   366→          s.reps >= 1 &&
   367→          s.reps <= repLimit &&
   368→          s.setType !== "FAILURE"
   369→        ) {
   370→          const e = estOneRmByMethod(s.weight, s.reps, method, s.rpe);
   371→          if (e > 0 && (bestE1rm == null || e > bestE1rm.v)) bestE1rm = { v: e, day };
   372→        }
   373→        if (s.distance != null && s.distance > 0 && (bestDistance == null || s.distance > bestDistance.v)) {
   374→          bestDistance = { v: s.distance, day };
   375→        }
   376→        if (s.timeSec != null && s.timeSec > 0 && (bestTime == null || s.timeSec > bestTime.v)) {
   377→          bestTime = { v: s.timeSec, day };
   378→        }
   379→      }
   380→    }
   381→
   382→    if (tracksWeightReps) {
   383→      if (bestWeight) {
   384→        out.push({ lift: "Best weight", value: weightLabel(bestWeight.w, unit), date: formatDayShort(bestWeight.day) });
   385→      }
   386→      if (bestVolume) {
   387→        out.push({ lift: "Best volume set", value: `${round1(bestVolume.v)} ${unit}`, date: formatDayShort(bestVolume.day) });
   388→      }
   389→      if (bestE1rm) {
   390→        // value prefers the server computation when loaded (authoritative)
   391→        const v = records && records.estimatedOneRm > 0 ? records.estimatedOneRm : bestE1rm.v;
   392→        out.push({ lift: "Estimated 1RM", value: `${round1(v)} ${unit}`, date: formatDayShort(bestE1rm.day) });
   393→      }
   394→    }
   395→    if (bestDistance) out.push({ lift: "Best distance", value: `${round2(bestDistance.v)} km`, date: formatDayShort(bestDistance.day) });
   396→    if (bestTime) out.push({ lift: "Best time", value: formatDuration(bestTime.v), date: formatDayShort(bestTime.day) });
   397→
   398→    // ---- per-reps actual PRs (records API, superseded dimmed) ----
   399→    const actual: RecordRow[] = (records?.actual ?? []).map((r) => ({
   400→      lift: `Best set at ${r.reps} ${r.reps === 1 ? "rep" : "reps"}`,
   401→      value: weightLabel(r.weight, unit),
   402→      date: formatDayShort(dayKeyOf(r.date)),
   403→      dim: r.superseded,
   404→    }));
   405→
   406→    // ---- estimated rep maxes (nRM, legacy grid → rows; 1RM row above) ----
   407→    const estimated: RecordRow[] = (records?.estimated ?? [])
   408→      .filter((r) => r.reps > 1)
   409→      .map((r) => ({ lift: `Estimated ${r.reps}RM`, value: `${round1(r.weight)} ${unit}`, date: null }));
   410→
   411→    return { out, actual, estimated };
   412→  }, [historyQuery.data, recordsQuery.data, settings, unit, tracksWeightReps]);
   413→
   414→  if (historyQuery.isLoading || recordsQuery.isLoading) {
   415→    return (
   416→      <div className="flex flex-col gap-2" aria-busy="true" aria-label="Loading records">
   417→        {[0, 1, 2, 3, 4, 5, 6].map((i) => (
   418→          <Skeleton key={i} className="h-10 rounded-lg" />
   419→        ))}
   420→      </div>
   421→    );
   422→  }
   423→
   424→  const nothing = rows.out.length === 0 && rows.actual.length === 0 && rows.estimated.length === 0;
   425→
   426→  return (
   427→    <div className="flex flex-col gap-3">
   428→      {nothing ? (
   429→        <div
   430→          data-row
   431→          className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"
   432→        >
   433→          No records yet — log a set to start tracking.
   434→        </div>
   435→      ) : null}
   436→
   437→      {rows.out.length > 0 ? (
   438→        <section className="flex flex-col gap-1" aria-label="Bests">
   439→          <SectionHeader label="Bests" />
   440→          {rows.out.map((r) => (
   441→            <RecordRowLine key={r.lift} row={r} />
   442→          ))}
   443→        </section>
   444→      ) : null}
   445→
   446→      {!tracksWeightReps && (rows.out.length > 0 || nothing) ? (
   447→        <p className="px-1 text-xs text-muted-foreground">
   448→          Weight × reps records need a weighted exercise — showing bests only.
   449→        </p>
   450→      ) : null}
   451→
   452→      {rows.actual.length > 0 ? (
   453→        <section className="flex flex-col gap-1" aria-label="Rep records">
   454→          <SectionHeader label="Rep records" />
   455→          {rows.actual.map((r) => (
   456→            <RecordRowLine key={r.lift} row={r} />
   457→          ))}
   458→        </section>
   459→      ) : null}
   460→
   461→      {rows.estimated.length > 0 ? (
   462→        <section className="flex flex-col gap-1" aria-label="Estimated maxes">
   463→          <SectionHeader label="Estimated maxes" />
   464→          {rows.estimated.map((r) => (
   465→            <RecordRowLine key={r.lift} row={r} />
   466→          ))}
   467→        </section>
   468→      ) : null}
   469→
   470→      {rows.actual.length > 0 ? (
   471→        <p className="px-1 text-xs text-muted-foreground">
   472→          A PR is superseded when a heavier set at the same-or-higher rep count was logged later.
   473→        </p>
   474→      ) : null}
   475→    </div>
   476→  );
   477→}

   521→      </button>

   524→          exercise={exercise}
   525→          unit={unit}

   536→        </div>

   540→            <button
   541→              type="button"
   542→              data-row
   543→              aria-expanded={expandedId === g.id}
   544→              onClick={() => setExpandedId((cur) => (cur === g.id ? null : g.id))}
   545→              className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap border-b border-border/50 text-left transition-colors hover:bg-accent/50"
   546→            >
   547→              <span className="flex min-w-0 flex-1 items-center gap-1.5">
   548→                <Target className="h-3.5 w-3.5 flex-none text-muted-foreground" aria-hidden />
   549→                <span className="min-w-0 truncate text-sm font-medium">
   550→                  {goalTypeLabel(g.type)}
   551→                  <span className="ml-1.5 font-normal text-muted-foreground">

   558→                  "w-[56px] flex-none text-right text-sm font-bold tabular-nums",
   559→                  g.achieved ? "text-primary" : "",
   560→                )}
   561→              >
   562→                {Math.round(g.pct)}%
   563→              </span>
   564→              <span className="flex w-[52px] flex-none items-center justify-end gap-1 text-xs font-semibold">
   565→                {g.achieved ? (
   566→                  <>
   567→                    <Trophy className="h-3.5 w-3.5 flex-none text-emerald-500" aria-hidden />
   568→                    <span className="text-emerald-500">hit</span>
   569→                  </>
   570→                ) : (
   571→                  <span className="text-muted-foreground">open</span>
   572→                )}
   573→              </span>
   574→            </button>
   575→
   576→            {expandedId === g.id ? (
   577→              <GoalExpansion key={g.id} goal={g} unit={unit} onDone={() => setExpandedId(null)} />
   578→            ) : null}
   579→          </div>
   580→        ))
   581→      )}
   582→    </div>
   583→  );
   584→}

   586→/** 96px inline goal creator (NOT a data-row): type select + target input. */
   587→function GoalCreator({
   588→  exercise,
   589→  unit,
   590→  onDone,
   591→}: {
   592→  exercise: ExerciseDTO;
   593→  unit: WeightUnit;
   594→  onDone: () => void;
   595→}) {
   596→  const run = useOfflineRun();
   597→  const inv = useInvalidate();
   598→  const [type, setType] = useState<string>("ONE_RM");
   599→  const [target, setTarget] = useState("");
   600→  const [saving, setSaving] = useState(false);
   601→
   602→  const field = goalTargetField(type);
   603→  const placeholder =
   604→    field === "targetWeight"
   605→      ? `Target ${unit}`
   606→      : field === "targetReps"
   607→        ? "Target reps"
   608→        : field === "targetDistance"
   609→          ? "Target km"
   610→          : "Target sec";
   611→
   612→  const save = async () => {
   613→    const n = Number(target);
   614→    if (target.trim() === "" || !Number.isFinite(n) || n <= 0) {
   615→      toast.error("Set a target value above zero");
   616→      return;
   617→    }
   618→    setSaving(true);
   619→    const payload: GoalInput = {
   620→      type,
   621→      targetWeight: field === "targetWeight" ? n : null,
   622→      targetReps: field === "targetReps" ? n : null,
   623→      targetDistance: field === "targetDistance" ? n : null,
   624→      targetTimeSec: field === "targetTimeSec" ? n : null,
   625→    };
   626→    const ok = await run({
   627→      label: "Goal",
   628→      path: "/api/goals",
   629→      method: "POST",
   630→      body: { ...payload, exerciseId: exercise.id },
   631→      run: () => goalsApi.create({ ...payload, exerciseId: exercise.id }),
   632→      successMsg: "Goal created",
   633→      onDone: () => inv.goals(),
   634→    });
   635→    setSaving(false);
   636→    if (ok) onDone();
   637→  };
   638→
   639→  return (
   640→    <div className="flex h-24 flex-none flex-col gap-1 rounded-lg border bg-card p-1.5">
   641→      <div className="flex min-h-0 flex-1 items-center overflow-hidden">
   642→        <Select value={type} onValueChange={setType}>
   643→          <SelectTrigger className="h-full w-full rounded-lg" aria-label="Goal type">
   644→            <SelectValue />
   645→          </SelectTrigger>
   646→          <SelectContent>
   647→            {["ONE_RM", "MAX_WEIGHT", "MAX_REPS", "MAX_DISTANCE", "MAX_TIME", "VOLUME"].map((t) => (
   648→              <SelectItem key={t} value={t}>
   649→                {goalTypeLabel(t)}
   650→              </SelectItem>
   651→            ))}
   652→          </SelectContent>
   653→        </Select>
   654→      </div>
   655→      <div className="flex min-h-0 flex-1 gap-2">
   656→        <Input
   657→          type="number"
   658→          inputMode="decimal"
   659→          step="any"
   660→          min="0"
   661→          className="h-full flex-1 rounded-lg tabular-nums"
   662→          value={target}
   663→          onChange={(e) => setTarget(e.target.value)}
   664→          placeholder={placeholder}
   665→          aria-label={placeholder}
   666→        />
   667→        <Button
   668→          type="button"
   669→          className="h-full flex-1 rounded-lg font-semibold"
   670→          disabled={saving}
   671→          onClick={() => void save()}
   672→        >
   673→          {saving ? "Saving…" : "Add goal"}
   674→        </Button>
   675→      </div>
   676→    </div>
   677→  );
   678→}

   680→/** 96px inline expansion (NOT a data-row): progress + target editor + delete. */
   681→function GoalExpansion({
   682→  goal,
   683→  unit,
   684→  onDone,
   685→}: {
   686→  goal: GoalDTO;
   687→  unit: WeightUnit;
   688→  onDone: () => void;
   689→}) {
   690→  const run = useOfflineRun();
   691→  const inv = useInvalidate();
   692→  const field = goalTargetField(goal.type);
   693→  const initial =
   694→    field === "targetWeight"
   695→      ? goal.targetWeight
   696→      : field === "targetReps"
   697→        ? goal.targetReps
   698→        : field === "targetDistance"
   699→          ? goal.targetDistance
   700→          : goal.targetTimeSec;
   701→
   702→  const [target, setTarget] = useState<string>(initial != null ? String(initial) : "");
   703→  const [saving, setSaving] = useState(false);
   704→  const [armed, setArmed] = useState(false);
   705→
   706→  // auto-disarm the two-tap delete after a pause
   707→  useEffect(() => {
   708→    if (!armed) return;
   709→    const t = window.setTimeout(() => setArmed(false), 4000);
   710→    return () => window.clearTimeout(t);
   711→  }, [armed]);
   712→
   713→  const save = async () => {
   714→    const n = Number(target);
   715→    if (target.trim() === "" || !Number.isFinite(n) || n <= 0) {
   716→      toast.error("Set a target value above zero");
   717→      return;
   718→    }
   719→    setSaving(true);
   720→    const payload: GoalInput = {
   721→      type: goal.type,
   722→      targetWeight: field === "targetWeight" ? n : null,
   723→      targetReps: field === "targetReps" ? n : null,
   724→      targetDistance: field === "targetDistance" ? n : null,
   725→      targetTimeSec: field === "targetTimeSec" ? n : null,
   726→    };
   727→    const ok = await run({
   728→      label: "Goal update",
   729→      path: `/api/goals/${goal.id}`,
   730→      method: "PATCH",
   731→      body: payload,
   732→      run: () => goalsApi.update(goal.id, payload),
   733→      successMsg: "Goal updated",
   734→      onDone: () => inv.goals(),
   735→    });
   736→    setSaving(false);
   737→    if (ok) onDone();
   738→  };
   739→
   740→  const remove = async () => {
   741→    if (!armed) {
   742→      setArmed(true);
   743→      return;
   744→    }
   745→    const ok = await run({
   746→      label: "Goal delete",
   747→      path: `/api/goals/${goal.id}`,
   748→      method: "DELETE",
   749→      run: () => goalsApi.remove(goal.id),
   750→      successMsg: "Goal deleted",
   751→      onDone: () => inv.goals(),
   752→    });
   753→    if (ok) onDone();
   754→  };
   755→
   756→  return (
   757→    <div className="flex h-24 flex-none flex-col gap-1 rounded-lg border bg-card p-1.5">
   758→      <div className="flex min-h-0 flex-1 items-center gap-2 overflow-hidden whitespace-nowrap">
   759→        <Progress
   760→          value={Math.min(100, goal.pct)}
   761→          className={cn("h-2 flex-1", goal.achieved && "*:data-[slot=progress-indicator]:bg-emerald-500")}
   762→          aria-label={`${goal.pct}% of goal`}
   763→        />
   764→        <span className="flex-none truncate text-xs font-semibold tabular-nums text-muted-foreground">
   765→          {goalValueLabel(goal.type, goal.current, unit)} / {goalValueLabel(goal.type, goal.target, unit)}
   766→        </span>
   767→      </div>
   768→      <div className="flex min-h-0 flex-1 gap-2">
   769→        <Input
   770→          type="number"
   771→          inputMode="decimal"
   772→          step="any"
   773→          min="0"
   774→          className="h-full flex-1 rounded-lg tabular-nums"
   775→          value={target}
   776→          onChange={(e) => setTarget(e.target.value)}
   777→          aria-label="Goal target"
   778→        />
   779→        <Button
   780→          type="button"
   781→          className="h-full flex-1 rounded-lg font-semibold"
   782→          disabled={saving}
   783→          onClick={() => void save()}
   784→        >
   785→          {saving ? "Saving…" : "Save target"}
   786→        </Button>
   787→        <Button
   788→          type="button"
   789→          variant="outline"
   790→          className={cn(
   791→            "h-full flex-none gap-1 rounded-lg",
   792→            armed ? "border-destructive/50 text-destructive" : "text-muted-foreground",
   793→          )}
   794→          onClick={() => void remove()}
   795→        >
   796→          <Trash2 className="h-3.5 w-3.5" aria-hidden />
   797→          {armed ? "Sure?" : "Delete"}
   798→        </Button>
   799→      </div>
   800→    </div>
   801→  );
   802→}

   805→// HISTORY — DateGroup×N (32px headers) + ONE ExerciseCard read mode per date

   808→function HistoryTab({
   809→  exercise,
   810→  visibleColumns,
   811→}: {
   812→  exercise: ExerciseDTO;
   813→  visibleColumns: CardVisibleColumns;
   814→}) {
   815→  const navigate = useApp((s) => s.navigate);
   816→  const settings = useApp((s) => s.settings);
   817→
   818→  const historyQuery = useQuery({
   819→    queryKey: qk.exerciseHistory(exercise.id),
   820→    queryFn: () => exercisesApi.history(exercise.id),
   821→    staleTime: 60_000,
   822→  });
   823→  // records → date-level PR badges (legacy history-tab pattern)
   824→  const recordsQuery = useQuery({
   825→    queryKey: qk.exerciseRecords(exercise.id),
   826→    queryFn: () => exercisesApi.records(exercise.id),
   827→    staleTime: 60_000,
   828→  });
   829→
   830→  const [collapsedIds, setCollapsedIds] = useState<Set<string>>(new Set());
   831→  const [notesEntryId, setNotesEntryId] = useState<string | null>(null);
   832→
   833→  const groups = useMemo(() => {
   834→    const map = new Map<string, HistoryEntry[]>();
   835→    for (const en of historyQuery.data ?? []) {
   836→      const key = dayKeyOf(en.date);
   837→      const list = map.get(key);
   838→      if (list) list.push(en);
   839→      else map.set(key, [en]);
   840→    }
   841→    return [...map.entries()].sort((a, b) => b[0].localeCompare(a[0]));
   842→  }, [historyQuery.data]);
   843→
   844→  const prDates = useMemo(
   845→    () => new Set((recordsQuery.data?.actual ?? []).map((r) => dayKeyOf(r.date))),
   846→    [recordsQuery.data],
   847→  );
   848→
   849→  if (historyQuery.isLoading) {
   850→    return (
   851→      <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading history">
   852→        {[0, 1, 2, 3].map((i) => (
   853→          <Skeleton key={i} className="h-14 rounded-lg" />
   854→        ))}
   855→      </div>
   856→    );
   857→  }
   858→
   859→  if (groups.length === 0) {
   860→    return (
   861→      <div
   862→        data-row
   863→        className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"
   864→      >
   865→        No history yet — every set you log appears here.
   866→      </div>
   867→    );
   868→  }
   869→
   870→  const cardEx: CardExercise = {
   871→    id: exercise.id,
   872→    name: exercise.name,
   873→    categoryLabel: exercise.category?.name ?? "Exercise",
   874→    categoryColour: exercise.category?.colour ?? "#71717a",
   875→    modality: exercise.type,
   876→    unit: exerciseUnit(exercise, settings),
   877→    weightIncrement: exercise.weightIncrement ?? settings?.defaultWeightIncrement ?? 2.5,
   878→  };
   879→
   880→  /** SetDTO → CardSet (distance km → CardSet metres, as in workout-block). */
   881→  const toCardSets = (sets: SetDTO[]): CardSet[] =>
   882→    [...sets]
   883→      .sort((a, b) => a.sortOrder - b.sortOrder)
   884→      .map((s, i) =>
   885→        toCardSet({ ...s, distance: s.distance != null ? s.distance * 1000 : null }, i + 1),
   886→      );
   887→
   888→  const toggleCollapsed = (weId: string) =>
   889→    setCollapsedIds((prev) => {
   890→      const next = new Set(prev);
   891→      if (next.has(weId)) next.delete(weId);
   892→      else next.add(weId);
   893→      return next;
   894→    });
   895→
   896→  return (
   897→    <div className="flex flex-col gap-3">
   898→      {groups.map(([day, entries]) => {
   899→        const setCount = entries.reduce((a, e) => a + e.sets.length, 0);
   900→        const isPr = prDates.has(day);
   901→        return (
   902→          <section key={day} className="flex flex-col gap-2" aria-label={formatDayLabel(day)}>
   903→            <h2 className="flex h-8 items-center gap-2 overflow-hidden px-1 text-xs font-bold uppercase tracking-wider text-muted-foreground">
   904→              <span className="flex-none truncate">{formatDayLabel(day)}</span>
   905→              {isPr ? (
   906→                <Trophy className="h-3.5 w-3.5 flex-none text-primary" aria-label="Personal record day" />
   907→              ) : null}
   908→              <span className="truncate text-[10px] font-medium normal-case tracking-normal text-muted-foreground/70">
   909→                {setCount} {setCount === 1 ? "set" : "sets"}
   910→              </span>
   911→              <span className="h-px min-w-0 flex-1 bg-border/60" aria-hidden />
   912→            </h2>
   913→
   914→            {entries.map((entry) => {
   915→              const collapsed = collapsedIds.has(entry.workoutExerciseId);
   916→              const notesOpen = notesEntryId === entry.workoutExerciseId;
   917→              return (
   918→                <div key={entry.workoutExerciseId} className="flex flex-col">
   919→                  <div
   920→                    role="button"
   921→                    tabIndex={0}
   922→                    aria-label={`${collapsed ? "Expand" : "Collapse"} sets from ${formatDayLabel(day)}`}
   923→                    className="flex-none cursor-pointer"
   924→                    onClick={(e) => {
   925→                      // taps on the card's own controls (buttons/inputs) never toggle
   926→                      if ((e.target as HTMLElement).closest("button, input, textarea, a")) return;
   927→                      toggleCollapsed(entry.workoutExerciseId);
   928→                    }}
   929→                    onKeyDown={(e) => {
   930→                      if (e.target !== e.currentTarget) return;
   931→                      if (e.key === "Enter" || e.key === " ") {
   932→                        e.preventDefault();
   933→                        toggleCollapsed(entry.workoutExerciseId);
   934→                      }
   935→                    }}
   936→                  >
   937→                    <ExerciseCard
   938→                      mode="read"
   939→                      collapsed={collapsed}
   940→                      exercise={cardEx}
   941→                      sets={toCardSets(entry.sets)}
   942→                      visibleColumns={visibleColumns}
   943→                      onAction={(action: CardAction) => {
   944→                        switch (action.type) {
   945→                          case "toggle-collapse":
   946→                            toggleCollapsed(entry.workoutExerciseId);
   947→                            break;
   948→                          case "open":
   949→                            navigate(`/today/${entry.workoutExerciseId}?date=${day}`);
   950→                            break;
   951→                          case "notes":
   952→                            setNotesEntryId(entry.workoutExerciseId);
   953→                            break;
   954→                          default:
   955→                            break;
   956→                        }
   957→                      }}
   958→                    />
   959→                  </div>
   960→                  {notesOpen ? (
   961→                    <div className="relative flex-none">
   962→                      <ExerciseNotesPopover
   963→                        exercise={exercise}
   964→                        open
   965→                        onClose={() => setNotesEntryId(null)}
   966→                      />
   967→                    </div>
   968→                  ) : null}
   969→                </div>
   970→              );
   971→            })}
   972→          </section>
   973→        );
   974→      })}
   975→      {(historyQuery.data?.length ?? 0) >= 100 ? (
   976→        <p className="px-1 text-xs text-muted-foreground">Showing the most recent 100 sessions.</p>
   977→      ) : null}
   978→    </div>
   979→  );
   980→}
