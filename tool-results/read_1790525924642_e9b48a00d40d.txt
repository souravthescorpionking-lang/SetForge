   105→function DaySection({ open, header, children }: { open: boolean; header: ReactNode; children: ReactNode }) {
   106→  return (

   108→      {header}
   109→      <div
   110→        className="grid transition-[grid-template-rows] duration-200 ease-out"
   111→        style={{ gridTemplateRows: open ? "1fr" : "0fr" }}
   112→      >
   113→        <div className="min-h-0 overflow-hidden">
   114→          {open ? children : null}
   115→        </div>
   116→      </div>
   117→    </section>
   118→  );
   119→}

   125→function ExerciseNotesPopover({
   126→  exerciseName,
   127→  notes,
   128→  open,
   129→  onClose,
   130→}: {
   131→  exerciseName: string;
   132→  notes: string | null;
   133→  open: boolean;
   134→  onClose: () => void;
   135→}) {
   136→  return (
   137→    <Popover open={open} onOpenChange={(o) => !o && onClose()}>
   138→      <PopoverTrigger asChild>
   139→        <span className="absolute bottom-2 left-2 h-1 w-1" aria-hidden />
   140→      </PopoverTrigger>
   141→      <PopoverContent
   142→        align="start"
   143→        className="w-72"
   144→        onOpenAutoFocus={(e) => e.preventDefault()}
   145→        // The ⋮ menu closes by returning focus to its trigger; without this
   146→        // the popover would be instantly dismissed (same Radix gotcha as the
   147→        // Today card popovers). Pointer-outside + Escape still close.
   148→        onFocusOutside={(e) => e.preventDefault()}
   149→      >
   150→        <p className="pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">
   151→          {exerciseName} — notes
   152→        </p>
   153→        <p className="text-sm leading-relaxed text-foreground">
   154→          {notes && notes.trim().length > 0 ? notes : "No notes on this exercise."}
   155→        </p>
   156→      </PopoverContent>
   157→    </Popover>
   158→  );
   159→}

   494→  // ---------- card action dispatch ----------
   495→  const handleCardAction = (day: RoutineDayDTO, re: RoutineExerciseDTO) => (action: CardAction): void => {
   496→    switch (action.type) {
   497→      case "toggle-collapse":
   498→        setCollapsedReIds((prev) => {
   499→          const next = new Set(prev);
   500→          if (next.has(re.id)) next.delete(re.id);
   501→          else next.add(re.id);
   502→          return next;
   503→        });
   504→        break;
   505→      case "add-set":
   506→        void addPredefinedSet(day, re);
   507→        break;
   508→      case "update-set":
   509→        void patchPredefinedSet(day, re, action.setId, action.patch);
   510→        break;
   511→      case "remove-set":
   512→        void removePredefinedSet(day, re, action.setId);
   513→        break;
   514→      case "copy-last":
   515→        void copyLastToSet(day, re, action.setId);
   516→        break;
   517→      case "move-up":
   518→      case "move-down": {
   519→        const ids = [...day.exercises].sort((a, b) => a.sortOrder - b.sortOrder).map((e) => e.id);
   520→        void reorderExercises(day.id, ids, re.id, action.type === "move-up" ? -1 : 1);
   521→        break;
   522→      }
   523→      case "remove":
   524→        void removeRoutineExercise(day, re);
   525→        break;
   526→      case "notes":
   527→        setNotesReId(re.id);
   528→        break;
   529→      case "add-to-group":
   530→        toast.info("Routine groups arrive with a later build");
