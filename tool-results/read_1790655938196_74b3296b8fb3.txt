    99→/** Map a CardSet patch (SetRow/ExerciseCard contract) onto the API's SetInput. */
   100→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   101→  const out: SetInput = {};
   102→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   103→  if ("reps" in patch) out.reps = patch.reps ?? null;
   104→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   105→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   106→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   107→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   108→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   109→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   110→  if ("note" in patch) out.comment = patch.note ?? null;
   111→  if ("done" in patch) out.isComplete = !!patch.done;
   112→  return out;
   113→}

   115→/** §4.11 showMaxWeightBar — "Best 110×5" from the exercise's record table
   116→ *  (max-weight entry; ties break to the higher rep count). */
   117→function bestSetLine(records: RecordsDTO | undefined): string | null {
   118→  if (!records) return null;
   119→  const usable = records.actual.filter((r) => r.weight != null && r.weight > 0 && r.reps != null);
   120→  const rows = usable.some((r) => !r.superseded) ? usable.filter((r) => !r.superseded) : usable;
   121→  if (rows.length === 0) return null;
   122→  let best = rows[0];
   123→  for (const r of rows) {
   124→    if (r.weight! > best.weight! || (r.weight === best.weight && r.reps! > best.reps!)) best = r;
   125→  }
   126→  return `Best ${trimNum(best.weight!)}×${best.reps!}`;
   127→}

   491→   *  skip-effect advances it once the cache catches up). */
   492→  const logGuidedSet = async () => {
   493→    const p = pointerOrder[guidedIdx];
   494→    if (!p) return;
   495→    const we = exercises.find((w) => w.id === p.exerciseId);
   496→    const set = sortedSetsByWe.get(p.exerciseId)?.[p.setIndex];
   497→    if (!we || !set || set.isComplete) return;
   498→    const logged = await toggleSetDone(we, set);
   499→    if (logged) advanceGuidedPast(guidedIdx + 1);
   500→  };
   501→
   502→  // card element registry — the guided pointer scrolls the current card into view
   503→  const cardRefs = useRef<Map<string, HTMLDivElement>>(new Map());
   504→  useEffect(() => {
   505→    if (!guidedActive) return;
   506→    const p = pointerOrder[guidedIdx];
   507→    if (!p) return;
   508→    const el = cardRefs.current.get(p.exerciseId);
   509→    if (!el) return;
   510→    const reduced =
   511→      typeof window.matchMedia === "function" &&
   512→      window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   513→    el.scrollIntoView({ block: "nearest", behavior: reduced ? "auto" : "smooth" });
   514→  }, [guidedActive, guidedIdx, pointerOrder]);
   515→
   516→  // ---------- §4.11 derived card data ----------
   517→
   518→  /** §4.10 group codes for today's cards (A1/A2… on superset members). */
   519→  const groupCodes = useMemo(() => computeGroupCodes(exercises), [exercises]);
   520→
   521→  const guidedCardWeId = guidedIdx < pointerOrder.length ? pointerOrder[guidedIdx].exerciseId : null;
   522→  const guidedAllDone = guidedActive && guidedIdx >= pointerOrder.length;
   523→  const guidedLabel = (() => {
   524→    if (!guidedActive) return null;
   525→    if (guidedAllDone) return "All sets done";
   526→    const p = pointerOrder[guidedIdx];
   527→    const code = groupCodes.codes.get(p.exerciseId);
   528→    const total = sortedSetsByWe.get(p.exerciseId)?.length ?? 0;
   529→    return code ? `${code} · set ${p.setIndex + 1}/${total}` : `set ${p.setIndex + 1}/${total}`;
   530→  })();
   531→
   532→  /** §4.11 sets progress — non-warmup performed sets / total sets. */
   533→  const setsProgress = useMemo(() => {
   534→    if (!workout) return null;
   535→    let total = 0;
   536→    let completed = 0;
   537→    for (const we of workout.exercises) {
   538→      for (const s of we.sets) {
   539→        total += 1;
   540→        if (s.isComplete && !s.isWarmup && (s.setType ?? "NORMAL") !== "WARMUP") completed += 1;
   541→      }
   542→    }
   543→    return total > 0 ? { completed, total } : null;
   544→  }, [workout]);
   545→
   546→  /** §4.11 showMaxWeightBar — "Best W×R" per exercise from its records. */
   547→  const showBestLine = !!settings?.showMaxWeightBar;
   548→  const recordsQueries = useQueries({
   549→    queries: exercises.map((we) => ({
   550→      queryKey: qk.exerciseRecords(we.exerciseId),
   551→      queryFn: () => exercisesApi.records(we.exerciseId),
   552→      enabled: showBestLine,
   553→      staleTime: 60_000,
   554→    })),
   555→  });
   556→  const bestByWeId = useMemo(() => {
   557→    const m = new Map<string, string>();
   558→    if (!showBestLine) return m;
   559→    exercises.forEach((we, i) => {
   560→      const line = bestSetLine(recordsQueries[i]?.data);
   561→      if (line) m.set(we.id, line);
   562→    });
   563→    return m;
   564→  }, [showBestLine, exercises, recordsQueries]);
   565→
   566→  /** Any card has a trainer tip → the MetaRow "Tip" chip renders. */
   567→  const anyTips = exercises.some((we) => !!we.exercise.trainerTip);
   568→
   569→  // ---------- Part 5: finish / undo-finish ----------
   570→  const finishWorkout = async () => {
   571→    if (!workout || !online) {
   572→      if (!online) toast.info("Finishing needs a connection");
   573→      return;
   574→    }
   575→    try {
   576→      const res = await workoutsApi.finish(workout.id);
   577→      invalidate.workout();
   578→      invalidate.dashboard();
   579→      invalidate.schedule();
   580→      invalidate.programs();
   581→      // "Day X up next" — index comes from the routine days (fallback: name)
   582→      let nextLabel = "Done";
   583→      if (res.nextDay) {
   584→        const routine = routinesQuery.data?.routines.find(
   585→          (r) => r.id === workout.sourceRoutineId,
   586→        );
   587→        const idx = routine?.days.findIndex((d) => d.id === res.nextDay?.id) ?? -1;
   588→        nextLabel = idx >= 0 ? `Day ${idx + 1} up next` : `${res.nextDay.name} up next`;
   589→      }
   590→      toast.success(`Workout finished · ${nextLabel}`, {
   591→        duration: 10_000,
   592→        action: {
   593→          label: "Undo",
   594→          onClick: () => void undoFinish(workout.id),
   595→        },
   596→      });
   597→    } catch (e) {
   598→      toast.error(e instanceof Error ? e.message : "Could not finish workout");
   599→    }
   600→  };
   601→
   602→  const undoFinish = async (workoutId: string) => {
   603→    try {
   604→      await workoutsApi.undoFinish(workoutId);
   605→      invalidate.workout();
   606→      invalidate.dashboard();
   607→      invalidate.schedule();
   608→      invalidate.programs();
   609→      toast.success("Finish undone — keep logging");
   610→    } catch (e) {
   611→      if (e instanceof ApiError && e.status === 404) {
   612→        toast.info("Too late to undo (past the 10s window)");
   613→      } else {
   614→        toast.error(e instanceof Error ? e.message : "Could not undo");
   615→      }
   616→    }
   617→  };
   618→
   619→  // §4.11 finishBehaviour — ALWAYS_SAVE (default) finishes directly; ASK opens
   620→  // the Save/Discard dialogs (finish-ask.tsx).
   621→  const handleFinishPressed = () => {
   622→    if (settings?.finishBehaviour === "ASK") setFinishAskOpen(true);
   623→    else void finishWorkout();
   624→  };
   625→
   626→  // ---------- card mapping ----------
   627→  const visibleColumns: CardVisibleColumns = {
   628→    setType: settings?.showSetType ?? true,
   629→    rpe: settings?.showRpe ?? true,
   630→    tempo: settings?.showTempo ?? true,
   631→    rest: settings?.showRest ?? true,
   632→  };
   633→
   634→  const toCardExercise = (we: WorkoutExerciseDTO): CardExercise => ({
   635→    id: we.id,
   636→    name: we.exercise.name,
   637→    categoryLabel: we.exercise.category?.name ?? "Exercise",
   638→    categoryColour: we.exercise.category?.colour ?? "#71717a",
   639→    modality: we.exercise.type,
   640→    unit: exerciseUnit(we.exercise, settings),
   641→    weightIncrement: we.exercise.weightIncrement ?? settings?.defaultWeightIncrement ?? 2.5,
   642→  });
   643→
   644→  const cardSets = (we: WorkoutExerciseDTO): CardSet[] =>
   645→    [...we.sets].sort((a, b) => a.sortOrder - b.sortOrder).map((s, i) => toCardSet(s, i + 1));
   646→
   647→  const handleCardAction =
   648→    (we: WorkoutExerciseDTO) =>
   649→    (action: CardAction): void => {
   650→      switch (action.type) {
   651→        case "toggle-collapse":
   652→          setCollapsedIds((prev) => {
   653→            const next = new Set(prev);
   654→            if (next.has(we.id)) next.delete(we.id);
   655→            else next.add(we.id);
   656→            return next;
   657→          });
   658→          break;
   659→        case "add-set":
   660→          void addSetToCard(we);
   661→          break;
   662→        case "update-set":
   663→          void patchSet(we, action.setId, cardPatchToSetInput(action.patch));
   664→          break;
   665→        case "toggle-done": {
   666→          const set = we.sets.find((s) => s.id === action.setId);
   667→          if (set) void toggleSetDone(we, set);
   668→          break;
   669→        }
   670→        case "copy-last":
   671→          void copyLast(we, action.setId);
   672→          break;
   673→        case "notes":
   674→          setNotesWeId(we.id);
   675→          break;
   676→        case "rest-timer":
   677→          rest.start(we.exercise.restSec ?? undefined, action.setId ?? null);
   678→          break;
   679→        case "move-up":
   680→          void moveExercise(we, -1);
   681→          break;
   682→        case "move-down":
   683→          void moveExercise(we, 1);
   684→          break;
   685→        case "add-to-group":
   686→          setGroupWeId(we.id);
   687→          break;
   688→        case "replace":
   689→          // p3-4: the picker opens in replace mode with the day context
   690→          navigate(`/exercises?replace=${we.id}&date=${dateKey}`);
   691→          break;
   692→        case "remove":
   693→          setRemoveWeId(we.id);
   694→          break;
   695→        case "select":
   696→          // multi-select omitted in the p3-3 Today rebuild (see report)
   697→          toast.info("Multi-select arrives with the next build");
   698→          break;
   699→        case "open":
   700→          // p3-4 "Focus view" — carry the day context into the training screen
   701→          navigate(`/today/${we.id}?date=${dateKey}`);
   702→          break;
   703→        case "toggle-select":
   704→          break; // preview-mode only; edit rows never emit it
   705→      }
   706→    };
   707→
   708→  const restActive = rest.remainingSec != null;
   709→  // §4.11 RING display — in-flow block while resting (RestBar stays too)
   710→  const ringRestActive = restActive && settings?.restDisplay === "RING";
   711→
   712→  // ---------- render ----------
   713→  return (
   714→    <Screen
   715→      topBar={
   716→        <TopBar
   717→          title="SetForge"
   718→          actions={
   719→            <>
   720→              <Button
   721→                type="button"
   722→                variant="ghost"
   723→                size="icon"
   724→                className="h-11 w-11 flex-none"
   725→                aria-label="Calendar"
   726→                onClick={() => navigate("/calendar")}
   727→              >
   728→                <CalendarDays className="h-5 w-5" aria-hidden />
   729→              </Button>
   730→              <DropdownMenu>
   731→                <DropdownMenuTrigger asChild>
   732→                  <Button
   733→                    type="button"
   734→                    variant="ghost"
   735→                    size="icon"
   736→                    className="h-11 w-11 flex-none"
   737→                    aria-label="More actions"
   738→                  >
   739→                    <MoreVertical className="h-5 w-5" aria-hidden />
   740→                  </Button>
   741→                </DropdownMenuTrigger>
   742→                <DropdownMenuContent align="end" className="w-52">
   743→                  {workout && workout.exercises.length > 0 ? (
   744→                    <>
   745→                      <DropdownMenuItem
   746→                        onClick={() => sessionFromWorkout.openFor(workout.id, dateKey)}
   747→                      >
   748→                        <Dumbbell className="h-4 w-4" aria-hidden /> Save as session
   749→                      </DropdownMenuItem>
   750→                      {/* §4.10c — today's arrange screen (6-d) */}
   751→                      <DropdownMenuItem
   752→                        onClick={() =>
   753→                          navigate(
   754→                            dateKey === todayKey() ? "/today/arrange" : `/today/arrange?date=${dateKey}`,
   755→                          )
   756→                        }
   757→                      >
   758→                        <ArrowDownUp className="h-4 w-4" aria-hidden /> Arrange exercises
   759→                      </DropdownMenuItem>
   760→                    </>
   761→                  ) : null}
   762→                  <DropdownMenuItem onClick={() => navigate("/history")}>
   763→                    <History className="h-4 w-4" aria-hidden /> History
   764→                  </DropdownMenuItem>
   765→                  <DropdownMenuItem onClick={() => navigate("/exercises")}>
   766→                    <Dumbbell className="h-4 w-4" aria-hidden /> Exercises
   767→                  </DropdownMenuItem>
   768→                  <DropdownMenuItem onClick={() => navigate("/tools")}>
   769→                    <Wrench className="h-4 w-4" aria-hidden /> Tools
   770→                  </DropdownMenuItem>
   771→                  <DropdownMenuItem onClick={() => navigate("/settings")}>
   772→                    <Settings className="h-4 w-4" aria-hidden /> Settings
   773→                  </DropdownMenuItem>
   774→                </DropdownMenuContent>
   775→              </DropdownMenu>
   776→            </>
   777→          }
   778→        />
   779→      }
   780→      subBar={
   781→        <DateStrip
   782→          dateKey={dateKey}
   783→          onChange={goTo}
   784→          workoutExists={!!workout}
   785→          progress={settings?.showSetsProgressBar !== false ? setsProgress : null}
   786→        />
   787→      }
   788→      bottomBar={
   789→        <BottomBar>
   790→          {restActive ? (
   791→            <RestBar
   792→              display={formatDuration(rest.remainingSec ?? 0)}
   793→              onAdjust={rest.adjust}
   794→              onSkip={rest.skip}
   795→            />
   796→          ) : workout?.finishedAt ? (
   797→            // finished — single disabled confirmation row
   798→            <FinishedBar />
   799→          ) : guidedActive ? (
   800→            // §4.11 guided mode — Prev / Log set / Next replaces Add/Finish
   801→            <GuidedBar
   802→              onPrev={() => moveGuided(-1)}
   803→              onNext={() => moveGuided(1)}
   804→              onLog={() => void logGuidedSet()}
   805→              onFinish={handleFinishPressed}
   806→              canLog={guidedIdx < pointerOrder.length && !isPointerComplete(guidedIdx)}
   807→              allDone={guidedAllDone}
   808→              atStart={guidedIdx <= 0}
   809→            />
   810→          ) : workout && workout.exercises.length > 0 ? (
   811→            // logging in progress — two equal actions
   812→            <FinishBar
   813→              onAdd={() => navigate(`/exercises?date=${dateKey}`)}
   814→              onFinish={handleFinishPressed}
   815→            />
   816→          ) : (
   817→            <AddExerciseBar onAdd={() => navigate(`/exercises?date=${dateKey}`)} />
   818→          )}
   819→        </BottomBar>
   820→      }
   821→    >
   822→      <ScrollBody>
   823→        {isLoading ? (
   824→          <DaySkeleton />
   825→        ) : !workout ? (
   826→          <TodayEmpty
   827→            hasPrevious={hasPrevious}
   828→            starting={creating}
   829→            onStartNew={() => void startNewWorkout()}
   830→            onCopyPrevious={() => void copyPreviousWorkout()}
   831→          />
   832→        ) : (
   833→          <>
   834→            {/* §4.11 RING rest display — FIRST child, pushes content down,
   835→                removed when rest ends (no overlay; RestBar stays in the BottomBar) */}
   836→            {ringRestActive ? (
   837→              <RestRingBlock
   838→                remainingSec={rest.remainingSec ?? 0}
   839→                totalSec={rest.totalSec}
   840→                onAdjust={rest.adjust}
   841→                onSkip={rest.skip}
   842→              />
   843→            ) : null}
   844→
   845→            <MetaRow
   846→              workout={workout}
   847→              source={source}
   848→              restRemainingSec={rest.remainingSec}
   849→              onToggleTimer={() => void toggleTimer()}
   850→              setsProgress={settings?.showSetsProgressBar !== false ? setsProgress : null}
   851→              guidedPointer={guidedLabel}
   852→              tipToggle={
   853→                anyTips ? { open: tipsOpen, onToggle: () => setTipsOpen((v) => !v) } : null
   854→              }
   855→            />
   856→
   857→            {exercises.map((we) => {
   858→              const group = we.groupId ? groupById.get(we.groupId) ?? null : null;
   859→              const isCurrentGuidedCard = guidedCardWeId === we.id;
   860→              // §4.11 guided video panel — only the CURRENT guided card, only
   861→              // when a video exists (MediaBlock itself collapses to 0px otherwise)
   862→              const showGuidedVideo =
   863→                isCurrentGuidedCard && !!settings?.showVideoPanel && !!we.exercise.videoUrl;
   864→              const showTipRow = tipsOpen && !!we.exercise.trainerTip;
   865→              const underHeader =
   866→                showGuidedVideo || showTipRow ? (
   867→                  <>
   868→                    {showGuidedVideo ? (
   869→                      <GuidedVideoPanel weId={we.id} videoUrl={we.exercise.videoUrl} height={180} />
   870→                    ) : null}
   871→                    {showTipRow ? <TrainerTipRow tip={we.exercise.trainerTip!} /> : null}
   872→                  </>
   873→                ) : undefined;
   874→              return (
   875→                <div
   876→                  key={we.id}
   877→                  ref={(el) => {
   878→                    if (el) cardRefs.current.set(we.id, el);
   879→                    else cardRefs.current.delete(we.id);
   880→                  }}
   881→                  className="relative flex-none"
   882→                >
   883→                  <ExerciseCard
   884→                    mode="edit"
   885→                    exercise={toCardExercise(we)}
   886→                    sets={cardSets(we)}
   887→                    collapsed={collapsedIds.has(we.id)}
   888→                    visibleColumns={visibleColumns}
   889→                    groupColour={group?.colour}
   890→                    groupName={group?.name}
   891→                    groupCode={groupCodes.codes.get(we.id)}
   892→                    metaExtra={bestByWeId.get(we.id)}
   893→                    underHeader={underHeader}
   894→                    className={isCurrentGuidedCard ? "ring-2 ring-primary/40" : undefined}
   895→                    onAction={handleCardAction(we)}
   896→                  />
   897→                  {notesWeId === we.id && (
   898→                    <ExerciseNotesPopover
   899→                      exercise={we.exercise}
   900→                      open
   901→                      onClose={() => setNotesWeId(null)}
   902→                    />
   903→                  )}
   904→                  {groupWeId === we.id && (
   905→                    <ExerciseGroupPopover
   906→                      workout={workout}
   907→                      we={we}
   908→                      open
   909→                      onClose={() => setGroupWeId(null)}
   910→                    />
   911→                  )}
   912→                </div>
   913→              );
   914→            })}
   915→
   916→            <SummaryRow workout={workout} unit={defaultUnitFor(settings)} />
   917→
   918→            {/* bottom breathing spacer — bars are flex siblings, nothing to clear */}
   919→            <div className="h-4 flex-none" aria-hidden />
   920→
   921→            <ConfirmRemoveExercise
   922→              exerciseName={removeWe?.exercise.name ?? ""}
   923→              open={removeWeId != null}
   924→              onClose={() => setRemoveWeId(null)}
   925→              onConfirm={async () => {
   926→                const id = removeWeId;
   927→                setRemoveWeId(null);
   928→                if (id) await removeExercise(id);
   929→              }}
   930→            />
   931→
   932→            {/* §4.11 finishBehaviour=ASK — Save/Discard dialogs (mounted only
   933→                while the ASK stage is open; ALWAYS_SAVE never reaches it) */}
   934→            {settings?.finishBehaviour === "ASK" ? (
   935→              <FinishAskFlow
   936→                workout={workout}
   937→                open={finishAskOpen}
   938→                onClose={() => setFinishAskOpen(false)}
   939→                onSave={() => void finishWorkout()}
   940→                dateKey={dateKey}
   941→                onDiscarded={() => {
   942→                  // a discarded day can't keep a rest countdown running
   943→                  rest.skip();
   944→                  setFinishAskOpen(false);
   945→                }}
   946→              />
   947→            ) : null}
   948→
   949→            {sessionFromWorkout.dialog}
   950→          </>
   951→        )}
   952→      </ScrollBody>
   953→    </Screen>
   954→  );
   955→}
   956→
   957→function DaySkeleton() {
   958→  return (
   959→    <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading day">
   960→      <div className="h-12 animate-pulse rounded-lg bg-muted/60" />
   961→      {[0, 1, 2].map((i) => (
   962→        <div key={i} className="h-14 animate-pulse rounded-lg bg-muted/40" />
   963→      ))}
   964→    </div>
   965→  );
   966→}
