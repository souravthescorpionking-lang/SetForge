   121→/** Map a CardSet patch (SetRow/ExerciseCard contract) onto the API's SetInput. */
   122→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   123→  const out: SetInput = {};
   124→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   125→  if ("reps" in patch) out.reps = patch.reps ?? null;
   126→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   127→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   128→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   129→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   130→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   131→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   132→  if ("note" in patch) out.comment = patch.note ?? null;
   133→  if ("done" in patch) out.isComplete = !!patch.done;
   134→  return out;
   135→}

   210→    staleTime: 60_000,
   211→  });
   212→
   213→  // full history (History tab)
   214→  const history = useQuery({
   215→    queryKey: qk.exerciseHistory(exercise?.id ?? ""),

   218→    staleTime: 60_000,

   221→  // ---------- ui state ----------
   222→  const [notesOpen, setNotesOpen] = useState(false);

   224→  const [groupOpen, setGroupOpen] = useState(false);
   225→  const [restOverrideOpen, setRestOverrideOpen] = useState(false);

   229→  // notes popover draft: edited override → stored notes (no effect needed)

   232→  // ---------- "Save set" focus detection (mobile keyboard bar) ----------

   234→    const check = () => {
   235→      const el = document.activeElement;

   237→    };
   238→    let timer = 0;
   239→    const onOut = () => {
   240→      window.clearTimeout(timer);
   241→      timer = window.setTimeout(check, 0);
   242→    };
   243→    document.addEventListener("focusin", check);
   244→    document.addEventListener("focusout", onOut);

   280→        path: `/api/workouts/${ctx.workoutId}/exercises/${ctx.weId}/sets`,
   281→        method: "POST",
   282→        body: payload,
   283→      },
   284→    });

   288→        description: `${created.reps} reps at ${created.weight} — best ever`,

   292→    }

   311→        else if (rest.everStarted) rest.start(undefined, set.id);
   312→      } else if (rest.everStarted) {
   313→        rest.start(undefined, set.id);
   314→      }

   325→    if (!src) {
   326→      toast.info("No earlier set to copy from");
   327→      return;
   328→    }
   329→    await patchSetIn(ctx, setId, {
   330→      weight: src.weight,
   331→      reps: src.reps,
   332→      distance: src.distance,
   333→      timeSec: src.timeSec,
   334→      rpe: src.rpe ?? null,
   335→      tempo: src.tempo ?? null,
   336→      restPlannedSec: src.restPlannedSec ?? null,
   337→    });

   458→              variant="ghost"

   920→// GRAPH — ControlRow → fixed chart box → reserved 72px DetailRow

   922→
   923→type GraphPoint = { date: string; value: number; prev?: number | null; next?: number | null };
   924→type ChartDatum = GraphPoint & {
   925→  label: string;
   926→  prev: number | null;
   927→  next: number | null;
   928→  trend?: number;
   929→};
   930→
   931→function GraphTab({ exercise }: { exercise: ExerciseDTO }) {
   932→  const settings = useApp((s) => s.settings);
   933→
   934→  const allowed = useMemo(
   935→    () =>
   936→      metricsForType(exercise.type)
   937→        .filter((m) => !OMITTED_METRICS.has(m))
   938→        .map((m) => ({ value: m, label: metricLabel(m) })),
   939→    [exercise.type],
   940→  );
   941→
   942→  const defaultMetric = allowed.some((m) => m.value === exercise.defaultGraph)
   943→    ? (exercise.defaultGraph as string)
   944→    : (allowed[0]?.value ?? "MAX_WEIGHT");
   945→
   946→  const [metric, setMetric] = useState<string>(defaultMetric);
   947→  const [range, setRange] = useState<RangeValue>("ALL");
   948→  const [showPoints, setShowPoints] = useState(true);
   949→  const [showTrend, setShowTrend] = useState(true);
   950→  const [fromZero, setFromZero] = useState(false);
   951→  const [selected, setSelected] = useState<GraphPoint | null>(null);
   952→  const onSelectPoint = useCallback((p: GraphPoint | null) => setSelected(p), []);
   953→
   954→  const rangeDays = RANGE_OPTIONS.find((r) => r.value === range)?.days ?? null;
   955→  const from = rangeDays ? addDays(todayKey(), -rangeDays) : undefined;
   956→
   957→  const graph = useQuery({
   958→    queryKey: qk.exerciseGraph(exercise.id, { metric, from: from ?? "" }),
   959→    queryFn: () => exercisesApi.graph(exercise.id, { metric, from }),
   960→    staleTime: 60_000,
   961→  });
   962→
   963→  const points = graph.data?.points ?? [];
   964→  const fmt = metricValueFmt(metric, exerciseUnit(exercise, settings));
   965→
   966→  const data = useMemo<ChartDatum[]>(() => {
   967→    const base: ChartDatum[] = points.map((p, i) => ({
   968→      ...p,
   969→      prev: i > 0 ? points[i - 1].value : null,
   970→      next: i < points.length - 1 ? points[i + 1].value : null,
   971→      label: formatDayShort(p.date),
   972→    }));
   973→    if (showTrend && base.length >= 3) {
   974→      // simple least-squares trend line (same math as the shared TrendChart)
   975→      const n = base.length;
   976→      const xs = base.map((_, i) => i);
   977→      const ys = base.map((p) => p.value);
   978→      const meanX = xs.reduce((a, b) => a + b, 0) / n;
   979→      const meanY = ys.reduce((a, b) => a + b, 0) / n;
   980→      let num = 0;
   981→      let den = 0;
   982→      for (let i = 0; i < n; i++) {
   983→        num += (xs[i] - meanX) * (ys[i] - meanY);
   984→        den += (xs[i] - meanX) ** 2;
   985→      }
   986→      const slope = den === 0 ? 0 : num / den;
   987→      const intercept = meanY - slope * meanX;
   988→      return base.map((p, i) => ({ ...p, trend: Math.round((intercept + slope * i) * 100) / 100 }));
   989→    }
   990→    return base;
   991→  }, [points, showTrend]);
   992→
   993→  return (
   994→    <div className="flex flex-col gap-3">
   995→      {/* ControlRow — 48px data-row: metric | range | ⋮ options */}
   996→      <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   997→        <Select value={metric} onValueChange={setMetric}>
   998→          <SelectTrigger
   999→            className="h-11 min-w-0 flex-1 rounded-lg text-sm font-semibold"
  1000→            aria-label="Graph metric"
  1001→          >
  1002→            <SelectValue />
  1003→          </SelectTrigger>
  1004→          <SelectContent>
  1005→            {allowed.map((m) => (
  1006→              <SelectItem key={m.value} value={m.value}>
  1007→                {m.label}
  1008→              </SelectItem>
  1009→            ))}
  1010→          </SelectContent>
  1011→        </Select>
  1012→        <Select value={range} onValueChange={(v) => setRange(v as RangeValue)}>
  1013→          <SelectTrigger
  1014→            className="h-11 w-[88px] flex-none rounded-lg text-sm font-semibold"
  1015→            aria-label="Graph range"
  1016→          >
  1017→            <SelectValue />
  1018→          </SelectTrigger>
  1019→          <SelectContent>
  1020→            {RANGE_OPTIONS.map((r) => (
  1021→              <SelectItem key={r.value} value={r.value}>
  1022→                {r.label}
  1023→              </SelectItem>
  1024→            ))}
  1025→          </SelectContent>
  1026→        </Select>
  1027→        <DropdownMenu>
  1028→          <DropdownMenuTrigger asChild>
  1029→            <Button
  1030→              type="button"
  1031→              variant="ghost"
  1032→              className="h-11 w-11 flex-none px-0"
  1033→              aria-label="Chart options"
  1034→            >
  1035→              <MoreVertical className="h-5 w-5" aria-hidden />
  1036→            </Button>
  1037→          </DropdownMenuTrigger>
  1038→          <DropdownMenuContent align="end" className="w-44">
  1039→            <DropdownMenuCheckboxItem checked={showPoints} onCheckedChange={(v) => setShowPoints(!!v)}>
  1040→              Show points
  1041→            </DropdownMenuCheckboxItem>
  1042→            <DropdownMenuCheckboxItem checked={showTrend} onCheckedChange={(v) => setShowTrend(!!v)}>
  1043→              Trend line
  1044→            </DropdownMenuCheckboxItem>
  1045→            <DropdownMenuCheckboxItem checked={fromZero} onCheckedChange={(v) => setFromZero(!!v)}>
  1046→              Y axis from zero
  1047→            </DropdownMenuCheckboxItem>
  1048→          </DropdownMenuContent>
  1049→        </DropdownMenu>
  1050→      </div>
  1051→
  1052→      {/* Chart box — fixed height; charts may scale internally (allowed exemption) */}
  1053→      <div className="flex h-[240px] flex-none items-center justify-center overflow-hidden rounded-lg border bg-card p-2 lg:h-[360px]">
  1054→        {graph.isLoading ? (
  1055→          <Skeleton className="h-full w-full rounded-lg" />
  1056→        ) : points.length === 0 ? (
  1057→          <p className="px-4 text-center text-sm text-muted-foreground">No data to chart yet</p>
  1058→        ) : (
  1059→          <ResponsiveContainer width="100%" height="100%">
  1060→            <LineChart data={data} margin={{ top: 8, right: 8, bottom: 0, left: 0 }}>
  1061→              <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
  1062→              <XAxis
  1063→                dataKey="label"
  1064→                tick={{ fontSize: 11, fill: "var(--muted-foreground)" }}
  1065→                tickLine={false}
  1066→                axisLine={{ stroke: "var(--border)" }}
  1067→                interval="preserveStartEnd"
  1068→                minTickGap={40}
  1069→              />
  1070→              <YAxis
  1071→                tick={{ fontSize: 11, fill: "var(--muted-foreground)" }}
  1072→                tickLine={false}
  1073→                axisLine={false}
  1074→                width={48}
  1075→                domain={fromZero ? [0, "auto"] : ["auto", "auto"]}
  1076→                tickFormatter={(v: number) => fmt(v)}
  1077→              />
  1078→              <Tooltip content={<GraphTooltip formatter={fmt} onSelect={onSelectPoint} />} />
  1079→              {showTrend && data[0]?.trend != null ? (
  1080→                <Line
  1081→                  type="linear"
  1082→                  dataKey="trend"
  1083→                  stroke="var(--muted-foreground)"

  1091→              <Line
  1092→                type="monotone"
  1093→                dataKey="value"
  1094→                stroke="var(--primary)"

  1265→function metricValueFmt(metric: string, unit: string): (v: number) => string {

  1269→    case "EST_1RM":

  1289→function addDays(key: string, days: number): string {
