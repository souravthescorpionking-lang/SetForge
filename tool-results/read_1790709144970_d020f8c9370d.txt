   134→function FacetChipRow({
   135→  options,
   136→  selected,
   137→  labelOf,
   138→  onToggle,
   139→  span,
   140→  ariaLabel,
   141→}: {
   142→  options: string[];
   143→  selected: Set<string>;
   144→  labelOf: (v: string) => string;
   145→  onToggle: (v: string) => void;
   146→  span?: boolean;
   147→  ariaLabel: string;
   148→}) {
   149→  if (options.length === 0) return null;
   150→  return (
   151→    <div
   152→      data-row
   153→      data-chip-scroller
   154→      role="group"
   155→      aria-label={ariaLabel}
   156→      className={cn(CHIP_ROW_CLS, span && "")}
   157→    >
   158→      {options.map((v) => {
   159→        const active = selected.has(v);
   160→        return (
   161→          <button
   162→            key={v}
   163→            type="button"
   164→            aria-pressed={active}
   165→            className={chipClass(active)}
   166→            {...tourAttrs({ skipTour: true, reason: "Data-driven facet chips for labels, muscles, equipment" })}
   167→            onClick={() => onToggle(v)}
   168→          >
   169→            {active ? <Check className="h-3.5 w-3.5" aria-hidden /> : null}
   170→            {labelOf(v)}
   171→          </button>
   172→        );
   173→      })}
   174→    </div>
   175→  );
   176→}
