    56→// ── local unit conversion helpers (metric is canonical on the wire) ──────────
    57→
    58→const CM_PER_IN = 2.54;
    59→const KG_PER_LB = 0.45359237;
    60→
    61→const roundTo1 = (n: number): number => Math.round(n * 10) / 10;
    62→
    63→/** 177.8 cm → "5 ft 10 in" (imperial height display). */
    64→function cmToFtIn(cm: number): string {
    65→  const totalIn = cm / CM_PER_IN;
    66→  let ft = Math.floor(totalIn / 12);
    67→  let inch = Math.round(totalIn - ft * 12);
    68→  if (inch >= 12) {
    69→    ft += 1;
    70→    inch -= 12;
    71→  }
    72→  return `${ft} ft ${inch} in`;
    73→}
    74→
    75→const cmToIn = (cm: number): number => cm / CM_PER_IN;
    76→const inToCm = (inches: number): number => inches * CM_PER_IN;
    77→const kgToLb = (kg: number): number => kg / KG_PER_LB;
    78→const lbToKg = (lb: number): number => lb * KG_PER_LB;
    79→const identity = (n: number): number => n;

   115→  const qc = useQueryClient();

   121→    queryFn: () => profileApi.get(),

   131→    try {
   132→      const dto = await profileApi.update(patch);
   133→      qc.setQueryData(["profile"], dto);
   134→      void qc.invalidateQueries({ queryKey: ["profile"] });
   135→      toast.success("Saved");
   136→      setEditing(null);
   137→    } catch (e) {
   138→      toast.error(e instanceof Error ? e.message : "Could not save");

   140→  };

   142→  if (!session || !settings) {
   143→    return (

   149→            <Skeleton className="h-3 w-56 rounded-lg" />
   150→          </div>
   151→          <Skeleton className="h-8 w-24 rounded-lg" />
   152→          {Array.from({ length: 6 }, (_, i) => (
   153→            <Skeleton key={i} className="h-14 w-full rounded-lg" />
   154→          ))}
   155→        </ScrollBody>

   158→  }

   205→function ProfileSection({

   207→  loading,
   208→  imperial,
   209→  editing,
   210→  setEditing,
   211→  savePatch,
   212→}: {
   213→  profile: UserProfileDTO | undefined;
   214→  loading: boolean;
   215→  imperial: boolean;
   216→  editing: EditKey | null;
   217→  setEditing: (k: EditKey | null) => void;

   227→      </section>
   228→    );
   229→  }
   230→
   231→  const level = asLevel(profile.level);
   232→  const goal = asGoal(profile.goal);
   233→  const toggle = (key: EditKey) => setEditing(editing === key ? null : key);
   234→
   235→  return (
   236→    <section aria-label="Profile" className="flex flex-col gap-2">
   237→      <SectionHeader title="Profile" />
   238→
   239→      <ProfileRow
   240→        label="Age"
   241→        value={profile.age != null ? `${profile.age} years` : null}
   242→        expanded={editing === "age"}
   243→        onToggle={() => toggle("age")}
   244→      />
   245→      {editing === "age" ? (
   246→        <StepperEditor
   247→          key={`age-${profile.age ?? "new"}`}
   248→          title="Age"
   249→          unitLabel="years"
   250→          min={AGE_MIN}
   251→          max={AGE_MAX}
   252→          initial={profile.age ?? 30}
   253→          formatValue={(n) => `${n}`}
   254→          ariaLabel="Age"
   255→          stepLabel="Age"
   256→          onSave={(n) => savePatch({ age: n })}
   257→          onCancel={() => setEditing(null)}
   258→        />
   259→      ) : null}
   260→
   261→      <ProfileRow
   262→        label="Height"
   263→        value={
   264→          profile.heightCm == null ? null
   265→          : imperial ? `${cmToFtIn(profile.heightCm)} · ${roundTo1(profile.heightCm)} cm`
   266→          : `${roundTo1(profile.heightCm)} cm`
   267→        }
   268→        expanded={editing === "height"}
   269→        onToggle={() => toggle("height")}
   270→      />
   271→      {editing === "height" ? (
   272→        <MeasureEditor
   273→          key={`height-${profile.heightCm ?? "new"}`}
   274→          title="Height"
   275→          unitLabel={imperial ? "inches" : "cm"}
   276→          placeholder={imperial ? "e.g. 70" : "e.g. 178"}
   277→          initialMetric={profile.heightCm}
   278→          imperial={imperial}
   279→          toDisplay={imperial ? cmToIn : identity}
   280→          toMetric={imperial ? inToCm : identity}
   281→          min={imperial ? HEIGHT_IN_MIN : HEIGHT_CM_MIN}
   282→          max={imperial ? HEIGHT_IN_MAX : HEIGHT_CM_MAX}
   283→          canonicalUnit="cm"
   284→          onSave={(cm) => savePatch({ heightCm: cm })}
   285→          onCancel={() => setEditing(null)}
   286→        />
   287→      ) : null}
   288→
   289→      <ProfileRow
   290→        label="Weight"
   291→        value={
   292→          profile.weightKg == null ? null
   293→          : imperial ? `${Math.round(kgToLb(profile.weightKg))} lb · ${roundTo1(profile.weightKg)} kg`
   294→          : `${roundTo1(profile.weightKg)} kg`
   295→        }
   296→        expanded={editing === "weight"}
   297→        onToggle={() => toggle("weight")}
   298→      />
   299→      {editing === "weight" ? (
   300→        <MeasureEditor
   301→          key={`weight-${profile.weightKg ?? "new"}`}
   302→          title="Weight"
   303→          unitLabel={imperial ? "pounds" : "kg"}
   304→          placeholder={imperial ? "e.g. 180" : "e.g. 80"}
   305→          initialMetric={profile.weightKg}
   306→          imperial={imperial}
   307→          toDisplay={imperial ? kgToLb : identity}
   308→          toMetric={imperial ? lbToKg : identity}
   309→          min={imperial ? WEIGHT_LB_MIN : WEIGHT_KG_MIN}
   310→          max={imperial ? WEIGHT_LB_MAX : WEIGHT_KG_MAX}
   311→          canonicalUnit="kg"
   312→          onSave={(kg) => savePatch({ weightKg: kg })}
   313→          onCancel={() => setEditing(null)}
   314→        />
   315→      ) : null}
   316→
   317→      <ProfileRow
   318→        label="Level"
   319→        value={level ? LEVEL_LABELS[level] : null}
   320→        expanded={editing === "level"}
   321→        onToggle={() => toggle("level")}
   322→      />
   323→      {editing === "level" ? (
   324→        <ChoiceEditor
   325→          key={`level-${profile.level ?? "new"}`}
   326→          title="Experience level"
   327→          options={PROFILE_LEVELS.map((l) => ({ value: l, label: LEVEL_LABELS[l] }))}
   328→          selected={level}
   329→          onSelect={(v) => {
   330→            if (level !== v) void savePatch({ level: v });
   331→          }}
   332→          onCancel={() => setEditing(null)}
   333→        />
   334→      ) : null}
   335→
   336→      <ProfileRow
   337→        label="Goal"
   338→        value={goal ? PROFILE_GOAL_LABELS[goal] : null}
   339→        expanded={editing === "goal"}
   340→        onToggle={() => toggle("goal")}
   341→      />
   342→      {editing === "goal" ? (
   343→        <ChoiceEditor
   344→          key={`goal-${profile.goal ?? "new"}`}
   345→          title="Main goal"
   346→          options={PROFILE_GOALS.map((g) => ({ value: g, label: PROFILE_GOAL_LABELS[g] }))}
   347→          selected={goal}
   348→          onSelect={(v) => {
   349→            if (goal !== v) void savePatch({ goal: v });
   350→          }}
   351→          onCancel={() => setEditing(null)}
   352→        />
   353→      ) : null}
   354→
   355→      <ProfileRow
   356→        label="Days per week"
   357→        value={profile.daysPerWeekTarget != null ? String(profile.daysPerWeekTarget) : null}
   358→        expanded={editing === "days"}
   359→        onToggle={() => toggle("days")}
   360→      />
   361→      {editing === "days" ? (
   362→        <StepperEditor
   363→          key={`days-${profile.daysPerWeekTarget ?? "new"}`}
   364→          title="Training days per week"
   365→          unitLabel="days"
   366→          min={1}
   367→          max={7}
   368→          initial={profile.daysPerWeekTarget ?? 3}
   369→          formatValue={(n) => `${n}`}
   370→          ariaLabel="Training days per week"
   371→          stepLabel="Days"
   372→          onSave={(n) => savePatch({ daysPerWeekTarget: n })}
   373→          onCancel={() => setEditing(null)}
   374→        />
   375→      ) : null}
   376→    </section>
   377→  );
   378→}

   429→/** Sign out + full local wipe (same behaviour as the settings screen). */
   430→function useSignOut() {
   431→  const setSession = useApp((s) => s.setSession);
   432→  const qc = useQueryClient();
   433→  return async () => {
   434→    try {
   435→      await authApi.logout();
   436→    } finally {
   437→      clearSwCaches();
   438→      wipeLocalData();
   439→      qc.clear();
   440→      setSession(null); // app-shell gate forces #/auth
   441→      toast.success("Signed out — local data cleared");
   442→    }
   443→  };
   444→}

   678→/** − n + stepper editor (age, days per week). */
   679→function StepperEditor({
   680→  title,
   681→  unitLabel,
   682→  min,
   683→  max,
   684→  initial,
   685→  formatValue,
   686→  ariaLabel,
   687→  stepLabel,
   688→  onSave,
   689→  onCancel,
   690→}: {
   691→  title: string;
   692→  unitLabel: string;
   693→  min: number;
   694→  max: number;
   695→  initial: number;
   696→  formatValue: (n: number) => string;
   697→  ariaLabel: string;
   698→  stepLabel: string;
   699→  onSave: (n: number) => Promise<void>;
   700→  onCancel: () => void;
   701→}) {
   702→  const [value, setValue] = useState(initial);
   703→  const [saving, setSaving] = useState(false);
   704→
   705→  const save = async () => {
   706→    setSaving(true);
   707→    try {
   708→      await onSave(value);
   709→    } finally {
   710→      setSaving(false);
   711→    }
   712→  };
   713→
   714→  return (
   715→    <EditorShell
   716→      title={`${title} · ${unitLabel}`}
   717→      footer={<SaveCancelButtons onSave={() => void save()} onCancel={onCancel} saving={saving} />}
   718→    >
   719→      <div className="flex items-center justify-between rounded-lg border bg-background px-2 py-1">
   720→        <span className="flex-none pl-1 text-xs text-muted-foreground">{stepLabel}</span>
   721→        <span className="flex items-center gap-1">
   722→          <Button
   723→            type="button"
   724→            variant="outline"
   725→            className="h-12 w-12 p-0"
   726→            disabled={value <= min || saving}
   727→            aria-label={`Decrease ${ariaLabel}`}
   728→            onClick={() => {
   729→              hapticTap();
   730→              setValue((v) => Math.max(min, v - 1));
   731→            }}
   732→          >
   733→            <Minus className="h-5 w-5" aria-hidden />
   734→          </Button>
   735→          <span
   736→            className="flex h-12 w-14 items-center justify-center text-base font-bold tabular-nums leading-none"
   737→            aria-live="polite"
   738→            aria-label={`${ariaLabel}: ${formatValue(value)}`}
   739→          >
   740→            {formatValue(value)}
   741→          </span>
   742→          <Button
   743→            type="button"
   744→            variant="outline"
   745→            className="h-12 w-12 p-0"
   746→            disabled={value >= max || saving}
   747→            aria-label={`Increase ${ariaLabel}`}
   748→            onClick={() => {
   749→              hapticTap();
   750→              setValue((v) => Math.min(max, v + 1));
   751→            }}
   752→          >
   753→            <Plus className="h-5 w-5" aria-hidden />
   754→          </Button>
   755→        </span>
   756→      </div>
   757→    </EditorShell>
   758→  );
   759→}

   761→/** Height/weight editor — number input in the display unit, stores metric. */
   762→function MeasureEditor({
   763→  title,
   764→  unitLabel,
   765→  placeholder,
   766→  initialMetric,
   767→  imperial,
   768→  toDisplay,
   769→  toMetric,
   770→  min,
   771→  max,
   772→  canonicalUnit,
   773→  onSave,
   774→  onCancel,
   775→}: {
   776→  title: string;
   777→  unitLabel: string;
   778→  placeholder: string;
   779→  initialMetric: number | null;
   780→  imperial: boolean;
   781→  toDisplay: (metric: number) => number;
   782→  toMetric: (display: number) => number;
   783→  min: number;
   784→  max: number;
   785→  canonicalUnit: string;
   786→  onSave: (metric: number) => Promise<void>;
   787→  onCancel: () => void;
   788→}) {
   789→  const [raw, setRaw] = useState(initialMetric != null ? String(roundTo1(toDisplay(initialMetric))) : "");
   790→  const [saving, setSaving] = useState(false);
   791→
   792→  const parsed = raw.trim() === "" ? null : Number(raw);
   793→  const metric = parsed == null || !Number.isFinite(parsed) ? null : toMetric(parsed);
   794→  const error =
   795→    parsed == null ? null
   796→    : !Number.isFinite(parsed) || parsed < min || parsed > max
   797→      ? `Enter ${min}–${max} ${unitLabel}`
   798→      : null;
   799→
   800→  const save = async () => {
   801→    if (metric == null || error) return;
   802→    setSaving(true);
   803→    try {
   804→      await onSave(roundTo1(metric));
   805→    } finally {
   806→      setSaving(false);
   807→    }
   808→  };
   809→
   810→  return (
   811→    <EditorShell
   812→      title={`${title} · ${unitLabel}`}
   813→      footer={
   814→        <SaveCancelButtons
   815→          onSave={() => void save()}
   816→          onCancel={onCancel}
   817→          saveDisabled={metric == null || error != null}
   818→          saving={saving}
   819→        />
   820→      }
   821→    >
   822→      <div className="flex flex-col gap-1.5">
   823→        <Input
   824→          type="number"
   825→          inputMode="decimal"
   826→          step="any"
   827→          min="0"
   828→          className={cn("h-12 rounded-lg text-base tabular-nums", error && "border-destructive")}
   829→          value={raw}
   830→          onChange={(e) => setRaw(e.target.value)}
   831→          placeholder={placeholder ?? ""}
   832→          aria-label={`${title} in ${unitLabel}`}
   833→          aria-invalid={error != null}
   834→        />
   835→        {error ? (
   836→          <p className="px-1 text-xs text-destructive" role="alert">
   837→            {error}
   838→          </p>
   839→        ) : imperial && metric != null ? (
   840→          <p className="px-1 text-xs tabular-nums text-muted-foreground">
   841→            ≈ {roundTo1(metric)} {canonicalUnit}
   842→          </p>
   843→        ) : (
   844→          <p className="px-1 text-xs text-muted-foreground">Stored as {canonicalUnit}.</p>
   845→        )}
   846→      </div>
   847→    </EditorShell>
   848→  );
   849→}
