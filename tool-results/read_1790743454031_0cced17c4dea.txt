    30→export class ApiError extends Error {
    31→  constructor(
    32→    public status: number,
    33→    public code: string,
    34→    message: string,
    35→    public details?: unknown,
    36→  ) {
    37→    super(message);
    38→  }
    39→}
    40→
    41→async function request<T>(path: string, init?: RequestInit): Promise<T> {
    42→  const res = await fetch(path, {
    43→    ...init,
    44→    headers: { "Content-Type": "application/json", ...(init?.headers ?? {}) },
    45→    credentials: "same-origin",
    46→  });
    47→  if (!res.ok) {
    48→    let code = "INTERNAL";
    49→    let message = res.statusText || "Request failed";
    50→    let details: unknown;
    51→    try {
    52→      const json = await res.json();
    53→      if (json?.error) {
    54→        code = json.error.code ?? code;
    55→        message = json.error.message ?? message;
    56→        details = json.error.details;
    57→      }
    58→    } catch {
    59→      /* non-JSON error */
    60→    }
    61→    throw new ApiError(res.status, code, message, details);
    62→  }
    63→  if (res.status === 204) return undefined as T;
    64→  return (await res.json()) as T;
    65→}
    66→
    67→const body = (data: unknown) => JSON.stringify(data);
    68→const qs = (params: Record<string, string | number | boolean | undefined>) => {
    69→  const search = new URLSearchParams();
    70→  for (const [k, v] of Object.entries(params)) {
    71→    if (v !== undefined && v !== "") search.set(k, String(v));
    72→  }
    73→  const s = search.toString();
    74→  return s ? `?${s}` : "";
    75→};
