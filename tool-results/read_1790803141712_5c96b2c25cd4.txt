    39→export class ApiError extends Error {
    40→  constructor(
    41→    public status: number,
    42→    public code: string,
    43→    message: string,
    44→    public details?: unknown,
    45→  ) {
    46→    super(message);
    47→  }
    48→}
    49→
    50→async function request<T>(path: string, init?: RequestInit): Promise<T> {
    51→  const res = await fetch(path, {
    52→    ...init,
    53→    headers: { "Content-Type": "application/json", ...(init?.headers ?? {}) },
    54→    credentials: "same-origin",
    55→  });
    56→  if (!res.ok) {
    57→    let code = "INTERNAL";
    58→    let message = res.statusText || "Request failed";
    59→    let details: unknown;
    60→    try {
    61→      const json = await res.json();
    62→      if (json?.error) {
    63→        code = json.error.code ?? code;
    64→        message = json.error.message ?? message;
    65→        details = json.error.details;
    66→      }
    67→    } catch {
    68→      /* non-JSON error */
    69→    }
    70→    throw new ApiError(res.status, code, message, details);
    71→  }
    72→  if (res.status === 204) return undefined as T;
    73→  return (await res.json()) as T;
    74→}
    75→
    76→const body = (data: unknown) => JSON.stringify(data);
    77→const qs = (params: Record<string, string | number | boolean | string[] | undefined>) => {
    78→  const search = new URLSearchParams();
    79→  for (const [k, v] of Object.entries(params)) {
    80→    if (Array.isArray(v)) {
    81→      const csv = v.filter(Boolean).join(",");
    82→      if (csv !== "") search.set(k, csv);
    83→    } else if (v !== undefined && v !== "") {
    84→      search.set(k, String(v));
    85→    }
    86→  }
    87→  const s = search.toString();
    88→  return s ? `?${s}` : "";
    89→};
