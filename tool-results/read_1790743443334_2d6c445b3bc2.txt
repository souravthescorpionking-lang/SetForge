   147→/**
   148→ * `Intermediate · 6 days · ~50 min` — the §3.5 meta line shared by the list's
   149→ * Following row and the detail screen's meta row (defined here; both files
   150→ * belong to the same Part 8 §3.5 task). `days` = daysPerWeek when seeded,
   151→ * else the day count.
   152→ */
   153→export function programMetaLine(input: {
   154→  difficulty?: string | null;
   155→  daysPerWeek?: number | null;
   156→  dayCount: number;
   157→  estMinutes?: number | null;
   158→}): string {
   159→  const parts: string[] = [];
   160→  const dl = difficultyLabel(input.difficulty ?? null);
   161→  if (dl) parts.push(dl);
   162→  const days = input.daysPerWeek ?? input.dayCount;
   163→  if (days > 0) parts.push(`${days} ${days === 1 ? "day" : "days"}`);
   164→  if (input.estMinutes != null && input.estMinutes > 0) parts.push(`~${input.estMinutes} min`);
   165→  return parts.join(" · ");
   166→}

   168→const toggleSet = (setter: React.Dispatch<React.SetStateAction<Set<string>>>) => (v: string) => {
   169→  setter((prev) => {
   170→    const next = new Set(prev);
   171→    if (next.has(v)) next.delete(v);
   172→    else next.add(v);
   173→    return next;
   174→  });
   175→};

   531→      topBar={
