   326→const MAX_IMPORT_BYTES = 50 * 1024 * 1024; // 50 MB
   327→
   328→export async function importBackup(userId: string, mode: "replace" | "merge", data: unknown) {
   329→  const size = JSON.stringify(data ?? {}).length;
   330→  if (size > MAX_IMPORT_BYTES) throw badRequest("Import file too large (max 50 MB)");
   331→
   332→  const parsed = backupSchema.safeParse(data);
   333→  if (!parsed.success) {
   334→    throw badRequest("Invalid backup file", parsed.error.issues.slice(0, 5).map((i) => ({ path: i.path.join("."), message: i.message })));
   335→  }
   336→  const backup = parsed.data;
   337→
   338→  if (mode === "replace") {
   339→    await db.$transaction(async (tx) => {
   340→      await tx.workout.deleteMany({ where: { userId } });
   341→      await tx.routine.deleteMany({ where: { userId } });
   342→      await tx.measurementRecord.deleteMany({ where: { userId } });
   343→      await tx.measurement.deleteMany({ where: { userId } });
   344→      await tx.goal.deleteMany({ where: { userId } });
   345→      // ---- Part 6 ----
   346→      await tx.progressPhoto.deleteMany({ where: { userId } });
   347→    });
   348→  }
   349→
   350→  const createdWorkouts: string[] = [];
   351→  const touchedExercises = new Set<string>();
   352→
   353→  for (const w of backup.workouts) {
   354→    const date = toDayUtc(w.date);
   355→    const exists = await db.workout.findFirst({ where: { userId, date }, orderBy: { createdAt: "desc" } });
   356→    if (exists && mode === "merge") continue; // keep existing day, skip duplicates
   357→    if (exists && mode === "replace") continue; // replaced above; skip
   358→
   359→    const workout = await db.workout.create({
   360→      data: {
   361→        id: uuid7(),
   362→        userId,
   363→        date,
   364→        comment: w.comment ?? null,
   365→        startAt: w.startAt ? new Date(w.startAt) : null,
   366→        endAt: w.endAt ? new Date(w.endAt) : null,
   367→      },
   368→    });
   369→    createdWorkouts.push(workout.id);
   370→
   371→    const groupIds = new Map<string, string>();
   372→    for (const ex of w.exercises) {
   373→      const exercise = await db.exercise.findFirst({ where: { userId, name: ex.exerciseName } });
   374→      if (!exercise) continue; // unknown exercise in this catalogue — skip
   375→      touchedExercises.add(exercise.id);
   376→      let groupId: string | null = null;
   377→      if (ex.groupName) {
   378→        if (!groupIds.has(ex.groupName)) {
   379→          const g = await db.workoutGroup.create({
   380→            data: { id: uuid7(), userId, workoutId: workout.id, name: ex.groupName },
   381→          });
   382→          groupIds.set(ex.groupName, g.id);
   383→        }
   384→        groupId = groupIds.get(ex.groupName)!;
   385→      }
   386→      const we = await db.workoutExercise.create({
   387→        data: { id: uuid7(), userId, workoutId: workout.id, exerciseId: exercise.id, sortOrder: ex.sortOrder ?? 0, groupId },
   388→      });
   389→      for (const s of ex.sets) {
   390→        const setType = s.setType ?? (s.isWarmup ? "WARMUP" : "NORMAL");
   391→        await db.trainingSet.create({
   392→          data: {
   393→            id: uuid7(),
   394→            userId,
   395→            workoutExerciseId: we.id,
   396→            weight: s.weight ?? null,
   397→            reps: s.reps ?? null,
   398→            distance: s.distance ?? null,
   399→            timeSec: s.timeSec ?? null,
   400→            comment: s.comment ?? null,
   401→            isComplete: s.isComplete ?? false,
   402→            isWarmup: s.isWarmup ?? false,
   403→            setType,
   404→            rpe: s.rpe ?? null,
   405→            tempo: s.tempo ?? null,
   406→            restPlannedSec: s.restPlannedSec ?? null,
   407→            restActualSec: s.restActualSec ?? null,
   408→            completedAt: s.completedAt ? new Date(s.completedAt) : null,
   409→            sortOrder: s.sortOrder ?? 0,
   410→          },
   411→        });
   412→      }
   413→    }
   414→  }
   415→
   416→  // measurements (merge by name)
   417→  for (const m of backup.measurements ?? []) {
   418→    const unit = await db.measurementUnit.findFirst({ where: { OR: [{ userId: null, name: m.unitName }, { userId, name: m.unitName }] } });
   419→    if (!unit) continue;
   420→    const existing = await db.measurement.findFirst({ where: { userId, name: m.name } });
   421→    if (existing) {
   422→      for (const r of m.records ?? []) {
   423→        const dup = await db.measurementRecord.findFirst({
   424→          where: { measurementId: existing.id, recordedAt: new Date(r.recordedAt), value: r.value },
   425→        });
   426→        if (!dup) {
   427→          await db.measurementRecord.create({
   428→            data: { id: uuid7(), userId, measurementId: existing.id, value: r.value, recordedAt: new Date(r.recordedAt), comment: r.comment ?? null },
   429→          });
   430→        }
   431→      }
   432→    } else {
   433→      const created = await db.measurement.create({
   434→        data: {
   435→          id: uuid7(),
   436→          userId,
   437→          unitId: unit.id,
   438→          name: m.name,
   439→          goalType: m.goalType ?? "NONE",
   440→          targetValue: m.targetValue ?? null,
   441→          isEnabled: m.isEnabled ?? true,
   442→          sortOrder: m.sortOrder ?? 0,
   443→        },
   444→      });
   445→      for (const r of m.records ?? []) {
   446→        await db.measurementRecord.create({
   447→          data: { id: uuid7(), userId, measurementId: created.id, value: r.value, recordedAt: new Date(r.recordedAt), comment: r.comment ?? null },
   448→        });
   449→      }
   450→    }
   451→  }
   452→
   453→  await db.$transaction(async (tx) => {
   454→    for (const exerciseId of touchedExercises) await recomputePRs(tx, userId, exerciseId);
   455→  });
   456→
   457→  // ---- Part 6: profile (photos export as a keys list only; not re-importable) ----
   458→  if (backup.profile) {
   459→    await db.userProfile.upsert({
   460→      where: { userId },
   461→      create: {
   462→        id: uuid7(),
   463→        userId,
   464→        age: backup.profile.age ?? null,
   465→        heightCm: backup.profile.heightCm ?? null,
   466→        weightKg: backup.profile.weightKg ?? null,
   467→        level: backup.profile.level ?? null,
   468→        goal: backup.profile.goal ?? null,
   469→        daysPerWeekTarget: backup.profile.daysPerWeekTarget ?? null,
   470→      },
   471→      update: {
   472→        age: backup.profile.age ?? null,
   473→        heightCm: backup.profile.heightCm ?? null,
   474→        weightKg: backup.profile.weightKg ?? null,
   475→        level: backup.profile.level ?? null,
   476→        goal: backup.profile.goal ?? null,
   477→        daysPerWeekTarget: backup.profile.daysPerWeekTarget ?? null,
   478→      },
   479→    });
   480→  }
   481→  // Part 8 §4: DailyCalories was dropped. Old backups still carry a `calories`
   482→  // array — accepted by the schema for compatibility but no longer imported
   483→  // (calorie data was exported at migration time; see scripts/migrate-part8.ts).
   484→
   485→  return { ok: true, importedWorkouts: createdWorkouts.length };
   486→}

   490→export async function exportCsv(userId: string, type: "workouts" | "body"): Promise<string> {
   491→  const esc = (v: unknown) => {
   492→    const s = String(v ?? "");
   493→    return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
   494→  };
   495→  if (type === "body") {
   496→    const rows = await db.measurement.findMany({
   497→      where: { userId },
   498→      include: { unit: true, records: { orderBy: { recordedAt: "asc" } } },
   499→      orderBy: { sortOrder: "asc" },
   500→    });
   501→    const lines = ["date,measurement,value,unit,comment"];
   502→    for (const m of rows) {
   503→      for (const r of m.records) {
   504→        lines.push([r.recordedAt.toISOString(), m.name, r.value, m.unit.name, r.comment ?? ""].map(esc).join(","));
   505→      }
   506→    }
   507→    return lines.join("\n");
   508→  }
   509→  const workouts = await db.workout.findMany({
   510→    where: { userId },
   511→    orderBy: { date: "asc" },
   512→    include: {
   513→      exercises: { orderBy: { sortOrder: "asc" }, include: { exercise: true, sets: { orderBy: { sortOrder: "asc" } } } },
   514→    },
   515→  });
   516→  const lines = ["date,exercise,category,set,weight,reps,distance,time_sec,comment"];
   517→  for (const w of workouts) {
   518→    for (const we of w.exercises) {
   519→      we.sets.forEach((s, i) => {
   520→        lines.push(
   521→          [
   522→            w.date.toISOString().slice(0, 10),
   523→            we.exercise.name,
   524→            "",
   525→            i + 1,
   526→            s.weight ?? "",
   527→            s.reps ?? "",
   528→            s.distance ?? "",
   529→            s.timeSec ?? "",
   530→            s.comment ?? "",
   531→          ]
   532→            .map(esc)
   533→            .join(","),
   534→        );
   535→      });
   536→    }
   537→  }
   538→  return lines.join("\n");
   539→}

   543→export async function deleteWorkoutHistory(
   544→  userId: string,
   545→  opts: { mode: "all" | "range" | "exercise"; from?: string; to?: string; exerciseId?: string },
   546→) {
   547→  let workoutIds: string[] = [];
   548→  if (opts.mode === "all") {
   549→    workoutIds = (await db.workout.findMany({ where: { userId }, select: { id: true } })).map((w) => w.id);
   550→    await db.workout.deleteMany({ where: { userId } });
   551→  } else if (opts.mode === "range") {
   552→    if (!opts.from || !opts.to) throw badRequest("Range deletion requires from and to dates");
   553→    const where = { userId, date: { gte: toDayUtc(opts.from), lte: toDayUtc(opts.to) } };
   554→    workoutIds = (await db.workout.findMany({ where, select: { id: true } })).map((w) => w.id);
   555→    await db.workout.deleteMany({ where });
   556→  } else if (opts.mode === "exercise") {
   557→    if (!opts.exerciseId) throw badRequest("Exercise deletion requires exerciseId");
   558→    const ex = await db.exercise.findFirst({ where: { id: opts.exerciseId, userId } });
   559→    if (!ex) throw notFound("Exercise not found");
   560→    const wes = await db.workoutExercise.findMany({
   561→      where: { userId, exerciseId: opts.exerciseId },
   562→      select: { workoutId: true },
   563→    });
   564→    const candidateIds = [...new Set(wes.map((w) => w.workoutId))];
   565→    await db.workoutExercise.deleteMany({ where: { userId, exerciseId: opts.exerciseId } });
   566→    // remove now-empty workouts
   567→    for (const wid of candidateIds) {
   568→      const remaining = await db.workoutExercise.count({ where: { workoutId: wid } });
   569→      const groupCount = await db.workoutGroup.count({ where: { workoutId: wid } });
   570→      if (remaining === 0) {
   571→        if (groupCount > 0) await db.workoutGroup.deleteMany({ where: { workoutId: wid } });
   572→        await db.workout.delete({ where: { id: wid } }).catch(() => undefined);
   573→        workoutIds.push(wid);
   574→      }
   575→    }
   576→  }
   577→  // full PR recompute
   578→  const exerciseIds = (await db.exercise.findMany({ where: { userId }, select: { id: true } })).map((e) => e.id);
   579→  await db.$transaction(async (tx) => {
   580→    for (const exerciseId of exerciseIds) await recomputePRs(tx, userId, exerciseId);
   581→  });
   582→  return { ok: true, deletedWorkouts: workoutIds.length };
   583→}

   611→// ---------- social sign-in providers (§9; env-gated, no OAuth in this build) ----------
   612→
   613→const PROVIDER_LABELS: Record<string, string> = {
   614→  google: "Google",
   615→  apple: "Apple",
   616→};
   617→
   618→/** Comma list from AUTH_SOCIAL_PROVIDERS, normalised (lowercase, deduped, non-empty). */
   619→function socialProviderIds(env: Env): string[] {
   620→  const seen = new Set<string>();
   621→  const ids: string[] = [];
   622→  for (const raw of env.AUTH_SOCIAL_PROVIDERS.split(",")) {
   623→    const id = raw.trim().toLowerCase();
   624→    if (id && !seen.has(id)) {
   625→      seen.add(id);
   626→      ids.push(id);
   627→    }
   628→  }
   629→  return ids;
   630→}
   631→
   632→function providerLabel(id: string): string {
   633→  return PROVIDER_LABELS[id] ?? id.charAt(0).toUpperCase() + id.slice(1);
   634→}
   635→
   636→/** True only when the provider's OAuth client credentials exist in the environment. */
   637→function oauthProviderConfigured(provider: string, env: Env): boolean {
   638→  if (provider === "google") return Boolean(env.AUTH_GOOGLE_ID && env.AUTH_GOOGLE_SECRET);
   639→  return false; // no other provider has credentials wired in this build
   640→}
   641→
   642→/**
   643→ * GET /api/account/social. Surfaces the env-configured provider list with an
   644→ * honest `linked` state: this deployment has no OAuth sign-in flow, so no
   645→ * account can ever have a linked provider — `linked` is always false.
   646→ */
   647→export function listSocialProviders() {
   648→  const env = getEnv();
   649→  return {
   650→    providers: socialProviderIds(env).map((id) => ({ id, label: providerLabel(id), linked: false })),
   651→  };
   652→}
   653→
   654→/**
   655→ * POST /api/account/social {provider}. Linking needs a live OAuth flow; this
   656→ * build has none, so a 200 {ok:false} with the honest reason is returned —
   657→ * never a faked success.
   658→ */
   659→export function linkSocialProvider(provider: string): { ok: boolean; message: string } {
   660→  const env = getEnv();
   661→  if (!socialProviderIds(env).includes(provider)) throw badRequest("Unknown provider");
   662→  if (!oauthProviderConfigured(provider, env)) {
   663→    return { ok: false, message: "Social sign-in is not configured on this deployment" };
   664→  }
   665→  // Credentials present, but this build ships no OAuth round-trip — still honest.
   666→  return { ok: false, message: "Social sign-in is not available in this build" };
   667→}
   668→
   669→/**
   670→ * DELETE /api/account/social {provider}. With no OAuth flow nothing can be
   671→ * linked, so unlinking is always rejected with a 400 (honest state).
   672→ */
   673→export function unlinkSocialProvider(provider: string): { ok: true } {
   674→  const env = getEnv();
   675→  if (!socialProviderIds(env).includes(provider)) throw badRequest("Unknown provider");
   676→  throw badRequest("No linked account");
   677→}

   679→// ---------- soft delete + 30-day purge ----------
   680→
   681→const PURGE_AFTER_DAYS = 30;
   682→
   683→/**
   684→ * §9 Delete account: SOFT delete. The email is anonymized to a unique
   685→ * `deleted+{userId}@setforge.invalid` (frees the original address for reuse —
   686→ * the anonymized one is rejected naturally by the unique constraint on signup),
   687→ * the name is cleared, and every Session row is destroyed (signed out
   688→ * everywhere). login() rejects `deletedAt != null` users, so the account is
   689→ * unreachable from that moment. purgeDeletedAccounts() hard-deletes the row
   690→ * (cascading every related record) after 30 days.
   691→ */
   692→export async function softDeleteAccount(userId: string): Promise<void> {
   693→  const user = await db.user.findUnique({ where: { id: userId } });
   694→  if (!user) throw notFound("Account not found");
   695→  if (user.deletedAt) return; // idempotent — already soft-deleted
   696→  await db.$transaction([
   697→    db.session.deleteMany({ where: { userId } }),
   698→    db.user.update({
   699→      where: { id: userId },
   700→      data: {
   701→        deletedAt: new Date(),
   702→        email: `deleted+${userId}@setforge.invalid`,
   703→        name: null,
   704→      },
   705→    }),
   706→  ]);
   707→  console.log(`[account] soft-deleted: user ${userId} (purge in ${PURGE_AFTER_DAYS} days)`);
   708→}
   709→
   710→/**
   711→ * §9 purge job: hard-delete users soft-deleted more than 30 days ago. The
   712→ * database-level cascades on User relations wipe all owned data. Idempotent —
   713→ * runs on every server boot (bootstrap.ts), safe to re-run.
   714→ */
   715→export async function purgeDeletedAccounts(): Promise<{ purged: number }> {
   716→  const cutoff = new Date(Date.now() - PURGE_AFTER_DAYS * 24 * 60 * 60 * 1000);
   717→  const result = await db.user.deleteMany({ where: { deletedAt: { lt: cutoff } } });
   718→  if (result.count > 0) {
   719→    console.log(`[account] purged ${result.count} deleted account(s) past the ${PURGE_AFTER_DAYS}-day window`);
   720→  }
   721→  return { purged: result.count };
   722→}
