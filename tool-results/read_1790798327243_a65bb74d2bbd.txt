     3→// ─────────────────────────────────────────────────────────────────────────────
     4→// SetRow — THE single source of truth set row (Part 3 §SET ROW).
     5→//
     6→// One 40px CSS-grid row for every mode (edit / read / preview / template).
     7→// The grid template is COMPUTED, never styled ad hoc:
     8→//   • spec track widths:  #24 · type32 · f1:72 · f2:56 · rpe:52 · tempo:64 ·
     9→//                          rest:56 · done:40 · more:32
    10→//   • missing modality fields or settings-hidden columns REMOVE tracks from
    11→//     the template (they are never "hidden" cells).
    12→//   • <468px viewport: rpe + tempo tracks are REMOVED — their values/editors
    13→//     move into the `more` ⋯ popover. (PLATFORM ADAPTATION: the spec's bottom
    14→//     sheet is forbidden here; an anchored popover — shadcn Popover on
    15→//     Floating UI with collision flip — is the allowed replacement.)
    16→//   • <360px viewport: f1/f2 drop their floors to spec−16 (minmax) so the row
    17→//     still fits a 320px screen. All small tracks stay exact px.
    18→//   • f1/f2 carry `minmax(spec, 1fr)` so value columns fill leftover card width
    19→//     at every viewport — the row can never overflow its card and never leaves
    20→//     a dead right margin.
    21→//
    22→// Grid math (weight+reps modality, all columns on):
    23→//   wide fixed tracks 24+32+72+56+52+64+56+40+32 = 428px
    24→//   mobile (<468, rpe+tempo dropped)             = 312px ≤ 328px (360−32 law)
    25→//   320px viewport floors 24+32+56+40+56+40+32   = 280px ≤ 282px available
    26→//   (The spec contract names 460 as the wide threshold; the corrected value is
    27→//   468 — 428px of tracks + 32px page gutters + 6px card chrome (border×2 +
    28→//   4px colour bar) need ≥466px, so the wide layout only engages at 468.)

    89→function buildGrid(
    90→  fields: SetField[],
    91→  visible: CardVisibleColumns,
    92→  mode: CardMode,
    93→  vw: number,
    94→  moreTrack: boolean,
    95→): GridSpec {
    96→  const keys: TrackKey[] = [];
    97→  const parts: string[] = [];
    98→  const push = (key: TrackKey, css: string) => {
    99→    keys.push(key);
   100→    parts.push(css);
   101→  };
   102→
   103→  push("index", `${TRACK.index}px`);
   104→  if (visible.setType) push("setType", `${TRACK.setType}px`);
   105→
   106→  // Value tracks: spec floor (−16 below 360px — documented adaptation so the
   107→  // fixed small tracks still fit a 320px screen) + 1fr growth to fill the card.
   108→  const narrow = vw < 360;
   109→  if (fields[0]) push("f1", `minmax(${TRACK.f1 - (narrow ? 16 : 0)}px, 1fr)`);
   110→  if (fields[1]) push("f2", `minmax(${TRACK.f2 - (narrow ? 16 : 0)}px, 1fr)`);
   111→
   112→  if (visible.rpe && vw >= WIDE_VIEWPORT) push("rpe", `${TRACK.rpe}px`);
   113→  if (visible.tempo && vw >= WIDE_VIEWPORT) push("tempo", `${TRACK.tempo}px`);
   114→  if (visible.rest) push("rest", `${TRACK.rest}px`);
   115→
   116→  // `done` never applies to routine templates → track removed.
   117→  if (mode !== "template") push("done", `${TRACK.done}px`);
   118→  if (mode === "log" || mode === "edit" || mode === "template") push("more", `${TRACK.more}px`);
   119→  else if (moreTrack) push("more", `${TRACK.more}px`);
   120→
   121→  return { keys, template: parts.join(" ") };
   122→}

   178→function TempoEditor({

   373→function TypeCell({
   374→  set,
   375→  mode,
   376→  editable,
   377→  onAction,
   378→}: {
   379→  set: CardSet;
   380→  mode: CardMode;
   381→  editable: boolean;
   382→  onAction?: (a: CardAction) => void;
   383→}) {
   384→  const [open, setOpen] = useState(false);
   385→  const holdRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   386→  const startRef = useRef<{ x: number; y: number } | null>(null);
   387→  const longPressed = useRef(false);
   388→
   389→  const endHold = () => {
   390→    if (holdRef.current) clearTimeout(holdRef.current);
   391→    holdRef.current = null;
   392→    startRef.current = null;
   393→  };
   394→
   395→  useEffect(() => () => endHold(), []);
   396→
   397→  const value = set.setType ?? null;
   398→  const meta = value ? SET_TYPE_META[value as SetType] : null;
   399→  const templateBlank = mode === "template" && value == null;
   400→
   401→  // read / preview / read-only fallback → static glyph
   402→  if (!editable || !onAction) {
   403→    if (meta) {
   404→      return (
   405→        <span
   406→          className={cn("flex h-6 w-6 items-center justify-center rounded-md text-[11px] font-bold", meta.className)}
   407→          title={`${meta.label} — ${meta.description}`}
   408→        >
   409→          {meta.letter}
   410→        </span>
   411→      );
   412→    }
   413→    return (
   414→      <span className="text-xs text-muted-foreground/40" title="Set type not set">
   415→        –
   416→      </span>
   417→    );
   418→  }
   419→
   420→  const cycle = () => {
   421→    const current = (value ?? "NORMAL") as SetType;
   422→    const next = TYPE_CYCLE[(TYPE_CYCLE.indexOf(current) + 1) % TYPE_CYCLE.length];
   423→    onAction({ type: "update-set", setId: set.id, patch: { setType: next } });
   424→  };
   425→
   426→  return (
   427→    <Popover open={open} onOpenChange={setOpen}>
   428→      <PopoverTrigger asChild>
   429→        <button
   430→          type="button"
   431→          {...tourAttrs({ id: "setRow.type", label: "Set type", help: "Tap to cycle; hold to open the picker.", order: 110 })}
   432→          aria-label={
   433→            meta
   434→              ? `Set type ${meta.label}. Tap to cycle, hold for picker`
   435→              : "Set type not set. Tap to choose"
   436→          }
   437→          title={meta ? `${meta.label} — ${meta.description}` : "Choose set type"}
   438→          onClick={(e) => {
   439→            // Fully controlled popover: preventDefault suppresses Radix's own
   440→            // click-toggle (composeEventHandlers skips when defaultPrevented),
   441→            // so tap = cycle · hold / right-click / Enter = picker.
   442→            e.preventDefault();
   443→            if (longPressed.current) {
   444→              longPressed.current = false;
   445→              return;
   446→            }
   447→            if (templateBlank) {
   448→              setOpen(true);
   449→              return;
   450→            }
   451→            cycle();
   452→          }}
   453→          onPointerDown={(e) => {
   454→            startRef.current = { x: e.clientX, y: e.clientY };
   455→            longPressed.current = false;
   456→            holdRef.current = setTimeout(() => {
   457→              longPressed.current = true;
   458→              setOpen(true);
   459→            }, 420);
   460→          }}
   461→          onPointerMove={(e) => {
   462→            const s = startRef.current;
   463→            if (s && (Math.abs(e.clientX - s.x) > 8 || Math.abs(e.clientY - s.y) > 8)) endHold();
   464→          }}
   465→          onPointerUp={endHold}
   466→          onPointerCancel={endHold}
   467→          onPointerLeave={endHold}
   468→          onContextMenu={(e) => {
   469→            e.preventDefault();
   470→            endHold();
   471→            setOpen(true);
   472→          }}
   473→          className={cn(
   474→            "flex h-7 w-7 items-center justify-center rounded-md text-[11px] font-bold transition-transform active:scale-95",
   475→            templateBlank ? "border border-dashed border-border text-muted-foreground" : meta?.className,
   476→          )}
   477→        >
   478→          {templateBlank ? "–" : (meta?.letter ?? "–")}
   479→        </button>
   480→      </PopoverTrigger>
   481→      <PopoverContent align="center" className="w-56 p-2">
   482→        <p className="px-1 pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Set type</p>
   483→        <div className="flex flex-col gap-1">
   484→          {SET_TYPES.map((t) => (
   485→            <button
   486→              key={t}
   487→              type="button"
   488→              {...tourAttrs({ skipTour: true, reason: "Set-type choices inside the type popover" })}
   489→              onClick={() => {
   490→                onAction({ type: "update-set", setId: set.id, patch: { setType: t } });
   491→                setOpen(false);
   492→              }}
   493→              className="flex w-full items-center gap-2 rounded-lg px-2 py-1 text-left transition-colors hover:bg-accent"
   494→            >
   495→              <span
   496→                className={cn(
   497→                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[11px] font-bold",
   498→                  SET_TYPE_META[t].className,
   499→                )}
   500→              >
   501→                {SET_TYPE_META[t].letter}
   502→              </span>
   503→              <span className="min-w-0 flex-1">
   504→                <span className="block text-sm font-semibold leading-tight">{SET_TYPE_META[t].label}</span>
   505→                <span className="block text-[11px] leading-tight text-muted-foreground">
   506→                  {SET_TYPE_META[t].description}
   507→                </span>
   508→              </span>
   509→            </button>
   510→          ))}
   511→        </div>
   512→      </PopoverContent>
   513→    </Popover>
   514→  );
   515→}

   517→function ValueCell({
   518→  field,
   519→  set,
   520→  exercise,
   521→  mode,
   522→  editable,
   523→  onAction,
   524→}: {
   525→  field: SetField;

   527→  exercise: CardExercise;

   531→}) {
   532→  const key = FIELD_TO_KEY[field];
   533→  const value = (set[key] as number | null | undefined) ?? null;
   534→  const [draft, setDraft] = useState<string | null>(null);
   535→  const [focused, setFocused] = useState(false);
   536→  const step = stepForField(field, exercise);
   537→  const templateBlank = mode === "template" && value == null;
   538→  const label = FIELD_LABEL[field];
   539→
   540→  if (!editable || !onAction) {
   541→    return (
   542→      <span
   543→        className="w-full truncate px-1 text-right text-sm tabular-nums text-foreground"
   544→        title={formatFieldValue(field, value, exercise.unit)}
   545→      >
   546→        {formatFieldValue(field, value, exercise.unit)}
   547→      </span>
   548→    );
   549→  }
   550→
   551→  const display = draft ?? (value == null ? "" : trimNum(value));
   552→
   553→  const commit = () => {
   554→    if (draft != null) {
   555→      const parsed = parseFieldValue(field, draft);
   556→      if (parsed !== undefined && parsed !== value) {
   557→        onAction({ type: "update-set", setId: set.id, patch: { [key]: parsed } as Partial<CardSet> });
   558→      }
   559→    }
   560→    setDraft(null);
   561→  };
   562→
   563→  const stepBy = (dir: 1 | -1) => {
   564→    const parsedDraft = draft != null ? parseFieldValue(field, draft) : undefined;
   565→    const base = (parsedDraft != null ? parsedDraft : value) ?? 0;
   566→    const next = Math.max(0, Math.round((base + dir * step) * 100) / 100);
   567→    onAction({ type: "update-set", setId: set.id, patch: { [key]: next } as Partial<CardSet> });
   568→    setDraft(null);
   569→  };
   570→
   571→  const stepperButton = (dir: 1 | -1) => (
   572→    <button
   573→      type="button"
   574→      aria-label={dir < 0 ? `Decrease ${label}` : `Increase ${label}`}
   575→      // Keep focus on the input so the steppers don't unmount before click.
   576→      onMouseDown={(e) => e.preventDefault()}
   577→      {...tourAttrs(
   578→        dir === -1
   579→          ? { id: "setRow.stepper", label: "Stepper", help: "Tap −/+ to nudge the focused value.", order: 190, hint: true }
   580→          : { skipTour: true, reason: "The + half of the focused-cell stepper pair" },
   581→      )}
   582→      onClick={() => stepBy(dir)}
   583→      className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   584→    >
   585→      {dir < 0 ? <Minus className="h-3 w-3" aria-hidden /> : <Plus className="h-3 w-3" aria-hidden />}
   586→    </button>
   587→  );
   588→
   589→  // Grid keyboard navigation (audit D6/M2): Enter/↓ → same column one row down
   590→  // (Enter on the last row adds a set); ↑ → one row up.
   591→  const moveFocus = (dir: 1 | -1, input: HTMLInputElement): boolean => {
   592→    const cell = input.parentElement?.parentElement;
   593→    const row = cell?.parentElement;
   594→    if (!cell || !row || !row.parentElement) return false;
   595→    const colIndex = Array.prototype.indexOf.call(row.children, cell);
   596→    const rows = Array.from(row.parentElement.querySelectorAll<HTMLElement>("[data-row]"));
   597→    const rowIndex = rows.indexOf(row);
   598→    const target = rows[rowIndex + dir];
   599→    if (!target) return false;
   600→    const targetCell = target.children[colIndex] as HTMLElement | undefined;
   601→    const focusable = targetCell?.querySelector<HTMLElement>("input, button");
   602→    if (focusable) {
   603→      focusable.focus();
   604→      return true;
   605→    }
   606→    return false;
   607→  };
   608→
   609→  const onInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   610→    if (e.key === "ArrowUp") {
   611→      e.preventDefault();
   612→      commit();
   613→      moveFocus(-1, e.currentTarget);
   614→    } else if (e.key === "ArrowDown" || e.key === "Enter") {
   615→      e.preventDefault();
   616→      commit();
   617→      const input = e.currentTarget;
   618→      if (!moveFocus(1, input)) {
   619→        // Last row: Enter adds a new set (prefilled by the card) and focuses it.
   620→        onAction({ type: "add-set" });
   621→        requestAnimationFrame(() => {
   622→          const row = input.parentElement?.parentElement?.parentElement;
   623→          const rows = row?.parentElement
   624→            ? Array.from(row.parentElement.querySelectorAll<HTMLElement>("[data-row]"))
   625→            : [];
   626→          const cell = input.parentElement?.parentElement;
   627→          const colIndex = cell && row ? Array.prototype.indexOf.call(row.children, cell) : -1;
   628→          const last = rows[rows.length - 1];
   629→          const focusable =
   630→            colIndex >= 0 ? (last?.children[colIndex] as HTMLElement | undefined)?.querySelector<HTMLElement>("input") : null;
   631→          focusable?.focus();
   632→        });
   633→      }
   634→    }
   635→  };
   636→
   637→  return (
   638→    <div className="flex h-full min-w-0 items-center">
   639→      {/* template blank: ↺ = copy-last (typed value replaces it) */}
   640→      {templateBlank && !focused ? (
   641→        <button
   642→          type="button"
   643→          {...tourAttrs({ skipTour: true, reason: "Copy-previous shortcut on blank template cells" })}
   644→          aria-label={`Copy previous set ${label.toLowerCase()}`}
   645→          title="Copy previous set"
   646→          onClick={() => onAction({ type: "copy-last", setId: set.id })}
   647→          className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   648→        >
   649→          <RotateCcw className="h-3 w-3" aria-hidden />
   650→        </button>
   651→      ) : null}
   652→      {/* focused: inline steppers inside the cell ([− 20px | input | + 20px]) */}
   653→      {focused ? stepperButton(-1) : null}
   654→      <input
   655→        aria-label={label}
   656→        {...tourAttrs({ skipTour: true, reason: "Input covered by the setRow.field cell anchor" })}
   657→        inputMode={field === "reps" ? "numeric" : "decimal"}
   658→        value={display}
   659→        onFocus={() => setFocused(true)}
   660→        onBlur={() => {
   661→          commit();
   662→          setFocused(false);
   663→        }}
   664→        onKeyDown={onInputKeyDown}
   665→        onChange={(e) => setDraft(e.target.value)}
   666→        className="h-full min-w-0 flex-1 truncate rounded-sm border-0 bg-transparent px-1 text-right text-sm tabular-nums text-foreground outline-none focus:bg-muted/40"
   667→      />
   668→      {focused ? stepperButton(1) : null}
   669→    </div>
   670→  );

   832→/** One selectable field in the ⋯ popover's "Apply to all sets" expander. */
   833→type ApplyField = { key: keyof ApplyToAllFields; label: string; enabled: boolean };

   835→function applyFieldsFor(exercise: CardExercise, set: CardSet): ApplyField[] {
   836→  const fields = fieldsForType(exercise.modality);
   837→  const out: ApplyField[] = fields.map((f) => ({
   838→    key: FIELD_TO_KEY[f] as keyof ApplyToAllFields,
   839→    label: FIELD_LABEL[f],
   840→    enabled: (set[FIELD_TO_KEY[f]] as number | null | undefined) != null,
   841→  }));
   842→  out.push({ key: "restPlannedSec", label: "Rest", enabled: set.restPlannedSec != null });
   843→  out.push({ key: "tempo", label: "Tempo", enabled: !!set.tempo });
   844→  out.push({ key: "rpe", label: "RPE", enabled: set.rpe != null });
   845→  return out;
   846→}

   848→function ApplyToAllSection({
   849→  exercise,
   850→  set,
   851→  onAction,
   852→}: {
   853→  exercise: CardExercise;
   854→  set: CardSet;
   855→  onAction: (a: CardAction) => void;
   856→}) {
   857→  const [open, setOpen] = useState(false);
   858→  const [selected, setSelected] = useState<Set<string>>(new Set());
   859→  const fields = applyFieldsFor(exercise, set);
   860→  const anyEnabled = fields.some((f) => f.enabled);
   861→
   862→  const toggle = (key: string) => {
   863→    hapticTap();
   864→    setSelected((prev) => {
   865→      const next = new Set(prev);
   866→      if (next.has(key)) next.delete(key);
   867→      else next.add(key);
   868→      return next;
   869→    });
   870→  };
   871→
   872→  const apply = () => {
   873→    // keys are constrained to keyof ApplyToAllFields by applyFieldsFor, so the
   874→    // record-then-cast keeps each property's value type sound.
   875→    const out: Record<string, unknown> = {};
   876→    for (const f of fields) {
   877→      if (!selected.has(f.key) || !f.enabled) continue;
   878→      // copy the value straight off the source row (undefined → null = clear)
   879→      out[f.key] = set[f.key] ?? null;
   880→    }
   881→    onAction({ type: "apply-to-all", fields: out as ApplyToAllFields });
   882→  };
   883→
   884→  if (!anyEnabled) {
   885→    return (
   886→      <p className="text-[11px] leading-snug text-muted-foreground/70">
   887→        Fill weight, reps, rest, tempo or RPE on this set to copy it to every set.
   888→      </p>
   889→    );
   890→  }
   891→
   892→  return (
   893→    <div>
   894→      <button
   895→        type="button"
   896→        aria-expanded={open}
   897→        {...tourAttrs({ id: "setRow.applyAll", label: "Apply to all", help: "Copy this set's fields onto every set.", order: 200, hint: true })}
   898→        aria-label="Apply to all sets"
   899→        onClick={() => {
   900→          hapticTap();
   901→          setOpen((o) => !o);
   902→        }}
   903→        className="flex w-full items-center gap-2 rounded-md px-1 py-1 text-left text-xs font-bold uppercase tracking-wider text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   904→      >
   905→        <Copy className="h-3.5 w-3.5 flex-none" aria-hidden />
   906→        <span className="min-w-0 flex-1">Apply to all sets</span>
   907→        <ChevronDown
   908→          className={cn("h-3.5 w-3.5 flex-none transition-transform", open && "rotate-180")}
   909→          aria-hidden
   910→        />
   911→      </button>
   912→      {open ? (
   913→        <div className="pt-2">
   914→          <div className="flex flex-wrap gap-1" role="group" aria-label="Fields to apply">
   915→            {fields.map((f) => {
   916→              const on = selected.has(f.key);
   917→              return (
   918→                <button
   919→                  key={f.key}
   920→                  type="button"
   921→                  disabled={!f.enabled}
   922→                  {...tourAttrs({ skipTour: true, reason: "Field chips inside the apply-to-all section" })}
   923→                  aria-pressed={on}
   924→                  aria-disabled={!f.enabled}
   925→                  aria-label={`Apply ${f.label}${f.enabled ? "" : " — no value on this set"}`}
   926→                  className={cn(
   927→                    "flex h-8 items-center rounded-full border px-3 text-xs font-bold transition-colors",
   928→                    on
   929→                      ? "border-primary/60 bg-primary/10 text-primary"
   930→                      : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   931→                    !f.enabled && "cursor-not-allowed opacity-40 hover:bg-transparent hover:text-muted-foreground",
   932→                  )}
   933→                  onClick={() => toggle(f.key)}
   934→                >
   935→                  {f.label}
   936→                  {on ? <Check className="ml-1 h-3 w-3" aria-hidden /> : null}
   937→                </button>
   938→              );
   939→            })}
   940→          </div>
   941→          <Button
   942→            type="button"
   943→            size="sm"
   944→            tour={{ skipTour: true, reason: "Apply action inside the apply-to-all section" }}
   945→            className="mt-2 h-8 w-full rounded-lg text-xs font-bold"
   946→            disabled={selected.size === 0}
   947→            onClick={apply}
   948→          >
   949→            Apply to all sets
   950→          </Button>
   951→        </div>
   952→      ) : null}
   953→    </div>
   954→  );
   955→}
