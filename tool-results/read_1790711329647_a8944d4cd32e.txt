   107→  return (

   109→      {header}
   110→      <div

   112→        style={{ gridTemplateRows: open ? "1fr" : "0fr" }}
   113→      >
   114→        <div className="min-h-0 overflow-hidden">
   115→          {open ? children : null}
   116→        </div>
   117→      </div>
   118→    </section>
   119→  );
   120→}

   126→function ExerciseNotesPopover({
   127→  exerciseName,
   128→  notes,
   129→  open,
   130→  onClose,
   131→}: {
   132→  exerciseName: string;
   133→  notes: string | null;
   134→  open: boolean;
   135→  onClose: () => void;
   136→}) {
   137→  return (
   138→    <Popover open={open} onOpenChange={(o) => !o && onClose()}>
   139→      <PopoverTrigger asChild>
   140→        <span className="absolute bottom-2 left-2 h-1 w-1" aria-hidden />
   141→      </PopoverTrigger>
   142→      <PopoverContent
   143→        align="start"
   144→        className="w-72"
   145→        onOpenAutoFocus={(e) => e.preventDefault()}
   146→        // The ⋮ menu closes by returning focus to its trigger; without this
   147→        // the popover would be instantly dismissed (same Radix gotcha as the
   148→        // Today card popovers). Pointer-outside + Escape still close.
   149→        onFocusOutside={(e) => e.preventDefault()}
   150→      >
   151→        <p className="pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">
   152→          {exerciseName} — notes
   153→        </p>
   154→        <p className="text-sm leading-relaxed text-foreground">
   155→          {notes && notes.trim().length > 0 ? notes : "No notes on this exercise."}
   156→        </p>
   157→      </PopoverContent>
   158→    </Popover>
   159→  );
   160→}
