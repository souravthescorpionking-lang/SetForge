   296→function RecordsTab({ exercise }: { exercise: ExerciseDTO }) {
   297→  const settings = useApp((s) => s.settings);
   298→  const unit = exerciseUnit(exercise, settings);
   299→  const fields = useMemo(() => new Set(fieldsForType(exercise.type)), [exercise.type]);
   300→  const tracksWeightReps = fields.has("weight") && fields.has("reps");
   301→
   302→  const recordsQuery = useQuery({
   303→    queryKey: qk.exerciseRecords(exercise.id),
   304→    queryFn: () => exercisesApi.records(exercise.id),
   305→    staleTime: 60_000,
   306→  });
   307→  const historyQuery = useQuery({
   308→    queryKey: qk.exerciseHistory(exercise.id),
   309→    queryFn: () => exercisesApi.history(exercise.id),
   310→    staleTime: 60_000,
   311→  });
   312→
   313→  const rows = useMemo<{ out: RecordRow[]; actual: RecordRow[]; estimated: RecordRow[] }>(() => {
   314→    const out: RecordRow[] = [];
   315→    const entries = historyQuery.data ?? [];
   316→    const records: RecordsDTO | undefined = recordsQuery.data;
   317→
   318→    // ---- bests from the logged sets (legacy records-tab computations) ----
   319→    let bestWeight: { w: number; day: string } | null = null;
   320→    let bestVolume: { v: number; day: string } | null = null;
   321→    let bestE1rm: { v: number; day: string } | null = null;
   322→    let bestDistance: { v: number; day: string } | null = null;
   323→    let bestTime: { v: number; day: string } | null = null;
   324→    const repLimit = settings?.estOneRmRepLimit ?? 10;
   325→    const method = settings?.e1rmMethod ?? "BRZYCKI";
   326→
   327→    for (const en of entries) {
   328→      const day = dayKeyOf(en.date);
   329→      for (const s of en.sets) {
   330→        if (s.weight != null && (bestWeight == null || s.weight > bestWeight.w)) {
   331→          bestWeight = { w: s.weight, day };
   332→        }
   333→        if (
   334→          s.weight != null &&
   335→          s.reps != null &&
   336→          s.weight > 0 &&
   337→          s.reps > 0 &&
   338→          (bestVolume == null || s.weight * s.reps > bestVolume.v)
   339→        ) {
   340→          bestVolume = { v: s.weight * s.reps, day };
   341→        }
   342→        // same guard as the server: failure sets are not e1RM inputs
   343→        if (
   344→          s.weight != null &&
   345→          s.reps != null &&
   346→          s.reps >= 1 &&
   347→          s.reps <= repLimit &&
   348→          s.setType !== "FAILURE"
   349→        ) {
   350→          const e = estOneRmByMethod(s.weight, s.reps, method, s.rpe);
   351→          if (e > 0 && (bestE1rm == null || e > bestE1rm.v)) bestE1rm = { v: e, day };
   352→        }
   353→        if (s.distance != null && s.distance > 0 && (bestDistance == null || s.distance > bestDistance.v)) {
   354→          bestDistance = { v: s.distance, day };
   355→        }
   356→        if (s.timeSec != null && s.timeSec > 0 && (bestTime == null || s.timeSec > bestTime.v)) {
   357→          bestTime = { v: s.timeSec, day };
   358→        }
   359→      }
   360→    }
   361→
   362→    if (tracksWeightReps) {
   363→      if (bestWeight) {
   364→        out.push({ lift: "Best weight", value: weightLabel(bestWeight.w, unit), date: formatDayShort(bestWeight.day) });
   365→      }
   366→      if (bestVolume) {
   367→        out.push({ lift: "Best volume set", value: `${round1(bestVolume.v)} ${unit}`, date: formatDayShort(bestVolume.day) });
   368→      }
   369→      if (bestE1rm) {
   370→        // value prefers the server computation when loaded (authoritative)
   371→        const v = records && records.estimatedOneRm > 0 ? records.estimatedOneRm : bestE1rm.v;
   372→        out.push({ lift: "Estimated 1RM", value: `${round1(v)} ${unit}`, date: formatDayShort(bestE1rm.day) });
   373→      }
   374→    }
   375→    if (bestDistance) out.push({ lift: "Best distance", value: `${round2(bestDistance.v)} km`, date: formatDayShort(bestDistance.day) });
   376→    if (bestTime) out.push({ lift: "Best time", value: formatDuration(bestTime.v), date: formatDayShort(bestTime.day) });
   377→
   378→    // ---- per-reps actual PRs (records API, superseded dimmed) ----
   379→    const actual: RecordRow[] = (records?.actual ?? []).map((r) => ({
   380→      lift: `Best set at ${r.reps} ${r.reps === 1 ? "rep" : "reps"}`,
   381→      value: weightLabel(r.weight, unit),
   382→      date: formatDayShort(dayKeyOf(r.date)),
   383→      dim: r.superseded,
   384→    }));
   385→
   386→    // ---- estimated rep maxes (nRM, legacy grid → rows; 1RM row above) ----
   387→    const estimated: RecordRow[] = (records?.estimated ?? [])
   388→      .filter((r) => r.reps > 1)
   389→      .map((r) => ({ lift: `Estimated ${r.reps}RM`, value: `${round1(r.weight)} ${unit}`, date: null }));
   390→
   391→    return { out, actual, estimated };
   392→  }, [historyQuery.data, recordsQuery.data, settings, unit, tracksWeightReps]);
   393→
   394→  if (historyQuery.isLoading || recordsQuery.isLoading) {
   395→    return (
   396→      <div className="flex flex-col gap-2" aria-busy="true" aria-label="Loading records">
   397→        {[0, 1, 2, 3, 4, 5, 6].map((i) => (
   398→          <Skeleton key={i} className="h-10 rounded-lg" />
   399→        ))}
   400→      </div>
   401→    );
   402→  }
   403→
   404→  const nothing = rows.out.length === 0 && rows.actual.length === 0 && rows.estimated.length === 0;
   405→
   406→  return (
   407→    <div className="flex flex-col gap-3">
   408→      {nothing ? (
   409→        <div
   410→          data-row
   411→          className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"
   412→        >
   413→          No records yet — log a set to start tracking.
   414→        </div>
   415→      ) : null}
   416→
   417→      {rows.out.length > 0 ? (
   418→        <section className="flex flex-col gap-1" aria-label="Bests">
   419→          <SectionHeader label="Bests" />
   420→          {rows.out.map((r) => (
   421→            <RecordRowLine key={r.lift} row={r} />
   422→          ))}
   423→        </section>
   424→      ) : null}
   425→
   426→      {!tracksWeightReps && (rows.out.length > 0 || nothing) ? (
   427→        <p className="px-1 text-xs text-muted-foreground">
   428→          Weight × reps records need a weighted exercise — showing bests only.
   429→        </p>
   430→      ) : null}
   431→
   432→      {rows.actual.length > 0 ? (
   433→        <section className="flex flex-col gap-1" aria-label="Rep records">
   434→          <SectionHeader label="Rep records" />
   435→          {rows.actual.map((r) => (
   436→            <RecordRowLine key={r.lift} row={r} />
   437→          ))}
   438→        </section>
   439→      ) : null}
   440→
   441→      {rows.estimated.length > 0 ? (
   442→        <section className="flex flex-col gap-1" aria-label="Estimated maxes">
   443→          <SectionHeader label="Estimated maxes" />
   444→          {rows.estimated.map((r) => (
   445→            <RecordRowLine key={r.lift} row={r} />
   446→          ))}
   447→        </section>
   448→      ) : null}
   449→
   450→      {rows.actual.length > 0 ? (
   451→        <p className="px-1 text-xs text-muted-foreground">
   452→          A PR is superseded when a heavier set at the same-or-higher rep count was logged later.
   453→        </p>
   454→      ) : null}
   455→    </div>
   456→  );
   457→}

   483→    );
   484→  }
   485→
   486→  return (
   487→    <div className="flex flex-col gap-1">
   488→      {/* add-goal row (48px data-row) → inline 96px creator */}
   489→      <button
   490→        type="button"
   491→        data-row
   492→        aria-expanded={creating}
   493→        onClick={() => {
   494→          setCreating((c) => !c);
   495→          setExpandedId(null);
   496→        }}
   497→        className="flex h-12 w-full items-center gap-2 overflow-hidden whitespace-nowrap rounded-lg border border-dashed px-3 text-left text-sm font-semibold text-primary transition-colors hover:bg-accent/50"
   498→      >
   499→        <Plus className="h-4 w-4 flex-none" aria-hidden />
   500→        <span className="min-w-0 flex-1 truncate">Add goal for {exercise.name}</span>
   501→      </button>
   502→      {creating ? (
   503→        <GoalCreator
   504→          exercise={exercise}
   505→          unit={unit}
   506→          onDone={() => setCreating(false)}
   507→        />
   508→      ) : null}
   509→
   510→      {goals.length === 0 ? (
   511→        <div
   512→          data-row
   513→          className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"
   514→        >
   515→          No goals for this exercise yet.
   516→        </div>
   517→      ) : (
   518→        goals.map((g) => (
   519→          <div key={g.id} className="flex flex-col">
   520→            <button
   521→              type="button"
   522→              data-row
   523→              aria-expanded={expandedId === g.id}
   524→              onClick={() => setExpandedId((cur) => (cur === g.id ? null : g.id))}

   566→/** 96px inline goal creator (NOT a data-row): type select + target input. */
   567→function GoalCreator({
   568→  exercise,
   569→  unit,
   570→  onDone,
   571→}: {
   572→  exercise: ExerciseDTO;
   573→  unit: WeightUnit;
   574→  onDone: () => void;
   575→}) {
   576→  const run = useOfflineRun();
   577→  const inv = useInvalidate();
   578→  const [type, setType] = useState<string>("ONE_RM");
   579→  const [target, setTarget] = useState("");
   580→  const [saving, setSaving] = useState(false);
   581→
   582→  const field = goalTargetField(type);
   583→  const placeholder =
   584→    field === "targetWeight"
   585→      ? `Target ${unit}`
   586→      : field === "targetReps"
   587→        ? "Target reps"
   588→        : field === "targetDistance"
   589→          ? "Target km"
   590→          : "Target sec";
   591→
   592→  const save = async () => {
   593→    const n = Number(target);
   594→    if (target.trim() === "" || !Number.isFinite(n) || n <= 0) {
   595→      toast.error("Set a target value above zero");
   596→      return;

   598→    setSaving(true);

   600→      type,
   601→      targetWeight: field === "targetWeight" ? n : null,
   602→      targetReps: field === "targetReps" ? n : null,
   603→      targetDistance: field === "targetDistance" ? n : null,
   604→      targetTimeSec: field === "targetTimeSec" ? n : null,
   605→    };
   606→    const ok = await run({
   607→      label: "Goal",
   608→      path: "/api/goals",
   609→      method: "POST",
   610→      body: { ...payload, exerciseId: exercise.id },

   612→      successMsg: "Goal created",
   613→      onDone: () => inv.goals(),
   614→    });
   615→    setSaving(false);
   616→    if (ok) onDone();
   617→  };
   618→
   619→  return (
   620→    <div className="flex h-24 flex-none flex-col gap-1 rounded-lg border bg-card p-1.5">
   621→      <div className="flex min-h-0 flex-1 items-center overflow-hidden">

   627→            {["ONE_RM", "MAX_WEIGHT", "MAX_REPS", "MAX_DISTANCE", "MAX_TIME", "VOLUME"].map((t) => (

   636→        <Input
   637→          type="number"
   638→          inputMode="decimal"
   639→          step="any"
   640→          min="0"
   641→          className="h-full flex-1 rounded-lg tabular-nums"
   642→          value={target}
   643→          onChange={(e) => setTarget(e.target.value)}
   644→          placeholder={placeholder}
   645→          aria-label={placeholder}
   646→        />
   647→        <Button
   648→          type="button"
   649→          className="h-full flex-1 rounded-lg font-semibold"
   650→          disabled={saving}
   651→          onClick={() => void save()}
   652→        >
   653→          {saving ? "Saving…" : "Add goal"}
   654→        </Button>
   655→      </div>
   656→    </div>
   657→  );
   658→}

   660→/** 96px inline expansion (NOT a data-row): progress + target editor + delete. */
   661→function GoalExpansion({
   662→  goal,
   663→  unit,
   664→  onDone,
   665→}: {
   666→  goal: GoalDTO;
   667→  unit: WeightUnit;
   668→  onDone: () => void;
   669→}) {
   670→  const run = useOfflineRun();
   671→  const inv = useInvalidate();
   672→  const field = goalTargetField(goal.type);
   673→  const initial =
   674→    field === "targetWeight"
   675→      ? goal.targetWeight
   676→      : field === "targetReps"
   677→        ? goal.targetReps
   678→        : field === "targetDistance"
   679→          ? goal.targetDistance
   680→          : goal.targetTimeSec;
   681→
   682→  const [target, setTarget] = useState<string>(initial != null ? String(initial) : "");
   683→  const [saving, setSaving] = useState(false);
   684→  const [armed, setArmed] = useState(false);
   685→
   686→  // auto-disarm the two-tap delete after a pause
   687→  useEffect(() => {
   688→    if (!armed) return;
   689→    const t = window.setTimeout(() => setArmed(false), 4000);
   690→    return () => window.clearTimeout(t);
   691→  }, [armed]);
   692→
   693→  const save = async () => {
   694→    const n = Number(target);
   695→    if (target.trim() === "" || !Number.isFinite(n) || n <= 0) {
   696→      toast.error("Set a target value above zero");
   697→      return;
   698→    }
   699→    setSaving(true);
   700→    const payload: GoalInput = {
   701→      type: goal.type,
   702→      targetWeight: field === "targetWeight" ? n : null,
   703→      targetReps: field === "targetReps" ? n : null,
   704→      targetDistance: field === "targetDistance" ? n : null,
   705→      targetTimeSec: field === "targetTimeSec" ? n : null,
   706→    };
   707→    const ok = await run({
   708→      label: "Goal update",
   709→      path: `/api/goals/${goal.id}`,
   710→      method: "PATCH",
   711→      body: payload,
   712→      run: () => goalsApi.update(goal.id, payload),
   713→      successMsg: "Goal updated",
   714→      onDone: () => inv.goals(),
   715→    });
   716→    setSaving(false);
   717→    if (ok) onDone();
   718→  };
   719→
   720→  const remove = async () => {
   721→    if (!armed) {
   722→      setArmed(true);
   723→      return;
   724→    }
   725→    const ok = await run({
   726→      label: "Goal delete",
   727→      path: `/api/goals/${goal.id}`,
   728→      method: "DELETE",
   729→      run: () => goalsApi.remove(goal.id),
   730→      successMsg: "Goal deleted",
   731→      onDone: () => inv.goals(),
   732→    });
   733→    if (ok) onDone();
   734→  };
   735→
   736→  return (
   737→    <div className="flex h-24 flex-none flex-col gap-1 rounded-lg border bg-card p-1.5">
   738→      <div className="flex min-h-0 flex-1 items-center gap-2 overflow-hidden whitespace-nowrap">
   739→        <Progress
   740→          value={Math.min(100, goal.pct)}
   741→          className={cn("h-2 flex-1", goal.achieved && "*:data-[slot=progress-indicator]:bg-emerald-500")}
   742→          aria-label={`${goal.pct}% of goal`}
   743→        />
   744→        <span className="flex-none truncate text-xs font-semibold tabular-nums text-muted-foreground">
   745→          {goalValueLabel(goal.type, goal.current, unit)} / {goalValueLabel(goal.type, goal.target, unit)}
   746→        </span>
   747→      </div>
   748→      <div className="flex min-h-0 flex-1 gap-2">
   749→        <Input
   750→          type="number"
   751→          inputMode="decimal"
   752→          step="any"
   753→          min="0"
   754→          className="h-full flex-1 rounded-lg tabular-nums"
   755→          value={target}
   756→          onChange={(e) => setTarget(e.target.value)}
   757→          aria-label="Goal target"
   758→        />
   759→        <Button
   760→          type="button"
   761→          className="h-full flex-1 rounded-lg font-semibold"
   762→          disabled={saving}
   763→          onClick={() => void save()}
   764→        >
   765→          {saving ? "Saving…" : "Save target"}
   766→        </Button>
   767→        <Button
   768→          type="button"
   769→          variant="outline"
   770→          className={cn(
   771→            "h-full flex-none gap-1 rounded-lg",
   772→            armed ? "border-destructive/50 text-destructive" : "text-muted-foreground",
   773→          )}
   774→          onClick={() => void remove()}
   775→        >
   776→          <Trash2 className="h-3.5 w-3.5" aria-hidden />
   777→          {armed ? "Sure?" : "Delete"}
   778→        </Button>
   779→      </div>
   780→    </div>
   781→  );
   782→}

   785→// HISTORY — DateGroup×N (32px headers) + ONE ExerciseCard read mode per date

   788→function HistoryTab({
   789→  exercise,
   790→  visibleColumns,
   791→}: {
   792→  exercise: ExerciseDTO;
   793→  visibleColumns: CardVisibleColumns;
   794→}) {
   795→  const navigate = useApp((s) => s.navigate);
   796→  const settings = useApp((s) => s.settings);
   797→
   798→  const historyQuery = useQuery({
   799→    queryKey: qk.exerciseHistory(exercise.id),
   800→    queryFn: () => exercisesApi.history(exercise.id),
   801→    staleTime: 60_000,
   802→  });
   803→  // records → date-level PR badges (legacy history-tab pattern)
   804→  const recordsQuery = useQuery({
   805→    queryKey: qk.exerciseRecords(exercise.id),
   806→    queryFn: () => exercisesApi.records(exercise.id),
   807→    staleTime: 60_000,
   808→  });
   809→
   810→  const [collapsedIds, setCollapsedIds] = useState<Set<string>>(new Set());
   811→  const [notesEntryId, setNotesEntryId] = useState<string | null>(null);
   812→
   813→  const groups = useMemo(() => {
   814→    const map = new Map<string, HistoryEntry[]>();
   815→    for (const en of historyQuery.data ?? []) {
   816→      const key = dayKeyOf(en.date);
   817→      const list = map.get(key);
   818→      if (list) list.push(en);
   819→      else map.set(key, [en]);
   820→    }
   821→    return [...map.entries()].sort((a, b) => b[0].localeCompare(a[0]));
   822→  }, [historyQuery.data]);
   823→
   824→  const prDates = useMemo(
   825→    () => new Set((recordsQuery.data?.actual ?? []).map((r) => dayKeyOf(r.date))),
   826→    [recordsQuery.data],
   827→  );
   828→
   829→  if (historyQuery.isLoading) {
   830→    return (
   831→      <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading history">
   832→        {[0, 1, 2, 3].map((i) => (
   833→          <Skeleton key={i} className="h-14 rounded-lg" />
   834→        ))}
   835→      </div>
   836→    );
   837→  }
   838→
   839→  if (groups.length === 0) {
   840→    return (
   841→      <div
   842→        data-row
   843→        className="flex h-12 items-center overflow-hidden whitespace-nowrap px-1 text-sm text-muted-foreground"
   844→      >
   845→        No history yet — every set you log appears here.
   846→      </div>
   847→    );
   848→  }
   849→
   850→  const cardEx: CardExercise = {
   851→    id: exercise.id,
   852→    name: exercise.name,
   853→    categoryLabel: exercise.category?.name ?? "Exercise",
   854→    categoryColour: exercise.category?.colour ?? "#71717a",
   855→    modality: exercise.type,
   856→    unit: exerciseUnit(exercise, settings),
   857→    weightIncrement: exercise.weightIncrement ?? settings?.defaultWeightIncrement ?? 2.5,
   858→  };
   859→
   860→  /** SetDTO → CardSet (distance km → CardSet metres, as in workout-block). */
   861→  const toCardSets = (sets: SetDTO[]): CardSet[] =>
   862→    [...sets]
   863→      .sort((a, b) => a.sortOrder - b.sortOrder)
   864→      .map((s, i) =>
   865→        toCardSet({ ...s, distance: s.distance != null ? s.distance * 1000 : null }, i + 1),
   866→      );
   867→
   868→  const toggleCollapsed = (weId: string) =>
   869→    setCollapsedIds((prev) => {
   870→      const next = new Set(prev);
   871→      if (next.has(weId)) next.delete(weId);
   872→      else next.add(weId);
   873→      return next;
   874→    });
   875→
   876→  return (
   877→    <div className="flex flex-col gap-3">
   878→      {groups.map(([day, entries]) => {
   879→        const setCount = entries.reduce((a, e) => a + e.sets.length, 0);
   880→        const isPr = prDates.has(day);
   881→        return (
   882→          <section key={day} className="flex flex-col gap-2" aria-label={formatDayLabel(day)}>
   883→            <h2 className="flex h-8 items-center gap-2 overflow-hidden px-1 text-xs font-bold uppercase tracking-wider text-muted-foreground">
   884→              <span className="flex-none truncate">{formatDayLabel(day)}</span>
   885→              {isPr ? (
   886→                <Trophy className="h-3.5 w-3.5 flex-none text-primary" aria-label="Personal record day" />
   887→              ) : null}
   888→              <span className="truncate text-[10px] font-medium normal-case tracking-normal text-muted-foreground/70">
   889→                {setCount} {setCount === 1 ? "set" : "sets"}
   890→              </span>
   891→              <span className="h-px min-w-0 flex-1 bg-border/60" aria-hidden />
   892→            </h2>
   893→
   894→            {entries.map((entry) => {
   895→              const collapsed = collapsedIds.has(entry.workoutExerciseId);
   896→              const notesOpen = notesEntryId === entry.workoutExerciseId;
   897→              return (
   898→                <div key={entry.workoutExerciseId} className="flex flex-col">
   899→                  <div
   900→                    role="button"
   901→                    tabIndex={0}
   902→                    aria-label={`${collapsed ? "Expand" : "Collapse"} sets from ${formatDayLabel(day)}`}
   903→                    className="flex-none cursor-pointer"
   904→                    onClick={(e) => {
   905→                      // taps on the card's own controls (buttons/inputs) never toggle
   906→                      if ((e.target as HTMLElement).closest("button, input, textarea, a")) return;
   907→                      toggleCollapsed(entry.workoutExerciseId);
   908→                    }}
   909→                    onKeyDown={(e) => {
   910→                      if (e.target !== e.currentTarget) return;
   911→                      if (e.key === "Enter" || e.key === " ") {
   912→                        e.preventDefault();
   913→                        toggleCollapsed(entry.workoutExerciseId);
   914→                      }
   915→                    }}
   916→                  >
   917→                    <ExerciseCard
   918→                      mode="read"
   919→                      collapsed={collapsed}
   920→                      exercise={cardEx}
   921→                      sets={toCardSets(entry.sets)}
   922→                      visibleColumns={visibleColumns}
   923→                      onAction={(action: CardAction) => {
   924→                        switch (action.type) {
   925→                          case "toggle-collapse":
   926→                            toggleCollapsed(entry.workoutExerciseId);
   927→                            break;
   928→                          case "open":
   929→                            navigate(`/today/${entry.workoutExerciseId}?date=${day}`);
   930→                            break;
   931→                          case "notes":
   932→                            setNotesEntryId(entry.workoutExerciseId);
   933→                            break;
   934→                          default:
   935→                            break;
   936→                        }
   937→                      }}
   938→                    />
   939→                  </div>
   940→                  {notesOpen ? (
   941→                    <div className="relative flex-none">
   942→                      <ExerciseNotesPopover
   943→                        exercise={exercise}
   944→                        open
   945→                        onClose={() => setNotesEntryId(null)}
   946→                      />
   947→                    </div>
   948→                  ) : null}
   949→                </div>
   950→              );
   951→            })}
   952→          </section>
   953→        );
   954→      })}
   955→      {(historyQuery.data?.length ?? 0) >= 100 ? (
   956→        <p className="px-1 text-xs text-muted-foreground">Showing the most recent 100 sessions.</p>
   957→      ) : null}
   958→    </div>
   959→  );
   960→}
