   122→/** 56px data-row: label | current value + chevron; whole row opens a small menu. */
   123→export function MenuRow<T extends string | number>({
   124→  label,
   125→  hint,
   126→  value,
   127→  options,
   128→  onSelect,
   129→}: {
   130→  label: string;
   131→  hint?: string;
   132→  value: T;
   133→  options: Array<{ value: T; label: string }>;
   134→  onSelect: (v: T) => void;
   135→}) {
   136→  const current = options.find((o) => o.value === value)?.label ?? String(value);
   137→  return (
   138→    <DropdownMenu>
   139→      <DropdownMenuTrigger asChild>
   140→        <button
   141→          type="button"
   142→          data-row
   143→          className={cn(ROW_CLS, "w-full text-left hover:bg-accent/40")}
   144→          aria-label={`${label}: ${current}`}
   145→        >
   146→          <span className="min-w-0 flex-1 truncate text-sm font-medium">
   147→            {label}
   148→            {hint ? <span className="ml-1.5 text-xs font-normal text-muted-foreground">{hint}</span> : null}
   149→          </span>
   150→          <span className="flex-none text-sm tabular-nums text-muted-foreground">{current}</span>
   151→          <ChevronRight className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   152→        </button>
   153→      </DropdownMenuTrigger>
   154→      <DropdownMenuContent align="end" className="w-44">
   155→        {options.map((o) => (
   156→          <DropdownMenuItem key={String(o.value)} onClick={() => onSelect(o.value)}>
   157→            <span className="flex-1">{o.label}</span>
   158→            {o.value === value ? <Check className="h-4 w-4 text-primary" aria-hidden /> : null}
   159→          </DropdownMenuItem>
   160→        ))}
   161→      </DropdownMenuContent>
   162→    </DropdownMenu>
   163→  );
   164→}

   337→/** Sign out + full local wipe (logic ported from legacy useLogout). */
   338→function useSignOut() {
   339→  const setSession = useApp((s) => s.setSession);
   340→  const qc = useQueryClient();
   341→  return async () => {
   342→    try {
   343→      await authApi.logout();
   344→    } finally {
   345→      clearSwCaches();
   346→      wipeLocalData();
   347→      qc.clear();
   348→      setSession(null); // AppShell gate forces #/auth
   349→      toast.success("Signed out — local data cleared");
   350→    }
   351→  };
   352→}

   354→export function AccountSection() {
   355→  const session = useApp((s) => s.session);
   356→  const online = useOnline();
   357→  const logout = useSignOut();
   358→
   359→  const [pwOpen, setPwOpen] = useState(false);
   360→  const [currentPw, setCurrentPw] = useState("");
   361→  const [newPw, setNewPw] = useState("");
   362→  const [confirmPw, setConfirmPw] = useState("");
   363→  const [pwError, setPwError] = useState<string | null>(null);
   364→  const [pwBusy, setPwBusy] = useState(false);
   365→
   366→  const [delOpen, setDelOpen] = useState(false);
   367→  const [delTyped, setDelTyped] = useState("");
   368→  const [deleting, setDeleting] = useState(false);
   369→
   370→  const [signOutOpen, setSignOutOpen] = useState(false);
   371→
   372→  if (!session) return null;
   373→
   374→  const initials =
   375→    (session.user.name ?? session.user.email)
   376→      .split(/[\s@.]/)
   377→      .filter(Boolean)
   378→      .slice(0, 2)
   379→      .map((p) => p[0]?.toUpperCase())
   380→      .join("") || "SF";
   381→
   382→  const submitPassword = async () => {
   383→    setPwError(null);
   384→    if (newPw.length < 8) {
   385→      setPwError("New password must be at least 8 characters.");
   386→      return;
   387→    }
   388→    if (newPw !== confirmPw) {
   389→      setPwError("New passwords don't match.");
   390→      return;
   391→    }
   392→    setPwBusy(true);
   393→    try {
   394→      await accountApi.changePassword({ currentPassword: currentPw, newPassword: newPw });
   395→      toast.success("Password changed — other sessions signed out");
   396→      setPwOpen(false);
   397→      setCurrentPw("");
   398→      setNewPw("");
   399→      setConfirmPw("");
   400→    } catch (e) {
   401→      setPwError(e instanceof Error ? e.message : "Could not change password");
   402→    } finally {
   403→      setPwBusy(false);
   404→    }
   405→  };
   406→
   407→  const confirmDeleteAccount = async () => {
   408→    setDeleting(true);
   409→    try {
   410→      await accountApi.delete();
   411→      toast.success("Account deleted — all data removed");
   412→    } catch (e) {
   413→      toast.error(e instanceof Error ? e.message : "Could not delete account");
   414→      setDeleting(false);
   415→      return;
   416→    }
   417→    await logout();
   418→  };
   419→
   420→  return (
   421→    <section aria-label="Account" className="flex flex-col">
   422→      <SectionHeader title="Account" />
   423→
   424→      {/* signed-in identity */}
   425→      <div data-row className={ROW_CLS}>
   426→        <span
   427→          aria-hidden
   428→          className="flex h-8 w-8 flex-none items-center justify-center rounded-full bg-primary/15 text-xs font-bold text-primary"
   429→        >
   430→          {initials}
   431→        </span>
   432→        <span className="min-w-0 flex-1 truncate text-sm font-medium">{session.user.email}</span>
   433→        <span className="max-w-24 flex-none truncate text-xs text-muted-foreground">{session.user.name ?? "Athlete"}</span>
   434→      </div>
   435→
   436→      {/* change password — inline expansion (legacy Dialog → inline block) */}
   437→      <ActionRow label="Change password" onClick={() => setPwOpen((v) => !v)} />
   438→      {pwOpen ? (
   439→        <div className="flex flex-none flex-col gap-1 border-b border-border/50 bg-muted/20 p-2" aria-label="Change password form">
   440→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   441→            <span className="w-20 flex-none text-xs text-muted-foreground">Current</span>
   442→            <Input
   443→              type="password"
   444→              autoComplete="current-password"
   445→              className="h-12 flex-1 rounded-lg"
   446→              value={currentPw}
   447→              onChange={(e) => setCurrentPw(e.target.value)}
   448→              aria-label="Current password"
   449→              placeholder="••••••••"
   450→            />
   451→          </div>
   452→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   453→            <span className="w-20 flex-none text-xs text-muted-foreground">New</span>
   454→            <Input
   455→              type="password"
   456→              autoComplete="new-password"
   457→              className="h-12 flex-1 rounded-lg"
   458→              value={newPw}
   459→              onChange={(e) => setNewPw(e.target.value)}
   460→              aria-label="New password"
   461→              placeholder="At least 8 characters"
   462→            />
   463→          </div>
   464→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   465→            <span className="w-20 flex-none text-xs text-muted-foreground">Confirm</span>
   466→            <Input
   467→              type="password"
   468→              autoComplete="new-password"
   469→              className="h-12 flex-1 rounded-lg"
   470→              value={confirmPw}
   471→              onChange={(e) => setConfirmPw(e.target.value)}
   472→              aria-label="Confirm new password"
   473→              placeholder="Repeat new password"
   474→            />
   475→            <Button

   479→              onClick={() => void submitPassword()}
   480→            >
   481→              {pwBusy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}

   487→              {pwError}

   489→          ) : (
   490→            <p className="flex-none px-1 text-xs text-muted-foreground">
   491→              Minimum 8 characters · other sessions are signed out.
   492→            </p>
   493→          )}
   494→        </div>
   495→      ) : null}
   496→
   497→      {/* sign out — confirm-destructive */}
   498→      <ActionRow label="Sign out" onClick={() => setSignOutOpen(true)} />
   499→      <AlertDialog open={signOutOpen} onOpenChange={setSignOutOpen}>
   500→        <AlertDialogContent>
   501→          <AlertDialogHeader>
   502→            <AlertDialogTitle>Sign out?</AlertDialogTitle>

   504→              This ends your session and clears offline data stored on this device. You can sign back in

   506→            </AlertDialogDescription>
   507→          </AlertDialogHeader>
   508→          <AlertDialogFooter>
   509→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   510→            <AlertDialogAction
   511→              className="bg-destructive text-white hover:bg-destructive/90"
   512→              onClick={(e) => {
   513→                e.preventDefault();
   514→                setSignOutOpen(false);
   515→                void logout();

   517→            >
   518→              <LogOut className="h-4 w-4" aria-hidden /> Sign out

   520→          </AlertDialogFooter>
   521→        </AlertDialogContent>
   522→      </AlertDialog>
   523→
   524→      {/* delete account — inline typed-DELETE confirm (no dialog) */}
   525→      <ActionRow label="Delete account" destructive onClick={() => { setDelTyped(""); setDelOpen((v) => !v); }} />
   526→      {delOpen ? (
   527→        <div className="flex flex-none flex-col gap-1 border-b border-border/50 bg-destructive/5 p-2" aria-label="Delete account confirm">
   528→          <p className="flex flex-none items-start gap-1.5 px-1 text-xs leading-relaxed text-destructive">
   529→            <TriangleAlert className="mt-0.5 h-3.5 w-3.5 flex-none" aria-hidden />
   530→            Permanently deletes your account and every piece of data in it — workouts, PRs, routines,
   531→            body logs. No undo.
   532→          </p>
   533→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">

   562→function downloadBlob(blob: Blob, filename: string) {
   563→  const url = URL.createObjectURL(blob);
   564→  const a = document.createElement("a");
   565→  a.href = url;
   566→  a.download = filename;
   567→  document.body.appendChild(a);
   568→  a.click();
   569→  a.remove();
   570→  setTimeout(() => URL.revokeObjectURL(url), 2000);
   571→}
