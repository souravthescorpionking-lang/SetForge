    84→  return (
    85→    <div
    86→      data-row
    87→      role="group"
    88→      aria-label={`Step ${step}: ${title}`}
    89→      className="flex h-8 w-full flex-none items-center gap-2 overflow-hidden whitespace-nowrap px-1"
    90→    >
    91→      <span className="flex h-6 w-6 flex-none items-center justify-center rounded bg-primary/10 text-[11px] font-bold tabular-nums text-primary">
    92→        {step}
    93→      </span>
    94→      <span className="min-w-0 flex-1 truncate text-xs font-bold uppercase tracking-wider text-muted-foreground">
    95→        {title}
    96→      </span>
    97→      {hint ? (
    98→        <span className="flex-none text-[11px] font-medium text-muted-foreground/70">{hint}</span>
    99→      ) : null}
   100→    </div>
   101→  );
   102→}

   104→function Stepper({
   105→  value,
   106→  min,
   107→  max,
   108→  formatValue,
   109→  ariaLabel,
   110→  onDecrement,
   111→  onIncrement,
   112→}: {
   113→  value: number | null;
   114→  min: number;
   115→  max: number;
   116→  formatValue?: (v: number | null) => string;
   117→  ariaLabel: string;
   118→  onDecrement: () => void;
   119→  onIncrement: () => void;
   120→}) {
   121→  const text = formatValue ? formatValue(value) : value == null ? "–" : String(value);
   122→  return (
   123→    <span className="flex flex-none items-center gap-1">
   124→      <Button
   125→        type="button"
   126→        variant="outline"
   127→        className="h-10 w-10 flex-none p-0"
   128→        disabled={value != null && value <= min}
   129→        aria-label={`Decrease ${ariaLabel}`}
   130→        tour={{ id: "programBuilder.stepper", label: "Stepper", help: "Adjust days per week and estimated minutes.", order: 80 }}
   131→        onClick={() => {
   132→          hapticTap();
   133→          onDecrement();
   134→        }}
   135→      >
   136→        <Minus className="h-4 w-4" aria-hidden />
   137→      </Button>
   138→      <span
   139→        className="flex h-10 w-12 flex-none items-center justify-center text-sm font-bold tabular-nums leading-none"
   140→        aria-live="polite"
   141→        aria-label={`${ariaLabel}: ${text}`}
   142→      >
   143→        {text}
   144→      </span>
   145→      <Button
   146→        type="button"
   147→        variant="outline"
   148→        className="h-10 w-10 flex-none p-0"
   149→        disabled={value != null && value >= max}
   150→        aria-label={`Increase ${ariaLabel}`}
   151→        tour={{ skipTour: true, reason: "The + half of the builder stepper pair" }}
   152→        onClick={() => {
   153→          hapticTap();
   154→          onIncrement();
   155→        }}
   156→      >
   157→        <Plus className="h-4 w-4" aria-hidden />
   158→      </Button>
   159→    </span>
   160→  );
   161→}

   163→// ---------- screen ----------
   164→
   165→export default function ProgramBuilderScreen() {
   166→  const navigate = useApp((s) => s.navigate);
   167→  const online = useOnline();
   168→  const invalidate = useInvalidate();
   169→
   170→  // ---------- draft state (local only — nothing persists before Create) ----------
   171→  // #/programs/new/builder?name=… deep-links can pre-fill the name.
   172→  const [name, setName] = useState(() => {
   173→    if (typeof window === "undefined") return "";
   174→    const m = window.location.hash.match(/[?&]name=([^&]+)/);
   175→    return m ? decodeURIComponent(m[1]).slice(0, 80) : "";
   176→  });
   177→  const [difficulty, setDifficulty] = useState<Difficulty>("BEGINNER");
   178→  const [daysManual, setDaysManual] = useState<number | null>(null);
   179→  const [estMinutes, setEstMinutes] = useState<number | null>(null);
   180→  const [phases, setPhases] = useState<PhaseDraft[]>([{ name: "Phase 1", weeks: 1 }]);
   181→  const [weekly, setWeekly] = useState<DayTemplate[]>(DEFAULT_WEEKLY);
   182→  const [labels, setLabels] = useState<string[]>([]);
   183→  const [addingLabel, setAddingLabel] = useState(false);
   184→  /** exercises per unique template name (draft; 3 blank sets attach on build). */
   185→  const [exercises, setExercises] = useState<Record<string, Array<{ exerciseId: string; name: string }>>>({});
   186→  const [pickerFor, setPickerFor] = useState<string | null>(null);
   187→  const [pickerSearch, setPickerSearch] = useState("");
   188→  const [creating, setCreating] = useState(false);
   189→  const searchFocusRef = useRef<HTMLInputElement | null>(null);
   190→
   191→  // ---------- derived ----------
   192→  const workoutDayCount = weekly.filter((w) => w.type === "WORKOUT").length;
   193→  const daysPerWeek = daysManual ?? workoutDayCount;
   194→  const templateNames = useMemo(() => {
   195→    const seen: string[] = [];
   196→    for (const w of weekly) {
   197→      if (w.type !== "WORKOUT") continue;
   198→      const n = w.name.trim();
   199→      if (n && !seen.includes(n)) seen.push(n);
   200→    }
   201→    return seen;
   202→  }, [weekly]);
   203→  const unnamedWorkoutDays = weekly.some((w) => w.type === "WORKOUT" && !w.name.trim());
   204→  const totalWeeks = phases.reduce((n, p) => n + p.weeks, 0);
   205→  const totalDays = totalWeeks * 7;
   206→  const canCreate =
   207→    name.trim().length > 0 && templateNames.length > 0 && !unnamedWorkoutDays && !creating;
   208→
   209→  // debounced picker search (250ms) — feeds useExercises
   210→  const [debouncedSearch, setDebouncedSearch] = useState("");
   211→  useEffect(() => {
   212→    const t = setTimeout(() => setDebouncedSearch(pickerSearch.trim()), 250);
   213→    return () => clearTimeout(t);
   214→  }, [pickerSearch]);
   215→  const exercisesQuery = useExercises(debouncedSearch ? { search: debouncedSearch } : undefined);
   216→  const allExercises = exercisesQuery.data ?? [];
   217→
   218→  // ---------- mutations ----------
   219→  const setDayType = (index: number, type: "WORKOUT" | "REST") => {
   220→    hapticTap();
   221→    setWeekly((prev) => prev.map((w, i) => (i === index ? { ...w, type } : w)));
   222→  };
   223→
   224→  const setDayName = (index: number, dayName: string) => {
   225→    setWeekly((prev) => prev.map((w, i) => (i === index ? { ...w, name: dayName.slice(0, 40) } : w)));
   226→  };
   227→
   228→  const addPhase = () => {
   229→    hapticTap();
   230→    setPhases((prev) =>
   231→      prev.length >= 12 ? prev : [...prev, { name: `Phase ${prev.length + 1}`, weeks: 1 }],
   232→    );
   233→  };
   234→
   235→  const removePhase = (index: number) => {
   236→    setPhases((prev) => (prev.length <= 1 ? prev : prev.filter((_, i) => i !== index)));
   237→  };
   238→
   239→  const openPicker = (templateName: string) => {
   240→    hapticTap();
   241→    setPickerSearch("");
   242→    setPickerFor((prev) => (prev === templateName ? null : templateName));
   243→    if (pickerFor !== templateName) {
   244→      requestAnimationFrame(() => searchFocusRef.current?.focus());
   245→    }
   246→  };
   247→
   248→  const addExerciseTo = (templateName: string, exerciseId: string, exerciseName: string) => {
   249→    hapticTap();
   250→    setExercises((prev) => {
   251→      const list = prev[templateName] ?? [];
   252→      if (list.some((e) => e.exerciseId === exerciseId)) return prev;
   253→      return { ...prev, [templateName]: [...list, { exerciseId, name: exerciseName }] };
   254→    });
   255→  };
   256→
   257→  const removeExerciseFrom = (templateName: string, exerciseId: string) => {
   258→    setExercises((prev) => {
   259→      const list = (prev[templateName] ?? []).filter((e) => e.exerciseId !== exerciseId);
   260→      const next = { ...prev, [templateName]: list };
   261→      return next;
   262→    });
   263→  };
   264→
   265→  const addLabel = (label: string) => {
   266→    const clean = label.trim().slice(0, 30);
   267→    if (!clean) return;
   268→    if (labels.includes(clean)) {
   269→      toast.info("That label already exists");
   270→      return;
   271→    }
   272→    if (labels.length >= 20) {
   273→      toast.info("Labels are limited to 20");
   274→      return;
   275→    }
   276→    setLabels((prev) => [...prev, clean]);
   277→  };
   278→
   279→  const create = async () => {
   280→    if (!canCreate) return;
   281→    if (!online) {
   282→      toast.info("Creating a program needs a connection");
   283→      return;
   284→    }
   285→    setCreating(true);
   286→    try {
   287→      const exercisesPayload: BuilderProgramInput["exercises"] = {};
   288→      for (const templateName of templateNames) {
   289→        const list = exercises[templateName] ?? [];
   290→        if (list.length === 0) continue;
   291→        exercisesPayload[templateName] = list.map((e) => ({
   292→          exerciseId: e.exerciseId,
   293→          sets: BLANK_BUILDER_SETS,
   294→        }));
   295→      }
   296→      const input: BuilderProgramInput = {
   297→        name: name.trim(),
   298→        difficulty,
   299→        ...(daysManual != null || workoutDayCount > 0 ? { daysPerWeek: Math.max(1, Math.min(7, daysPerWeek)) } : {}),
   300→        ...(estMinutes != null ? { estMinutes } : {}),
   301→        ...(labels.length > 0 ? { labels } : {}),
   302→        phases: phases.map((p, i) => ({ name: p.name.trim() || `Phase ${i + 1}`, weeks: p.weeks })),
   303→        weekly: weekly.map((w, i) => ({
   304→          weekday: i,
   305→          type: w.type,
   306→          ...(w.type === "WORKOUT" ? { name: w.name.trim() } : { name: null }),
   307→        })),
   308→        ...(exercisesPayload && Object.keys(exercisesPayload).length > 0 ? { exercises: exercisesPayload } : {}),
   309→      };
   310→      const res = await programsMetaApi.build(input);
   311→      invalidate.programs();
   312→      invalidate.routines();
   313→      hapticSuccess();
   314→      toast.success(`Program created · ${res.dayCount} days`);
   315→      replaceHash(`#/programs/${res.id}`);
   316→    } catch (e) {
   317→      toast.error(errorMessage(e));
   318→    } finally {
   319→      setCreating(false);
   320→    }
   321→  };
   322→
   323→  // ---------- render helpers ----------
   324→  const renderWeeklyRow = (index: number) => {
   325→    const day = weekly[index];
   326→    return (
   327→      <div
   328→        key={WEEKDAYS[index]}
   329→        data-row
   330→        className="flex h-12 w-full items-center gap-1 overflow-hidden whitespace-nowrap"
   331→      >
   332→        <span className="w-10 flex-none text-xs font-bold uppercase tracking-wide text-muted-foreground">
   333→          {WEEKDAYS[index]}
   334→        </span>
   335→        {day.type === "WORKOUT" ? (
   336→          <Input
   337→            value={day.name}
   338→            placeholder="Push, Pull…"
   339→            aria-label={`${WEEKDAYS[index]} workout template name`}
   340→            {...tourAttrs({ id: "programBuilder.templateName", label: "Template name", help: "Name the day's template; same names share.", order: 50 })}
   341→            onChange={(e) => setDayName(index, e.target.value)}
   342→            className="h-10 min-w-0 flex-1 rounded-md"
   343→          />
   344→        ) : (
   345→          <span className="flex h-10 min-w-0 flex-1 items-center gap-2 rounded-md border border-dashed border-border px-3 text-sm text-muted-foreground">
   346→            <Moon className="h-4 w-4 flex-none text-muted-foreground/60" aria-hidden />
   347→            Rest day
   348→          </span>
   349→        )}
   350→        <DropdownMenu>
   351→          <DropdownMenuTrigger asChild>
   352→            <Button
   353→              type="button"
   354→              variant="outline"
   355→              className="h-10 w-24 flex-none justify-between gap-1 px-2 text-xs font-bold"
   356→              aria-label={`${WEEKDAYS[index]} day type: ${day.type === "WORKOUT" ? "workout" : "rest"}`}
   357→              tour={{ id: "programBuilder.dayType", label: "Day type", help: "Switch this weekday between workout and rest.", order: 60 }}
   358→            >
   359→              {day.type === "WORKOUT" ? (
   360→                <Dumbbell className="h-4 w-4 flex-none text-primary" aria-hidden />
   361→              ) : (
   362→                <Moon className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   363→              )}
   364→              <span className="min-w-0 flex-1 truncate">
   365→                {day.type === "WORKOUT" ? "Workout" : "Rest"}
   366→              </span>
   367→              <ChevronDown className="h-3.5 w-3.5 flex-none text-muted-foreground" aria-hidden />
   368→            </Button>
   369→          </DropdownMenuTrigger>
   370→          <DropdownMenuContent align="end" className="w-36">
   371→            <DropdownMenuItem onClick={() => setDayType(index, "WORKOUT")}>
   372→              <Dumbbell className="h-4 w-4" aria-hidden /> Workout
   373→            </DropdownMenuItem>
   374→            <DropdownMenuItem onClick={() => setDayType(index, "REST")}>
   375→              <Moon className="h-4 w-4" aria-hidden /> Rest
   376→            </DropdownMenuItem>
   377→          </DropdownMenuContent>
   378→        </DropdownMenu>
   379→      </div>
   380→    );
   381→  };
   382→
   383→  const renderTemplateSection = (templateName: string) => {
   384→    const list = exercises[templateName] ?? [];
   385→    const addedIds = new Set(list.map((e) => e.exerciseId));
   386→    const pickerOpen = pickerFor === templateName;
   387→    const candidates = allExercises
   388→      .filter((ex) => !addedIds.has(ex.id))
   389→      .slice(0, 30);
   390→    return (
   391→      <div key={templateName} className="flex flex-none flex-col overflow-hidden rounded-lg border bg-card">
   392→        {/* template header 40px */}
   393→        <div
   394→          data-row
   395→          className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap border-b border-border px-3"
   396→        >
   397→          <Dumbbell className="h-4 w-4 flex-none text-primary" aria-hidden />
   398→          <span className="min-w-0 flex-1 truncate text-sm font-semibold leading-none" title={templateName}>
   399→            {templateName}
   400→          </span>
   401→          <span className="flex-none text-[11px] font-semibold tabular-nums text-muted-foreground">
   402→            {list.length} exercise{list.length === 1 ? "" : "s"}
   403→          </span>
   404→        </div>
   405→        {/* added exercises — 40px rows with × remove */}
   406→        {list.map((e) => (
   407→          <div
   408→            key={e.exerciseId}
   409→            data-row
   410→            className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap border-b border-border/60 pl-4 pr-1"
   411→          >
   412→            <span className="min-w-0 flex-1 truncate text-sm leading-none" title={e.name}>
   413→              {e.name}
   414→            </span>
   415→            <Button
   416→              type="button"
   417→              variant="ghost"
   418→              className="h-10 w-10 flex-none p-0 text-muted-foreground"
   419→              aria-label={`Remove ${e.name} from ${templateName}`}
   420→              tour={{ skipTour: true, reason: "Remove chip on an added exercise row" }}
   421→              onClick={() => removeExerciseFrom(templateName, e.exerciseId)}
   422→            >
   423→              <X className="h-4 w-4" aria-hidden />
   424→            </Button>
   425→          </div>
   426→        ))}
   427→        {/* + Add exercise → inline searchable picker (0fr→1fr expansion) */}
   428→        <button
   429→          type="button"
   430→          data-row
   431→          aria-expanded={pickerOpen}
   432→          aria-label={`Add exercise to ${templateName}`}
   433→          {...tourAttrs({ id: "programBuilder.addExercise", label: "Add exercise", help: "Open the searchable picker for this template.", order: 70 })}
   434→          className={cn(
   435→            "flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap px-3 text-left text-sm font-medium transition-colors",
   436→            pickerOpen
   437→              ? "bg-primary/5 text-primary"
   438→              : "text-muted-foreground hover:bg-accent/40",
   439→          )}
   440→          onClick={() => openPicker(templateName)}
   441→        >
   442→          {pickerOpen ? (
   443→            <ChevronDown className="h-4 w-4 flex-none rotate-180" aria-hidden />
   444→          ) : (
   445→            <Plus className="h-4 w-4 flex-none" aria-hidden />
   446→          )}
   447→          <span className="min-w-0 flex-1 truncate">
   448→            {pickerOpen ? "Close picker" : "Add exercise"}
   449→          </span>
   450→        </button>
   451→        <div
   452→          className="grid transition-[grid-template-rows] duration-200 ease-out"
   453→          style={{ gridTemplateRows: pickerOpen ? "1fr" : "0fr" }}
   454→        >
   455→          <div className="min-h-0 overflow-hidden">
   456→            {pickerOpen ? (
   457→              <div className="flex flex-col gap-2 border-t border-border p-2">
   458→                <div className="relative flex-none">
   459→                  <Search
   460→                    className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   461→                    aria-hidden
   462→                  />
   463→                  <Input
   464→                    ref={searchFocusRef}
   465→                    value={pickerSearch}
   466→                    placeholder="Search your exercises…"
   467→                    aria-label={`Search exercises for ${templateName}`}
   468→                    {...tourAttrs({ skipTour: true, reason: "Search input inside the inline picker" })}
   469→                    onChange={(e) => setPickerSearch(e.target.value)}
   470→                    className="h-10 rounded-md pl-9"
   471→                  />
   472→                </div>
   473→                <div
   474→                  className="scroll-slim flex max-h-96 flex-none flex-col overflow-y-auto"
   475→                  aria-label="Exercise results"
   476→                >
   477→                  {exercisesQuery.isLoading ? (
   478→                    <p className="px-1 py-2 text-xs text-muted-foreground" aria-busy="true">
   479→                      Loading exercises…
   480→                    </p>
   481→                  ) : candidates.length === 0 ? (
   482→                    <p className="px-1 py-2 text-xs text-muted-foreground">
   483→                      {debouncedSearch ? "No exercises match that search." : "No exercises yet — create some first."}
   484→                    </p>
   485→                  ) : (
   486→                    candidates.map((ex) => (
   487→                      <button
   488→                        key={ex.id}
   489→                        type="button"
   490→                        data-row
   491→                        aria-label={`Add ${ex.name} to ${templateName}`}
   492→                        {...tourAttrs({ skipTour: true, reason: "Data-driven result rows in the inline picker" })}
   493→                        onClick={() => addExerciseTo(templateName, ex.id, ex.name)}
   494→                        className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap rounded-md px-2 text-left transition-colors hover:bg-accent/50"
   495→                      >
   496→                        <CategoryDot colour={ex.category?.colour ?? null} />
   497→                        <span className="min-w-0 flex-1 truncate text-sm leading-none">{ex.name}</span>
   498→                        <span className="flex-none text-[11px] text-muted-foreground">
   499→                          {ex.category?.name ?? "Exercise"}
   500→                        </span>
   501→                        <Plus className="h-3.5 w-3.5 flex-none text-primary" aria-hidden />
   502→                      </button>
   503→                    ))
