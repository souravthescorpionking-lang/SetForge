    93→/** 48px set row grid: # 32 · f1 minmax(72,1fr) · f2 minmax(56,1fr) · rest 56 · … 32 */
    94→function gridTemplate(fields: SetField[], restVisible: boolean): string {
    95→  const parts: string[] = ["32px"];
    96→  if (fields[0]) parts.push("minmax(72px,1fr)");
    97→  if (fields[1]) parts.push("minmax(56px,1fr)");
    98→  if (restVisible) parts.push("56px");
    99→  parts.push("32px");
   100→  return parts.join(" ");
   101→}

   103→function fieldLabel(field: SetField, unit: string | null): string {
   104→  switch (field) {
   105→    case "weight":
   106→      return unit ?? "kg";
   107→    case "reps":
   108→      return "reps";

   116→function NumberInput({
   117→  value,
   118→  onCommit,
   119→  ariaLabel,
   120→  className,
   121→}: {
   122→  value: number | null;
   123→  onCommit: (next: number | null) => void;
   124→  ariaLabel: string;
   125→  className?: string;
   126→}) {
   127→  const [draft, setDraft] = useState<string | null>(null);
   128→  const display = draft ?? (value == null ? "" : trimNum(value));
   129→  const commit = () => {
   130→    if (draft != null) {
   131→      const parsed = parseFieldValue("reps", draft);
   132→      if (parsed !== undefined && parsed !== value) onCommit(parsed);
   133→    }
   134→    setDraft(null);
   135→  };
   136→  return (
   137→    <input
   138→      inputMode="decimal"
   139→      aria-label={ariaLabel}
   140→      {...tourAttrs({ skipTour: true, reason: "Covered by the setsEditor row cell anchors" })}
   141→      value={display}
   142→      onChange={(e) => setDraft(e.target.value)}
   143→      onBlur={commit}
   144→      onKeyDown={(e) => {
   145→        if (e.key === "Enter") {
   146→          e.preventDefault();
   147→          e.currentTarget.blur();
   148→        }
   149→      }}
   150→      className={cn(
   151→        "h-10 min-w-0 w-full rounded-md border-0 bg-transparent px-1 text-right text-sm tabular-nums text-foreground outline-none focus:bg-muted/40",
   152→        className,
   153→      )}
   154→    />
   155→  );
   156→}

   160→function WeightKindCell({
   161→  set,
   162→  onApply,
   163→}: {
   164→  set: PredefinedSetDTO;
   165→  onApply: (input: PredefinedSetInput) => void;
   166→}) {
   167→  const kind: WeightKind = (set.weightKind ?? (set.weight != null ? "FIXED" : "COPY_LAST")) as WeightKind;
   168→  const [open, setOpen] = useState(false);
   169→  const [mode, setMode] = useState<WeightKind>(kind);
   170→  const [value, setValue] = useState("");
   171→
   172→  const reseed = () => {
   173→    setMode(kind);
   174→    setValue(
   175→      kind === "PERCENT_1RM"
   176→        ? set.pct != null
   177→          ? trimNum(set.pct)
   178→          : ""
   179→        : set.weight != null
   180→          ? trimNum(set.weight)
   181→          : "",
   182→    );
   183→  };
   184→
   185→  const text =
   186→    kind === "PERCENT_1RM"
   187→      ? set.pct != null
   188→        ? `${trimNum(set.pct)}%`
   189→        : "↺ %"
   190→      : kind === "COPY_LAST"
   191→        ? "↺"
   192→        : set.weight != null
   193→          ? trimNum(set.weight)
   194→          : "–";
   195→
   196→  const apply = () => {
   197→    if (mode === "COPY_LAST") {
   198→      onApply({ weightKind: "COPY_LAST", weight: null, pct: null });
   199→    } else if (mode === "PERCENT_1RM") {
   200→      const pct = parseFieldValue("reps", value);
   201→      if (pct == null || pct <= 0 || pct > 100) {
   202→        toast.info("Enter a % of your 1RM between 1 and 100");
   203→        return;
   204→      }
   205→      onApply({ weightKind: "PERCENT_1RM", pct, weight: null });
   206→    } else {
   207→      const weight = parseFieldValue("reps", value);
   208→      onApply({ weightKind: "FIXED", weight: weight ?? null, pct: null });
   209→    }
   210→    setOpen(false);
   211→  };
   212→
   213→  const seg = (m: WeightKind, label: string) => (
   214→    <Button
   215→      key={m}
   216→      type="button"
   217→      variant={mode === m ? "default" : "outline"}
   218→      size="sm"
   219→      tour={{ skipTour: true, reason: "Weight-kind segment inside the prescription popover" }}
   220→      className="h-8 min-w-0 flex-1 rounded-lg px-1 text-xs font-bold"
   221→      onClick={() => setMode(m)}
   222→    >
   223→      <span className="truncate">{label}</span>
   224→    </Button>
   225→  );
   226→
   227→  return (
   228→    <Popover
   229→      open={open}
   230→      onOpenChange={(o) => {
   231→        if (o) reseed();
   232→        setOpen(o);
   233→      }}
   234→    >
   235→      <PopoverTrigger asChild>
   236→        <button
   237→          type="button"
   238→          {...tourAttrs({ id: "setsEditor.weightKind", label: "Weight", help: "Fixed weight, copy last workout, or a % of your 1RM.", order: 40 })}
   239→          aria-label={`Weight prescription for set ${set.sortOrder + 1}: ${kind}. Tap to change`}
   240→          className="flex h-10 w-full items-center justify-center rounded-md px-1 text-sm font-semibold tabular-nums transition-colors hover:bg-accent"
   241→        >
   242→          {text}
   243→        </button>
   244→      </PopoverTrigger>
   245→      <PopoverContent align="center" className="w-60 p-3">
   246→        <p className="pb-2 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">
   247→          Weight
   248→        </p>
   249→        <div className="flex items-center gap-1">
   250→          {seg("FIXED", "Fixed")}
   251→          {seg("COPY_LAST", "Copy last")}
   252→          {seg("PERCENT_1RM", "% 1RM")}
   253→        </div>
   254→        {mode === "FIXED" ? (
   255→          <div className="mt-3 flex items-center gap-1">
   256→            <Input
   257→              autoFocus
   258→              inputMode="decimal"
   259→              value={value}
   260→              aria-label="Fixed weight in kg"
   261→              {...tourAttrs({ skipTour: true, reason: "Fixed-weight input inside the prescription popover" })}
   262→              onChange={(e) => setValue(e.target.value)}
   263→              onKeyDown={(e) => e.key === "Enter" && apply()}
   264→              placeholder="kg"
   265→              className="h-8 min-w-0 flex-1"
   266→            />
   267→            <Button type="button" size="sm" tour={{ skipTour: true, reason: "Apply inside the prescription popover" }} className="h-8 rounded-lg" onClick={apply}>
   268→              Set
   269→            </Button>
   270→          </div>
   271→        ) : mode === "COPY_LAST" ? (
   272→          <p className="mt-3 text-xs leading-relaxed text-muted-foreground">
   273→            ↺ No stored weight — the set copies your last workout when the day starts.
   274→          </p>
   275→        ) : (
   276→          <div className="mt-3 flex items-center gap-1">
   277→            <Input
   278→              autoFocus
   279→              inputMode="decimal"
   280→              value={value}
   281→              aria-label="Percent of one-rep max"
   282→              {...tourAttrs({ skipTour: true, reason: "Percent input inside the prescription popover" })}
   283→              onChange={(e) => setValue(e.target.value)}
   284→              onKeyDown={(e) => e.key === "Enter" && apply()}
   285→              placeholder="% of 1RM"
   286→              className="h-8 min-w-0 flex-1"
   287→            />
   288→            <Button type="button" size="sm" tour={{ skipTour: true, reason: "Apply inside the prescription popover" }} className="h-8 rounded-lg" onClick={apply}>
   289→              Set
   290→            </Button>
   291→          </div>
   292→        )}
   293→      </PopoverContent>
   294→    </Popover>
   295→  );
   296→}

   600→function SetsEditorInner({ routineId, reId }: { routineId: string; reId: string }) {
   601→  const navigate = useApp((s) => s.navigate);
   602→  const settings = useApp((s) => s.settings);

   604→  const { run } = useRoutineRun();
   605→  const prefillFor = useLastSetsPrefill();
   606→  const qc = useQueryClient();

   608→  // ---------- data (reId is unique across the routine — find its day) ----------
   609→  const { data: routine, isLoading } = useQuery({
   610→    queryKey: qk.routine(routineId),
   611→    queryFn: () => routinesApi.get(routineId),
   612→    retry: 1,
   613→  });
   614→
   615→  const located = useMemo(() => {

   618→      const re = day.exercises.find((e) => e.id === reId);
   619→      if (re) return { day, re };
   620→    }
   621→    return null;
   622→  }, [routine, reId]);
   623→
   624→  const metaKey = ["exercise-meta", routineId, reId] as const;
   625→  const metaQuery = useQuery({
   626→    queryKey: metaKey,
   627→    queryFn: () => exerciseMetaApi.get(routineId, reId),
   628→    enabled: !!located,
   629→  });
   630→
   631→  const day = located?.day ?? null;
   632→  const re = located?.re ?? null;
   633→  const kindSeg = routine ? kindSegment(routine) : "program";
   634→  const backHref = `/builder/${kindSeg}/${routineId}`;
   635→
   636→  const fields = useMemo(
   637→    () => (re ? (fieldsForType(re.exercise.type) as SetField[]) : []),
   638→    [re],
   639→  );
   640→  const unit = useMemo(() => (re ? exerciseUnit(re.exercise, settings) : "kg"), [re, settings]);
   641→  const restVisible = settings?.showRest ?? true;
   642→
   643→  const sortedSets = useMemo(
   644→    () => (re ? [...re.sets].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   645→    [re],
   646→  );
   647→
   648→  // ---------- ui state ----------
   649→  const [warmupOpen, setWarmupOpen] = useState(false);
   650→  const [progressionOpen, setProgressionOpen] = useState(false);
   651→  const [tipOpen, setTipOpen] = useState(false);
   652→  const [tipDraft, setTipDraft] = useState<string | null>(null);

   654→
   655→  // tip draft syncs from the server value once
   656→  useEffect(() => {
   657→    if (re && tipDraft == null) setTipDraft(re.tip ?? "");
   658→  }, [re, tipDraft]);
   659→
   660→  // ---------- set mutations (immediate persist) ----------
   661→  const patchSet = async (setId: string, input: PredefinedSetInput) => {
   662→    if (!day || !re) return;
   663→    await run(() => routinesApi.updateSet(routineId, day.id, reId, setId, input), {
   664→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${reId}/sets/${setId}`,
   665→      method: "PATCH",
   666→      body: input,
   667→      label: "Set updated",
   668→    });
   669→  };
   670→
   671→  const addSet = async () => {
   672→    if (!day || !re) return;
   673→    await run(() => routinesApi.addSet(routineId, day.id, reId, {}), {
   674→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${reId}/sets`,
   675→      method: "POST",
   676→      body: {},
   677→      label: "Set added",
   678→    });
   679→  };
   680→
   681→  const removeSet = async (setId: string) => {
   682→    if (!day || !re) return;
   683→    await run(() => routinesApi.removeSet(routineId, day.id, reId, setId), {
   684→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${reId}/sets/${setId}`,
   685→      method: "DELETE",
   686→      label: "Set removed",
   687→    });
   688→  };
   689→
   690→  const copyLastToSet = async (setId: string, index0: number) => {
   691→    if (!day || !re) return;
   692→    const src = await prefillFor(re.exerciseId, index0);
   693→    if (!src) {
   694→      toast.info("No previous workout to copy from");
   695→      return;
   696→    }
   697→    await run(
   698→      () =>
   699→        routinesApi.updateSet(routineId, day.id, reId, setId, {
   700→          weight: src.weight,
   701→          reps: src.reps,
   702→          distance: src.distance,
   703→          timeSec: src.timeSec,
   704→          weightKind: src.weight != null ? "FIXED" : null,
   705→        }),
   706→      {
   707→        path: `/api/routines/${routineId}/days/${day.id}/exercises/${reId}/sets/${setId}`,
   708→        method: "PATCH",
   709→        body: { weight: src.weight, reps: src.reps, distance: src.distance, timeSec: src.timeSec },
   710→        label: "Copied from last workout",
   711→      },
   712→    );
   713→    toast.success("Copied from last workout");
   714→  };
   715→
   716→  const toggleWarmupSet = (set: PredefinedSetDTO) => {
   717→    void patchSet(set.id, {
   718→      setType: (set.setType ?? "NORMAL") === "WARMUP" ? "NORMAL" : "WARMUP",
   719→    });
   720→  };
   721→
   722→  /** §12 per-set AMRAP — both fields stay in sync (server law mirrors it). */
   723→  const toggleAmrapSet = (set: PredefinedSetDTO) => {
   724→    const isAmrap = set.isAmrap || (set.setType ?? "NORMAL") === "AMRAP";
   725→    void patchSet(set.id, isAmrap ? { setType: "NORMAL", isAmrap: false } : { setType: "AMRAP", isAmrap: true });
   726→  };
   727→
   728→  /** Apply ONE set's everything onto every other set. */
   729→  const applySetToAll = (source: PredefinedSetDTO) => {
   730→    for (const s of sortedSets) {
   731→      if (s.id === source.id) continue;
   732→      void patchSet(s.id, {
   733→        weight: source.weight,
   734→        weightKind: source.weightKind ?? null,
   735→        pct: source.pct ?? null,
   736→        reps: source.reps,
   737→        distance: source.distance,
   738→        timeSec: source.timeSec,
   739→        restPlannedSec: source.restPlannedSec,
   740→      });
   741→    }
   742→    toast.success(`Applied set ${source.sortOrder + 1} to all ${sortedSets.length} sets`);
   743→  };
   744→
   745→  /** Apply the FIRST set's chosen fields onto every other set (bottom ▾). */
   746→  const applyFirstFields = (which: "weight" | "reps" | "rest" | "everything") => {
   747→    const first = sortedSets[0];
   748→    if (!first || sortedSets.length < 2) {
   749→      toast.info("Add at least two sets first");
   750→      return;
   751→    }
   752→    for (const s of sortedSets.slice(1)) {
   753→      const input: PredefinedSetInput = {};
   754→      if (which === "weight" || which === "everything") {
   755→        input.weight = first.weight;
   756→        input.weightKind = first.weightKind ?? null;
   757→        input.pct = first.pct ?? null;
   758→      }
   759→      if (which === "reps" || which === "everything") {
   760→        input.reps = first.reps;
   761→        input.distance = first.distance;
   762→        input.timeSec = first.timeSec;
   763→      }
   764→      if (which === "rest" || which === "everything") {
   765→        input.restPlannedSec = first.restPlannedSec;
   766→      }
   767→      void patchSet(s.id, input);
   768→    }
   769→    toast.success(`Applied ${which} from set 1 to the other sets`);
   770→  };
   771→
   772→  // ---------- exercise meta (warm-up + progression) ----------
   773→  const warmupScheme = metaQuery.data?.warmupScheme ?? "NONE";
   774→  const progressionMeta = metaQuery.data?.progression ?? null;
   775→
   776→  const [prog, setProg] = useState<ExerciseMetaProgression | null>(null);
   777→  useEffect(() => {
   778→    setProg(progressionMeta);
   779→  }, [progressionMeta]);
   780→
   781→  const putMeta = async (body: Parameters<typeof exerciseMetaApi.put>[2]) => {
   782→    if (!located) return;
   783→    if (!online) {
   784→      toast.info("Saving needs a connection");
   785→      return;
   786→    }
   787→    try {
   788→      await exerciseMetaApi.put(routineId, reId, body);
   789→      await qc.invalidateQueries({ queryKey: metaKey });
   790→      qc.invalidateQueries({ queryKey: qk.routine(routineId) });
   791→    } catch (e) {
   792→      toast.error(errorMessage(e));
   793→    }
   794→  };
   795→
   796→  const putWarmupScheme = (scheme: "NONE" | "STANDARD" | "LIGHT") => {
   797→    void putMeta({ warmupScheme: scheme });
   798→    toast.success(scheme === "NONE" ? "Warm-up off" : `Warm-up · ${scheme === "STANDARD" ? "Standard" : "Light"}`);
   799→  };
   800→
   801→  const putProgression = (next: ExerciseMetaProgression) => {
   802→    setProg(next);
   803→    void putMeta({ progression: next });
   804→  };
   805→
   806→  const setProgField = (patch: Partial<ExerciseMetaProgression>) => {
   807→    const base = prog ?? DEFAULT_PROGRESSION;
   808→    putProgression({ ...base, ...patch });
   809→  };
   810→
   811→  const progSummary = !prog || prog.type === "NONE" ? "Off" : `${prog.type === "DOUBLE" ? "Double" : "Linear"} · +${trimNum(prog.increment)} ${prog.unit}`;
   812→  const warmupSummary = warmupScheme === "NONE" ? "Off" : warmupScheme === "STANDARD" ? "Standard" : "Light";
   813→
   814→  // ---------- §12 trainer tip + restNone (persist via the exercise PATCH) ----------
   815→  const restNone = re?.restNone ?? false;
   816→
   817→  const saveTip = async () => {
   818→    if (!day || !re || tipDraft == null) return;
   819→    const next = tipDraft.trim();
   820→    if (next === (re.tip ?? "")) return;
   821→    setTipBusy(true);

   823→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${reId}`,
   824→      method: "PATCH",
   825→      body: { tip: next || null },
   826→      label: "Tip saved",
   827→    });
   828→    setTipBusy(false);
   829→    if (ok) toast.success("Tip saved");
   830→  };

   832→  const putRestNone = async (on: boolean) => {
   833→    if (!day || !re) return;

   835→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${reId}`,
   836→      method: "PATCH",
   837→      body: { restNone: on },
   838→      label: "Rest setting saved",
   839→    });

   845→    navigate(backHref);
   846→  };

   848→  // ---------- render ----------
   849→  const headerKeys: string[] = ["index"];
   850→  if (fields[0]) headerKeys.push("f1");
   851→  if (fields[1]) headerKeys.push("f2");
   852→  if (restVisible) headerKeys.push("rest");
   853→  headerKeys.push("more");
   854→
   855→  return (
   856→    <Screen
   857→      topBar={
   858→        <TopBar
   859→          leading={<BackButton fallbackHash={`#${backHref}`} label={`Back to ${routine?.name ?? "editor"}`} />}
   860→          title={
   861→            <span className="flex min-w-0 items-center gap-1">
   862→              <span className="min-w-0 truncate">{re ? re.exercise.name : "Sets"}</span>
   863→              <span className="flex-none text-xs font-medium text-muted-foreground">· sets</span>
   864→            </span>
   865→          }
   866→          actions={
   867→            <>
   868→              <Button
   869→                type="button"
   870→                className="h-11 flex-none gap-1.5 px-4 text-sm font-bold"
   871→                disabled={!located}
   872→                tour={{ id: "setsEditor.done", label: "Done", help: "Finish and return to the editor; edits already persist.", order: 10 }}
   873→                onClick={done}
   874→              >
   875→                <Check className="h-4 w-4" aria-hidden />
   876→                Done
   877→              </Button>
   878→              <TopBarHelp />
   879→            </>
   880→          }
   881→        />
   882→      }
   883→    >
   884→      <ScrollBody>
   885→        {isLoading ? (
   886→          <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading exercise">
   887→            <div className="h-8 animate-pulse rounded-lg bg-muted/40" />
   888→            <div className="h-12 animate-pulse rounded-lg bg-muted/40" />
   889→            <div className="h-12 animate-pulse rounded-lg bg-muted/40" />
   890→            <div className="h-12 animate-pulse rounded-lg bg-muted/40" />
   891→          </div>
   892→        ) : !located || !re || !day ? (
   893→          <div className="flex h-[200px] flex-col items-center justify-center gap-3 rounded-lg border border-dashed border-border">
   894→            <p className="text-sm font-semibold">Exercise not found in this program</p>
   895→            <Button
   896→              type="button"
   897→              variant="outline"
   898→              tour={{ skipTour: true, reason: "Error-state back link for a missing exercise" }}
   899→              onClick={() => navigate(backHref)}
   900→            >
   901→              Back to editor
   902→            </Button>
   903→          </div>
   904→        ) : (
   905→          <>
   906→            {/* ---------- header row 32px (# · f1 · f2 · rest) ---------- */}
   907→            <div
   908→              data-row
   909→              role="group"
   910→              aria-label="Set columns"

   913→            >

   952→                onClick={() => void addSet()}

   954→              >
   955→                <Plus className="h-4 w-4 flex-none" aria-hidden />

   963→                    aria-label="Apply to all sets"
   964→                    className="flex h-10 min-w-0 flex-1 items-center justify-center gap-1 overflow-hidden whitespace-nowrap rounded-lg border border-dashed border-border px-3 text-sm font-medium text-muted-foreground transition-colors hover:border-primary/40 hover:bg-accent/40 hover:text-foreground"

   998→              />
   999→              <span className="min-w-0 flex-1 truncate text-sm font-semibold leading-none">Rest: none</span>
