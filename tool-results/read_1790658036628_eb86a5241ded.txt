     1→"use client";
     2→
     3→// ─────────────────────────────────────────────────────────────────────────────
     4→// OnboardingScreen — #/onboarding (Part 6 §4.16). The 6-step welcome wizard.
     5→//
     6→//   TopBar (56)  : "Welcome to SetForge" + [Skip] ghost h-11 (always available)
     7→//   Progress     : 4px primary bar under the TopBar — step N of 6
     8→//   ScrollBody   : one step at a time (subtle slide/fade transition)
     9→//                  1 Welcome · 2 Units · 3 Goal · 4 Level · 5 Schedule+body ·
    10→//                  6 Review (choice summary + "change" jump links)
    11→//   BottomBar    : [Back] ghost + [Get started / Continue / Start training]
    12→//
    13→// Gate: the app-shell redirects here while profile.onboardingCompletedAt is
    14→// null. Completion writes through profileApi.completeOnboarding (profile +
    15→// unitSystem setting + first Body-Weight record), then the ["profile"] cache
    16→// is updated synchronously (setQueryData) BEFORE navigating so the shell gate
    17→// never bounces the user back.
    18→//
    19→// Units: heights/weights are METRIC canonically on the wire. When Imperial is
    20→// selected in step 2 the step-5 inputs are inches/pounds and convert live
    21→// (heightCm = in × 2.54 · weightKg = lb × 0.45359237) with the converted
    22→// metric value shown as a hint under each input.
    23→//
    24→// No Dialogs; step 5 fields are optional (empty = skip) but must be valid
    25→// (age 13–99 · height 100–250 cm / 39–98 in · weight 30–300 kg / 66–660 lb).
    26→// ─────────────────────────────────────────────────────────────────────────────
    27→
    28→import { useMemo, useState } from "react";
    29→import { useQueryClient } from "@tanstack/react-query";
    30→import { Screen, TopBar, ScrollBody, BottomBar } from "@/components/layout";
    31→import { Button } from "@/components/ui/button";
    32→import { Input } from "@/components/ui/input";
    33→import {
    34→  Check,
    35→  Dumbbell,
    36→  Flame,
    37→  Heart,
    38→  Loader2,
    39→  Minus,
    40→  Plus,
    41→  SkipForward,
    42→  Sprout,
    43→  TrendingUp,
    44→  Trophy,
    45→  Zap,
    46→} from "lucide-react";
    47→import { toast } from "sonner";
    48→import { profileApi } from "@/lib/client/api";
    49→import { hapticSelection, hapticSuccess, hapticTap } from "@/lib/client/haptics";
    50→import { useApp } from "@/lib/client/store";
    51→import { PROFILE_GOALS, PROFILE_GOAL_LABELS, PROFILE_LEVELS } from "@/lib/constants";
    52→import type { UserProfileDTO } from "@/lib/types";
    53→import { replaceHash } from "@/features/shell/router";
    54→import { cn } from "@/lib/utils";
    55→
    56→// ── constants ────────────────────────────────────────────────────────────────
    57→
    58→const TOTAL_STEPS = 6;
    59→const STEP_INDEX = { welcome: 0, units: 1, goal: 2, level: 3, body: 4, review: 5 } as const;
    60→
    61→const CM_PER_IN = 2.54;
    62→const KG_PER_LB = 0.45359237;
    63→
    64→type UnitSystem = "metric" | "imperial";
    65→type Goal = (typeof PROFILE_GOALS)[number];
    66→type Level = (typeof PROFILE_LEVELS)[number];
    67→
    68→const GOAL_ICONS: Record<Goal, typeof Zap> = {
    69→  STRENGTH: Zap,
    70→  MUSCLE: Dumbbell,
    71→  FAT_LOSS: Flame,
    72→  GENERAL: Heart,
    73→};
    74→
    75→const LEVEL_META: Array<{ value: Level; label: string; hint: string; icon: typeof Sprout }> = [
    76→  { value: "BEGINNER", label: "Beginner", hint: "New to training", icon: Sprout },
    77→  { value: "INTERMEDIATE", label: "Intermediate", hint: "6 months – 2 years", icon: TrendingUp },
    78→  { value: "ADVANCED", label: "Advanced", hint: "2+ years", icon: Trophy },
    79→];
    80→
    81→const LEVEL_LABELS: Record<Level, string> = {
    82→  BEGINNER: "Beginner",
    83→  INTERMEDIATE: "Intermediate",
    84→  ADVANCED: "Advanced",
    85→};
    86→
    87→// Validation bands (client-side; the server accepts a slightly wider range).
    88→const AGE_MIN = 13;
    89→const AGE_MAX = 99;
    90→const HEIGHT_CM_MIN = 100;
    91→const HEIGHT_CM_MAX = 250;
    92→const HEIGHT_IN_MIN = 39;
    93→const HEIGHT_IN_MAX = 98;
    94→const WEIGHT_KG_MIN = 30;
    95→const WEIGHT_KG_MAX = 300;
    96→const WEIGHT_LB_MIN = 66;
    97→const WEIGHT_LB_MAX = 660;
    98→
    99→const roundTo1 = (n: number): number => Math.round(n * 10) / 10;
   100→
   101→// ── screen ───────────────────────────────────────────────────────────────────
   102→
   103→export default function OnboardingScreen() {
   104→  const qc = useQueryClient();
   105→  const updateSettings = useApp((s) => s.updateSettings);
   106→
   107→  const [step, setStep] = useState<number>(STEP_INDEX.welcome);
   108→  const [saving, setSaving] = useState(false);
   109→
   110→  // draft state (local only — nothing persists before the final step)
   111→  const [unit, setUnit] = useState<UnitSystem>("metric");
   112→  const [goal, setGoal] = useState<Goal | null>(null);
   113→  const [level, setLevel] = useState<Level | null>(null);
   114→  const [days, setDays] = useState(3);
   115→  const [heightInput, setHeightInput] = useState(""); // cm (metric) | inches (imperial)
   116→  const [weightInput, setWeightInput] = useState(""); // kg (metric) | pounds (imperial)
   117→  const [ageInput, setAgeInput] = useState("");
   118→
   119→  const imperial = unit === "imperial";
   120→  const stepNumber = step + 1;
   121→
   122→  // ---------- step 5 field parsing / validation ----------
   123→
   124→  const height = useMemo(() => parseNumber(heightInput), [heightInput]);
   125→  const weight = useMemo(() => parseNumber(weightInput), [weightInput]);
   126→  const age = useMemo(() => parseNumber(ageInput), [ageInput]);
   127→
   128→  const heightCm = height == null ? null : imperial ? height * CM_PER_IN : height;
   129→  const weightKg = weight == null ? null : imperial ? weight * KG_PER_LB : weight;
   130→
   131→  const heightError = height == null ? null : imperial
   132→    ? height < HEIGHT_IN_MIN || height > HEIGHT_IN_MAX ? `Enter ${HEIGHT_IN_MIN}–${HEIGHT_IN_MAX} inches` : null
   133→    : height < HEIGHT_CM_MIN || height > HEIGHT_CM_MAX ? `Enter ${HEIGHT_CM_MIN}–${HEIGHT_CM_MAX} cm` : null;
   134→  const weightError = weight == null ? null : imperial
   135→    ? weight < WEIGHT_LB_MIN || weight > WEIGHT_LB_MAX ? `Enter ${WEIGHT_LB_MIN}–${WEIGHT_LB_MAX} lb` : null
   136→    : weight < WEIGHT_KG_MIN || weight > WEIGHT_KG_MAX ? `Enter ${WEIGHT_KG_MIN}–${WEIGHT_KG_MAX} kg` : null;
   137→  const ageError = age == null ? null
   138→    : !Number.isInteger(age) || age < AGE_MIN || age > AGE_MAX ? `Enter ${AGE_MIN}–${AGE_MAX} years` : null;
   139→
   140→  /** Step 5 is the only gate: fields are optional, but non-empty ones must be valid. */
   141→  const step5Valid = heightError == null && weightError == null && ageError == null;
   142→  const nextDisabled = step === STEP_INDEX.body && !step5Valid;
   143→
   144→  // ---------- navigation ----------
   145→
   146→  const goTo = (n: number) => {
   147→    setStep(Math.min(Math.max(n, STEP_INDEX.welcome), STEP_INDEX.review));
   148→    hapticTap();
   149→  };
   150→
   151→  const next = () => {
   152→    if (nextDisabled || saving) return;
   153→    if (step === STEP_INDEX.review) {
   154→      void finish();
   155→    } else {
   156→      goTo(step + 1);
   157→    }
   158→  };
   159→
   160→  // ---------- completion ----------
   161→
   162→  /** Update the ["profile"] cache synchronously so the shell gate lifts instantly. */
   163→  const applyProfile = (dto: UserProfileDTO) => {
   164→    qc.setQueryData(["profile"], dto);
   165→    void qc.invalidateQueries({ queryKey: ["profile"] });
   166→  };
   167→
   168→  const finish = async () => {
   169→    setSaving(true);
   170→    try {
   171→      const dto = await profileApi.completeOnboarding({
   172→        unitSystem: unit,
   173→        goal,
   174→        level,
   175→        daysPerWeekTarget: days,
   176→        ...(heightCm != null ? { heightCm: roundTo1(heightCm) } : {}),
   177→        ...(weightKg != null ? { weightKg: roundTo1(weightKg) } : {}),
   178→        ...(age != null ? { age } : {}),
   179→      });
   180→      applyProfile(dto);
   181→      // The service persisted unitSystem, but the client settings/session in
   182→      // the store still carry the signup default — sync them so every unit-aware
   183→      // screen (weights, plates, profile) shows the chosen system without a
   184→      // reload. Idempotent PATCH; failure is non-fatal (reload would resync).
   185→      void updateSettings({ unitSystem: unit }).catch(() => undefined);
   186→      hapticSuccess();
   187→      toast.success("Welcome to SetForge");
   188→      replaceHash("#/home");
   189→    } catch (e) {
   190→      toast.error(e instanceof Error ? e.message : "Could not finish setup — try again");
   191→    } finally {
   192→      setSaving(false);
   193→    }
   194→  };
   195→
   196→  const skip = async () => {
   197→    if (saving) return;
   198→    setSaving(true);
   199→    try {
   200→      const dto = await profileApi.completeOnboarding({ skipped: true });
   201→      applyProfile(dto);
   202→      toast.info("Setup skipped — finish any time from Profile");
   203→      replaceHash("#/home");
   204→    } catch (e) {
   205→      toast.error(e instanceof Error ? e.message : "Could not skip setup — try again");
   206→    } finally {
   207→      setSaving(false);
   208→    }
   209→  };
   210→
   211→  const nextLabel =
   212→    step === STEP_INDEX.welcome ? "Get started"
   213→    : step === STEP_INDEX.review ? "Start training"
   214→    : "Continue";
   215→
   216→  // ---------- render ----------
   217→
   218→  return (
   219→    <Screen
   220→      nav={false}
   221→      topBar={
   222→        <TopBar
   223→          title="Welcome to SetForge"
   224→          actions={
   225→            <Button
   226→              type="button"
   227→              variant="ghost"
   228→              className="h-11 gap-1 px-3 text-xs"
   229→              onClick={() => void skip()}
   230→              disabled={saving}
   231→            >
   232→              <SkipForward className="h-4 w-4" aria-hidden /> Skip
   233→            </Button>
   234→          }
   235→        />
   236→      }
   237→      bottomBar={
   238→        <BottomBar>
   239→          {step > STEP_INDEX.welcome ? (
   240→            <Button
   241→              type="button"
   242→              variant="ghost"
   243→              className="h-11 flex-none px-4"
   244→              onClick={() => goTo(step - 1)}
   245→              disabled={saving}
   246→            >
   247→              Back
   248→            </Button>
   249→          ) : null}
   250→          <Button
   251→            type="button"
   252→            className="h-12 min-w-0 flex-1 text-base font-semibold"
   253→            onClick={next}
   254→            disabled={nextDisabled || saving}
   255→          >
   256→            {saving && step === STEP_INDEX.review ? (
   257→              <Loader2 className="h-5 w-5 animate-spin" aria-hidden />
   258→            ) : null}
   259→            {nextLabel}
   260→          </Button>
   261→        </BottomBar>
   262→      }
   263→    >
   264→      {/* 4px progress bar — step N of 6 */}
   265→      <div
   266→        className="h-1 flex-none overflow-hidden bg-primary/10"
   267→        role="progressbar"
   268→        aria-label="Onboarding progress"
   269→        aria-valuemin={1}
   270→        aria-valuemax={TOTAL_STEPS}
   271→        aria-valuenow={stepNumber}
   272→      >
   273→        <div
   274→          className="h-full bg-primary transition-all duration-300 ease-out"
   275→          style={{ width: `${(stepNumber / TOTAL_STEPS) * 100}%` }}
   276→        />
   277→      </div>
   278→
   279→      <ScrollBody contentClassName="flex flex-col gap-5 py-6">
   280→        <p className="flex-none text-right text-xs font-medium tabular-nums text-muted-foreground">
   281→          Step {stepNumber} of {TOTAL_STEPS}
   282→        </p>
   283→
   284→        <div key={step} className="flex flex-col gap-5 duration-300 animate-in fade-in slide-in-from-bottom-2">
   285→          {step === STEP_INDEX.welcome ? (
   286→            <WelcomeStep />
   287→          ) : null}
   288→
   289→          {step === STEP_INDEX.units ? (
   290→            <section aria-label="Units" className="flex flex-col gap-2">
   291→              <StepHeading title="Choose your units" subtitle="You can change this any time in Settings." />
   292→              <div className="grid grid-cols-2 gap-3">
   293→                {(["metric", "imperial"] as const).map((u) => (
   294→                  <SelectTile
   295→                    key={u}
   296→                    selected={unit === u}
   297→                    onPress={() => {
   298→                      setUnit(u);
   299→                      hapticSelection();
   300→                    }}
   301→                    ariaLabel={`Units: ${u === "metric" ? "Metric — kilograms and centimetres" : "Imperial — pounds and inches"}`}
   302→                    className="h-24 flex-col justify-center gap-1"
   303→                  >
   304→                    <span className="flex w-full items-center justify-between gap-2">
   305→                      <span className="text-lg font-bold leading-none">{u === "metric" ? "Metric" : "Imperial"}</span>
   306→                      <TileCheck selected={unit === u} />
   307→                    </span>
   308→                    <span className="text-sm text-muted-foreground">{u === "metric" ? "kg · cm" : "lb · in"}</span>
   309→                  </SelectTile>
   310→                ))}
   311→              </div>
   312→            </section>
   313→          ) : null}
   314→
   315→          {step === STEP_INDEX.goal ? (
   316→            <section aria-label="Goal" className="flex flex-col gap-2">
   317→              <StepHeading title="What's your main goal?" subtitle="This tunes your dashboard and suggestions." />
   318→              <div className="flex flex-col gap-2">
   319→                {PROFILE_GOALS.map((g) => {
   320→                  const Icon = GOAL_ICONS[g];
   321→                  return (
   322→                    <SelectTile
   323→                      key={g}
   324→                      selected={goal === g}
   325→                      onPress={() => {
   326→                        setGoal(g);
   327→                        hapticSelection();
   328→                      }}
   329→                      ariaLabel={`Goal: ${PROFILE_GOAL_LABELS[g]}`}
   330→                      className="h-16 items-center gap-3"
   331→                    >
   332→                      <span className="flex h-10 w-10 flex-none items-center justify-center rounded-lg bg-primary/10 text-primary">
   333→                        <Icon className="h-5 w-5" aria-hidden />
   334→                      </span>
   335→                      <span className="min-w-0 flex-1 truncate text-base font-semibold leading-none">
   336→                        {PROFILE_GOAL_LABELS[g]}
   337→                      </span>
   338→                      <TileCheck selected={goal === g} />
   339→                    </SelectTile>
   340→                  );
   341→                })}
   342→              </div>
   343→            </section>
   344→          ) : null}
   345→
   346→          {step === STEP_INDEX.level ? (
   347→            <section aria-label="Experience level" className="flex flex-col gap-2">
   348→              <StepHeading title="How much training experience?" subtitle="Be honest — you can change this later." />
   349→              <div className="grid grid-cols-3 gap-2">
   350→                {LEVEL_META.map((l) => {
   351→                  const Icon = l.icon;
   352→                  return (
   353→                    <SelectTile
   354→                      key={l.value}
   355→                      selected={level === l.value}
   356→                      onPress={() => {
   357→                        setLevel(l.value);
   358→                        hapticSelection();
   359→                      }}
   360→                      ariaLabel={`Level: ${l.label}`}
   361→                      className="h-24 flex-col items-center justify-center gap-1 text-center"
   362→                    >
   363→                      <TileCheck selected={level === l.value} className="absolute right-2 top-2" />
   364→                      <Icon className="h-6 w-6 flex-none text-primary" aria-hidden />
   365→                      <span className="text-sm font-bold leading-none">{l.label}</span>
   366→                      <span className="text-[11px] leading-tight text-muted-foreground">{l.hint}</span>
   367→                    </SelectTile>
   368→                  );
   369→                })}
   370→              </div>
   371→            </section>
   372→          ) : null}
   373→
   374→          {step === STEP_INDEX.body ? (
   375→            <BodyStep
   376→              imperial={imperial}
   377→              days={days}
   378→              setDays={(n) => {
   379→                setDays(n);
   380→                hapticTap();
   381→              }}
   382→              heightInput={heightInput}
   383→              setHeightInput={setHeightInput}
   384→              heightError={heightError}
   385→              heightCm={heightCm}
   386→              weightInput={weightInput}
   387→              setWeightInput={setWeightInput}
   388→              weightError={weightError}
   389→              weightKg={weightKg}
   390→              ageInput={ageInput}
   391→              setAgeInput={setAgeInput}
   392→              ageError={ageError}
   393→            />
   394→          ) : null}
   395→
   396→          {step === STEP_INDEX.review ? (
   397→            <ReviewStep
   398→              unit={unit}
   399→              goal={goal}
   400→              level={level}
   401→              days={days}
   402→              heightCm={heightCm}
   403→              weightKg={weightKg}
   404→              age={age}
   405→              onChangeStep={(n) => goTo(n)}
   406→            />
   407→          ) : null}
   408→        </div>
   409→      </ScrollBody>
   410→    </Screen>
   411→  );
   412→}

   440→function BodyStep({
   441→  imperial,
   442→  days,
   443→  setDays,
   444→  heightInput,
   445→  setHeightInput,
   446→  heightError,
   447→  heightCm,
   448→  weightInput,
   449→  setWeightInput,
   450→  weightError,
   451→  weightKg,
   452→  ageInput,
   453→  setAgeInput,
   454→  ageError,
   455→}: {
   456→  imperial: boolean;
   457→  days: number;
   458→  setDays: (n: number) => void;
   459→  heightInput: string;
   460→  setHeightInput: (v: string) => void;
   461→  heightError: string | null;
   462→  heightCm: number | null;
   463→  weightInput: string;
   464→  setWeightInput: (v: string) => void;
   465→  weightError: string | null;
   466→  weightKg: number | null;
   467→  ageInput: string;
   468→  setAgeInput: (v: string) => void;
   469→  ageError: string | null;
   470→}) {
   471→  return (
   472→    <section aria-label="Schedule and body" className="flex flex-col gap-4">
   473→      <StepHeading title="Schedule & body" subtitle="Optional — leave anything blank to skip it for now." />
   474→
   475→      {/* days per week — stepper */}
   476→      <div className="flex flex-col gap-1.5">
   477→        <p className="px-1 text-sm font-semibold">Training days per week</p>
   478→        <div className="flex h-12 items-center justify-between rounded-lg border bg-card px-2">
   479→          <span className="flex-none pl-1 text-xs text-muted-foreground">Days</span>
   480→          <span className="flex items-center gap-1">
   481→            <Button
   482→              type="button"
   483→              variant="outline"
   484→              className="h-12 w-12 p-0"
   485→              disabled={days <= 1}
   486→              aria-label="Decrease training days per week"
   487→              onClick={() => setDays(days - 1)}
   488→            >
   489→              <Minus className="h-5 w-5" aria-hidden />
   490→            </Button>
   491→            <span
   492→              className="flex h-12 w-14 items-center justify-center text-base font-bold tabular-nums leading-none"
   493→              aria-live="polite"
   494→              aria-label={`Training days per week: ${days}`}
   495→            >
   496→              {days}
   497→            </span>
   498→            <Button
   499→              type="button"
   500→              variant="outline"
   501→              className="h-12 w-12 p-0"
   502→              disabled={days >= 7}
   503→              aria-label="Increase training days per week"
   504→              onClick={() => setDays(days + 1)}
   505→            >
   506→              <Plus className="h-5 w-5" aria-hidden />
   507→            </Button>
   508→          </span>
   509→        </div>
   510→      </div>
   511→
   512→      <NumberField
   513→        id="onboarding-height"
   514→        label="Height"
   515→        unitLabel={imperial ? "inches" : "cm"}
   516→        inputMode="decimal"
   517→        value={heightInput}
   518→        onChange={setHeightInput}
   519→        error={heightError}
   520→        hint={imperial && heightCm != null ? `≈ ${roundTo1(heightCm)} cm` : null}
   521→        placeholder={imperial ? "e.g. 70" : "e.g. 178"}
   522→      />
   523→      <NumberField
   524→        id="onboarding-weight"
   525→        label="Weight"
   526→        unitLabel={imperial ? "pounds" : "kg"}
   527→        inputMode="decimal"
   528→        value={weightInput}
   529→        onChange={setWeightInput}
   530→        error={weightError}
   531→        hint={imperial && weightKg != null ? `≈ ${roundTo1(weightKg)} kg` : null}
   532→        placeholder={imperial ? "e.g. 180" : "e.g. 80"}
   533→      />
   534→      <NumberField
   535→        id="onboarding-age"
   536→        label="Age"
   537→        unitLabel="years"
   538→        inputMode="numeric"
   539→        value={ageInput}
   540→        onChange={setAgeInput}
   541→        error={ageError}
   542→        placeholder="e.g. 30"
   543→      />
   544→    </section>
   545→  );
   546→}

   548→// ── step 6 — review ──────────────────────────────────────────────────────────
   549→
   550→function ReviewStep({
   551→  unit,
   552→  goal,
   553→  level,
   554→  days,
   555→  heightCm,
   556→  weightKg,
   557→  age,
   558→  onChangeStep,
   559→}: {
   560→  unit: UnitSystem;
   561→  goal: Goal | null;
   562→  level: Level | null;
   563→  days: number;
   564→  heightCm: number | null;
   565→  weightKg: number | null;
   566→  age: number | null;
   567→  onChangeStep: (step: number) => void;
   568→}) {
   569→  const imperial = unit === "imperial";
   570→  const heightLabel =
   571→    heightCm == null ? null
   572→    : imperial ? `${roundTo1(heightCm / CM_PER_IN)} in (${roundTo1(heightCm)} cm)`
   573→    : `${roundTo1(heightCm)} cm`;
   574→  const weightLabel =
   575→    weightKg == null ? null
   576→    : imperial ? `${roundTo1(weightKg / KG_PER_LB)} lb (${roundTo1(weightKg)} kg)`
   577→    : `${roundTo1(weightKg)} kg`;
   578→
   579→  const rows: Array<{ label: string; value: string; step: number }> = [
   580→    { label: "Units", value: imperial ? "Imperial (lb · in)" : "Metric (kg · cm)", step: STEP_INDEX.units },
   581→    { label: "Goal", value: goal ? PROFILE_GOAL_LABELS[goal] : "Not set", step: STEP_INDEX.goal },
   582→    { label: "Experience", value: level ? LEVEL_LABELS[level] : "Not set", step: STEP_INDEX.level },
   583→    { label: "Days per week", value: String(days), step: STEP_INDEX.body },
   584→    { label: "Height", value: heightLabel ?? "Skipped", step: STEP_INDEX.body },
   585→    { label: "Weight", value: weightLabel ?? "Skipped", step: STEP_INDEX.body },
   586→    { label: "Age", value: age != null ? `${age} years` : "Skipped", step: STEP_INDEX.body },
   587→  ];
   588→
   589→  return (
   590→    <section aria-label="Review" className="flex flex-col gap-2">
   591→      <StepHeading title="Ready to forge." subtitle="Check your setup — tap change to jump back." />
   592→      <div className="flex flex-col gap-2">
   593→        {rows.map((r) => (
   594→          <div
   595→            key={r.label}
   596→            data-row
   597→            className="flex h-14 items-center gap-2 overflow-hidden whitespace-nowrap rounded-lg border bg-card px-3"
   598→          >
   599→            <span className="min-w-0 flex-1 truncate text-sm font-medium">{r.label}</span>
   600→            <span className="max-w-[55%] flex-none truncate text-sm tabular-nums text-muted-foreground">
   601→              {r.value}
   602→            </span>
   603→            <button
   604→              type="button"
   605→              onClick={() => onChangeStep(r.step)}
   606→              className="h-11 flex-none rounded-md px-2 text-xs font-semibold text-primary hover:bg-primary/10 focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   607→            >
   608→              change
   609→            </button>
   610→          </div>
   611→        ))}
   612→      </div>
   613→    </section>
   614→  );
   615→}

   728→/** "" → null; otherwise the finite number value (NaN → null via band checks). */
   729→function parseNumber(raw: string): number | null {
   730→  const trimmed = raw.trim();
   731→  if (trimmed === "") return null;
   732→  const n = Number(trimmed);
   733→  return Number.isFinite(n) ? n : null;
   734→}
