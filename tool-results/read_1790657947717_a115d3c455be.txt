   133→function FacetChipRow({
   134→  options,
   135→  selected,
   136→  labelOf,
   137→  onToggle,
   138→  span,
   139→  ariaLabel,
   140→}: {
   141→  options: string[];
   142→  selected: Set<string>;
   143→  labelOf: (v: string) => string;
   144→  onToggle: (v: string) => void;
   145→  span?: boolean;
   146→  ariaLabel: string;
   147→}) {
   148→  if (options.length === 0) return null;
   149→  return (
   150→    <div
   151→      data-row
   152→      data-chip-scroller
   153→      role="group"
   154→      aria-label={ariaLabel}
   155→      className={cn(CHIP_ROW_CLS, span && "lg:col-span-2")}
   156→    >
   157→      {options.map((v) => {
   158→        const active = selected.has(v);
   159→        return (
   160→          <button
   161→            key={v}
   162→            type="button"
   163→            aria-pressed={active}
   164→            className={chipClass(active)}
   165→            onClick={() => onToggle(v)}
   166→          >
   167→            {active ? <Check className="h-3.5 w-3.5" aria-hidden /> : null}
   168→            {labelOf(v)}
   169→          </button>
   170→        );
   171→      })}
   172→    </div>
   173→  );
   174→}
