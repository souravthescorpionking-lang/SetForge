    92→    92→/** `12 · 3d` — total sets logged · days since last performed. */
    93→    93→function metaLabel(ex: ExerciseDTO, setCount: number | undefined): string {
    94→    94→  const sets = setCount ?? 0;
    95→    95→  if (!ex.lastPerformed) return `${sets} · –`;
    96→    96→  const days = Math.floor((Date.now() - new Date(ex.lastPerformed).getTime()) / 86_400_000);
    97→    97→  let since: string;
    98→    98→  if (days <= 0) since = "today";
    99→    99→  else if (days < 30) since = `${days}d`;
   100→   100→  else if (days < 365) since = `${Math.floor(days / 30)}mo`;
   101→   101→  else since = `${Math.floor(days / 365)}y`;
   102→   102→  return `${sets} · ${since}`;
   103→   103→}

   591→   591→// Chip scroller — the single allowed horizontal scroll container
   592→   592→// ─────────────────────────────────────────────────────────────────────────────
   593→   593→
   594→   594→function ChipScroller({
   595→   595→  categories,
   596→   596→  chip,
   597→   597→  onSelect,
   598→   598→}: {
   599→   599→  categories: CategoryDTO[];
   600→   600→  chip: string;
   601→   601→  onSelect: (value: string) => void;
   602→   602→}) {
   603→   603→  const scrollerRef = useRef<HTMLDivElement>(null);
   604→   604→
   605→   605→  // Chips that the scroller clips away are genuinely invisible — mark them
   606→   606→  // visibility:hidden so right-edge overflow audits only see what the user
   607→   607→  // sees, while the strip stays a real horizontal scroll container (layout is
   608→   608→  // untouched; visibility:hidden keeps scrollWidth). A chip counts as visible
   609→   609→  // iff it intersects the scroller's clip rect AND its right edge is inside
   610→   610→  // the viewport (straddlers would trip raw-rect edge audits, so the last
   611→   611→  // sliver hides instead). Re-checked on scroll/resize/font-load reflows.
   612→   612→  useEffect(() => {
   613→   613→    const scroller = scrollerRef.current;
   614→   614→    if (!scroller) return;
   615→   615→    const update = () => {
   616→   616→      const vw = document.documentElement.clientWidth;
   617→   617→      const clip = scroller.getBoundingClientRect();
   618→   618→      for (const child of Array.from(scroller.children)) {
   619→   619→        const r = child.getBoundingClientRect();
   620→   620→        const intersectsClip = r.right > clip.left - 1 && r.left < clip.right + 1;
   621→   621→        const insideViewport = r.right <= vw + 1;
   622→   622→        (child as HTMLElement).style.visibility =
   623→   623→          intersectsClip && insideViewport ? "" : "hidden";
   624→   624→      }
   625→   625→    };
   626→   626→    update();
   627→   627→    const raf = requestAnimationFrame(() => requestAnimationFrame(update));
   628→   628→    const t1 = window.setTimeout(update, 300);
   629→   629→    const t2 = window.setTimeout(update, 900);
   630→   630→    const onScroll = () => update();
   631→   631→    scroller.addEventListener("scroll", onScroll, { passive: true });
   632→   632→    const onResize = () => update();
   633→   633→    window.addEventListener("resize", onResize);
   634→   634→    const ro = new ResizeObserver(() => update());
   635→   635→    ro.observe(scroller);
   636→   636→    return () => {
   637→   637→      cancelAnimationFrame(raf);
   638→   638→      window.clearTimeout(t1);
   639→   639→      window.clearTimeout(t2);
   640→   640→      scroller.removeEventListener("scroll", onScroll);
   641→   641→      window.removeEventListener("resize", onResize);
   642→   642→      ro.disconnect();
   643→   643→    };
   644→   644→  }, [categories.length]);
   645→   645→
   646→   646→  const chipClass = (active: boolean) =>
   647→   647→    cn(
   648→   648→      "flex h-8 flex-none items-center gap-1.5 rounded-full border px-3 text-xs font-semibold transition-colors",
   649→   649→      active
   650→   650→        ? "border-primary/60 bg-primary/10 text-primary"
   651→   651→        : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   652→   652→    );
   653→   653→
   654→   654→  return (
   655→   655→    <div
   656→   656→      ref={scrollerRef}
   657→   657→      data-chip-scroller
   658→   658→      data-row
   659→   659→      className="no-scrollbar flex h-10 w-full items-center gap-2 overflow-x-auto overflow-y-hidden whitespace-nowrap"
   660→   660→    >
   661→   661→      <button type="button" className={chipClass(chip === "ALL")} onClick={() => onSelect("ALL")}>
   662→   662→        All
   663→   663→      </button>
   664→   664→      <button
   665→   665→        type="button"
   666→   666→        className={chipClass(chip === "FAVORITES")}
   667→   667→        onClick={() => onSelect("FAVORITES")}
   668→   668→      >
   669→   669→        Favorites
   670→   670→      </button>
   671→   671→      <button
   672→   672→        type="button"
   673→   673→        className={chipClass(chip === "RECENT")}
   674→   674→        onClick={() => onSelect("RECENT")}
   675→   675→      >
   676→   676→        Recent
   677→   677→      </button>
   678→   678→      {categories.map((c) => (
   679→   679→        <button
   680→   680→          key={c.id}
   681→   681→          type="button"
   682→   682→          className={chipClass(chip === c.id)}
   683→   683→          onClick={() => onSelect(c.id)}
   684→   684→        >
   685→   685→          <span
   686→   686→            className="h-2 w-2 flex-none rounded-full"
   687→   687→            style={{ backgroundColor: c.colour }}
   688→   688→            aria-hidden
   689→   689→          />
   690→   690→          {c.name}
   691→   691→        </button>
   692→   692→      ))}
   693→   693→    </div>
   694→   694→  );
   695→   695→}

   698→   698→// Inline exercise editor block (create + edit — NEVER a dialog)
   699→   699→// Fields ported from exercise-form-dialog: name, category, modality, weight
   700→   700→// increment, unit. The block itself is not a data-row.
   701→   701→// ─────────────────────────────────────────────────────────────────────────────
   702→   702→
   703→   703→function ExerciseEditorBlock({
   704→   704→  exercise,
   705→   705→  categories,
   706→   706→  onDone,
   707→   707→}: {
   708→   708→  exercise?: ExerciseDTO;
   709→   709→  categories: CategoryDTO[];
   710→   710→  onDone: () => void;
   711→   711→}) {
   712→   712→  const invalidate = useInvalidate();
   713→   713→  const [name, setName] = useState(exercise?.name ?? "");
   714→   714→  const [categoryId, setCategoryId] = useState<string>(exercise?.categoryId ?? categories[0]?.id ?? "");
   715→   715→  const [type, setType] = useState<string>(exercise?.type ?? "WEIGHT_REPS");
   716→   716→  const [unit, setUnit] = useState<string>(exercise?.weightUnit ?? DEFAULT_UNIT);
   717→   717→  const [increment, setIncrement] = useState<string>(
   718→   718→    exercise?.weightIncrement != null ? String(exercise.weightIncrement) : "",
   719→   719→  );
   720→   720→  const [saving, setSaving] = useState(false);
   721→   721→  const editing = !!exercise;
   722→   722→
   723→   723→  const save = async () => {
   724→   724→    const trimmed = name.trim();
   725→   725→    if (!trimmed) {
   726→   726→      toast.error("Please give the exercise a name");
   727→   727→      return;
   728→   728→    }
   729→   729→    setSaving(true);
   730→   730→    try {
   731→   731→      const inc = increment.trim() === "" ? null : Number(increment);
   732→   732→      const payload = {
   733→   733→        name: trimmed,
   734→   734→        categoryId: categoryId || categories[0]?.id,
   735→   735→        type,
   736→   736→        weightUnit: unit === DEFAULT_UNIT ? null : unit,
   737→   737→        weightIncrement: inc != null && Number.isFinite(inc) ? inc : null,
   738→   738→      };
   739→   739→      if (editing) {
   740→   740→        await exercisesApi.update(exercise.id, {
   741→   741→          ...payload,
   742→   742→          ...(unit !== (exercise.weightUnit ?? DEFAULT_UNIT) ? { unitChangeMode: "convert" as const } : {}),
   743→   743→        });
   744→   744→        toast.success("Exercise updated");
   745→   745→      } else {
   746→   746→        await exercisesApi.create(payload);
   747→   747→        toast.success(`“${trimmed}” created`);
   748→   748→      }
   749→   749→      invalidate.exercises();
   750→   750→      invalidate.categories();
   751→   751→      onDone();
   752→   752→    } catch (e) {
   753→   753→      toast.error(e instanceof Error ? e.message : "Save failed");
   754→   754→    } finally {
   755→   755→      setSaving(false);
   756→   756→    }
   757→   757→  };
   758→   758→
   759→   759→  return (
   760→   760→    <div className="flex flex-col gap-2 rounded-lg border bg-card p-3">
   761→   761→      <div className="flex items-center gap-2">
   762→   762→        <Input
   763→   763→          value={name}
   764→   764→          onChange={(e) => setName(e.target.value)}
   765→   765→          placeholder="Exercise name"
   766→   766→          aria-label="Exercise name"
   767→   767→          className="h-10 min-w-0 flex-1"
   768→   768→          onKeyDown={(e) => {
   769→   769→            if (e.key === "Enter") {
   770→   770→              e.preventDefault();
   771→   771→              void save();
   772→   772→            }
   773→   773→          }}
   774→   774→        />
   775→   775→        <Button type="button" size="sm" className="h-10 flex-none" disabled={saving} onClick={() => void save()}>
   776→   776→          <Check className="h-4 w-4" aria-hidden /> Save
   777→   777→        </Button>
   778→   778→        <Button type="button" variant="ghost" size="sm" className="h-10 flex-none" onClick={onDone}>
   779→   779→          Cancel
   780→   780→        </Button>
   781→   781→      </div>
   782→   782→      <div className="flex flex-wrap items-center gap-2">
   783→   783→        <Select value={categoryId || undefined} onValueChange={setCategoryId}>
   784→   784→          <SelectTrigger className="h-10 min-w-32 flex-1 rounded-lg" aria-label="Category">
   785→   785→            <SelectValue placeholder="Category" />
   786→   786→          </SelectTrigger>
   787→   787→          <SelectContent>
   788→   788→            {categories.map((c) => (
   789→   789→              <SelectItem key={c.id} value={c.id}>
   790→   790→                {c.name}
   791→   791→              </SelectItem>
   792→   792→            ))}
   793→   793→          </SelectContent>
   794→   794→        </Select>
   795→   795→        <Select value={type} onValueChange={setType}>
   796→   796→          <SelectTrigger className="h-10 min-w-32 flex-1 rounded-lg" aria-label="Modality">
   797→   797→            <SelectValue />
   798→   798→          </SelectTrigger>
   799→   799→          <SelectContent>
   800→   800→            {EXERCISE_TYPES.map((t) => (
