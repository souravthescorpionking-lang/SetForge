    83→  return (
    84→    <div
    85→      data-row
    86→      role="group"
    87→      aria-label={`Step ${step}: ${title}`}
    88→      className="flex h-8 w-full flex-none items-center gap-2 overflow-hidden whitespace-nowrap px-1"
    89→    >
    90→      <span className="flex h-6 w-6 flex-none items-center justify-center rounded bg-primary/10 text-[11px] font-bold tabular-nums text-primary">
    91→        {step}
    92→      </span>
    93→      <span className="min-w-0 flex-1 truncate text-xs font-bold uppercase tracking-wider text-muted-foreground">
    94→        {title}
    95→      </span>
    96→      {hint ? (
    97→        <span className="flex-none text-[11px] font-medium text-muted-foreground/70">{hint}</span>
    98→      ) : null}
    99→    </div>
   100→  );
   101→}

   103→function Stepper({
   104→  value,
   105→  min,
   106→  max,
   107→  formatValue,
   108→  ariaLabel,
   109→  onDecrement,
   110→  onIncrement,
   111→}: {
   112→  value: number | null;
   113→  min: number;
   114→  max: number;
   115→  formatValue?: (v: number | null) => string;
   116→  ariaLabel: string;
   117→  onDecrement: () => void;
   118→  onIncrement: () => void;
   119→}) {
   120→  const text = formatValue ? formatValue(value) : value == null ? "–" : String(value);
   121→  return (
   122→    <span className="flex flex-none items-center gap-1">
   123→      <Button
   124→        type="button"
   125→        variant="outline"
   126→        className="h-10 w-10 flex-none p-0"
   127→        disabled={value != null && value <= min}
   128→        aria-label={`Decrease ${ariaLabel}`}
   129→        onClick={() => {
   130→          hapticTap();
   131→          onDecrement();
   132→        }}
   133→      >
   134→        <Minus className="h-4 w-4" aria-hidden />
   135→      </Button>
   136→      <span
   137→        className="flex h-10 w-12 flex-none items-center justify-center text-sm font-bold tabular-nums leading-none"
   138→        aria-live="polite"
   139→        aria-label={`${ariaLabel}: ${text}`}
   140→      >
   141→        {text}
   142→      </span>
   143→      <Button
   144→        type="button"
   145→        variant="outline"
   146→        className="h-10 w-10 flex-none p-0"
   147→        disabled={value != null && value >= max}
   148→        aria-label={`Increase ${ariaLabel}`}
   149→        onClick={() => {
   150→          hapticTap();
   151→          onIncrement();
   152→        }}
   153→      >
   154→        <Plus className="h-4 w-4" aria-hidden />
   155→      </Button>
   156→    </span>
   157→  );
   158→}

   160→// ---------- screen ----------
   161→
   162→export default function ProgramBuilderScreen() {
   163→  const navigate = useApp((s) => s.navigate);
   164→  const online = useOnline();
   165→  const invalidate = useInvalidate();
   166→
   167→  // ---------- draft state (local only — nothing persists before Create) ----------
   168→  // #/programs/new/builder?name=… deep-links can pre-fill the name.
   169→  const [name, setName] = useState(() => {
   170→    if (typeof window === "undefined") return "";
   171→    const m = window.location.hash.match(/[?&]name=([^&]+)/);
   172→    return m ? decodeURIComponent(m[1]).slice(0, 80) : "";
   173→  });
   174→  const [difficulty, setDifficulty] = useState<Difficulty>("BEGINNER");
   175→  const [daysManual, setDaysManual] = useState<number | null>(null);
   176→  const [estMinutes, setEstMinutes] = useState<number | null>(null);
   177→  const [phases, setPhases] = useState<PhaseDraft[]>([{ name: "Phase 1", weeks: 1 }]);
   178→  const [weekly, setWeekly] = useState<DayTemplate[]>(DEFAULT_WEEKLY);
   179→  const [labels, setLabels] = useState<string[]>([]);
   180→  const [addingLabel, setAddingLabel] = useState(false);
   181→  /** exercises per unique template name (draft; 3 blank sets attach on build). */
   182→  const [exercises, setExercises] = useState<Record<string, Array<{ exerciseId: string; name: string }>>>({});
   183→  const [pickerFor, setPickerFor] = useState<string | null>(null);
   184→  const [pickerSearch, setPickerSearch] = useState("");
   185→  const [creating, setCreating] = useState(false);
   186→  const searchFocusRef = useRef<HTMLInputElement | null>(null);
   187→
   188→  // ---------- derived ----------
   189→  const workoutDayCount = weekly.filter((w) => w.type === "WORKOUT").length;
   190→  const daysPerWeek = daysManual ?? workoutDayCount;
   191→  const templateNames = useMemo(() => {
   192→    const seen: string[] = [];
   193→    for (const w of weekly) {
   194→      if (w.type !== "WORKOUT") continue;
   195→      const n = w.name.trim();
   196→      if (n && !seen.includes(n)) seen.push(n);
   197→    }
   198→    return seen;
   199→  }, [weekly]);
   200→  const unnamedWorkoutDays = weekly.some((w) => w.type === "WORKOUT" && !w.name.trim());
   201→  const totalWeeks = phases.reduce((n, p) => n + p.weeks, 0);
   202→  const totalDays = totalWeeks * 7;
   203→  const canCreate =
   204→    name.trim().length > 0 && templateNames.length > 0 && !unnamedWorkoutDays && !creating;
   205→
   206→  // debounced picker search (250ms) — feeds useExercises
   207→  const [debouncedSearch, setDebouncedSearch] = useState("");
   208→  useEffect(() => {
   209→    const t = setTimeout(() => setDebouncedSearch(pickerSearch.trim()), 250);
   210→    return () => clearTimeout(t);
   211→  }, [pickerSearch]);
   212→  const exercisesQuery = useExercises(debouncedSearch ? { search: debouncedSearch } : undefined);
   213→  const allExercises = exercisesQuery.data ?? [];
   214→
   215→  // ---------- mutations ----------
   216→  const setDayType = (index: number, type: "WORKOUT" | "REST") => {
   217→    hapticTap();
   218→    setWeekly((prev) => prev.map((w, i) => (i === index ? { ...w, type } : w)));
   219→  };
   220→
   221→  const setDayName = (index: number, dayName: string) => {
   222→    setWeekly((prev) => prev.map((w, i) => (i === index ? { ...w, name: dayName.slice(0, 40) } : w)));
   223→  };
   224→
   225→  const addPhase = () => {
   226→    hapticTap();
   227→    setPhases((prev) =>
   228→      prev.length >= 12 ? prev : [...prev, { name: `Phase ${prev.length + 1}`, weeks: 1 }],
   229→    );
   230→  };
   231→
   232→  const removePhase = (index: number) => {
   233→    setPhases((prev) => (prev.length <= 1 ? prev : prev.filter((_, i) => i !== index)));
   234→  };
   235→
   236→  const openPicker = (templateName: string) => {
   237→    hapticTap();
   238→    setPickerSearch("");
   239→    setPickerFor((prev) => (prev === templateName ? null : templateName));
   240→    if (pickerFor !== templateName) {
   241→      requestAnimationFrame(() => searchFocusRef.current?.focus());
   242→    }
   243→  };
   244→
   245→  const addExerciseTo = (templateName: string, exerciseId: string, exerciseName: string) => {
   246→    hapticTap();
   247→    setExercises((prev) => {
   248→      const list = prev[templateName] ?? [];
   249→      if (list.some((e) => e.exerciseId === exerciseId)) return prev;
   250→      return { ...prev, [templateName]: [...list, { exerciseId, name: exerciseName }] };
   251→    });
   252→  };
   253→
   254→  const removeExerciseFrom = (templateName: string, exerciseId: string) => {
   255→    setExercises((prev) => {
   256→      const list = (prev[templateName] ?? []).filter((e) => e.exerciseId !== exerciseId);
   257→      const next = { ...prev, [templateName]: list };
   258→      return next;
   259→    });
   260→  };
   261→
   262→  const addLabel = (label: string) => {
   263→    const clean = label.trim().slice(0, 30);
   264→    if (!clean) return;
   265→    if (labels.includes(clean)) {
   266→      toast.info("That label already exists");
   267→      return;
   268→    }
   269→    if (labels.length >= 20) {
   270→      toast.info("Labels are limited to 20");
   271→      return;
   272→    }
   273→    setLabels((prev) => [...prev, clean]);
   274→  };
   275→
   276→  const create = async () => {
   277→    if (!canCreate) return;
   278→    if (!online) {
   279→      toast.info("Creating a program needs a connection");
   280→      return;
   281→    }
   282→    setCreating(true);
   283→    try {
   284→      const exercisesPayload: BuilderProgramInput["exercises"] = {};
   285→      for (const templateName of templateNames) {
   286→        const list = exercises[templateName] ?? [];
   287→        if (list.length === 0) continue;
   288→        exercisesPayload[templateName] = list.map((e) => ({
   289→          exerciseId: e.exerciseId,
   290→          sets: BLANK_BUILDER_SETS,
   291→        }));
   292→      }
   293→      const input: BuilderProgramInput = {
   294→        name: name.trim(),
   295→        difficulty,
   296→        ...(daysManual != null || workoutDayCount > 0 ? { daysPerWeek: Math.max(1, Math.min(7, daysPerWeek)) } : {}),
   297→        ...(estMinutes != null ? { estMinutes } : {}),
   298→        ...(labels.length > 0 ? { labels } : {}),
   299→        phases: phases.map((p, i) => ({ name: p.name.trim() || `Phase ${i + 1}`, weeks: p.weeks })),
   300→        weekly: weekly.map((w, i) => ({
   301→          weekday: i,
   302→          type: w.type,
   303→          ...(w.type === "WORKOUT" ? { name: w.name.trim() } : { name: null }),
   304→        })),
   305→        ...(exercisesPayload && Object.keys(exercisesPayload).length > 0 ? { exercises: exercisesPayload } : {}),
   306→      };
   307→      const res = await programsMetaApi.build(input);
   308→      invalidate.programs();
   309→      invalidate.routines();
   310→      hapticSuccess();
   311→      toast.success(`Program created · ${res.dayCount} days`);
   312→      replaceHash(`#/programs/${res.id}`);
   313→    } catch (e) {
   314→      toast.error(errorMessage(e));
   315→    } finally {
   316→      setCreating(false);
   317→    }
   318→  };
   319→
   320→  // ---------- render helpers ----------
   321→  const renderWeeklyRow = (index: number) => {
   322→    const day = weekly[index];
   323→    return (
   324→      <div
   325→        key={WEEKDAYS[index]}
   326→        data-row
   327→        className="flex h-12 w-full items-center gap-1 overflow-hidden whitespace-nowrap"
   328→      >
   329→        <span className="w-10 flex-none text-xs font-bold uppercase tracking-wide text-muted-foreground">
   330→          {WEEKDAYS[index]}
   331→        </span>
   332→        {day.type === "WORKOUT" ? (
   333→          <Input
   334→            value={day.name}
   335→            placeholder="Push, Pull…"
   336→            aria-label={`${WEEKDAYS[index]} workout template name`}
   337→            onChange={(e) => setDayName(index, e.target.value)}
   338→            className="h-10 min-w-0 flex-1 rounded-md"
   339→          />
   340→        ) : (
   341→          <span className="flex h-10 min-w-0 flex-1 items-center gap-2 rounded-md border border-dashed border-border px-3 text-sm text-muted-foreground">
   342→            <Moon className="h-4 w-4 flex-none text-muted-foreground/60" aria-hidden />
   343→            Rest day
   344→          </span>
   345→        )}
   346→        <DropdownMenu>
   347→          <DropdownMenuTrigger asChild>
   348→            <Button
   349→              type="button"
   350→              variant="outline"
   351→              className="h-10 w-24 flex-none justify-between gap-1 px-2 text-xs font-bold"
   352→              aria-label={`${WEEKDAYS[index]} day type: ${day.type === "WORKOUT" ? "workout" : "rest"}`}
   353→            >
   354→              {day.type === "WORKOUT" ? (
   355→                <Dumbbell className="h-4 w-4 flex-none text-primary" aria-hidden />
   356→              ) : (
   357→                <Moon className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   358→              )}
   359→              <span className="min-w-0 flex-1 truncate">
   360→                {day.type === "WORKOUT" ? "Workout" : "Rest"}
   361→              </span>
   362→              <ChevronDown className="h-3.5 w-3.5 flex-none text-muted-foreground" aria-hidden />
   363→            </Button>
   364→          </DropdownMenuTrigger>
   365→          <DropdownMenuContent align="end" className="w-36">
   366→            <DropdownMenuItem onClick={() => setDayType(index, "WORKOUT")}>
   367→              <Dumbbell className="h-4 w-4" aria-hidden /> Workout
   368→            </DropdownMenuItem>
   369→            <DropdownMenuItem onClick={() => setDayType(index, "REST")}>
   370→              <Moon className="h-4 w-4" aria-hidden /> Rest
   371→            </DropdownMenuItem>
   372→          </DropdownMenuContent>
   373→        </DropdownMenu>
   374→      </div>
   375→    );
   376→  };
   377→
   378→  const renderTemplateSection = (templateName: string) => {
   379→    const list = exercises[templateName] ?? [];
   380→    const addedIds = new Set(list.map((e) => e.exerciseId));
   381→    const pickerOpen = pickerFor === templateName;
   382→    const candidates = allExercises
   383→      .filter((ex) => !addedIds.has(ex.id))
   384→      .slice(0, 30);
   385→    return (
   386→      <div key={templateName} className="flex flex-none flex-col overflow-hidden rounded-lg border bg-card">
   387→        {/* template header 40px */}
   388→        <div
   389→          data-row
   390→          className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap border-b border-border px-3"
   391→        >
   392→          <Dumbbell className="h-4 w-4 flex-none text-primary" aria-hidden />
   393→          <span className="min-w-0 flex-1 truncate text-sm font-semibold leading-none" title={templateName}>
   394→            {templateName}
   395→          </span>
   396→          <span className="flex-none text-[11px] font-semibold tabular-nums text-muted-foreground">
   397→            {list.length} exercise{list.length === 1 ? "" : "s"}
   398→          </span>
   399→        </div>
   400→        {/* added exercises — 40px rows with × remove */}
   401→        {list.map((e) => (
   402→          <div
   403→            key={e.exerciseId}
   404→            data-row
   405→            className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap border-b border-border/60 pl-4 pr-1"
   406→          >
   407→            <span className="min-w-0 flex-1 truncate text-sm leading-none" title={e.name}>
   408→              {e.name}
   409→            </span>
   410→            <Button
   411→              type="button"
   412→              variant="ghost"
   413→              className="h-10 w-10 flex-none p-0 text-muted-foreground"
   414→              aria-label={`Remove ${e.name} from ${templateName}`}
   415→              onClick={() => removeExerciseFrom(templateName, e.exerciseId)}
   416→            >
   417→              <X className="h-4 w-4" aria-hidden />
   418→            </Button>
   419→          </div>
   420→        ))}
   421→        {/* + Add exercise → inline searchable picker (0fr→1fr expansion) */}
   422→        <button
   423→          type="button"
   424→          data-row
   425→          aria-expanded={pickerOpen}
   426→          aria-label={`Add exercise to ${templateName}`}
   427→          className={cn(
   428→            "flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap px-3 text-left text-sm font-medium transition-colors",
   429→            pickerOpen
   430→              ? "bg-primary/5 text-primary"
   431→              : "text-muted-foreground hover:bg-accent/40",
   432→          )}
   433→          onClick={() => openPicker(templateName)}
   434→        >
   435→          {pickerOpen ? (
   436→            <ChevronDown className="h-4 w-4 flex-none rotate-180" aria-hidden />
   437→          ) : (
   438→            <Plus className="h-4 w-4 flex-none" aria-hidden />
   439→          )}
   440→          <span className="min-w-0 flex-1 truncate">
   441→            {pickerOpen ? "Close picker" : "Add exercise"}
   442→          </span>
   443→        </button>
   444→        <div
   445→          className="grid transition-[grid-template-rows] duration-200 ease-out"
   446→          style={{ gridTemplateRows: pickerOpen ? "1fr" : "0fr" }}
   447→        >
   448→          <div className="min-h-0 overflow-hidden">
   449→            {pickerOpen ? (
   450→              <div className="flex flex-col gap-2 border-t border-border p-2">
   451→                <div className="relative flex-none">
   452→                  <Search
   453→                    className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   454→                    aria-hidden
   455→                  />
   456→                  <Input
   457→                    ref={searchFocusRef}
   458→                    value={pickerSearch}
   459→                    placeholder="Search your exercises…"
   460→                    aria-label={`Search exercises for ${templateName}`}
   461→                    onChange={(e) => setPickerSearch(e.target.value)}
   462→                    className="h-10 rounded-md pl-9"
   463→                  />
   464→                </div>
   465→                <div
   466→                  className="scroll-slim flex max-h-96 flex-none flex-col overflow-y-auto"
   467→                  aria-label="Exercise results"
   468→                >
   469→                  {exercisesQuery.isLoading ? (
   470→                    <p className="px-1 py-2 text-xs text-muted-foreground" aria-busy="true">
   471→                      Loading exercises…
   472→                    </p>
   473→                  ) : candidates.length === 0 ? (
   474→                    <p className="px-1 py-2 text-xs text-muted-foreground">
   475→                      {debouncedSearch ? "No exercises match that search." : "No exercises yet — create some first."}
   476→                    </p>
   477→                  ) : (
   478→                    candidates.map((ex) => (
   479→                      <button
   480→                        key={ex.id}
   481→                        type="button"
   482→                        data-row
   483→                        aria-label={`Add ${ex.name} to ${templateName}`}
   484→                        onClick={() => addExerciseTo(templateName, ex.id, ex.name)}
   485→                        className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap rounded-md px-2 text-left transition-colors hover:bg-accent/50"
   486→                      >
   487→                        <CategoryDot colour={ex.category?.colour ?? null} />
   488→                        <span className="min-w-0 flex-1 truncate text-sm leading-none">{ex.name}</span>
   489→                        <span className="flex-none text-[11px] text-muted-foreground">
   490→                          {ex.category?.name ?? "Exercise"}
   491→                        </span>
   492→                        <Plus className="h-3.5 w-3.5 flex-none text-primary" aria-hidden />
   493→                      </button>
   494→                    ))
   495→                  )}
   496→                </div>
   497→              </div>
   498→            ) : null}
   499→          </div>
   500→        </div>
   501→      </div>
   502→    );
   503→  };
   504→
   505→  // ---------- render ----------
   506→  return (
   507→    <Screen
   508→      topBar={
   509→        <TopBar
   510→          leading={
   511→            <Button
   512→              type="button"
   513→              variant="ghost"
   514→              size="icon"
   515→              className="h-11 w-11 flex-none"
   516→              onClick={() => navigate("/programs")}
   517→              aria-label="Back to programs"
   518→            >
   519→              <ChevronLeft className="h-5 w-5" aria-hidden />
   520→            </Button>
   521→          }
   522→          title={
   523→            <span className="flex min-w-0 items-center gap-1.5">
