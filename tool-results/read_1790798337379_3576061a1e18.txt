     4→// GroupCard — THE single source of truth group card (Part 8 §2). One card per
     5→// GROUP (a superset group, or an ungrouped exercise as its own group of 1):
     6→//
     7→//   |▌ 4px colour bar full height (group colour; category colour on singletons)
     8→//   | A1                Superset | 32  code (12px bold muted) · label right
     9→//   | Deadlift                   | 40  name 16px
    10→//   | Reps:  5  5  5     💡 …    | 40  per-set reps tabular · tip · options
    11→//   | Tempo: 4/0/1/0              | 32  hidden when no tempo anywhere
    12→//   | ⏱ ›                        | 40  collapsed rest (toggle remembered)
    13→//   |    3× · Rest sec: 90 90 90 | 40  expanded rest row
    14→//   |                              12  gap between exercises inside group (no line)
    15→//
    16→// Modes (§2.3):
    17→//   view — Program detail / Session detail: Reps / Tempo / ⏱ rows.
    18→//   read — Log detail: SetRow list read-only ("100 · 5 🏆").
    19→//   edit — Builder editors: view layout, fields tappable, ≡ drag replaces 💡.
    20→//   log  — Logging screen: Tempo row + SetRow list + "+ set"; collapsed
    21→//          exercises show "0/3" right.
    22→//
    23→// Laws honoured: ONE border per GROUP; colour = the 4px bar ONLY (Law 7);
    24→// rows 56/48/40/32, single line, nowrap + ellipsis (Law 4); card never
    25→// scrolls and never contains inner cards; width = container width.
    26→// Divider BETWEEN groups (16px gap + 1px 30%-muted line) is rendered by
    27→// <GroupCardStack> — never inside the card, never after the last group.

   173→// ---------- reps cell text (§2.2 markers) ----------
   174→
   175→function repsCellText(set: CardSet, fields: Array<"weight" | "reps" | "distance" | "timeSec">): string {
   176→  // Part 9 §5: AMRAP prescribed sets spell out "AMRAP" (tap → Term definition).
   177→  // Part 9 §8: performed AMRAP sets carry their actual reps → "AMRAP→{n}".
   178→  const reps =
   179→    set.setType === "AMRAP"
   180→      ? set.reps == null
   181→        ? "AMRAP"
   182→        : `AMRAP→${set.reps}`
   183→      : set.reps == null
   184→        ? "∞"
   185→        : String(set.reps);
   186→  if (!fields.includes("weight")) {
   187→    if (fields.includes("distance") && set.distanceM != null) return `${trimNum(set.distanceM)} m · ${reps}`;
   188→    if (fields.includes("timeSec") && set.timeSec != null) return `${set.timeSec}s · ${reps}`;
   189→    return reps;
   190→  }
   191→  const weight =
   192→    set.weightKind === "COPY_LAST"
   193→      ? "↺"
   194→      : set.weightKind === "PERCENT_1RM" && set.pct != null
   195→        ? `${trimNum(set.pct)}%`
   196→        : set.weightKg != null
   197→          ? trimNum(set.weightKg)
   198→          : "↺";
   199→  return `${weight}·${reps}`;
   200→}

   202→// ---------- GroupCard ----------
   203→
   204→const CODE_ROW = "flex h-8 items-center gap-2 px-3";
   205→const NAME_ROW = "flex h-10 items-center gap-1 px-3";
   206→const NOTE_ROW = "flex h-8 items-center gap-2 px-3";
   207→const REPS_ROW = "flex h-10 items-center gap-1 px-3";
   208→const TEMPO_ROW = "flex h-8 items-center gap-2 px-3";
   209→const REST_ROW = "flex h-10 items-center gap-2 px-3";
   210→
   211→export function GroupCard({
   212→  mode,
   213→  group,
   214→  entries,
   215→  onAction,
   216→  visibleColumns,
   217→  restExpanded = false,
   218→  onToggleRest,
   219→  menuItems,
   220→  className,
   221→}: GroupCardProps) {
   222→  const [tipOpen, setTipOpen] = useState<string | null>(null);
   223→  const [noteOpen, setNoteOpen] = useState<string | null>(null);
   224→  const hasActions = onAction != null;
   225→  const barColour = group.colour ?? entries[0]?.exercise.categoryColour ?? "#f97316";
   226→  const menu = hasActions ? (menuItems ?? menuForMode(mode)) : [];
   227→  // Legacy Part 3 modes (template/preview/summary/edit-legacy) render through
   228→  // their own SetRow list below — the Part 8 layout branches do not run.
   229→  const isLegacyRows =
   230→    mode === "template" || mode === "preview" || mode === "summary" || mode === "edit-legacy";
   231→  const layoutMode: "view" | "read" | "edit" | "log" =
   232→    mode === "view" || mode === "read" || mode === "edit" || mode === "log" ? mode : "read";
   233→
   234→  // Rest row hidden entirely when no entry carries rest data (§5: restNone
   235→  // exercises always show their "Rest: none" row).
   236→  const anyRest = entries.some((e) => e.restNone || e.sets.some((s) => s.restPlannedSec != null));
   237→  const anyTempo = entries.some((e) => e.sets.some((s) => s.tempo != null && s.tempo !== ""));
   238→
   239→  const dispatch = (action: CardAction, entryIndex: number, exerciseId?: string) => {
   240→    if (!onAction) return;
   241→    // exerciseId-bearing actions get the concrete id injected here.
   242→    const a =
   243→      exerciseId && ("exerciseId" in action && (action as { exerciseId?: string }).exerciseId === "")
   244→        ? ({ ...action, exerciseId } as CardAction)
   245→        : action;
   246→    onAction(a, entryIndex);
   247→  };
   248→
   249→  return (
   250→    <article
   251→      className={cn("flex overflow-hidden rounded-lg border bg-card", className)}
   252→      aria-label={group.label ? `${group.label} ${group.code}` : (entries[0]?.exercise.name ?? "Group")}
   253→    >
   254→      <div aria-hidden className="w-1 flex-none" style={{ backgroundColor: barColour }} />
   255→      <div className="min-w-0 flex-1 py-1">
   256→        {entries.map((entry, i) => {
   257→          const fields = fieldsForType(entry.exercise.modality) as Array<"weight" | "reps" | "distance" | "timeSec">;
   258→          const e = entry.exercise;
   259→          const collapsed = layoutMode === "log" && entry.collapsed;
   260→          const doneCount = entry.sets.filter((s) => s.done).length;
   261→          const menuForEntry = menu.map((item) => ({
   262→            ...item,
   263→            action:
   264→              "exerciseId" in item.action && (item.action as { exerciseId?: string }).exerciseId === ""
   265→                ? ({ ...item.action, exerciseId: e.id } as CardAction)
   266→                : item.action,
   267→          }));
   268→          return (
   269→            <div key={e.id} className={cn("min-w-0", i > 0 && "mt-3")}>
   270→              {/* 32px code row */}
   271→              <div {...tourAttrs({ id: "groupCard.code", label: "Series code", help: "Group letter and position, e.g. A1.", order: 100 })} className={CODE_ROW}>
   272→                <span className="text-xs font-bold tabular-nums leading-none text-muted-foreground">{entry.code || group.code}</span>
   273→                {group.label ? (
   274→                  <span className="ml-auto text-xs leading-none text-muted-foreground">
   275→                    <Term id={group.label as "Superset" | "Triset" | "Giant set"}>{group.label}</Term>
   276→                  </span>
   277→                ) : null}
   278→              </div>
   279→
   280→              {/* 40px name row */}
   281→              <div {...tourAttrs({ id: "groupCard.name", label: "Exercise name", help: "Exercise name; long-press options via the menu.", order: 110 })} className={NAME_ROW}>
   282→                <h3 className="min-w-0 flex-1 truncate text-base font-semibold leading-none">{e.name}</h3>
   283→                {entry.note != null ? (
   284→                  <span className="flex flex-none items-center gap-1 text-xs leading-none text-muted-foreground" aria-label="Has a note">
   285→                    <StickyNote className="h-3.5 w-3.5" aria-hidden />
   286→                  </span>
   287→                ) : null}
   288→                {e.progressionDelta != null && e.progressionDelta !== 0 ? (
   289→                  <span className="flex-none text-xs font-medium leading-none text-primary">
   290→                    {e.progressionDeload ? "↓ deload" : `↑ +${trimNum(e.progressionDelta)} next`}
   291→                  </span>
   292→                ) : null}
   293→                {collapsed ? (
   294→                  <span className="flex-none text-xs tabular-nums leading-none text-muted-foreground">{doneCount}/{entry.sets.length}</span>
   295→                ) : null}
   296→                {hasActions && menuForEntry.length > 0 ? (
   297→                  <span className="flex flex-none" onClick={(ev) => ev.stopPropagation()}>
   298→                    <DropdownMenu>
   299→                      <DropdownMenuTrigger asChild>
   300→                        <Button
   301→                          type="button"
   302→                          variant="ghost"
   303→                          tour={{ id: "groupCard.menu", label: "Options", help: "Detail, history, graph, records, notes, replace or remove.", order: 120 }}
   304→                          className="h-11 w-11 p-0"
   305→                          aria-label={`Options for ${e.name}`}
   306→                        >
   307→                          <MoreVertical className="h-5 w-5" aria-hidden />
   308→                        </Button>
   309→                      </DropdownMenuTrigger>
   310→                      <DropdownMenuContent align="end" className="w-44">
   311→                        {menuForEntry.map((item) => (
   312→                          <DropdownMenuItem
   313→                            key={item.label}
   314→                            onClick={() => dispatch(item.action, i)}
   315→                            className={item.destructive ? "text-destructive focus:text-destructive" : undefined}
   316→                          >
   317→                            <item.icon className="h-4 w-4" aria-hidden />
   318→                            {item.label}
   319→                          </DropdownMenuItem>
   320→                        ))}
   321→                      </DropdownMenuContent>
   322→                    </DropdownMenu>
   323→                  </span>
   324→                ) : null}
   325→              </div>
   326→
   327→              {/* 📝 note row (§5.4) — 32px single line under the header; tap
   328→                  expands the full text (prose block, exempt from nowrap). */}
   329→              {entry.note ? (
   330→                <>
   331→                  <button
   332→                    type="button"
   333→                    {...tourAttrs({ id: "groupCard.note", label: "Note", help: "Your coaching note for this exercise; tap to expand.", order: 120 })}
   334→                    aria-expanded={noteOpen === e.id}
   335→                    aria-label={`Note for ${e.name}`}
   336→                    onClick={() => setNoteOpen((cur) => (cur === e.id ? null : e.id))}
   337→                    className={cn(NOTE_ROW, "w-full gap-2 text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground")}
   338→                  >
   339→                    <StickyNote className="h-3.5 w-3.5 flex-none text-primary" aria-hidden />
   340→                    <span className="truncate">{noteOpen === e.id ? "Hide note" : "Note"}</span>
   341→                    <span className="min-w-0 flex-1 truncate text-muted-foreground/70">{entry.note}</span>
   342→                    <ChevronDown className={cn("h-3.5 w-3.5 flex-none text-muted-foreground/60 transition-transform", noteOpen !== e.id && "-rotate-90")} aria-hidden />
   343→                  </button>
   344→                  {noteOpen === e.id ? (
   345→                    <p className="mx-3 mb-1 max-h-24 overflow-y-auto whitespace-pre-wrap rounded-md bg-muted/60 px-3 py-2 text-xs leading-relaxed text-muted-foreground">
   346→                      {entry.note}
   347→                    </p>
   348→                  ) : null}
   349→                </>
   350→              ) : null}
   351→
   352→              {/* 💡 trainer tip toggle (edit mode: drag handle replaces it) */}
   353→              {mode === "edit" && entry.dragHandle != null ? (
   354→                <div className="flex h-8 items-center px-3">{entry.dragHandle}</div>
   355→              ) : entry.tip ? (
   356→                <>
   357→                  <button
   358→                    type="button"
   359→                    {...tourAttrs({ id: "groupCard.tip", label: "Trainer tip", help: "Expand this exercise's trainer tip.", order: 130 })}
   360→                    aria-expanded={tipOpen === e.id}
   361→                    aria-label={`Trainer tip for ${e.name}`}
   362→                    onClick={() => setTipOpen((cur) => (cur === e.id ? null : e.id))}
   363→                    className={cn(TEMPO_ROW, "w-full gap-2 text-left text-xs font-medium text-muted-foreground transition-colors hover:text-foreground")}
   364→                  >
   365→                    <Lightbulb className="h-4 w-4 flex-none text-amber-500" aria-hidden />
   366→                    <span className="truncate">{tipOpen === e.id ? "Hide tip" : "Trainer tip"}</span>
   367→                  </button>
   368→                  {tipOpen === e.id ? (
   369→                    <p className="mx-3 mb-1 line-clamp-3 rounded-md bg-muted/60 px-3 py-2 text-xs leading-relaxed text-muted-foreground transition-all duration-150">
   370→                      {entry.tip}
   371→                    </p>
   372→                  ) : null}
   373→                </>
   374→              ) : null}
   375→
   376→              {/* mode bodies */}
   377→              {(mode === "view" || mode === "edit") && !collapsed ? (
   378→                <>
   379→                  <div {...tourAttrs({ id: "groupCard.reps", label: "Reps row", help: "Planned reps per set; ↺ copies last time, % is 1RM-based.", order: 140 })} className={REPS_ROW}>
   380→                    <span className="flex-none text-xs font-medium text-muted-foreground">Reps:</span>
   381→                    <span className="flex min-w-0 flex-1 items-center gap-3 overflow-x-auto text-sm font-medium tabular-nums leading-none">
   382→                      {entry.sets.map((s) => (
   383→                        <span key={s.id} className="flex-none whitespace-nowrap">
   384→                          {s.setType === "AMRAP" ? (
   385→                            <Term id="AMRAP" className="tabular-nums">{repsCellText(s, fields)}</Term>
   386→                          ) : (
   387→                            repsCellText(s, fields)
   388→                          )}
   389→                        </span>
   390→                      ))}
   391→                      {entry.sets.length === 0 ? <span className="text-muted-foreground">–</span> : null}
   392→                    </span>
   393→                  </div>
   394→                  {anyTempo ? (
   395→                    <div {...tourAttrs({ id: "groupCard.tempo", label: "Tempo row", help: "Planned tempo, e.g. 4/0/1/0 — eccentric, pause, concentric, pause.", order: 150 })} className={TEMPO_ROW}>
   396→                      <Term id="Tempo" className="flex-none text-xs font-medium text-muted-foreground"><span className="text-xs font-medium text-muted-foreground">Tempo:</span></Term>
   397→                      <span className="truncate text-sm tabular-nums leading-none text-muted-foreground">
   398→                        {entry.sets.find((s) => s.tempo)?.tempo?.replace(/-/g, "/") ?? "–"}
   399→                      </span>
   400→                    </div>
   401→                  ) : null}
   402→                  {anyRest ? (
   403→                    <button
   404→                      type="button"
   405→                      {...tourAttrs({ id: "groupCard.rest", label: "Rest row", help: "Planned rest between sets; tap to expand per-set seconds.", order: 160 })}
   406→                      onClick={() => onToggleRest?.()}
   407→                      aria-expanded={restExpanded}
   408→                      className={cn(REST_ROW, "w-full gap-2 text-left")}
   409→                    >
   410→                      <Timer className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   411→                      {entry.restNone ? (
   412→                        <span className="truncate text-sm leading-none text-muted-foreground">Rest: none</span>
   413→                      ) : restExpanded ? (
   414→                        <span className="truncate text-sm tabular-nums leading-none text-muted-foreground">
   415→                          {entry.sets.length}× · Rest sec: {entry.sets.map((s) => s.restPlannedSec ?? "–").join(" ")}
   416→                        </span>
   417→                      ) : (
   418→                        <span className="text-sm leading-none text-muted-foreground">Rest {entry.sets[0]?.restPlannedSec ?? "–"}s</span>
   419→                      )}
   420→                      <ChevronDown className={cn("ml-auto h-4 w-4 flex-none text-muted-foreground transition-transform", !restExpanded && "-rotate-90")} aria-hidden />
   421→                    </button>
   422→                  ) : null}
   423→                </>
   424→              ) : null}
   425→
   426→              {(mode === "read" || isLegacyRows) && mode !== "summary" ? (
   427→                <div className="px-1 pb-1">
   428→                  {entry.sets.map((s) => (
   429→                    <SetRow key={s.id} mode={mode} exercise={e} set={s} visibleColumns={effectiveCols(e, visibleColumns)} />
   430→                  ))}
   431→                </div>
   432→              ) : null}
   433→
   434→              {(mode === "log" || mode === "template" || mode === "edit-legacy") && !collapsed ? (
   435→                <div className="px-1">
   436→                  {anyTempo ? (
   437→                    <div className={cn(TEMPO_ROW, "px-2")}>
   438→                      <span className="flex-none text-xs font-medium text-muted-foreground">Tempo:</span>
   439→                      <span className="truncate text-sm tabular-nums leading-none text-muted-foreground">
   440→                        {entry.sets.find((s) => s.tempo)?.tempo?.replace(/-/g, "/") ?? "–"}
   441→                      </span>
   442→                    </div>
   443→                  ) : null}
   444→                  {entry.sets.map((s) => (
   445→                    <SetRow
   446→                      key={s.id}
   447→                      mode={mode === "log" ? "log" : mode}
   448→                      exercise={e}
   449→                      set={s}
   450→                      visibleColumns={effectiveCols(e, visibleColumns)}
   451→                      current={entry.currentSetIndex === s.index - 1}
   452→                      onAction={onAction != null ? (a) => dispatch(a, i) : undefined}
   453→                    />
   454→                  ))}
   455→                  {hasActions ? (
   456→                    <button
   457→                      type="button"
   458→                      data-row
   459→                      {...tourAttrs({ id: "groupCard.addSet", label: "Add set", help: "Append another set to this exercise.", order: 170 })}
   460→                      aria-label={`Add set to ${e.name}`}
   461→                      className={cn(rowBase, "w-full gap-2 px-2 text-sm font-medium text-muted-foreground transition-colors hover:bg-accent/50")}
   462→                      onClick={() => dispatch({ type: "add-set" }, i)}
   463→                    >
   464→                      <Plus className="h-4 w-4" aria-hidden />
   465→                      Add set
   466→                    </button>
   467→                  ) : null}
   468→                </div>
   469→              ) : null}
   470→
   471→              {/* log mode collapsed tap target */}
   472→              {layoutMode === "log" && collapsed && hasActions ? (
   473→                <button
   474→                  type="button"
   475→                  {...tourAttrs({ id: "groupCard.expand", label: "Expand", help: "Show this exercise's sets.", order: 180 })}
   476→                  aria-label={`Expand ${e.name}`}
   477→                  className={cn(REST_ROW, "w-full px-3 text-left text-sm font-medium text-muted-foreground")}
   478→                  onClick={() => dispatch({ type: "toggle-collapse" }, i)}
   479→                >
   480→                  <ChevronDown className="h-4 w-4 -rotate-90" aria-hidden />
   481→                  Expand sets
   482→                </button>
   483→              ) : null}
   484→
   485→              {/* §8 per-entry footer — consumer-owned rows (performed data) */}
   486→              {entry.footer != null ? <div className="mt-1">{entry.footer}</div> : null}
   487→            </div>
   488→          );
   489→        })}
   490→      </div>
   491→    </article>
   492→  );
   493→}

   509→function divideChildren(children: ReactNode): ReactNode[] {
   510→  const items = Children.toArray(children).filter(isValidElement);
   511→  return items.map((child, i) => (
   512→    <div key={child.key ?? String(i)} className="flex flex-col gap-4">
   513→      {i > 0 ? <div aria-hidden className="h-px flex-none bg-muted/30" /> : null}
   514→      {child}
   515→    </div>
   516→  ));
   517→}
