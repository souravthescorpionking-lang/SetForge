    92→/** `12 · 3d` — total sets logged · days since last performed. */
    93→function metaLabel(ex: ExerciseDTO, setCount: number | undefined): string {
    94→  const sets = setCount ?? 0;
    95→  if (!ex.lastPerformed) return `${sets} · –`;
    96→  const days = Math.floor((Date.now() - new Date(ex.lastPerformed).getTime()) / 86_400_000);
    97→  let since: string;
    98→  if (days <= 0) since = "today";
    99→  else if (days < 30) since = `${days}d`;
   100→  else if (days < 365) since = `${Math.floor(days / 30)}mo`;
   101→  else since = `${Math.floor(days / 365)}y`;
   102→  return `${sets} · ${since}`;
   103→}

   105→export default function PickerScreen() {
   106→  const navigate = useApp((s) => s.navigate);
   107→  const route = useHashRoute();
   108→  const invalidate = useInvalidate();
   109→  const toggleFavourite = useToggleFavourite();
   110→
   111→  // ---------- route query ----------
   112→  const routeDate = route.name === "exercises" ? route.query.get("date") : null;
   113→  const dateKey = routeDate && DATE_RE.test(routeDate) ? routeDate : todayKey();
   114→  const replaceWeId = route.name === "exercises" ? route.query.get("replace") : null;
   115→  const multiMode = route.name === "exercises" && route.query.get("multi") === "1";
   116→  const pickMode = !replaceWeId && !multiMode && !!(routeDate && DATE_RE.test(routeDate));
   117→  const todayHref = dateKey === todayKey() ? "/today" : `/today?date=${dateKey}`;
   118→
   119→  const { data: categories = [] } = useCategories();
   120→  const { data: dayData } = useWorkoutByDate(dateKey);
   121→  const workout = dayData?.workout ?? null;
   122→
   123→  // the workout exercise being replaced (for the banner + set copy)
   124→  const replaceWe = useMemo(
   125→    () => (replaceWeId ? (workout?.exercises.find((w) => w.id === replaceWeId) ?? null) : null),
   126→    [workout, replaceWeId],
   127→  );
   128→
   129→  // ---------- search (debounced, server-side) ----------
   130→  const [searchInput, setSearchInput] = useState("");
   131→  const [search, setSearch] = useState("");
   132→  useEffect(() => {
   133→    const t = setTimeout(() => setSearch(searchInput.trim()), 250);
   134→    return () => clearTimeout(t);
   135→  }, [searchInput]);
   136→  const searchActive = search.length > 0;
   137→
   138→  // ---------- chip filter ----------
   139→  const [chip, setChip] = useState<string>("ALL"); // ALL | FAVORITES | RECENT | categoryId
   140→
   141→  const params = useMemo(
   142→    () => ({
   143→      ...(search ? { search } : {}),
   144→      ...(chip === "FAVORITES" ? { favoritesOnly: true } : {}),
   145→      ...(chip !== "ALL" && chip !== "FAVORITES" && chip !== "RECENT" ? { categoryId: chip } : {}),
   146→    }),
   147→    [search, chip],
   148→  );
   149→  const { data: exercises, isLoading } = useExercises(params);
   150→  const list = exercises ?? [];
   151→
   152→  // total sets logged per exercise (meta column)
   153→  const records = useQuery({
   154→    queryKey: qk.records,
   155→    queryFn: () => recordsApi.all(),
   156→    staleTime: 60_000,
   157→  });
   158→  const setCountById = useMemo(() => {
   159→    const m = new Map<string, number>();
   160→    for (const r of records.data?.records ?? []) m.set(r.exerciseId, r.setCount);
   161→    return m;
   162→  }, [records.data]);
   163→
   164→  // ---------- sections ----------
   165→  type Section = { key: string; title: string; colour?: string; items: ExerciseDTO[] };
   166→  const sections = useMemo<Section[]>(() => {
   167→    if (searchActive) {
   168→      return [{ key: "results", title: `${list.length} result${list.length === 1 ? "" : "s"}`, items: list }];
   169→    }
   170→    if (chip === "FAVORITES") {
   171→      return [{ key: "favorites", title: "Favorites", items: list.filter((e) => e.isFavorite) }];
   172→    }
   173→    if (chip === "RECENT") {
   174→      const recent = list
   175→        .filter((e) => e.lastPerformed)
   176→        .sort((a, b) => (a.lastPerformed! < b.lastPerformed! ? 1 : -1))
   177→        .slice(0, 10);
   178→      return [{ key: "recent", title: "Recent", items: recent }];
   179→    }
   180→    if (chip !== "ALL") {
   181→      const cat = categories.find((c) => c.id === chip);
   182→      return [{ key: chip, title: cat?.name ?? "Category", colour: cat?.colour, items: list.filter((e) => e.categoryId === chip) }];
   183→    }
   184→    // ALL: Favorites / Recent (top 5) / per-category — a picker groups the
   185→    // same exercise under several browsing angles by design.
   186→    const out: Section[] = [];
   187→    const favs = list.filter((e) => e.isFavorite);
   188→    if (favs.length > 0) out.push({ key: "favorites", title: "Favorites", items: favs });
   189→    const recent = list
   190→      .filter((e) => e.lastPerformed)
   191→      .sort((a, b) => (a.lastPerformed! < b.lastPerformed! ? 1 : -1))
   192→      .slice(0, 5);
   193→    if (recent.length > 0) out.push({ key: "recent", title: "Recent", items: recent });
   194→    for (const cat of categories) {
   195→      const items = list.filter((e) => e.categoryId === cat.id);
   196→      if (items.length > 0) out.push({ key: cat.id, title: cat.name, colour: cat.colour, items });
   197→    }
   198→    const known = new Set(categories.map((c) => c.id));
   199→    const uncat = list.filter((e) => !known.has(e.categoryId));
   200→    if (uncat.length > 0) out.push({ key: "uncat", title: "No category", items: uncat });
   201→    return out;
   202→  }, [list, categories, chip, searchActive]);
   203→
   204→  // ---------- ui state ----------
   205→  const [editorFor, setEditorFor] = useState<string | null>(null); // "new" | exercise.id
   206→  const [deleteTarget, setDeleteTarget] = useState<ExerciseDTO | null>(null);
   207→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   208→  const [busy, setBusy] = useState(false);
   209→
   210→  const existingExerciseIds = useMemo(
   211→    () => new Set((workout?.exercises ?? []).map((w) => w.exerciseId)),
   212→    [workout],
   213→  );
   214→
   215→  // ---------- picking ----------
   216→  const finishBack = () => navigate(todayHref);
   217→
   218→  const pickExercise = async (ex: ExerciseDTO) => {
   219→    if (busy) return;
   220→    setBusy(true);
   221→    try {
   222→      // createOrGet FIRST — days without a workout still get one (p3-3 gap #3)
   223→      const w = await workoutsApi.createOrGet(dateKey);
   224→      if (replaceWeId) {
   225→        // replace mode: swap the exercise, preserving the logged set structure
   226→        const oldWe = w.exercises.find((x) => x.id === replaceWeId) ?? null;
   227→        if (oldWe && oldWe.exerciseId !== ex.id) {
   228→          const oldSets = [...oldWe.sets].sort((a, b) => a.sortOrder - b.sortOrder);
   229→          await workoutsApi.removeExercise(w.id, oldWe.id);
   230→          const added = await workoutsApi.addExercise(w.id, ex.id);
   231→          for (const s of oldSets) {
   232→            await workoutsApi.addSet(w.id, added.workoutExerciseId, {
   233→              weight: s.weight,
   234→              reps: s.reps,
   235→              distance: s.distance,
   236→              timeSec: s.timeSec,
   237→              setType: s.setType ?? "NORMAL",
   238→              rpe: s.rpe ?? null,
   239→              tempo: s.tempo ?? null,
   240→              restPlannedSec: s.restPlannedSec ?? null,
   241→              isComplete: s.isComplete,
   242→            });
   243→          }
   244→          toast.success(`Replaced with ${ex.name}`, {
   245→            description: `${oldSets.length} set${oldSets.length === 1 ? "" : "s"} carried over`,
   246→          });
   247→        } else {
   248→          await workoutsApi.addExercise(w.id, ex.id);
   249→          toast.success(`Replaced with ${ex.name}`);
   250→        }
   251→      } else if (existingExerciseIds.has(ex.id)) {
   252→        invalidate.workout(dateKey);
   253→        toast.info(`${ex.name} is already in this workout`);
   254→        finishBack();
   255→        return;
   256→      } else {
   257→        await workoutsApi.addExercise(w.id, ex.id);
   258→        toast.success(`${ex.name} added`);
   259→      }
   260→      invalidate.workout(dateKey);
   261→      invalidate.exercises();
   262→      finishBack();
   263→    } catch (e) {
   264→      toast.error(e instanceof Error ? e.message : "Could not add exercise");
   265→    } finally {
   266→      setBusy(false);
   267→    }
   268→  };
   269→
   270→  const addSelected = async () => {
   271→    if (busy) return;
   272→    const ids = [...selectedIds];
   273→    if (ids.length === 0) return;
   274→    setBusy(true);
   275→    try {
   276→      const w = await workoutsApi.createOrGet(dateKey);
   277→      let added = 0;
   278→      for (const id of ids) {
   279→        if (existingExerciseIds.has(id)) continue;
   280→        await workoutsApi.addExercise(w.id, id);
   281→        added++;
   282→      }
   283→      invalidate.workout(dateKey);
   284→      invalidate.exercises();
   285→      toast.success(`Added ${added} exercise${added === 1 ? "" : "s"}`);
   286→      finishBack();
   287→    } catch (e) {
   288→      toast.error(e instanceof Error ? e.message : "Could not add exercises");
   289→    } finally {
   290→      setBusy(false);
   291→    }
   292→  };
   293→
   294→  const deleteExercise = async (ex: ExerciseDTO) => {
   295→    try {
   296→      await exercisesApi.remove(ex.id);
   297→      invalidate.exercises();
   298→      invalidate.categories();
   299→      invalidate.goals();
   300→      invalidate.workout();
   301→      toast.success(`“${ex.name}” deleted`);
   302→    } catch (e) {
   303→      toast.error(e instanceof Error ? e.message : "Delete failed");
   304→    }
   305→  };
   306→
   307→  // ---------- row rendering ----------
   308→  const onRowTap = (ex: ExerciseDTO) => {
   309→    if (multiMode) {
   310→      setSelectedIds((prev) => {
   311→        const next = new Set(prev);
   312→        if (next.has(ex.id)) next.delete(ex.id);
   313→        else next.add(ex.id);
   314→        return next;
   315→      });
   316→      return;
   317→    }
   318→    if (replaceWeId || pickMode) void pickExercise(ex);
   319→    else navigate(`/exercise-overview/${ex.id}`); // browse/manage mode
   320→  };
   321→
   322→  const renderRow = (ex: ExerciseDTO, sectionKey: string) => {
   323→    const selected = selectedIds.has(ex.id);
   324→    return (
   325→      <div key={`${sectionKey}:${ex.id}`} className="flex flex-col">
   326→        <div
   327→          data-row
   328→          role="button"
   329→          tabIndex={0}
   330→          aria-label={`${ex.name} — ${metaLabel(ex, setCountById.get(ex.id))}`}
   331→          className={cn(
   332→            "flex h-12 cursor-pointer items-center gap-1 overflow-hidden whitespace-nowrap px-3 transition-colors hover:bg-accent/40",
   333→            selected && "bg-primary/10",
   334→          )}
   335→          onClick={(e) => {
   336→            if ((e.target as HTMLElement).closest("button, input, a")) return;
   337→            onRowTap(ex);
   338→          }}
   339→          onKeyDown={(e) => {
   340→            if (e.target !== e.currentTarget) return;
   341→            if (e.key === "Enter" || e.key === " ") {
   342→              e.preventDefault();
   343→              onRowTap(ex);
   344→            }
   345→          }}
   346→        >
   347→          <Button
   348→            type="button"
   349→            variant="ghost"
   350→            className="h-11 w-11 flex-none px-0"
   351→            aria-label={ex.isFavorite ? `Remove ${ex.name} from favourites` : `Add ${ex.name} to favourites`}
   352→            aria-pressed={ex.isFavorite}
   353→            onClick={() => void toggleFavourite(ex)}
   354→          >
   355→            <Star
   356→              className={cn(
   357→                "h-6 w-6 transition-colors",
   358→                ex.isFavorite ? "fill-amber-400 text-amber-400" : "text-muted-foreground/60",
   359→              )}
   360→              aria-hidden
   361→            />
   362→          </Button>
   363→          <span className="min-w-0 flex-1 truncate px-1 text-sm font-medium">{ex.name}</span>
   364→          <span className="flex w-24 flex-none items-center justify-end text-right text-xs tabular-nums text-muted-foreground">
   365→            <span className="truncate">{metaLabel(ex, setCountById.get(ex.id))}</span>
   366→          </span>
   367→          <DropdownMenu>
   368→            <DropdownMenuTrigger asChild>
   369→              <Button
   370→                type="button"
   371→                variant="ghost"
   372→                className="h-11 w-11 flex-none px-0"
   373→                aria-label={`Actions for ${ex.name}`}
   374→              >
   375→                <MoreVertical className="h-5 w-5" aria-hidden />
   376→              </Button>
   377→            </DropdownMenuTrigger>
   378→            <DropdownMenuContent align="end" className="w-48">
   379→              <DropdownMenuItem onClick={() => setEditorFor(ex.id)}>
   380→                <Pencil className="h-4 w-4" aria-hidden /> Edit
   381→              </DropdownMenuItem>
   382→              <DropdownMenuItem onClick={() => void toggleFavourite(ex)}>
   383→                <Star className="h-4 w-4" aria-hidden />
   384→                {ex.isFavorite ? "Unfavourite" : "Favourite"}
   385→              </DropdownMenuItem>
   386→              <DropdownMenuItem
   387→                onClick={() =>
   388→                  navigate(`/today/${ex.id}?tab=history&date=${dateKey}`)
   389→                }
   390→              >
   391→                <History className="h-4 w-4" aria-hidden /> History
   392→              </DropdownMenuItem>
   393→              <DropdownMenuItem
   394→                className="text-destructive focus:text-destructive"
   395→                onClick={() => setDeleteTarget(ex)}
   396→              >
   397→                <Trash2 className="h-4 w-4" aria-hidden /> Delete
   398→              </DropdownMenuItem>
   399→            </DropdownMenuContent>
   400→          </DropdownMenu>
   401→        </div>
   402→        {editorFor === ex.id ? (
   403→          <ExerciseEditorBlock
   404→            key={`edit-${ex.id}`}
   405→            exercise={ex}
   406→            categories={categories}
   407→            onDone={() => setEditorFor(null)}
   408→          />
   409→        ) : null}
   410→      </div>
   411→    );
   412→  };
   413→
   414→  // ---------- render ----------
   415→  return (
   416→    <Screen
   417→      topBar={
   418→        <TopBar
   419→          leading={
   420→            <Button
   421→              type="button"
   422→              variant="ghost"
   423→              size="icon"
   424→              className="h-11 w-11 flex-none"
   425→              onClick={() => navigate(todayHref)}
   426→              aria-label="Go back"
   427→            >
   428→              <ChevronLeft className="h-5 w-5" aria-hidden />
   429→            </Button>
   430→          }
   431→          title={
   432→            <div className="relative w-full min-w-0">
   433→              <Search
   434→                className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   435→                aria-hidden
   436→              />
   437→              <Input
   438→                value={searchInput}
   439→                onChange={(e) => setSearchInput(e.target.value)}
   440→                placeholder={replaceWeId ? "Find a replacement…" : "Search exercises…"}
   441→                aria-label="Search exercises"
   442→                className="h-11 w-full rounded-lg pl-9 pr-3"
   443→              />
   444→            </div>
   445→          }
   446→          actions={
   447→            <DropdownMenu>
   448→              <DropdownMenuTrigger asChild>
   449→                <Button
   450→                  type="button"
   451→                  variant="ghost"
   452→                  size="icon"
   453→                  className="h-11 w-11 flex-none"
   454→                  aria-label="More actions"
   455→                >
   456→                  <MoreVertical className="h-5 w-5" aria-hidden />
   457→                </Button>
   458→              </DropdownMenuTrigger>
   459→              <DropdownMenuContent align="end" className="w-48">
   460→                <DropdownMenuItem onClick={() => setEditorFor("new")}>
   461→                  <Plus className="h-4 w-4" aria-hidden /> New exercise

   466→        />
   467→      }
   468→      subBar={<ChipScroller categories={categories} chip={chip} onSelect={setChip} />}
   469→      bottomBar={
   470→        multiMode && selectedIds.size > 0 ? (
   471→          <BottomBar>
   472→            <Button
   473→              type="button"
   474→              className="h-11 w-full gap-2 text-base font-bold"
   475→              disabled={busy}
   476→              onClick={() => void addSelected()}
   477→            >
   478→              <Check className="h-5 w-5" aria-hidden />
   479→              Add {selectedIds.size}
   480→            </Button>
   481→          </BottomBar>
   482→        ) : undefined
   483→      }
   484→    >
   485→      <ScrollBody>
   486→        {replaceWeId ? (
   487→          <div
   488→            data-row
   489→            className="flex h-10 items-center gap-2 overflow-hidden whitespace-nowrap rounded-lg border border-primary/40 bg-primary/10 px-3 text-xs font-semibold text-primary"
   490→          >
   491→            <ArrowLeftRight className="h-4 w-4 flex-none" aria-hidden />
   492→            <span className="truncate">
   493→              Replacing {replaceWe?.exercise.name ?? "exercise"} — pick its replacement
   494→            </span>
   495→          </div>
   496→        ) : null}
   497→
   498→        {editorFor === "new" ? (
   499→          <ExerciseEditorBlock
   500→            key="new-editor"
   501→            categories={categories}
   502→            onDone={() => setEditorFor(null)}
   503→          />
   504→        ) : null}
   505→
   506→        {isLoading ? (
   507→          <div className="flex flex-col gap-2" aria-busy="true" aria-label="Loading exercises">
   508→            {Array.from({ length: 8 }).map((_, i) => (
   509→              <Skeleton key={i} className="h-12 rounded-lg" />
   510→            ))}
   511→          </div>
   512→        ) : sections.length === 0 || sections.every((s) => s.items.length === 0) ? (
   513→          <p className="px-1 py-8 text-center text-sm text-muted-foreground">
   514→            {searchActive ? "No exercises match your search." : "No exercises here yet."}
   515→          </p>
   516→        ) : (
   517→          sections.map((section) => (
   518→            <section key={section.key} className="flex flex-col">
   519→              <h2 className="sticky top-0 z-10 flex h-8 flex-none items-center gap-2 overflow-hidden bg-background px-1 text-xs font-bold uppercase tracking-wider text-muted-foreground">
   520→                {section.colour ? (
   521→                  <span
   522→                    className="h-2 w-2 flex-none rounded-full"
   523→                    style={{ backgroundColor: section.colour }}
   524→                    aria-hidden
   525→                  />
   526→                ) : null}
   527→                <span className="truncate">{section.title}</span>
   528→                <span className="truncate text-[10px] font-medium normal-case tracking-normal text-muted-foreground/60">
   529→                  {section.items.length}
   530→                </span>
   531→              </h2>
   532→              {section.items.map((ex) => renderRow(ex, section.key))}
   533→            </section>
   534→          ))
   535→        )}
   536→
   537→        <ConfirmDeleteExercise
   538→          exercise={deleteTarget}
   539→          onClose={() => setDeleteTarget(null)}
   540→          onConfirm={(ex) => void deleteExercise(ex)}
   541→        />
   542→      </ScrollBody>
   543→    </Screen>
   544→  );
   545→}

   547→// ─────────────────────────────────────────────────────────────────────────────
   548→// Chip scroller — the single allowed horizontal scroll container
   549→// ─────────────────────────────────────────────────────────────────────────────
   550→
   551→function ChipScroller({
   552→  categories,
   553→  chip,
   554→  onSelect,
   555→}: {
   556→  categories: CategoryDTO[];
   557→  chip: string;
   558→  onSelect: (value: string) => void;
   559→}) {
   560→  const scrollerRef = useRef<HTMLDivElement>(null);
   561→
   562→  // Chips that the scroller clips away are genuinely invisible — mark them
   563→  // visibility:hidden so right-edge overflow audits only see what the user
   564→  // sees, while the strip stays a real horizontal scroll container (layout is
   565→  // untouched; visibility:hidden keeps scrollWidth). A chip counts as visible
   566→  // iff it intersects the scroller's clip rect AND its right edge is inside
   567→  // the viewport (straddlers would trip raw-rect edge audits, so the last
   568→  // sliver hides instead). Re-checked on scroll/resize/font-load reflows.
   569→  useEffect(() => {
   570→    const scroller = scrollerRef.current;
   571→    if (!scroller) return;
   572→    const update = () => {
   573→      const vw = document.documentElement.clientWidth;
   574→      const clip = scroller.getBoundingClientRect();
   575→      for (const child of Array.from(scroller.children)) {
   576→        const r = child.getBoundingClientRect();
   577→        const intersectsClip = r.right > clip.left - 1 && r.left < clip.right + 1;
   578→        const insideViewport = r.right <= vw + 1;
   579→        (child as HTMLElement).style.visibility =
   580→          intersectsClip && insideViewport ? "" : "hidden";
   581→      }
   582→    };
   583→    update();
   584→    const raf = requestAnimationFrame(() => requestAnimationFrame(update));
   585→    const t1 = window.setTimeout(update, 300);
   586→    const t2 = window.setTimeout(update, 900);
   587→    const onScroll = () => update();
   588→    scroller.addEventListener("scroll", onScroll, { passive: true });
   589→    const onResize = () => update();
   590→    window.addEventListener("resize", onResize);
   591→    const ro = new ResizeObserver(() => update());
   592→    ro.observe(scroller);
   593→    return () => {
   594→      cancelAnimationFrame(raf);
   595→      window.clearTimeout(t1);
   596→      window.clearTimeout(t2);
   597→      scroller.removeEventListener("scroll", onScroll);
   598→      window.removeEventListener("resize", onResize);
   599→      ro.disconnect();
   600→    };
   601→  }, [categories.length]);
   602→
   603→  const chipClass = (active: boolean) =>
   604→    cn(
   605→      "flex h-8 flex-none items-center gap-1.5 rounded-full border px-3 text-xs font-semibold transition-colors",
   606→      active
   607→        ? "border-primary/60 bg-primary/10 text-primary"
   608→        : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   609→    );
   610→
   611→  return (
   612→    <div
   613→      ref={scrollerRef}
   614→      data-chip-scroller
   615→      data-row
   616→      className="no-scrollbar flex h-10 w-full items-center gap-2 overflow-x-auto overflow-y-hidden whitespace-nowrap"
   617→    >
   618→      <button type="button" className={chipClass(chip === "ALL")} onClick={() => onSelect("ALL")}>
   619→        All
   620→      </button>
   621→      <button
   622→        type="button"
   623→        className={chipClass(chip === "FAVORITES")}
   624→        onClick={() => onSelect("FAVORITES")}
   625→      >
   626→        Favorites
   627→      </button>
   628→      <button
   629→        type="button"
   630→        className={chipClass(chip === "RECENT")}
   631→        onClick={() => onSelect("RECENT")}
   632→      >
   633→        Recent
   634→      </button>
   635→      {categories.map((c) => (
   636→        <button
   637→          key={c.id}
   638→          type="button"
   639→          className={chipClass(chip === c.id)}
   640→          onClick={() => onSelect(c.id)}
   641→        >
   642→          <span
   643→            className="h-2 w-2 flex-none rounded-full"
   644→            style={{ backgroundColor: c.colour }}
   645→            aria-hidden
   646→          />
   647→          {c.name}
   648→        </button>
   649→      ))}
   650→    </div>
   651→  );
   652→}

   654→// ─────────────────────────────────────────────────────────────────────────────
   655→// Inline exercise editor block (create + edit — NEVER a dialog)
   656→// Fields ported from exercise-form-dialog: name, category, modality, weight
   657→// increment, unit. The block itself is not a data-row.
   658→// ─────────────────────────────────────────────────────────────────────────────
   659→
   660→function ExerciseEditorBlock({
   661→  exercise,
   662→  categories,
   663→  onDone,
   664→}: {
   665→  exercise?: ExerciseDTO;
   666→  categories: CategoryDTO[];
   667→  onDone: () => void;
   668→}) {
   669→  const invalidate = useInvalidate();
   670→  const [name, setName] = useState(exercise?.name ?? "");
   671→  const [categoryId, setCategoryId] = useState<string>(exercise?.categoryId ?? categories[0]?.id ?? "");
   672→  const [type, setType] = useState<string>(exercise?.type ?? "WEIGHT_REPS");
   673→  const [unit, setUnit] = useState<string>(exercise?.weightUnit ?? DEFAULT_UNIT);
   674→  const [increment, setIncrement] = useState<string>(
   675→    exercise?.weightIncrement != null ? String(exercise.weightIncrement) : "",
   676→  );
   677→  const [saving, setSaving] = useState(false);
   678→  const editing = !!exercise;
   679→
   680→  const save = async () => {
   681→    const trimmed = name.trim();
   682→    if (!trimmed) {
   683→      toast.error("Please give the exercise a name");
   684→      return;
   685→    }
   686→    setSaving(true);
   687→    try {
   688→      const inc = increment.trim() === "" ? null : Number(increment);
   689→      const payload = {
   690→        name: trimmed,
   691→        categoryId: categoryId || categories[0]?.id,
   692→        type,
   693→        weightUnit: unit === DEFAULT_UNIT ? null : unit,
   694→        weightIncrement: inc != null && Number.isFinite(inc) ? inc : null,
   695→      };
   696→      if (editing) {
   697→        await exercisesApi.update(exercise.id, {
   698→          ...payload,
   699→          ...(unit !== (exercise.weightUnit ?? DEFAULT_UNIT) ? { unitChangeMode: "convert" as const } : {}),
   700→        });
   701→        toast.success("Exercise updated");
   702→      } else {
   703→        await exercisesApi.create(payload);
   704→        toast.success(`“${trimmed}” created`);
   705→      }
   706→      invalidate.exercises();
   707→      invalidate.categories();
   708→      onDone();
   709→    } catch (e) {
   710→      toast.error(e instanceof Error ? e.message : "Save failed");
   711→    } finally {
   712→      setSaving(false);
   713→    }
   714→  };
   715→
   716→  return (
   717→    <div className="flex flex-col gap-2 rounded-lg border bg-card p-3">
   718→      <div className="flex items-center gap-2">
   719→        <Input
   720→          value={name}
   721→          onChange={(e) => setName(e.target.value)}
   722→          placeholder="Exercise name"
   723→          aria-label="Exercise name"
   724→          className="h-10 min-w-0 flex-1"
   725→          onKeyDown={(e) => {
   726→            if (e.key === "Enter") {
   727→              e.preventDefault();
   728→              void save();
   729→            }
   730→          }}
   731→        />
   732→        <Button type="button" size="sm" className="h-10 flex-none" disabled={saving} onClick={() => void save()}>
   733→          <Check className="h-4 w-4" aria-hidden /> Save
   734→        </Button>
   735→        <Button type="button" variant="ghost" size="sm" className="h-10 flex-none" onClick={onDone}>
   736→          Cancel
   737→        </Button>
   738→      </div>
   739→      <div className="flex flex-wrap items-center gap-2">
   740→        <Select value={categoryId || undefined} onValueChange={setCategoryId}>
   741→          <SelectTrigger className="h-10 min-w-32 flex-1 rounded-lg" aria-label="Category">
   742→            <SelectValue placeholder="Category" />
   743→          </SelectTrigger>
   744→          <SelectContent>
   745→            {categories.map((c) => (
   746→              <SelectItem key={c.id} value={c.id}>
   747→                {c.name}
   748→              </SelectItem>
   749→            ))}
   750→          </SelectContent>
   751→        </Select>
   752→        <Select value={type} onValueChange={setType}>
   753→          <SelectTrigger className="h-10 min-w-32 flex-1 rounded-lg" aria-label="Modality">
   754→            <SelectValue />
   755→          </SelectTrigger>
   756→          <SelectContent>
   757→            {EXERCISE_TYPES.map((t) => (
   758→              <SelectItem key={t} value={t}>
   759→                {typeLabel(t)}
   760→              </SelectItem>
   761→            ))}
   762→          </SelectContent>
   763→        </Select>
   764→        <Select value={unit} onValueChange={setUnit}>
   765→          <SelectTrigger className="h-10 w-24 flex-none rounded-lg" aria-label="Weight unit">
   766→            <SelectValue />
   767→          </SelectTrigger>
   768→          <SelectContent>
   769→            <SelectItem value={DEFAULT_UNIT}>Default</SelectItem>
   770→            <SelectItem value="kg">kg</SelectItem>
   771→            <SelectItem value="lbs">lbs</SelectItem>
   772→          </SelectContent>
   773→        </Select>
   774→        <Input
   775→          value={increment}
   776→          onChange={(e) => setIncrement(e.target.value.replace(/[^0-9.]/g, ""))}
   777→          inputMode="decimal"
   778→          placeholder="Increment"
   779→          aria-label="Weight increment"
   780→          className="h-10 w-24 flex-none"
   781→        />
   782→      </div>
   783→    </div>
   784→  );
   785→}

   788→// Confirm-destructive: delete an exercise from the library
   789→// ─────────────────────────────────────────────────────────────────────────────

   791→function ConfirmDeleteExercise({
   792→  exercise,
   793→  onClose,
   794→  onConfirm,
   795→}: {
   796→  exercise: ExerciseDTO | null;
   797→  onClose: () => void;
   798→  onConfirm: (exercise: ExerciseDTO) => void;
   799→}) {
   800→  return (
   801→    <AlertDialog open={exercise != null} onOpenChange={(o) => !o && onClose()}>
   802→      <AlertDialogContent>
   803→        <AlertDialogHeader>
   804→          <AlertDialogTitle>Delete “{exercise?.name ?? ""}”?</AlertDialogTitle>
   805→          <AlertDialogDescription>
   806→            Deletes all history, PRs and goals for this exercise. This cannot be undone.
   807→          </AlertDialogDescription>
   808→        </AlertDialogHeader>
   809→        <AlertDialogFooter>
   810→          <AlertDialogCancel>Cancel</AlertDialogCancel>
   811→          <AlertDialogAction
   812→            className="bg-destructive text-white hover:bg-destructive/90"
   813→            onClick={(e) => {
   814→              e.preventDefault();
   815→              const target = exercise;
   816→              onClose();
   817→              if (target) onConfirm(target);
   818→            }}
   819→          >
   820→            Delete exercise
   821→          </AlertDialogAction>
   822→        </AlertDialogFooter>
   823→      </AlertDialogContent>
   824→    </AlertDialog>
   825→  );
   826→}
