   133→/**

   138→function toPerformedCardSets(sets: SetDTO[]): CardSet[] {

   149→/** Performed tonnage of one session (mirrors mapWorkoutSummary server-side). */
   150→function workoutStats(workout: WorkoutDTO): { setCount: number; volume: number; distance: number } {
   151→  let setCount = 0;
   152→  let volume = 0;
   153→  let distance = 0;
   154→  for (const we of workout.exercises) {
   155→    for (const s of we.sets) {
   156→      if (!s.isComplete || s.isWarmup) continue;
   157→      setCount++;
   158→      volume += (s.weight ?? 0) * (s.reps ?? 0);
   159→      distance += s.distance ?? 0;
   160→    }
   161→  }
   162→  return { setCount, volume, distance };
   163→}

   165→/** §8 duration: stored durationSec, else legacy startAt→(endAt|finishedAt). */
   166→function logDurationSec(workout: WorkoutDTO): number {
   167→  if (workout.durationSec != null) return workout.durationSec;
   168→  if (workout.startAt) {
   169→    const end = workout.endAt ?? workout.finishedAt;
   170→    if (end) return Math.max(0, Math.round((new Date(end).getTime() - new Date(workout.startAt).getTime()) / 1000));
   171→  }
   172→  return 0;
   173→}

   199→  const unit = exerciseUnit(we.exercise, settings);
   200→  const sets = useMemo(() => [...we.sets].sort((a, b) => a.sortOrder - b.sortOrder), [we.sets]);

   251→              s.reps != null
   252→                ? String(s.reps)
   253→                : s.timeSec != null
   254→                  ? formatRestSec(s.timeSec)
   255→                  : s.distance != null
   256→                    ? `${round2(s.distance)} km`
   257→                    : "—";
   258→            const weight = s.weight != null ? `${round1(s.weight)} ${unit}` : "—";
   259→            return (
   260→              <div
   261→                key={s.id}
   262→                data-row
   263→                aria-label={`Set ${i + 1}: ${setTypeShort(s.setType)}, ${reps}${weight !== "—" ? `, ${weight}` : ""}`}

   362→function LogDetailInner({ workoutId }: { workoutId: string }) {
   363→  const navigate = useApp((s) => s.navigate);

   365→  const invalidate = useInvalidate();

   369→  const workoutQuery = useQuery({
   370→    queryKey: ["workout", "byId", workoutId],
   371→    queryFn: () => workoutsApi.get(workoutId),
   372→    retry: false, // 404s should surface immediately
   373→  });
   374→  const workout = workoutQuery.data ?? null;
   375→
   376→  const dayKey = workout ? dayKeyOf(workout.date) : "";
   377→  const isRemoved = workout?.removedAt != null;
   378→  const isToday = !!workout && dayKey === todayKey();
   379→  const isLive = !!workout && isToday && !workout.finishedAt && !isRemoved;
   380→
   381→  const logName = workout ? logDetailName(workout) : "Log detail";
   382→  const stats = useMemo(() => (workout ? workoutStats(workout) : null), [workout]);
   383→  const durationSec = workout ? logDurationSec(workout) : 0;
   384→  const muscles = useMemo(() => {
   385→    const seen: string[] = [];
   386→    for (const we of workout?.exercises ?? []) {
   387→      for (const m of we.exercise.primaryMuscles ?? []) {
   388→        if (!seen.includes(m)) seen.push(m);

   391→    return seen;

   393→
   394→  // ---------- read-mode group cards ----------
   395→  const visibleColumns: CardVisibleColumns = {

   403→    () => (workout ? [...workout.exercises].sort((a, b) => a.sortOrder - b.sortOrder) : []),

   405→  );

   407→  const groupById = useMemo(() => {

   411→  }, [workout]);

   419→  // ---------- log-scoped replace flow (§5.2 pattern, writes the WORKOUT) ----------
   420→  // target == null → "pick an exercise" step; target set → suggestions step.
   421→  const [replaceTarget, setReplaceTarget] = useState<WorkoutExerciseDTO | null>(null);
   422→  const [replaceOpen, setReplaceOpen] = useState(false);
   423→  const [replaceSelection, setReplaceSelection] = useState<ExerciseSuggestionDTO | null>(null);
   424→  const [replaceBusy, setReplaceBusy] = useState(false);
   425→

   427→    queryKey: qk.exerciseSuggestions(replaceTarget?.exerciseId ?? ""),
   428→    queryFn: () => exerciseSuggestionsApi.list(replaceTarget!.exerciseId),
   429→    enabled: replaceOpen && replaceTarget != null,
   430→  });
   431→
   432→  const openReplace = (we: WorkoutExerciseDTO | null) => {
   433→    setReplaceOpen(true);
   434→    setReplaceTarget(we);
   435→    setReplaceSelection(null);
   436→  };
   437→  const closeReplace = () => {
   438→    setReplaceOpen(false);
   439→    setReplaceTarget(null);
   440→    setReplaceSelection(null);
   441→  };
   442→  const applyReplace = async () => {
   443→    const target = replaceTarget;
   444→    const selection = replaceSelection;
   445→    if (!workout || !target || !selection || replaceBusy || !online) return;
   446→    setReplaceBusy(true);

   448→      await workoutsApi.updateExercise(workout.id, target.id, { exerciseId: selection.id });
   449→      invalidate.workout(dayKey);
   450→      toast.success(`Replaced with ${selection.name}`, { description: "Sets are kept — this log only" });
   451→      closeReplace();
   452→    } catch (e) {
   453→      toast.error(errorMessage(e));
   454→    } finally {
   455→      setReplaceBusy(false);
   456→    }
   457→  };

   475→      toast.success("Note saved", { description: `${logName} · ${formatDayLabel(dayKey)}` });
   476→      setNoteEditorOpen(false);
   477→    } catch (e) {
   478→      toast.error(errorMessage(e));
   479→    } finally {
   480→      setNoteBusy(false);
   481→    }
   482→  };

   484→  // ---------- other mutations (kept from Part 8) ----------
