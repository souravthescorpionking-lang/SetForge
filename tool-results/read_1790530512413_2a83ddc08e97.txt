     3→// ─────────────────────────────────────────────────────────────────────────────
     4→// SetRow — THE single source of truth set row (Part 3 §SET ROW).
     5→//
     6→// One 40px CSS-grid row for every mode (edit / read / preview / template).
     7→// The grid template is COMPUTED, never styled ad hoc:
     8→//   • spec track widths:  #24 · type32 · f1:72 · f2:56 · rpe:52 · tempo:64 ·
     9→//                          rest:56 · done:40 · more:32
    10→//   • missing modality fields or settings-hidden columns REMOVE tracks from
    11→//     the template (they are never "hidden" cells).
    12→//   • <468px viewport: rpe + tempo tracks are REMOVED — their values/editors
    13→//     move into the `more` ⋯ popover. (PLATFORM ADAPTATION: the spec's bottom
    14→//     sheet is forbidden here; an anchored popover — shadcn Popover on
    15→//     Floating UI with collision flip — is the allowed replacement.)
    16→//   • <360px viewport: f1/f2 drop their floors to spec−16 (minmax) so the row
    17→//     still fits a 320px screen. All small tracks stay exact px.
    18→//   • f1/f2 carry `minmax(spec, 1fr)` so value columns fill leftover card width
    19→//     at every viewport — the row can never overflow its card and never leaves
    20→//     a dead right margin.
    21→//
    22→// Grid math (weight+reps modality, all columns on):
    23→//   wide fixed tracks 24+32+72+56+52+64+56+40+32 = 428px
    24→//   mobile (<468, rpe+tempo dropped)             = 312px ≤ 328px (360−32 law)
    25→//   320px viewport floors 24+32+56+40+56+40+32   = 280px ≤ 282px available
    26→//   (The spec contract names 460 as the wide threshold; the corrected value is
    27→//   468 — 428px of tracks + 32px page gutters + 6px card chrome (border×2 +
    28→//   4px colour bar) need ≥466px, so the wide layout only engages at 468.)

    84→type TrackKey = "index" | "setType" | "f1" | "f2" | "rpe" | "tempo" | "rest" | "done" | "more";

    88→function buildGrid(
    89→  fields: SetField[],
    90→  visible: CardVisibleColumns,
    91→  mode: CardMode,
    92→  vw: number,
    93→  moreTrack: boolean,
    94→): GridSpec {
    95→  const keys: TrackKey[] = [];
    96→  const parts: string[] = [];
    97→  const push = (key: TrackKey, css: string) => {
    98→    keys.push(key);
    99→    parts.push(css);
   100→  };
   101→
   102→  push("index", `${TRACK.index}px`);
   103→  if (visible.setType) push("setType", `${TRACK.setType}px`);
   104→
   105→  // Value tracks: spec floor (−16 below 360px — documented adaptation so the
   106→  // fixed small tracks still fit a 320px screen) + 1fr growth to fill the card.
   107→  const narrow = vw < 360;
   108→  if (fields[0]) push("f1", `minmax(${TRACK.f1 - (narrow ? 16 : 0)}px, 1fr)`);
   109→  if (fields[1]) push("f2", `minmax(${TRACK.f2 - (narrow ? 16 : 0)}px, 1fr)`);
   110→
   111→  if (visible.rpe && vw >= WIDE_VIEWPORT) push("rpe", `${TRACK.rpe}px`);
   112→  if (visible.tempo && vw >= WIDE_VIEWPORT) push("tempo", `${TRACK.tempo}px`);
   113→  if (visible.rest) push("rest", `${TRACK.rest}px`);
   114→
   115→  // `done` never applies to routine templates → track removed.
   116→  if (mode !== "template") push("done", `${TRACK.done}px`);
   117→  if (mode === "edit" || mode === "template") push("more", `${TRACK.more}px`);
   118→  else if (moreTrack) push("more", `${TRACK.more}px`);
   119→
   120→  return { keys, template: parts.join(" ") };
   121→}

   175→function TempoEditor({

   351→function TypeCell({
   352→  set,
   353→  mode,
   354→  editable,
   355→  onAction,
   356→}: {
   357→  set: CardSet;
   358→  mode: CardMode;
   359→  editable: boolean;
   360→  onAction?: (a: CardAction) => void;
   361→}) {
   362→  const [open, setOpen] = useState(false);
   363→  const holdRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   364→  const startRef = useRef<{ x: number; y: number } | null>(null);
   365→  const longPressed = useRef(false);
   366→
   367→  const endHold = () => {
   368→    if (holdRef.current) clearTimeout(holdRef.current);
   369→    holdRef.current = null;
   370→    startRef.current = null;
   371→  };
   372→
   373→  useEffect(() => () => endHold(), []);
   374→
   375→  const value = set.setType ?? null;
   376→  const meta = value ? SET_TYPE_META[value as SetType] : null;
   377→  const templateBlank = mode === "template" && value == null;
   378→
   379→  // read / preview / read-only fallback → static glyph
   380→  if (!editable || !onAction) {
   381→    if (meta) {
   382→      return (
   383→        <span
   384→          className={cn("flex h-6 w-6 items-center justify-center rounded-md text-[11px] font-bold", meta.className)}
   385→          title={`${meta.label} — ${meta.description}`}
   386→        >
   387→          {meta.letter}
   388→        </span>
   389→      );
   390→    }
   391→    return (
   392→      <span className="text-xs text-muted-foreground/40" title="Set type not set">
   393→        –
   394→      </span>
   395→    );
   396→  }
   397→
   398→  const cycle = () => {
   399→    const current = (value ?? "NORMAL") as SetType;
   400→    const next = TYPE_CYCLE[(TYPE_CYCLE.indexOf(current) + 1) % TYPE_CYCLE.length];
   401→    onAction({ type: "update-set", setId: set.id, patch: { setType: next } });
   402→  };
   403→
   404→  return (
   405→    <Popover open={open} onOpenChange={setOpen}>
   406→      <PopoverTrigger asChild>
   407→        <button
   408→          type="button"
   409→          aria-label={
   410→            meta
   411→              ? `Set type ${meta.label}. Tap to cycle, hold for picker`
   412→              : "Set type not set. Tap to choose"
   413→          }
   414→          title={meta ? `${meta.label} — ${meta.description}` : "Choose set type"}
   415→          onClick={(e) => {
   416→            // Fully controlled popover: preventDefault suppresses Radix's own
   417→            // click-toggle (composeEventHandlers skips when defaultPrevented),
   418→            // so tap = cycle · hold / right-click / Enter = picker.
   419→            e.preventDefault();
   420→            if (longPressed.current) {
   421→              longPressed.current = false;
   422→              return;
   423→            }
   424→            if (templateBlank) {
   425→              setOpen(true);
   426→              return;
   427→            }
   428→            cycle();
   429→          }}
   430→          onPointerDown={(e) => {
   431→            startRef.current = { x: e.clientX, y: e.clientY };
   432→            longPressed.current = false;
   433→            holdRef.current = setTimeout(() => {
   434→              longPressed.current = true;
   435→              setOpen(true);
   436→            }, 420);
   437→          }}
   438→          onPointerMove={(e) => {
   439→            const s = startRef.current;
   440→            if (s && (Math.abs(e.clientX - s.x) > 8 || Math.abs(e.clientY - s.y) > 8)) endHold();
   441→          }}
   442→          onPointerUp={endHold}
   443→          onPointerCancel={endHold}
   444→          onPointerLeave={endHold}
   445→          onContextMenu={(e) => {
   446→            e.preventDefault();
   447→            endHold();
   448→            setOpen(true);
   449→          }}
   450→          className={cn(
   451→            "flex h-7 w-7 items-center justify-center rounded-md text-[11px] font-bold transition-transform active:scale-95",
   452→            templateBlank ? "border border-dashed border-border text-muted-foreground" : meta?.className,
   453→          )}
   454→        >
   455→          {templateBlank ? "–" : (meta?.letter ?? "–")}
   456→        </button>
   457→      </PopoverTrigger>
   458→      <PopoverContent align="center" className="w-56 p-2">
   459→        <p className="px-1 pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Set type</p>
   460→        <div className="flex flex-col gap-1">
   461→          {SET_TYPES.map((t) => (
   462→            <button
   463→              key={t}
   464→              type="button"
   465→              onClick={() => {
   466→                onAction({ type: "update-set", setId: set.id, patch: { setType: t } });
   467→                setOpen(false);
   468→              }}
   469→              className="flex w-full items-center gap-2 rounded-lg px-2 py-1 text-left transition-colors hover:bg-accent"
   470→            >
   471→              <span
   472→                className={cn(
   473→                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[11px] font-bold",
   474→                  SET_TYPE_META[t].className,
   475→                )}
   476→              >
   477→                {SET_TYPE_META[t].letter}
   478→              </span>
   479→              <span className="min-w-0 flex-1">
   480→                <span className="block text-sm font-semibold leading-tight">{SET_TYPE_META[t].label}</span>
   481→                <span className="block text-[11px] leading-tight text-muted-foreground">
   482→                  {SET_TYPE_META[t].description}
   483→                </span>
   484→              </span>
   485→            </button>
   486→          ))}
   487→        </div>
   488→      </PopoverContent>
   489→    </Popover>
   490→  );
   491→}

   493→function ValueCell({
   494→  field,
   495→  set,
   496→  exercise,
   497→  mode,
   498→  editable,
   499→  onAction,
   500→}: {
   501→  field: SetField;
   502→  set: CardSet;
   503→  exercise: CardExercise;
   504→  mode: CardMode;
   505→  editable: boolean;
   506→  onAction?: (a: CardAction) => void;
   507→}) {
   508→  const key = FIELD_TO_KEY[field];
   509→  const value = (set[key] as number | null | undefined) ?? null;
   510→  const [draft, setDraft] = useState<string | null>(null);
   511→  const [focused, setFocused] = useState(false);
   512→  const step = stepForField(field, exercise);
   513→  const templateBlank = mode === "template" && value == null;
   514→  const label = FIELD_LABEL[field];
   515→
   516→  if (!editable || !onAction) {
   517→    return (
   518→      <span
   519→        className="w-full truncate px-1 text-right text-sm tabular-nums text-foreground"
   520→        title={formatFieldValue(field, value, exercise.unit)}
   521→      >
   522→        {formatFieldValue(field, value, exercise.unit)}
   523→      </span>
   524→    );
   525→  }
   526→
   527→  const display = draft ?? (value == null ? "" : trimNum(value));
   528→
   529→  const commit = () => {
   530→    if (draft != null) {
   531→      const parsed = parseFieldValue(field, draft);
   532→      if (parsed !== undefined && parsed !== value) {
   533→        onAction({ type: "update-set", setId: set.id, patch: { [key]: parsed } as Partial<CardSet> });
   534→      }
   535→    }
   536→    setDraft(null);
   537→  };
   538→
   539→  const stepBy = (dir: 1 | -1) => {
   540→    const parsedDraft = draft != null ? parseFieldValue(field, draft) : undefined;
   541→    const base = (parsedDraft != null ? parsedDraft : value) ?? 0;
   542→    const next = Math.max(0, Math.round((base + dir * step) * 100) / 100);
   543→    onAction({ type: "update-set", setId: set.id, patch: { [key]: next } as Partial<CardSet> });
   544→    setDraft(null);
   545→  };
   546→
   547→  const stepperButton = (dir: 1 | -1) => (
   548→    <button
   549→      type="button"
   550→      aria-label={dir < 0 ? `Decrease ${label}` : `Increase ${label}`}
   551→      // Keep focus on the input so the steppers don't unmount before click.
   552→      onMouseDown={(e) => e.preventDefault()}
   553→      onClick={() => stepBy(dir)}
   554→      className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   555→    >
   556→      {dir < 0 ? <Minus className="h-3 w-3" aria-hidden /> : <Plus className="h-3 w-3" aria-hidden />}
   557→    </button>
   558→  );
   559→
   560→  // Grid keyboard navigation (audit D6/M2): Enter/↓ → same column one row down
   561→  // (Enter on the last row adds a set); ↑ → one row up.
   562→  const moveFocus = (dir: 1 | -1, input: HTMLInputElement): boolean => {
   563→    const cell = input.parentElement?.parentElement;
   564→    const row = cell?.parentElement;
   565→    if (!cell || !row || !row.parentElement) return false;
   566→    const colIndex = Array.prototype.indexOf.call(row.children, cell);
   567→    const rows = Array.from(row.parentElement.querySelectorAll<HTMLElement>("[data-row]"));
   568→    const rowIndex = rows.indexOf(row);
   569→    const target = rows[rowIndex + dir];
   570→    if (!target) return false;
   571→    const targetCell = target.children[colIndex] as HTMLElement | undefined;
   572→    const focusable = targetCell?.querySelector<HTMLElement>("input, button");
   573→    if (focusable) {
   574→      focusable.focus();
   575→      return true;
   576→    }
   577→    return false;
   578→  };
   579→
   580→  const onInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   581→    if (e.key === "ArrowUp") {
   582→      e.preventDefault();
   583→      commit();
   584→      moveFocus(-1, e.currentTarget);
   585→    } else if (e.key === "ArrowDown" || e.key === "Enter") {
   586→      e.preventDefault();
   587→      commit();
   588→      const input = e.currentTarget;
   589→      if (!moveFocus(1, input)) {
   590→        // Last row: Enter adds a new set (prefilled by the card) and focuses it.
   591→        onAction({ type: "add-set" });
   592→        requestAnimationFrame(() => {
   593→          const row = input.parentElement?.parentElement?.parentElement;
   594→          const rows = row?.parentElement
   595→            ? Array.from(row.parentElement.querySelectorAll<HTMLElement>("[data-row]"))
   596→            : [];
   597→          const cell = input.parentElement?.parentElement;
   598→          const colIndex = cell && row ? Array.prototype.indexOf.call(row.children, cell) : -1;
   599→          const last = rows[rows.length - 1];
   600→          const focusable =
   601→            colIndex >= 0 ? (last?.children[colIndex] as HTMLElement | undefined)?.querySelector<HTMLElement>("input") : null;
   602→          focusable?.focus();
   603→        });
   604→      }
   605→    }
   606→  };
   607→
   608→  return (
   609→    <div className="flex h-full min-w-0 items-center">
   610→      {/* template blank: ↺ = copy-last (typed value replaces it) */}
   611→      {templateBlank && !focused ? (
   612→        <button
   613→          type="button"
   614→          aria-label={`Copy previous set ${label.toLowerCase()}`}
   615→          title="Copy previous set"
   616→          onClick={() => onAction({ type: "copy-last", setId: set.id })}
   617→          className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   618→        >
   619→          <RotateCcw className="h-3 w-3" aria-hidden />
   620→        </button>
   621→      ) : null}
   622→      {/* focused: inline steppers inside the cell ([− 20px | input | + 20px]) */}
   623→      {focused ? stepperButton(-1) : null}
   624→      <input
   625→        aria-label={label}
   626→        inputMode={field === "reps" ? "numeric" : "decimal"}
   627→        value={display}
   628→        onFocus={() => setFocused(true)}
   629→        onBlur={() => {
   630→          commit();
   631→          setFocused(false);
   632→        }}
   633→        onKeyDown={onInputKeyDown}
   634→        onChange={(e) => setDraft(e.target.value)}
   635→        className="h-full min-w-0 flex-1 truncate rounded-sm border-0 bg-transparent px-1 text-right text-sm tabular-nums text-foreground outline-none focus:bg-muted/40"
   636→      />
   637→      {focused ? stepperButton(1) : null}
   638→    </div>
   639→  );

   798→type ApplyField = { key: keyof ApplyToAllFields; label: string; enabled: boolean };
   799→
   800→function applyFieldsFor(exercise: CardExercise, set: CardSet): ApplyField[] {
   801→  const fields = fieldsForType(exercise.modality);
   802→  const out: ApplyField[] = fields.map((f) => ({
   803→    key: FIELD_TO_KEY[f] as keyof ApplyToAllFields,
   804→    label: FIELD_LABEL[f],
   805→    enabled: (set[FIELD_TO_KEY[f]] as number | null | undefined) != null,
   806→  }));
   807→  out.push({ key: "restPlannedSec", label: "Rest", enabled: set.restPlannedSec != null });
   808→  out.push({ key: "tempo", label: "Tempo", enabled: !!set.tempo });
   809→  out.push({ key: "rpe", label: "RPE", enabled: set.rpe != null });
   810→  return out;
   811→}

   813→function ApplyToAllSection({
   814→  exercise,
   815→  set,
   816→  onAction,
   817→}: {
   818→  exercise: CardExercise;
   819→  set: CardSet;
   820→  onAction: (a: CardAction) => void;
   821→}) {
   822→  const [open, setOpen] = useState(false);
   823→  const [selected, setSelected] = useState<Set<string>>(new Set());
   824→  const fields = applyFieldsFor(exercise, set);
   825→  const anyEnabled = fields.some((f) => f.enabled);
   826→
   827→  const toggle = (key: string) => {
   828→    hapticTap();
   829→    setSelected((prev) => {
   830→      const next = new Set(prev);
   831→      if (next.has(key)) next.delete(key);
   832→      else next.add(key);
   833→      return next;
   834→    });
   835→  };
   836→
   837→  const apply = () => {
   838→    // keys are constrained to keyof ApplyToAllFields by applyFieldsFor, so the
   839→    // record-then-cast keeps each property's value type sound.
   840→    const out: Record<string, unknown> = {};
   841→    for (const f of fields) {
   842→      if (!selected.has(f.key) || !f.enabled) continue;
   843→      // copy the value straight off the source row (undefined → null = clear)
   844→      out[f.key] = set[f.key] ?? null;
   845→    }
   846→    onAction({ type: "apply-to-all", fields: out as ApplyToAllFields });
   847→  };
   848→
   849→  if (!anyEnabled) {
   850→    return (
   851→      <p className="text-[11px] leading-snug text-muted-foreground/70">
   852→        Fill weight, reps, rest, tempo or RPE on this set to copy it to every set.
   853→      </p>
   854→    );
   855→  }
   856→
   857→  return (
   858→    <div>
   859→      <button
   860→        type="button"
   861→        aria-expanded={open}
   862→        aria-label="Apply to all sets"
   863→        onClick={() => {
   864→          hapticTap();
   865→          setOpen((o) => !o);
   866→        }}
   867→        className="flex w-full items-center gap-2 rounded-md px-1 py-1 text-left text-xs font-bold uppercase tracking-wider text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   868→      >
   869→        <Copy className="h-3.5 w-3.5 flex-none" aria-hidden />
   870→        <span className="min-w-0 flex-1">Apply to all sets</span>
   871→        <ChevronDown
   872→          className={cn("h-3.5 w-3.5 flex-none transition-transform", open && "rotate-180")}
   873→          aria-hidden
   874→        />
   875→      </button>
   876→      {open ? (
   877→        <div className="pt-2">
   878→          <div className="flex flex-wrap gap-1" role="group" aria-label="Fields to apply">
   879→            {fields.map((f) => {
   880→              const on = selected.has(f.key);
   881→              return (
   882→                <button
   883→                  key={f.key}
   884→                  type="button"
   885→                  disabled={!f.enabled}
   886→                  aria-pressed={on}
   887→                  aria-disabled={!f.enabled}
   888→                  aria-label={`Apply ${f.label}${f.enabled ? "" : " — no value on this set"}`}
   889→                  className={cn(
   890→                    "flex h-8 items-center rounded-full border px-3 text-xs font-bold transition-colors",
   891→                    on
   892→                      ? "border-primary/60 bg-primary/10 text-primary"
   893→                      : "border-border text-muted-foreground hover:bg-accent hover:text-foreground",
   894→                    !f.enabled && "cursor-not-allowed opacity-40 hover:bg-transparent hover:text-muted-foreground",
   895→                  )}
   896→                  onClick={() => toggle(f.key)}
   897→                >
   898→                  {f.label}
   899→                  {on ? <Check className="ml-1 h-3 w-3" aria-hidden /> : null}
   900→                </button>
   901→              );
   902→            })}
   903→          </div>
   904→          <Button
   905→            type="button"
   906→            size="sm"
   907→            className="mt-2 h-8 w-full rounded-lg text-xs font-bold"
   908→            disabled={selected.size === 0}
   909→            onClick={apply}
   910→          >
   911→            Apply to all sets
   912→          </Button>
   913→        </div>
   914→      ) : null}
   915→    </div>
   916→  );
   917→}
