    34→export class ApiError extends Error {
    35→  constructor(
    36→    public status: number,
    37→    public code: string,
    38→    message: string,
    39→    public details?: unknown,
    40→  ) {
    41→    super(message);
    42→  }
    43→}
    44→
    45→async function request<T>(path: string, init?: RequestInit): Promise<T> {
    46→  const res = await fetch(path, {
    47→    ...init,
    48→    headers: { "Content-Type": "application/json", ...(init?.headers ?? {}) },
    49→    credentials: "same-origin",
    50→  });
    51→  if (!res.ok) {
    52→    let code = "INTERNAL";
    53→    let message = res.statusText || "Request failed";
    54→    let details: unknown;
    55→    try {
    56→      const json = await res.json();
    57→      if (json?.error) {
    58→        code = json.error.code ?? code;
    59→        message = json.error.message ?? message;
    60→        details = json.error.details;
    61→      }
    62→    } catch {
    63→      /* non-JSON error */
    64→    }
    65→    throw new ApiError(res.status, code, message, details);
    66→  }
    67→  if (res.status === 204) return undefined as T;
    68→  return (await res.json()) as T;
    69→}
    70→
    71→const body = (data: unknown) => JSON.stringify(data);
    72→const qs = (params: Record<string, string | number | boolean | undefined>) => {
    73→  const search = new URLSearchParams();
    74→  for (const [k, v] of Object.entries(params)) {
    75→    if (v !== undefined && v !== "") search.set(k, String(v));
    76→  }
    77→  const s = search.toString();
    78→  return s ? `?${s}` : "";
    79→};
