    95→function DaySection({ open, header, children }: { open: boolean; header: ReactNode; children: ReactNode }) {
    96→  return (
    97→    <section className="flex flex-none flex-col overflow-hidden rounded-lg border bg-card">
    98→      {header}
    99→      <div
   100→        className="grid transition-[grid-template-rows] duration-200 ease-out"
   101→        style={{ gridTemplateRows: open ? "1fr" : "0fr" }}
   102→      >
   103→        <div className="min-h-0 overflow-hidden">
   104→          {open ? children : null}
   105→        </div>
   106→      </div>
   107→    </section>
   108→  );
   109→}

   115→function ExerciseNotesPopover({
   116→  exerciseName,
   117→  notes,
   118→  open,
   119→  onClose,
   120→}: {
   121→  exerciseName: string;
   122→  notes: string | null;
   123→  open: boolean;
   124→  onClose: () => void;
   125→}) {
   126→  return (
   127→    <Popover open={open} onOpenChange={(o) => !o && onClose()}>
   128→      <PopoverTrigger asChild>
   129→        <span className="absolute bottom-2 left-2 h-1 w-1" aria-hidden />
   130→      </PopoverTrigger>
   131→      <PopoverContent
   132→        align="start"
   133→        className="w-72"
   134→        onOpenAutoFocus={(e) => e.preventDefault()}
   135→        // The ⋮ menu closes by returning focus to its trigger; without this
   136→        // the popover would be instantly dismissed (same Radix gotcha as the
   137→        // Today card popovers). Pointer-outside + Escape still close.
   138→        onFocusOutside={(e) => e.preventDefault()}
   139→      >
   140→        <p className="pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">
   141→          {exerciseName} — notes
   142→        </p>
   143→        <p className="text-sm leading-relaxed text-foreground">
   144→          {notes && notes.trim().length > 0 ? notes : "No notes on this exercise."}
   145→        </p>
   146→      </PopoverContent>
   147→    </Popover>
   148→  );
   149→}

   159→
   160→function RoutineDetailInner({ routineId }: { routineId: string }) {
   161→  const navigate = useApp((s) => s.navigate);
   162→  const settings = useApp((s) => s.settings);
   163→  const invalidate = useInvalidate();
   164→  const online = useOnline();
   165→  const { run } = useRoutineRun();
   166→  const prefillFor = useLastSetsPrefill();
   167→
   168→  // ---------- data ----------
   169→  const { data: routine, isLoading, error } = useQuery({
   170→    queryKey: qk.routine(routineId),
   171→    queryFn: () => routinesApi.get(routineId),
   172→    retry: 1,
   173→  });
   174→
   175→  const days = useMemo(
   176→    () => (routine ? [...routine.days].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   177→    [routine],
   178→  );
   179→
   180→  // ---------- ui state ----------
   181→  const vw = useViewportWidth();
   182→  const isWide = vw >= WIDE_DAY_VIEWPORT;
   183→  const [editing, setEditing] = useState(false);
   184→  // undefined = follow the default (first day open); "" = all collapsed.
   185→  const [openDayId, setOpenDayId] = useState<string | undefined>(undefined);
   186→  const [collapsedReIds, setCollapsedReIds] = useState<Set<string>>(new Set());
   187→  const [renamingRoutine, setRenamingRoutine] = useState(false);
   188→  const [renamingDayId, setRenamingDayId] = useState<string | null>(null);
   189→  const [deleteRoutineOpen, setDeleteRoutineOpen] = useState(false);
   190→  const [deleteDay, setDeleteDay] = useState<RoutineDayDTO | null>(null);
   191→  const [notesOpen, setNotesOpen] = useState(false);
   192→  const [notesDraft, setNotesDraft] = useState("");
   193→  const [notesReId, setNotesReId] = useState<string | null>(null);
   194→
   195→  const effectiveOpen = (dayId: string) =>
   196→    isWide ? true : (openDayId ?? days[0]?.id ?? "") === dayId;
   197→
   198→  const toggleDay = (dayId: string) => {
   199→    const current = openDayId ?? days[0]?.id ?? "";
   200→    setOpenDayId(current === dayId ? "" : dayId);
   201→  };
   202→
   203→  const visibleColumns: CardVisibleColumns = {
   204→    setType: settings?.showSetType ?? true,
   205→    rpe: settings?.showRpe ?? true,
   206→    tempo: settings?.showTempo ?? true,
   207→    rest: settings?.showRest ?? true,
   208→  };
   209→
   210→  // ---------- routine-level mutations ----------
   211→  const renameRoutine = async (name: string) => {
   212→    if (!routine || !name || name === routine.name) return;
   213→    const ok = await run(() => routinesApi.update(routineId, { name }), {
   214→      path: `/api/routines/${routineId}`,
   215→      method: "PATCH",
   216→      body: { name },
   217→      label: "Routine renamed",
   218→    });
   219→    if (ok) toast.success("Routine renamed");
   220→  };
   221→
   222→  const saveNotes = async () => {
   223→    if (!routine) return;
   224→    const next = notesDraft.trim();
   225→    if (next === (routine.notes ?? "")) return;
   226→    const ok = await run(() => routinesApi.update(routineId, { notes: next || null }), {
   227→      path: `/api/routines/${routineId}`,
   228→      method: "PATCH",
   229→      body: { notes: next || null },
   230→      label: "Notes saved",
   231→    });
   232→    if (ok) toast.success("Notes saved");
   233→  };
   234→
   235→  const copyRoutine = async () => {
   236→    if (!routine) return;
   237→    const ok = await run(() => routinesApi.copy(routineId), {
   238→      path: `/api/routines/${routineId}/copy`,
   239→      method: "POST",
   240→      label: "Duplicate routine",
   241→    });
   242→    if (ok) toast.success(`Duplicated “${routine.name}”`);
   243→  };
   244→
   245→  const removeRoutine = async () => {
   246→    const ok = await run(() => routinesApi.remove(routineId), {
   247→      path: `/api/routines/${routineId}`,
   248→      method: "DELETE",
   249→      label: `Deleted ${routine?.name ?? "routine"}`,
   250→    });
   251→    if (ok) {
   252→      toast.success(`Deleted “${routine?.name ?? "routine"}”`);
   253→      navigate("/routines");
   254→    }
   255→  };
   256→
   257→  // ---------- day-level mutations ----------
   258→  const renameDay = async (day: RoutineDayDTO, name: string) => {
   259→    if (!name || name === day.name) return;
   260→    const ok = await run(() => routinesApi.updateDay(routineId, day.id, { name }), {
   261→      path: `/api/routines/${routineId}/days/${day.id}`,
   262→      method: "PATCH",
   263→      body: { name },
   264→      label: "Day renamed",
   265→    });
   266→    if (ok) toast.success("Day renamed");
   267→  };
   268→
   269→  const removeDay = async (day: RoutineDayDTO) => {
   270→    const ok = await run(() => routinesApi.removeDay(routineId, day.id), {
   271→      path: `/api/routines/${routineId}/days/${day.id}`,
   272→      method: "DELETE",
   273→      label: `Deleted ${day.name}`,
   274→    });
   275→    if (ok) toast.success(`Deleted “${day.name}”`);
   276→  };
   277→
   278→  const reorderDay = useDayReorder(routineId);
   279→
   280→  const addDay = async () => {
   281→    if (!routine) return;
   282→    if (!online) {
   283→      toast.info("Adding a day needs a connection");
   284→      return;
   285→    }
   286→    try {
   287→      const updated = await routinesApi.addDay(routineId, nextDayName(routine.days));
   288→      invalidate.routines();
   289→      const oldIds = new Set(routine.days.map((d) => d.id));
   290→      const created = [...updated.days].sort((a, b) => b.sortOrder - a.sortOrder).find((d) => !oldIds.has(d.id)) ?? null;
   291→      if (created) {
   292→        setOpenDayId(created.id);
   293→        setRenamingDayId(created.id);
   294→      }
   295→      toast.success("Day added");
   296→    } catch (e) {
   297→      toast.error(errorMessage(e));
   298→    }
   299→  };
   300→
   301→  // ---------- card (template) mutations ----------
   302→  const reorderExercises = async (dayId: string, ids: string[], reId: string, delta: -1 | 1) => {
   303→    const i = ids.indexOf(reId);
   304→    const j = i + delta;
   305→    if (i < 0 || j < 0 || j >= ids.length) return;
   306→    const next = [...ids];
   307→    [next[i], next[j]] = [next[j], next[i]];
   308→    await run(() => routinesApi.reorderExercises(routineId, dayId, next), {
   309→      path: `/api/routines/${routineId}/days/${dayId}/exercises/order`,
   310→      method: "PUT",
   311→      body: { ids: next },
   312→      label: "Exercise order",
   313→    });
   314→  };
   315→
   316→  const patchPredefinedSet = async (
   317→    day: RoutineDayDTO,
   318→    re: RoutineExerciseDTO,
   319→    setId: string,
   320→    patch: Partial<CardSet>,
   321→  ) => {
   322→    const input = cardPatchToPredefinedInput(patch);
   323→    if (Object.keys(input).length === 0) return; // note-only: predefined sets carry no comments
   324→    await run(() => routinesApi.updateSet(routineId, day.id, re.id, setId, input), {
   325→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${re.id}/sets/${setId}`,
   326→      method: "PATCH",
   327→      body: input,
   328→      label: "Set updated",
   329→    });
   330→  };
   331→
   332→  const removePredefinedSet = async (day: RoutineDayDTO, re: RoutineExerciseDTO, setId: string) => {
   333→    await run(() => routinesApi.removeSet(routineId, day.id, re.id, setId), {
   334→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${re.id}/sets/${setId}`,
   335→      method: "DELETE",
   336→      label: "Set removed",
   337→    });
   338→  };
   339→
   340→  const addPredefinedSet = async (day: RoutineDayDTO, re: RoutineExerciseDTO) => {
   341→    await run(() => routinesApi.addSet(routineId, day.id, re.id, {}), {
   342→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${re.id}/sets`,
   343→      method: "POST",
   344→      body: {},
   345→      label: "Set added",
   346→    });
   347→  };
   348→
   349→  const copyLastToSet = async (day: RoutineDayDTO, re: RoutineExerciseDTO, setId: string) => {
   350→    const sorted = [...re.sets].sort((a, b) => a.sortOrder - b.sortOrder);
   351→    const idx = sorted.findIndex((s) => s.id === setId);
   352→    if (idx < 0) return;
   353→    const src = await prefillFor(re.exerciseId, idx);
   354→    if (!src) {
   355→      toast.info("No previous workout to copy from");
   356→      return;
   357→    }
   358→    await run(
   359→      () =>
   360→        routinesApi.updateSet(routineId, day.id, re.id, setId, {
   361→          weight: src.weight,
   362→          reps: src.reps,
   363→          distance: src.distance,
   364→          timeSec: src.timeSec,
   365→        }),
   366→      {
   367→        path: `/api/routines/${routineId}/days/${day.id}/exercises/${re.id}/sets/${setId}`,
   368→        method: "PATCH",
   369→        body: { weight: src.weight, reps: src.reps, distance: src.distance, timeSec: src.timeSec },
   370→        label: "Copied from last workout",
   371→      },
   372→    );
   373→    toast.success("Copied from last workout");
   374→  };
   375→
   376→  const removeRoutineExercise = async (day: RoutineDayDTO, re: RoutineExerciseDTO) => {
   377→    await run(() => routinesApi.removeExercise(routineId, day.id, re.id), {
   378→      path: `/api/routines/${routineId}/days/${day.id}/exercises/${re.id}`,
   379→      method: "DELETE",
   380→      label: `Removed ${re.exercise.name}`,
   381→    });
   382→    toast.success(`${re.exercise.name} removed from day`);
   383→  };
   384→
   385→  // ---------- card action dispatch ----------
   386→  const handleCardAction = (day: RoutineDayDTO, re: RoutineExerciseDTO) => (action: CardAction): void => {
   387→    switch (action.type) {
   388→      case "toggle-collapse":
   389→        setCollapsedReIds((prev) => {
   390→          const next = new Set(prev);
   391→          if (next.has(re.id)) next.delete(re.id);
   392→          else next.add(re.id);
   393→          return next;
   394→        });
   395→        break;
   396→      case "add-set":
   397→        void addPredefinedSet(day, re);
   398→        break;
   399→      case "update-set":
   400→        void patchPredefinedSet(day, re, action.setId, action.patch);
   401→        break;
   402→      case "remove-set":
   403→        void removePredefinedSet(day, re, action.setId);
   404→        break;
   405→      case "copy-last":
   406→        void copyLastToSet(day, re, action.setId);
   407→        break;
   408→      case "move-up":
   409→      case "move-down": {
   410→        const ids = [...day.exercises].sort((a, b) => a.sortOrder - b.sortOrder).map((e) => e.id);
   411→        void reorderExercises(day.id, ids, re.id, action.type === "move-up" ? -1 : 1);
   412→        break;
   413→      }
   414→      case "remove":
   415→        void removeRoutineExercise(day, re);
   416→        break;
   417→      case "notes":
   418→        setNotesReId(re.id);
   419→        break;
   420→      case "add-to-group":
   421→        toast.info("Routine groups arrive with a later build");
   422→        break;
   423→      case "replace":
   424→        toast.info("Replacing exercises in templates arrives with a later build");
   425→        break;
   426→      case "select":
   427→        toast.info("Multi-select arrives with the next build");
   428→        break;
   429→      case "rest-timer":
   430→        toast.info("The rest timer lives on the Today screen");
   431→        break;
   432→      case "toggle-done":
   433→      case "toggle-select":
   434→      case "open":
   435→        break; // preview/edit-mode-only actions; never emitted by template cards
   436→    }
   437→  };
   438→
   439→  // ---------- render helpers ----------
   440→  const renderDayHeader = (day: RoutineDayDTO, index: number) => {
   441→    const open = effectiveOpen(day.id);
   442→    const renaming = editing && renamingDayId === day.id;
   443→    return (
   444→      <div data-row className="flex h-12 items-center gap-1 overflow-hidden whitespace-nowrap pl-2 pr-1">
   445→        <Button
   446→          type="button"
   447→          variant="ghost"
   448→          className="h-11 w-6 flex-none p-0"
   449→          onClick={() => toggleDay(day.id)}
   450→          aria-label={open ? `Collapse ${day.name}` : `Expand ${day.name}`}
   451→          aria-expanded={open}
   452→        >
   453→          <ChevronDown
   454→            className={cn("h-4 w-4 transition-transform", open ? "" : "-rotate-90")}
   455→            aria-hidden
   456→          />
   457→        </Button>
   458→        {renaming ? (
   459→          <InlineInput
   460→            value={day.name}
   461→            ariaLabel={`Rename ${day.name}`}
   462→            onCommit={(name) => {
   463→              setRenamingDayId(null);
   464→              void renameDay(day, name);
   465→            }}
   466→            onCancel={() => setRenamingDayId(null)}
   467→            className="h-10 min-w-0 flex-1"
   468→          />
   469→        ) : (
   470→          <span className="min-w-0 flex-1 truncate text-sm font-semibold leading-none">{day.name}</span>
   471→        )}
   472→        {editing ? (
   473→          <span className="flex flex-none">
   474→            <DropdownMenu>
   475→              <DropdownMenuTrigger asChild>
   476→                <Button
   477→                  type="button"
   478→                  variant="ghost"
   479→                  className="h-11 w-11 p-0"
   480→                  aria-label={`Actions for ${day.name}`}
   481→                >
   482→                  <MoreVertical className="h-5 w-5" aria-hidden />
   483→                </Button>
   484→              </DropdownMenuTrigger>
   485→              <DropdownMenuContent align="end" className="w-44">
   486→                <DropdownMenuItem onClick={() => setRenamingDayId(day.id)}>
   487→                  <Pencil className="h-4 w-4" aria-hidden /> Rename
   488→                </DropdownMenuItem>
   489→                <DropdownMenuItem disabled={index === 0} onClick={() => void reorderDay(days, index, -1)}>
   490→                  <ArrowUp className="h-4 w-4" aria-hidden /> Move up
   491→                </DropdownMenuItem>
   492→                <DropdownMenuItem
   493→                  disabled={index === days.length - 1}
   494→                  onClick={() => void reorderDay(days, index, 1)}
   495→                >
   496→                  <ArrowDown className="h-4 w-4" aria-hidden /> Move down
   497→                </DropdownMenuItem>
   498→                <DropdownMenuItem
   499→                  className="text-destructive focus:text-destructive"
   500→                  onClick={() => setDeleteDay(day)}
   501→                >
   502→                  <Trash2 className="h-4 w-4" aria-hidden /> Delete
   503→                </DropdownMenuItem>
   504→              </DropdownMenuContent>
   505→            </DropdownMenu>
   506→          </span>
   507→        ) : (
   508→          <Button
   509→            type="button"
   510→            className="h-11 w-18 flex-none gap-1 px-0"
   511→            aria-label={`Log ${day.name} to today`}
   512→            onClick={() => navigate(`/routines/${routineId}/log/${day.id}`)}
   513→          >
   514→            <Zap className="h-4 w-4" aria-hidden />
   515→            Log
   516→          </Button>
   517→        )}
   518→      </div>
   519→    );
   520→  };
   521→
   522→  const renderDayBody = (day: RoutineDayDTO) => {
   523→    const exercises = [...day.exercises].sort((a, b) => a.sortOrder - b.sortOrder);
   524→    return (
   525→      <div className="flex flex-col gap-2 p-2">
   526→        {exercises.map((re) => {
   527→          const openNotes = notesReId === re.id;
   528→          return (
   529→            <div key={re.id} className="relative flex-none">
   530→              <ExerciseCard
   531→                mode="template"
   532→                exercise={toCardExercise(re, settings)}
   533→                sets={cardSetsOf(re)}
   534→                collapsed={collapsedReIds.has(re.id)}
   535→                visibleColumns={visibleColumns}
   536→                headerLeading={editing ? <DragGlyph /> : undefined}
   537→                onAction={handleCardAction(day, re)}
   538→              />
   539→              {openNotes ? (
   540→                <ExerciseNotesPopover
   541→                  exerciseName={re.exercise.name}
   542→                  notes={re.exercise.notes ?? null}
   543→                  open
   544→                  onClose={() => setNotesReId(null)}
   545→                />
   546→              ) : null}
   547→            </div>
   548→          );
   549→        })}
   550→        {exercises.length === 0 ? (
   551→          <p className="flex-none rounded-lg border border-dashed border-border px-3 py-2 text-xs text-muted-foreground">
   552→            No exercises in this day yet.
   553→          </p>
   554→        ) : null}
   555→        {editing ? (
   556→          <button
   557→            type="button"
   558→            data-row
   559→            className="flex h-10 w-full items-center gap-2 overflow-hidden whitespace-nowrap rounded-lg border border-dashed border-border px-3 text-sm font-medium text-muted-foreground transition-colors hover:bg-accent/40"
   560→            onClick={() => navigate(`/exercises?context=routine&routineId=${routineId}&dayId=${day.id}`)}
   561→          >
   562→            <Plus className="h-4 w-4 flex-none" aria-hidden />
   563→            Add exercise to day
   564→          </button>
   565→        ) : null}
   566→      </div>
   567→    );
   568→  };
   569→
   570→  const notesVisible = editing || !!(routine?.notes && routine.notes.trim().length > 0);
   571→
   572→  // ---------- render ----------
   573→  return (
   574→    <Screen
   575→      topBar={
   576→        <TopBar
   577→          leading={
   578→            <Button
   579→              type="button"
   580→              variant="ghost"
   581→              size="icon"
   582→              className="h-11 w-11 flex-none"
   583→              onClick={() => navigate("/routines")}
   584→              aria-label="Back to routines"
   585→            >
   586→              <ChevronLeft className="h-5 w-5" aria-hidden />
   587→            </Button>
   588→          }
   589→          title={
   590→            renamingRoutine || !routine ? (
   591→              routine ? (
   592→                <InlineInput
   593→                  value={routine.name}
   594→                  ariaLabel="Rename routine"
   595→                  onCommit={(name) => {
   596→                    setRenamingRoutine(false);
   597→                    void renameRoutine(name);
   598→                  }}
   599→                  onCancel={() => setRenamingRoutine(false)}
   600→                  className="h-11 min-w-0 w-full"
   601→                />
   602→              ) : (
   603→                "Routine"
   604→              )
   605→            ) : (
   606→              <button
   607→                type="button"
   608→                className="min-w-0 flex-1 truncate text-left text-base font-semibold leading-none"
   609→                aria-label={`Rename ${routine.name}`}
   610→                title="Tap to rename"
   611→                onClick={() => setRenamingRoutine(true)}
   612→              >
   613→                {routine.name}
   614→              </button>
   615→            )
   616→          }
   617→          actions={
   618→            routine ? (
   619→              <>
   620→                <Button
   621→                  type="button"
   622→                  variant={editing ? "default" : "outline"}
   623→                  className="h-11 flex-none px-4"
   624→                  aria-pressed={editing}
   625→                  onClick={() => {
   626→                    if (!editing) setNotesDraft(routine?.notes ?? ""); // draft starts from the saved notes
   627→                    setEditing((e) => !e);
   628→                  }}
   629→                >
   630→                  {editing ? "Done" : "Edit"}
   631→                </Button>
   632→                <DropdownMenu>
   633→                  <DropdownMenuTrigger asChild>
   634→                    <Button
   635→                      type="button"
   636→                      variant="ghost"
   637→                      size="icon"
   638→                      className="h-11 w-11 flex-none"
   639→                      aria-label="More actions"
   640→                    >
   641→                      <MoreVertical className="h-5 w-5" aria-hidden />
   642→                    </Button>
   643→                  </DropdownMenuTrigger>
   644→                  <DropdownMenuContent align="end" className="w-48">
   645→                    <DropdownMenuItem onClick={() => setRenamingRoutine(true)}>
   646→                      <Pencil className="h-4 w-4" aria-hidden /> Rename
   647→                    </DropdownMenuItem>
   648→                    <DropdownMenuItem onClick={() => void copyRoutine()}>
   649→                      <Copy className="h-4 w-4" aria-hidden /> Copy
   650→                    </DropdownMenuItem>
   651→                    <DropdownMenuItem
   652→                      onClick={() => {
   653→                        setEditing(true);
   654→                        toast.info("Edit mode on — use ⋮ on a day to move it up or down");
   655→                      }}
   656→                    >
   657→                      <Layers className="h-4 w-4" aria-hidden /> Reorder days
   658→                    </DropdownMenuItem>
   659→                    <DropdownMenuItem
   660→                      className="text-destructive focus:text-destructive"
   661→                      onClick={() => setDeleteRoutineOpen(true)}
   662→                    >
   663→                      <Trash2 className="h-4 w-4" aria-hidden /> Delete
   664→                    </DropdownMenuItem>
   665→                  </DropdownMenuContent>
   666→                </DropdownMenu>
   667→              </>
   668→            ) : undefined
   669→          }
   670→        />
   671→      }
   672→      subBar={
   673→        notesVisible ? (
   674→          <SubBar>
   675→            <button
   676→              type="button"
   677→              className="flex h-11 min-w-0 flex-1 items-center gap-2 text-left"
   678→              aria-label={editing ? "Routine notes editor below" : "Expand routine notes"}
   679→              aria-expanded={editing ? true : notesOpen}
   680→              onClick={() => {
   681→                if (!editing) setNotesOpen((o) => !o);
   682→              }}
   683→            >
   684→              <StickyNote className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   685→              <span className="min-w-0 flex-1 truncate text-sm text-muted-foreground">
   686→                {editing
   687→                  ? routine?.notes?.trim()
   688→                    ? "Notes — editing below"
   689→                    : "Tap the field below to add notes"
   690→                  : routine?.notes}
   691→              </span>
   692→              {editing ? null : (
   693→                <ChevronDown
   694→                  className={`h-4 w-4 flex-none text-muted-foreground transition-transform ${notesOpen ? "" : "-rotate-90"}`}
   695→                  aria-hidden
   696→                />
   697→              )}
   698→            </button>
   699→          </SubBar>
   700→        ) : undefined
   701→      }
   702→    >
   703→      <ScrollBody>
   704→        {error ? (
   705→          <div className="flex h-[200px] flex-col items-center justify-center gap-3 rounded-lg border border-dashed border-border">
   706→            <p className="text-sm font-semibold">Routine not found</p>
   707→            <Button type="button" variant="outline" onClick={() => navigate("/routines")}>
   708→              Back to routines
   709→            </Button>
   710→          </div>
   711→        ) : isLoading ? (
   712→          <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading routine">
   713→            <div className="h-12 animate-pulse rounded-lg bg-muted/40" />
   714→            <div className="h-14 animate-pulse rounded-lg bg-muted/40" />
   715→            <div className="h-14 animate-pulse rounded-lg bg-muted/40" />
   716→          </div>
   717→        ) : routine ? (
   718→          <>
   719→            {editing ? (
   720→              <textarea
   721→                aria-label="Routine notes"
   722→                rows={3}
   723→                value={notesDraft}
   724→                placeholder="Notes for this routine (e.g. progression scheme, rest rules)…"
   725→                onChange={(e) => setNotesDraft(e.target.value)}
   726→                onBlur={() => void saveNotes()}
   727→                className="flex-none w-full resize-none rounded-lg border border-input bg-background px-3 py-2 text-sm leading-relaxed outline-none focus:ring-2 focus:ring-ring/40"
   728→              />
   729→            ) : notesOpen && routine.notes ? (
   730→              <p className="line-clamp-3 flex-none px-1 text-sm leading-relaxed text-muted-foreground">
   731→                {routine.notes}
   732→              </p>
   733→            ) : null}
   734→
   735→            {days.map((day, index) => (
   736→              <DaySection key={day.id} open={effectiveOpen(day.id)} header={renderDayHeader(day, index)}>
   737→                {renderDayBody(day)}
   738→              </DaySection>
   739→            ))}
   740→
   741→            {days.length === 0 ? (
   742→              <div className="flex h-[200px] flex-none flex-col items-center justify-center gap-3 rounded-lg border border-dashed border-border">
   743→                <p className="text-sm font-semibold">No days yet</p>
   744→                <p className="max-w-[280px] text-center text-xs text-muted-foreground">
   745→                  Split this routine into training days, then log a whole day in one tap.
   746→                </p>
   747→                {editing ? (
   748→                  <Button type="button" className="gap-1.5" onClick={() => void addDay()}>
   749→                    <Plus className="h-4 w-4" aria-hidden /> Add first day
   750→                  </Button>
   751→                ) : (
   752→                  <Button
   753→                    type="button"
   754→                    variant="outline"
   755→                    className="gap-1.5"
   756→                    onClick={() => {
   757→                      setEditing(true);
   758→                      toast.info("Edit mode on — add your first day");
   759→                    }}
   760→                  >
   761→                    <Pencil className="h-4 w-4" aria-hidden /> Edit routine
   762→                  </Button>
   763→                )}
   764→              </div>
   765→            ) : null}
   766→
   767→            {editing ? (
   768→              <button
   769→                type="button"
   770→                data-row
   771→                className="flex h-12 w-full items-center justify-center gap-2 overflow-hidden whitespace-nowrap rounded-lg border border-dashed border-border text-sm font-medium text-muted-foreground transition-colors hover:bg-accent/40"
   772→                onClick={() => void addDay()}
   773→              >
   774→                <Plus className="h-4 w-4 flex-none" aria-hidden />
   775→                Add day
   776→              </button>
   777→            ) : null}
   778→
   779→            <div className="h-4 flex-none" aria-hidden />
   780→          </>
   781→        ) : null}
   782→      </ScrollBody>
   783→
   784→      {/* confirm-destructive: routine delete */}
   785→      <AlertDialog open={deleteRoutineOpen} onOpenChange={(o) => !o && setDeleteRoutineOpen(false)}>
   786→        <AlertDialogContent>
   787→          <AlertDialogHeader>
   788→            <AlertDialogTitle>Delete routine?</AlertDialogTitle>
   789→            <AlertDialogDescription>
   790→              “{routine?.name}” and its {days.length} day{days.length === 1 ? "" : "s"} will be removed. Logged
   791→              workouts stay untouched. This cannot be undone.
   792→            </AlertDialogDescription>
   793→          </AlertDialogHeader>
   794→          <AlertDialogFooter>
   795→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   796→            <AlertDialogAction
   797→              className="bg-destructive text-white hover:bg-destructive/90"
   798→              onClick={(e) => {
   799→                e.preventDefault();
   800→                setDeleteRoutineOpen(false);
   801→                void removeRoutine();
   802→              }}
   803→            >
   804→              Delete routine
   805→            </AlertDialogAction>
   806→          </AlertDialogFooter>
   807→        </AlertDialogContent>
   808→      </AlertDialog>
   809→
   810→      {/* confirm-destructive: day delete */}
   811→      <AlertDialog open={deleteDay != null} onOpenChange={(o) => !o && setDeleteDay(null)}>
   812→        <AlertDialogContent>
   813→          <AlertDialogHeader>
   814→            <AlertDialogTitle>Delete day?</AlertDialogTitle>
   815→            <AlertDialogDescription>
   816→              “{deleteDay?.name}” and its {deleteDay?.exercises.length ?? 0} exercise
   817→              {(deleteDay?.exercises.length ?? 0) === 1 ? "" : "s"} will be removed from this routine. This
   818→              cannot be undone.
   819→            </AlertDialogDescription>
   820→          </AlertDialogHeader>
   821→          <AlertDialogFooter>
   822→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   823→            <AlertDialogAction
   824→              className="bg-destructive text-white hover:bg-destructive/90"
   825→              onClick={(e) => {
   826→                e.preventDefault();
   827→                const day = deleteDay;
   828→                setDeleteDay(null);
   829→                if (day) void removeDay(day);
   830→              }}
   831→            >
   832→              Delete day
   833→            </AlertDialogAction>
   834→          </AlertDialogFooter>
   835→        </AlertDialogContent>
   836→      </AlertDialog>
   837→    </Screen>
   838→  );
