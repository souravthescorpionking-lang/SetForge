    78→  control,

   121→/** 56px data-row: label | current value + chevron; whole row opens a small menu. */
   122→export function MenuRow<T extends string | number>({
   123→  label,
   124→  hint,
   125→  value,
   126→  options,
   127→  onSelect,
   128→}: {
   129→  label: string;
   130→  hint?: string;
   131→  value: T;
   132→  options: Array<{ value: T; label: string }>;
   133→  onSelect: (v: T) => void;
   134→}) {
   135→  const current = options.find((o) => o.value === value)?.label ?? String(value);
   136→  return (
   137→    <DropdownMenu>
   138→      <DropdownMenuTrigger asChild>
   139→        <button
   140→          type="button"
   141→          data-row
   142→          className={cn(ROW_CLS, "w-full text-left hover:bg-accent/40")}
   143→          aria-label={`${label}: ${current}`}
   144→        >
   145→          <span className="min-w-0 flex-1 truncate text-sm font-medium">
   146→            {label}
   147→            {hint ? <span className="ml-1.5 text-xs font-normal text-muted-foreground">{hint}</span> : null}
   148→          </span>
   149→          <span className="flex-none text-sm tabular-nums text-muted-foreground">{current}</span>
   150→          <ChevronRight className="h-4 w-4 flex-none text-muted-foreground" aria-hidden />
   151→        </button>
   152→      </DropdownMenuTrigger>
   153→      <DropdownMenuContent align="end" className="w-44">
   154→        {options.map((o) => (
   155→          <DropdownMenuItem key={String(o.value)} onClick={() => onSelect(o.value)}>
   156→            <span className="flex-1">{o.label}</span>
   157→            {o.value === value ? <Check className="h-4 w-4 text-primary" aria-hidden /> : null}
   158→          </DropdownMenuItem>
   159→        ))}
   160→      </DropdownMenuContent>
   161→    </DropdownMenu>
   162→  );
   163→}

   223→  // Coalesce rapid numeric edits into one PATCH (ported from legacy).

   225→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   226→  useEffect(() => {
   227→    const update = updateSettings;
   228→    return () => {
   229→      if (timerRef.current) clearTimeout(timerRef.current);
   230→      if (pendingRef.current) void update(pendingRef.current).catch(() => undefined);
   231→      pendingRef.current = null;
   232→    };
   233→  }, [updateSettings]);

   235→  if (!settings) return null;

   336→/** Sign out + full local wipe (logic ported from legacy useLogout). */
   337→function useSignOut() {
   338→  const setSession = useApp((s) => s.setSession);
   339→  const qc = useQueryClient();
   340→  return async () => {
   341→    try {
   342→      await authApi.logout();
   343→    } finally {
   344→      clearSwCaches();
   345→      wipeLocalData();
   346→      qc.clear();
   347→      setSession(null); // AppShell gate forces #/auth
   348→      toast.success("Signed out — local data cleared");
   349→    }
   350→  };
   351→}

   353→export function AccountSection() {
   354→  const session = useApp((s) => s.session);
   355→  const online = useOnline();
   356→  const logout = useSignOut();
   357→
   358→  const [pwOpen, setPwOpen] = useState(false);
   359→  const [currentPw, setCurrentPw] = useState("");
   360→  const [newPw, setNewPw] = useState("");
   361→  const [confirmPw, setConfirmPw] = useState("");
   362→  const [pwError, setPwError] = useState<string | null>(null);
   363→  const [pwBusy, setPwBusy] = useState(false);
   364→
   365→  const [delOpen, setDelOpen] = useState(false);
   366→  const [delTyped, setDelTyped] = useState("");
   367→  const [deleting, setDeleting] = useState(false);
   368→
   369→  const [signOutOpen, setSignOutOpen] = useState(false);
   370→
   371→  if (!session) return null;
   372→
   373→  const initials =
   374→    (session.user.name ?? session.user.email)
   375→      .split(/[\s@.]/)
   376→      .filter(Boolean)
   377→      .slice(0, 2)
   378→      .map((p) => p[0]?.toUpperCase())
   379→      .join("") || "SF";
   380→
   381→  const submitPassword = async () => {
   382→    setPwError(null);
   383→    if (newPw.length < 8) {
   384→      setPwError("New password must be at least 8 characters.");
   385→      return;
   386→    }
   387→    if (newPw !== confirmPw) {
   388→      setPwError("New passwords don't match.");
   389→      return;
   390→    }
   391→    setPwBusy(true);
   392→    try {
   393→      await accountApi.changePassword({ currentPassword: currentPw, newPassword: newPw });
   394→      toast.success("Password changed — other sessions signed out");
   395→      setPwOpen(false);
   396→      setCurrentPw("");
   397→      setNewPw("");
   398→      setConfirmPw("");
   399→    } catch (e) {
   400→      setPwError(e instanceof Error ? e.message : "Could not change password");
   401→    } finally {
   402→      setPwBusy(false);
   403→    }
   404→  };
   405→
   406→  const confirmDeleteAccount = async () => {
   407→    setDeleting(true);
   408→    try {
   409→      await accountApi.delete();
   410→      toast.success("Account deleted — all data removed");
   411→    } catch (e) {
   412→      toast.error(e instanceof Error ? e.message : "Could not delete account");
   413→      setDeleting(false);
   414→      return;
   415→    }
   416→    await logout();
   417→  };
   418→
   419→  return (
   420→    <section aria-label="Account" className="flex flex-col">
   421→      <SectionHeader title="Account" />
   422→
   423→      {/* signed-in identity */}
   424→      <div data-row className={ROW_CLS}>
   425→        <span
   426→          aria-hidden
   427→          className="flex h-8 w-8 flex-none items-center justify-center rounded-full bg-primary/15 text-xs font-bold text-primary"
   428→        >
   429→          {initials}
   430→        </span>
   431→        <span className="min-w-0 flex-1 truncate text-sm font-medium">{session.user.email}</span>
   432→        <span className="max-w-24 flex-none truncate text-xs text-muted-foreground">{session.user.name ?? "Athlete"}</span>
   433→      </div>
   434→
   435→      {/* change password — inline expansion (legacy Dialog → inline block) */}
   436→      <ActionRow label="Change password" onClick={() => setPwOpen((v) => !v)} />
   437→      {pwOpen ? (
   438→        <div className="flex flex-none flex-col gap-1 border-b border-border/50 bg-muted/20 p-2" aria-label="Change password form">
   439→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   440→            <span className="w-20 flex-none text-xs text-muted-foreground">Current</span>
   441→            <Input
   442→              type="password"
   443→              autoComplete="current-password"
   444→              className="h-12 flex-1 rounded-lg"
   445→              value={currentPw}
   446→              onChange={(e) => setCurrentPw(e.target.value)}
   447→              aria-label="Current password"
   448→              placeholder="••••••••"
   449→            />
   450→          </div>
   451→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   452→            <span className="w-20 flex-none text-xs text-muted-foreground">New</span>
   453→            <Input
   454→              type="password"
   455→              autoComplete="new-password"
   456→              className="h-12 flex-1 rounded-lg"
   457→              value={newPw}
   458→              onChange={(e) => setNewPw(e.target.value)}
   459→              aria-label="New password"
   460→              placeholder="At least 8 characters"
   461→            />
   462→          </div>
   463→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   464→            <span className="w-20 flex-none text-xs text-muted-foreground">Confirm</span>
   465→            <Input
   466→              type="password"
   467→              autoComplete="new-password"
   468→              className="h-12 flex-1 rounded-lg"
   469→              value={confirmPw}
   470→              onChange={(e) => setConfirmPw(e.target.value)}
   471→              aria-label="Confirm new password"
   472→              placeholder="Repeat new password"
   473→            />
   474→            <Button
   475→              type="button"
   476→              className="h-12 flex-none rounded-lg px-4"
   477→              disabled={pwBusy || !currentPw || newPw.length < 8 || newPw !== confirmPw}
   478→              onClick={() => void submitPassword()}
   479→            >
   480→              {pwBusy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   481→              Update
   482→            </Button>
   483→          </div>
   484→          {pwError ? (
   485→            <p className="flex-none px-1 text-xs text-destructive" role="alert">
   486→              {pwError}
   487→            </p>
   488→          ) : (
   489→            <p className="flex-none px-1 text-xs text-muted-foreground">
   490→              Minimum 8 characters · other sessions are signed out.
   491→            </p>
   492→          )}
   493→        </div>
   494→      ) : null}
   495→
   496→      {/* sign out — confirm-destructive */}
   497→      <ActionRow label="Sign out" onClick={() => setSignOutOpen(true)} />
   498→      <AlertDialog open={signOutOpen} onOpenChange={setSignOutOpen}>
   499→        <AlertDialogContent>
   500→          <AlertDialogHeader>
   501→            <AlertDialogTitle>Sign out?</AlertDialogTitle>
   502→            <AlertDialogDescription>
   503→              This ends your session and clears offline data stored on this device. You can sign back in
   504→              any time.
   505→            </AlertDialogDescription>
   506→          </AlertDialogHeader>
   507→          <AlertDialogFooter>
   508→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   509→            <AlertDialogAction
   510→              className="bg-destructive text-white hover:bg-destructive/90"
   511→              onClick={(e) => {
   512→                e.preventDefault();
   513→                setSignOutOpen(false);
   514→                void logout();
   515→              }}
   516→            >
   517→              <LogOut className="h-4 w-4" aria-hidden /> Sign out
   518→            </AlertDialogAction>
   519→          </AlertDialogFooter>
   520→        </AlertDialogContent>
   521→      </AlertDialog>
   522→
   523→      {/* delete account — inline typed-DELETE confirm (no dialog) */}
   524→      <ActionRow label="Delete account" destructive onClick={() => { setDelTyped(""); setDelOpen((v) => !v); }} />
   525→      {delOpen ? (
   526→        <div className="flex flex-none flex-col gap-1 border-b border-border/50 bg-destructive/5 p-2" aria-label="Delete account confirm">
   527→          <p className="flex flex-none items-start gap-1.5 px-1 text-xs leading-relaxed text-destructive">
   528→            <TriangleAlert className="mt-0.5 h-3.5 w-3.5 flex-none" aria-hidden />
   529→            Permanently deletes your account and every piece of data in it — workouts, PRs, routines,
   530→            body logs. No undo.
   531→          </p>
   532→          <div data-row className="flex h-12 items-center gap-2 overflow-hidden whitespace-nowrap">
   533→            <span className="w-20 flex-none text-xs text-muted-foreground">Type DELETE</span>
   534→            <Input
   535→              className="h-12 flex-1 rounded-lg font-mono tracking-widest"
   536→              value={delTyped}
   537→              onChange={(e) => setDelTyped(e.target.value)}
   538→              placeholder="DELETE"
   539→              autoComplete="off"
   540→              aria-label="Type DELETE to confirm account deletion"
   541→            />
   542→            <Button
   543→              type="button"
   544→              variant="destructive"
   545→              className="h-12 flex-none rounded-lg px-4"
   546→              disabled={delTyped.trim().toUpperCase() !== "DELETE" || deleting || !online}
   547→              onClick={() => void confirmDeleteAccount()}
   548→            >
   549→              {deleting ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Trash2 className="h-4 w-4" aria-hidden />}
   550→              Delete
   551→            </Button>
   552→          </div>
   553→        </div>

   555→    </section>

   561→function downloadBlob(blob: Blob, filename: string) {
   562→  const url = URL.createObjectURL(blob);
   563→  const a = document.createElement("a");
   564→  a.href = url;
   565→  a.download = filename;
   566→  document.body.appendChild(a);
   567→  a.click();
   568→  a.remove();
   569→  setTimeout(() => URL.revokeObjectURL(url), 2000);
   570→}

   572→export function DataSection() {
   573→  const invalidate = useInvalidate();
   574→  const online = useOnline();
   575→
   576→  const [exporting, setExporting] = useState(false);
   577→  const [csvBusy, setCsvBusy] = useState<"workouts" | "body" | null>(null);
   578→  const [recalcing, setRecalcing] = useState(false);
   579→
   580→  // ----- import (legacy dialog → inline expansion) -----
   581→  const fileInputRef = useRef<HTMLInputElement>(null);
   582→  const [importOpen, setImportOpen] = useState(false);
   583→  const [pendingImport, setPendingImport] = useState<unknown | null>(null);
   584→  const [importMode, setImportMode] = useState<"replace" | "merge">("merge");
   585→  const [importing, setImporting] = useState(false);
   586→  const [importFileName, setImportFileName] = useState<string | null>(null);
   587→
   588→  const resetFileInput = () => {
   589→    setImportFileName(null);
   590→    if (fileInputRef.current) fileInputRef.current.value = "";
   591→  };
   592→
   593→  const onFileChosen = async (file: File) => {
   594→    setImportFileName(file.name);
   595→    if (file.size > MAX_IMPORT_BYTES) {
   596→      toast.error("File too large — backups are limited to 50 MB");
   597→      resetFileInput();
   598→      return;
   599→    }
   600→    try {
   601→      const text = await file.text();
   602→      const data = JSON.parse(text);
   603→      if (typeof data !== "object" || data === null || Array.isArray(data)) {
   604→        throw new Error("not an object");
   605→      }
   606→      setPendingImport(data);
   607→    } catch {
   608→      toast.error("That file is not valid JSON — expected a SetForge backup");
   609→      resetFileInput();
   610→    }
   611→  };
   612→
   613→  const confirmImport = async () => {
   614→    if (pendingImport == null) return;
   615→    setImporting(true);
   616→    try {
   617→      const res = await accountApi.importBackup(pendingImport, importMode);
   618→      toast.success(
   619→        `Import complete — ${res.importedWorkouts} workout${res.importedWorkouts === 1 ? "" : "s"} restored`,
   620→      );
   621→      setPendingImport(null);
   622→      resetFileInput();
   623→      invalidate.all();
   624→    } catch (e) {
   625→      toast.error(e instanceof Error ? e.message : "Import failed");
   626→    } finally {
   627→      setImporting(false);
   628→    }
   629→  };
   630→
   631→  // ----- clear data (delete workout history) — confirm-destructive -----
   632→  const [clearOpen, setClearOpen] = useState(false);
   633→  const [clearing, setClearing] = useState(false);
   634→
   635→  const confirmClear = async () => {
   636→    setClearing(true);
   637→    try {
   638→      const res = await accountApi.deleteHistory({ mode: "all" });
   639→      toast.success(
   640→        `Deleted ${res.deletedWorkouts} workout${res.deletedWorkouts === 1 ? "" : "s"} — PRs recalculated`,
   641→      );
   642→      setClearOpen(false);
   643→      invalidate.all();
   644→    } catch (e) {
   645→      toast.error(e instanceof Error ? e.message : "Could not delete history");
   646→    } finally {
   647→      setClearing(false);
   648→    }
   649→  };
   650→
   651→  // ----- actions -----
   652→  const exportBackup = async () => {
   653→    setExporting(true);
   654→    try {
   655→      const backup = await accountApi.exportBackup();
   656→      downloadBlob(
   657→        new Blob([JSON.stringify(backup, null, 2)], { type: "application/json" }),
   658→        `setforge-backup-${todayKey()}.json`,
   659→      );
   660→      toast.success("Backup downloaded — keep it somewhere safe");
   661→    } catch (e) {
   662→      toast.error(e instanceof Error ? e.message : "Export failed");
   663→    } finally {
   664→      setExporting(false);
   665→    }
   666→  };
   667→
   668→  const exportCsv = async (type: "workouts" | "body") => {
   669→    setCsvBusy(type);
   670→    try {
   671→      const blob = await accountApi.exportCsv(type);
   672→      downloadBlob(blob, `setforge-${type}-${todayKey()}.csv`);
   673→      toast.success(`${type === "workouts" ? "Workouts" : "Body"} CSV downloaded`);
   674→    } catch (e) {
   675→      toast.error(e instanceof Error ? e.message : "Export failed");
   676→    } finally {
   677→      setCsvBusy(null);
   678→    }
   679→  };
   680→
   681→  const recalculate = async () => {
   682→    setRecalcing(true);
   683→    try {
   684→      const res = await recordsApi.recalculate();
   685→      toast.success(`Recalculated ${res.recalculated} exercises`);
   686→      invalidate.all();
   687→    } catch (e) {
   688→      toast.error(e instanceof Error ? e.message : "Recalculation failed");
   689→    } finally {
   690→      setRecalcing(false);
   691→    }
   692→  };
   693→
   694→  return (
   695→    <section aria-label="Data" className="flex flex-col">
   696→      <SectionHeader title="Data" />
   697→
   698→      <ActionRow
   699→        label="Export data"
   700→        hint="JSON backup"
   701→        disabled={exporting || !online}
   702→        onClick={() => void exportBackup()}
   703→        trailing={
   704→          exporting ? (
   705→            <Loader2 className="h-4 w-4 flex-none animate-spin text-muted-foreground" aria-hidden />
   706→          ) : undefined
   707→        }
   708→      />
   709→      <ActionRow
   710→        label="Workouts CSV"
   711→        disabled={csvBusy !== null || !online}
   712→        onClick={() => void exportCsv("workouts")}
   713→        trailing={
   714→          csvBusy === "workouts" ? (
   715→            <Loader2 className="h-4 w-4 flex-none animate-spin text-muted-foreground" aria-hidden />
   716→          ) : undefined
   717→        }
   718→      />
   719→      <ActionRow
   720→        label="Body CSV"
   721→        disabled={csvBusy !== null || !online}
   722→        onClick={() => void exportCsv("body")}
   723→        trailing={
   724→          csvBusy === "body" ? (
   725→            <Loader2 className="h-4 w-4 flex-none animate-spin text-muted-foreground" aria-hidden />
   726→          ) : undefined
   727→        }
   728→      />
   729→
   730→      {/* import / restore — inline expansion */}
   731→      <ActionRow
   732→        label="Import backup"
   733→        hint={importFileName ? importFileName.slice(0, 18) : "merge / replace"}
   734→        onClick={() => setImportOpen((v) => !v)}
   735→      />
   736→      {importOpen ? (
   737→        <div className="flex flex-none flex-col gap-1 border-b border-border/50 bg-muted/20 p-2" aria-label="Import backup">
   738→          <input
   739→            ref={fileInputRef}
   740→            type="file"
   741→            accept=".json,application/json"
   742→            className="hidden"
   743→            aria-hidden
   744→            tabIndex={-1}
   745→            onChange={(e) => {
   746→              const file = e.target.files?.[0];
   747→              if (file) void onFileChosen(file);
   748→            }}
   749→          />
   750→          <ActionRow
   751→            label={importFileName ? `File · ${importFileName}` : "Choose backup file (.json)"}
   752→            ariaLabel="Choose backup file"
   753→            disabled={!online}
   754→            onClick={() => fileInputRef.current?.click()}
   755→          />
   756→          {pendingImport != null ? (
   757→            <>
   758→              <MenuRow
   759→                label="Mode"
   760→                value={importMode}
   761→                options={[
   762→                  { value: "merge", label: "Merge" },
   763→                  { value: "replace", label: "Replace all data" },
   764→                ]}
   765→                onSelect={(v) => setImportMode(v)}
   766→              />
   767→              <ActionRow
   768→                label={importMode === "replace" ? "Replace & import" : "Merge & import"}
   769→                disabled={importing || !online}
   770→                onClick={() => void confirmImport()}
   771→                trailing={
   772→                  importing ? (
   773→                    <Loader2 className="h-4 w-4 flex-none animate-spin text-muted-foreground" aria-hidden />
   774→                  ) : undefined
   775→                }
   776→              />
   777→              {importMode === "replace" ? (
   778→                <p className="flex-none px-1 text-xs text-destructive" role="alert">
   779→                  Replace deletes your current workouts, routines, body measurements and goals first.
   780→                  No undo — export a backup first.
   781→                </p>
   782→              ) : null}
   783→            </>
   784→          ) : (
   785→            <p className="flex-none px-1 text-xs text-muted-foreground">
   786→              Restore data from a SetForge backup file (max 50 MB).
   787→            </p>
   788→          )}
   789→        </div>
   790→      ) : null}
   791→
   792→      <ActionRow
   793→        label="Recalculate PRs"
   794→        disabled={recalcing || !online}
   795→        onClick={() => void recalculate()}
   796→        trailing={
   797→          recalcing ? (
   798→            <Loader2 className="h-4 w-4 flex-none animate-spin text-muted-foreground" aria-hidden />
   799→          ) : undefined
   800→        }
   801→      />
   802→
   803→      {/* clear data — confirm-destructive */}
   804→      <ActionRow label="Clear data" hint="all workouts" destructive onClick={() => setClearOpen(true)} />
   805→      <AlertDialog open={clearOpen} onOpenChange={(o) => !clearing && setClearOpen(o)}>
   806→        <AlertDialogContent>

   810→              This permanently deletes every logged workout and recalculates your personal records.

   828→          </AlertDialogFooter>
   829→        </AlertDialogContent>
   830→      </AlertDialog>
   831→    </section>

   837→export function AppSection() {
   838→  const online = useOnline();

   840→  const [standalone] = useState(() => isStandalone());

   843→  const clearCache = async () => {
   844→    setClearing(true);
   845→    try {
   846→      const regs = await navigator.serviceWorker?.getRegistrations();
   847→      const sw = regs?.[0]?.active;
   848→      sw?.postMessage({ type: "CLEAR_CACHES" });
   849→      if ("caches" in window) {
   850→        const names = await caches.keys();
   851→        await Promise.all(names.map((n) => caches.delete(n)));
   852→      }
   853→      toast.success("Offline cache cleared", {
   854→        description: "Everything re-downloads fresh on your next visit.",
   855→      });
   856→    } catch {
   857→      toast.error("Could not clear the offline cache");
   858→    } finally {
   859→      setClearing(false);
   860→    }
   861→  };
   862→
   863→  const isInstalled = installed || standalone;
   864→
   865→  return (

   867→      <SectionHeader title="App" />

   874→            <Button
   875→              type="button"
