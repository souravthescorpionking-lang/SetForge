     3→// ─────────────────────────────────────────────────────────────────────────────
     4→// SetRow — THE single source of truth set row (Part 3 §SET ROW).
     5→//
     6→// One 40px CSS-grid row for every mode (edit / read / preview / template).
     7→// The grid template is COMPUTED, never styled ad hoc:
     8→//   • spec track widths:  #24 · type32 · f1:72 · f2:56 · rpe:52 · tempo:64 ·
     9→//                          rest:56 · done:40 · more:32
    10→//   • missing modality fields or settings-hidden columns REMOVE tracks from
    11→//     the template (they are never "hidden" cells).
    12→//   • <468px viewport: rpe + tempo tracks are REMOVED — their values/editors
    13→//     move into the `more` ⋯ popover. (PLATFORM ADAPTATION: the spec's bottom
    14→//     sheet is forbidden here; an anchored popover — shadcn Popover on
    15→//     Floating UI with collision flip — is the allowed replacement.)
    16→//   • <360px viewport: f1/f2 drop their floors to spec−16 (minmax) so the row
    17→//     still fits a 320px screen. All small tracks stay exact px.
    18→//   • f1/f2 carry `minmax(spec, 1fr)` so value columns fill leftover card width
    19→//     at every viewport — the row can never overflow its card and never leaves
    20→//     a dead right margin.
    21→//
    22→// Grid math (weight+reps modality, all columns on):
    23→//   wide fixed tracks 24+32+72+56+52+64+56+40+32 = 428px
    24→//   mobile (<468, rpe+tempo dropped)             = 312px ≤ 328px (360−32 law)
    25→//   320px viewport floors 24+32+56+40+56+40+32   = 280px ≤ 282px available
    26→//   (The spec contract names 460 as the wide threshold; the corrected value is
    27→//   468 — 428px of tracks + 32px page gutters + 6px card chrome (border×2 +
    28→//   4px colour bar) need ≥466px, so the wide layout only engages at 468.)
    29→// ─────────────────────────────────────────────────────────────────────────────

    66→/** Spec track widths in px — the single source of truth for the row grid. */
    67→const TRACK = { index: 24, setType: 32, f1: 72, f2: 56, rpe: 52, tempo: 64, rest: 56, done: 40, more: 32 } as const;
    68→
    69→/** Viewport at which rpe/tempo tracks return to the grid (see file header). */
    70→const WIDE_VIEWPORT = 468;
    71→
    72→/** Tap→cycle order for the set-type chip. */
    73→const TYPE_CYCLE: SetType[] = ["NORMAL", "WARMUP", "DROP", "FAILURE", "AMRAP"];
    74→
    75→/** Rest presets (sec) — spec list. */
    76→const REST_PRESETS = [30, 60, 90, 120, 150, 180];
    77→
    78→/** RPE chip scale: 1–10 in half steps (spec). */
    79→const RPE_CHIPS: number[] = [];
    80→for (let v = 1; v <= 10; v += 0.5) RPE_CHIPS.push(v);

    82→type TrackKey = "index" | "setType" | "f1" | "f2" | "rpe" | "tempo" | "rest" | "done" | "more";
    83→
    84→type GridSpec = { keys: TrackKey[]; template: string };
    85→
    86→function buildGrid(
    87→  fields: SetField[],
    88→  visible: CardVisibleColumns,
    89→  mode: CardMode,
    90→  vw: number,
    91→  moreTrack: boolean,
    92→): GridSpec {
    93→  const keys: TrackKey[] = [];
    94→  const parts: string[] = [];
    95→  const push = (key: TrackKey, css: string) => {
    96→    keys.push(key);
    97→    parts.push(css);
    98→  };
    99→
   100→  push("index", `${TRACK.index}px`);
   101→  if (visible.setType) push("setType", `${TRACK.setType}px`);
   102→
   103→  // Value tracks: spec floor (−16 below 360px — documented adaptation so the
   104→  // fixed small tracks still fit a 320px screen) + 1fr growth to fill the card.
   105→  const narrow = vw < 360;
   106→  if (fields[0]) push("f1", `minmax(${TRACK.f1 - (narrow ? 16 : 0)}px, 1fr)`);
   107→  if (fields[1]) push("f2", `minmax(${TRACK.f2 - (narrow ? 16 : 0)}px, 1fr)`);
   108→
   109→  if (visible.rpe && vw >= WIDE_VIEWPORT) push("rpe", `${TRACK.rpe}px`);
   110→  if (visible.tempo && vw >= WIDE_VIEWPORT) push("tempo", `${TRACK.tempo}px`);
   111→  if (visible.rest) push("rest", `${TRACK.rest}px`);
   112→
   113→  // `done` never applies to routine templates → track removed.
   114→  if (mode !== "template") push("done", `${TRACK.done}px`);
   115→  if (mode === "edit" || mode === "template") push("more", `${TRACK.more}px`);
   116→  else if (moreTrack) push("more", `${TRACK.more}px`);
   117→
   118→  return { keys, template: parts.join(" ") };
   119→}

   310→function TypeCell({
   311→  set,
   312→  mode,
   313→  editable,
   314→  onAction,
   315→}: {
   316→  set: CardSet;
   317→  mode: CardMode;
   318→  editable: boolean;
   319→  onAction?: (a: CardAction) => void;
   320→}) {
   321→  const [open, setOpen] = useState(false);
   322→  const holdRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   323→  const startRef = useRef<{ x: number; y: number } | null>(null);
   324→  const longPressed = useRef(false);
   325→
   326→  const endHold = () => {
   327→    if (holdRef.current) clearTimeout(holdRef.current);
   328→    holdRef.current = null;
   329→    startRef.current = null;
   330→  };
   331→
   332→  useEffect(() => () => endHold(), []);
   333→
   334→  const value = set.setType ?? null;
   335→  const meta = value ? SET_TYPE_META[value as SetType] : null;
   336→  const templateBlank = mode === "template" && value == null;
   337→
   338→  // read / preview / read-only fallback → static glyph
   339→  if (!editable || !onAction) {
   340→    if (meta) {
   341→      return (
   342→        <span
   343→          className={cn("flex h-6 w-6 items-center justify-center rounded-md text-[11px] font-bold", meta.className)}
   344→          title={`${meta.label} — ${meta.description}`}
   345→        >
   346→          {meta.letter}
   347→        </span>
   348→      );
   349→    }
   350→    return (
   351→      <span className="text-xs text-muted-foreground/40" title="Set type not set">
   352→        –
   353→      </span>
   354→    );
   355→  }
   356→
   357→  const cycle = () => {
   358→    const current = (value ?? "NORMAL") as SetType;
   359→    const next = TYPE_CYCLE[(TYPE_CYCLE.indexOf(current) + 1) % TYPE_CYCLE.length];
   360→    onAction({ type: "update-set", setId: set.id, patch: { setType: next } });
   361→  };
   362→
   363→  return (
   364→    <Popover open={open} onOpenChange={setOpen}>
   365→      <PopoverTrigger asChild>
   366→        <button
   367→          type="button"
   368→          aria-label={
   369→            meta
   370→              ? `Set type ${meta.label}. Tap to cycle, hold for picker`
   371→              : "Set type not set. Tap to choose"
   372→          }
   373→          title={meta ? `${meta.label} — ${meta.description}` : "Choose set type"}
   374→          onClick={(e) => {
   375→            // Fully controlled popover: preventDefault suppresses Radix's own
   376→            // click-toggle (composeEventHandlers skips when defaultPrevented),
   377→            // so tap = cycle · hold / right-click / Enter = picker.
   378→            e.preventDefault();
   379→            if (longPressed.current) {
   380→              longPressed.current = false;
   381→              return;
   382→            }
   383→            if (templateBlank) {
   384→              setOpen(true);
   385→              return;
   386→            }
   387→            cycle();
   388→          }}
   389→          onPointerDown={(e) => {
   390→            startRef.current = { x: e.clientX, y: e.clientY };
   391→            longPressed.current = false;
   392→            holdRef.current = setTimeout(() => {
   393→              longPressed.current = true;
   394→              setOpen(true);
   395→            }, 420);
   396→          }}
   397→          onPointerMove={(e) => {
   398→            const s = startRef.current;
   399→            if (s && (Math.abs(e.clientX - s.x) > 8 || Math.abs(e.clientY - s.y) > 8)) endHold();
   400→          }}
   401→          onPointerUp={endHold}
   402→          onPointerCancel={endHold}
   403→          onPointerLeave={endHold}
   404→          onContextMenu={(e) => {
   405→            e.preventDefault();
   406→            endHold();
   407→            setOpen(true);
   408→          }}
   409→          className={cn(
   410→            "flex h-7 w-7 items-center justify-center rounded-md text-[11px] font-bold transition-transform active:scale-95",
   411→            templateBlank ? "border border-dashed border-border text-muted-foreground" : meta?.className,
   412→          )}
   413→        >
   414→          {templateBlank ? "–" : (meta?.letter ?? "–")}
   415→        </button>
   416→      </PopoverTrigger>
   417→      <PopoverContent align="center" className="w-56 p-2">
   418→        <p className="px-1 pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Set type</p>
   419→        <div className="flex flex-col gap-1">
   420→          {SET_TYPES.map((t) => (
   421→            <button
   422→              key={t}
   423→              type="button"
   424→              onClick={() => {
   425→                onAction({ type: "update-set", setId: set.id, patch: { setType: t } });
   426→                setOpen(false);
   427→              }}
   428→              className="flex w-full items-center gap-2 rounded-lg px-2 py-1 text-left transition-colors hover:bg-accent"
   429→            >
   430→              <span
   431→                className={cn(
   432→                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[11px] font-bold",
   433→                  SET_TYPE_META[t].className,
   434→                )}
   435→              >
   436→                {SET_TYPE_META[t].letter}
   437→              </span>
   438→              <span className="min-w-0 flex-1">
   439→                <span className="block text-sm font-semibold leading-tight">{SET_TYPE_META[t].label}</span>
   440→                <span className="block text-[11px] leading-tight text-muted-foreground">
   441→                  {SET_TYPE_META[t].description}
   442→                </span>
   443→              </span>
   444→            </button>
   445→          ))}
   446→        </div>
   447→      </PopoverContent>
   448→    </Popover>
   449→  );
   450→}

   452→function ValueCell({
   453→  field,
   454→  set,
   455→  exercise,
   456→  mode,
   457→  editable,
   458→  onAction,
   459→}: {
   460→  field: SetField;
   461→  set: CardSet;
   462→  exercise: CardExercise;
   463→  mode: CardMode;
   464→  editable: boolean;
   465→  onAction?: (a: CardAction) => void;
   466→}) {
   467→  const key = FIELD_TO_KEY[field];
   468→  const value = (set[key] as number | null | undefined) ?? null;
   469→  const [draft, setDraft] = useState<string | null>(null);
   470→  const [focused, setFocused] = useState(false);
   471→  const step = stepForField(field, exercise);
   472→  const templateBlank = mode === "template" && value == null;
   473→  const label = FIELD_LABEL[field];
   474→
   475→  if (!editable || !onAction) {
   476→    return (
   477→      <span
   478→        className="w-full truncate px-1 text-right text-sm tabular-nums text-foreground"
   479→        title={formatFieldValue(field, value, exercise.unit)}
   480→      >
   481→        {formatFieldValue(field, value, exercise.unit)}
   482→      </span>
   483→    );
   484→  }
   485→
   486→  const display = draft ?? (value == null ? "" : trimNum(value));
   487→
   488→  const commit = () => {
   489→    if (draft != null) {
   490→      const parsed = parseFieldValue(field, draft);
   491→      if (parsed !== undefined && parsed !== value) {
   492→        onAction({ type: "update-set", setId: set.id, patch: { [key]: parsed } as Partial<CardSet> });
   493→      }
   494→    }
   495→    setDraft(null);
   496→  };
   497→
   498→  const stepBy = (dir: 1 | -1) => {
   499→    const parsedDraft = draft != null ? parseFieldValue(field, draft) : undefined;
   500→    const base = (parsedDraft != null ? parsedDraft : value) ?? 0;
   501→    const next = Math.max(0, Math.round((base + dir * step) * 100) / 100);
   502→    onAction({ type: "update-set", setId: set.id, patch: { [key]: next } as Partial<CardSet> });
   503→    setDraft(null);
   504→  };
   505→
   506→  const stepperButton = (dir: 1 | -1) => (
   507→    <button
   508→      type="button"
   509→      aria-label={dir < 0 ? `Decrease ${label}` : `Increase ${label}`}
   510→      // Keep focus on the input so the steppers don't unmount before click.
   511→      onMouseDown={(e) => e.preventDefault()}
   512→      onClick={() => stepBy(dir)}
   513→      className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   514→    >
   515→      {dir < 0 ? <Minus className="h-3 w-3" aria-hidden /> : <Plus className="h-3 w-3" aria-hidden />}
   516→    </button>
   517→  );
   518→
   519→  // Grid keyboard navigation (audit D6/M2): Enter/↓ → same column one row down
   520→  // (Enter on the last row adds a set); ↑ → one row up.
   521→  const moveFocus = (dir: 1 | -1, input: HTMLInputElement): boolean => {
   522→    const cell = input.parentElement?.parentElement;
   523→    const row = cell?.parentElement;
   524→    if (!cell || !row || !row.parentElement) return false;
   525→    const colIndex = Array.prototype.indexOf.call(row.children, cell);
   526→    const rows = Array.from(row.parentElement.querySelectorAll<HTMLElement>("[data-row]"));
   527→    const rowIndex = rows.indexOf(row);
   528→    const target = rows[rowIndex + dir];
   529→    if (!target) return false;
   530→    const targetCell = target.children[colIndex] as HTMLElement | undefined;
   531→    const focusable = targetCell?.querySelector<HTMLElement>("input, button");
   532→    if (focusable) {
   533→      focusable.focus();
   534→      return true;
   535→    }
   536→    return false;
   537→  };
   538→
   539→  const onInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   540→    if (e.key === "ArrowUp") {
   541→      e.preventDefault();
   542→      commit();
   543→      moveFocus(-1, e.currentTarget);
   544→    } else if (e.key === "ArrowDown" || e.key === "Enter") {
   545→      e.preventDefault();
   546→      commit();
   547→      const input = e.currentTarget;
   548→      if (!moveFocus(1, input)) {
   549→        // Last row: Enter adds a new set (prefilled by the card) and focuses it.
   550→        onAction({ type: "add-set" });
   551→        requestAnimationFrame(() => {
   552→          const row = input.parentElement?.parentElement?.parentElement;
   553→          const rows = row?.parentElement
   554→            ? Array.from(row.parentElement.querySelectorAll<HTMLElement>("[data-row]"))
   555→            : [];
   556→          const cell = input.parentElement?.parentElement;
   557→          const colIndex = cell && row ? Array.prototype.indexOf.call(row.children, cell) : -1;
   558→          const last = rows[rows.length - 1];
   559→          const focusable =
   560→            colIndex >= 0 ? (last?.children[colIndex] as HTMLElement | undefined)?.querySelector<HTMLElement>("input") : null;
   561→          focusable?.focus();
   562→        });
   563→      }
   564→    }
   565→  };
   566→
   567→  return (
   568→    <div className="flex h-full min-w-0 items-center">
   569→      {/* template blank: ↺ = copy-last (typed value replaces it) */}
   570→      {templateBlank && !focused ? (
   571→        <button
   572→          type="button"
   573→          aria-label={`Copy previous set ${label.toLowerCase()}`}
   574→          title="Copy previous set"
   575→          onClick={() => onAction({ type: "copy-last", setId: set.id })}
   576→          className="flex h-8 w-5 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   577→        >
   578→          <RotateCcw className="h-3 w-3" aria-hidden />
   579→        </button>
   580→      ) : null}
   581→      {/* focused: inline steppers inside the cell ([− 20px | input | + 20px]) */}
   582→      {focused ? stepperButton(-1) : null}
   583→      <input
   584→        aria-label={label}
   585→        inputMode={field === "reps" ? "numeric" : "decimal"}
   586→        value={display}
   587→        onFocus={() => setFocused(true)}
   588→        onBlur={() => {
   589→          commit();
   590→          setFocused(false);
   591→        }}
   592→        onKeyDown={onInputKeyDown}
   593→        onChange={(e) => setDraft(e.target.value)}
   594→        className="h-full min-w-0 flex-1 truncate rounded-sm border-0 bg-transparent px-1 text-right text-sm tabular-nums text-foreground outline-none focus:bg-muted/40"
   595→      />
   596→      {focused ? stepperButton(1) : null}
   597→    </div>

   740→function MoreCell({
   741→  set,
   742→  editable,
   743→  showRpeEditor,
   744→  showTempoEditor,
   745→  allowRemoveSet,
   746→  onAction,
   747→}: {
   748→  set: CardSet;
   749→  editable: boolean;
   750→  showRpeEditor: boolean;
   751→  showTempoEditor: boolean;
   752→  /** template mode: predefined sets can be deleted from the row ⋯ popover. */
   753→  allowRemoveSet?: boolean;
   754→  onAction?: (a: CardAction) => void;
   755→}) {
   756→  const [open, setOpen] = useState(false);
   757→  const [noteDraft, setNoteDraft] = useState<string | null>(null);
   758→
   759→  // read / preview: PR/note glyph (inline — the preferred PR badge spot)
   760→  if (!editable || !onAction) {
   761→    if (set.isNewPr) return <Trophy className="h-3.5 w-3.5 text-amber-400" role="img" aria-label="New personal record" />;
   762→    if (set.note) return <StickyNote className="h-3.5 w-3.5 text-muted-foreground" role="img" aria-label="Has note" />;
   763→    return null;
   764→  }
   765→
   766→  const commitNote = () => {
   767→    if (noteDraft != null && noteDraft !== (set.note ?? "")) {
   768→      onAction({ type: "update-set", setId: set.id, patch: { note: noteDraft } });
   769→    }
   770→  };
   771→
   772→  const handleOpenChange = (next: boolean) => {
   773→    if (!next) commitNote();
   774→    setOpen(next);
   775→    if (next) setNoteDraft(null);

   778→  const dropped = showRpeEditor || showTempoEditor;
   779→
   780→  return (
   781→    <Popover open={open} onOpenChange={handleOpenChange}>

   812→          <textarea

   817→            onBlur={commitNote}

   819→            className="w-full resize-none rounded-md border border-input bg-background px-2 py-1 text-xs outline-none focus:ring-2 focus:ring-ring/40"

   832→              type="button"

   843→          </div>
