    54→// ── local unit conversion helpers (metric is canonical on the wire) ──────────
    55→
    56→const CM_PER_IN = 2.54;
    57→const KG_PER_LB = 0.45359237;
    58→
    59→const roundTo1 = (n: number): number => Math.round(n * 10) / 10;
    60→
    61→/** 177.8 cm → "5 ft 10 in" (imperial height display). */
    62→function cmToFtIn(cm: number): string {
    63→  const totalIn = cm / CM_PER_IN;
    64→  let ft = Math.floor(totalIn / 12);
    65→  let inch = Math.round(totalIn - ft * 12);
    66→  if (inch >= 12) {
    67→    ft += 1;
    68→    inch -= 12;
    69→  }
    70→  return `${ft} ft ${inch} in`;
    71→}
    72→
    73→const cmToIn = (cm: number): number => cm / CM_PER_IN;
    74→const inToCm = (inches: number): number => inches * CM_PER_IN;
    75→const kgToLb = (kg: number): number => kg / KG_PER_LB;
    76→const lbToKg = (lb: number): number => lb * KG_PER_LB;
    77→const identity = (n: number): number => n;

   110→export default function ProfileScreen() {
   111→  const session = useApp((s) => s.session);
   112→  const settings = useApp((s) => s.settings);
   113→  const qc = useQueryClient();
   114→
   115→  // Same key as the app-shell onboarding gate (staleTime Infinity — the cache
   116→  // is written synchronously after each save, then background-revalidated).
   117→  const { data: profile, isLoading } = useQuery({
   118→    queryKey: ["profile"],
   119→    queryFn: () => profileApi.get(),
   120→    staleTime: Infinity,
   121→  });
   122→
   123→  const [editing, setEditing] = useState<EditKey | null>(null);
   124→
   125→  const imperial = settings?.unitSystem === "imperial";
   126→
   127→  /** Save a profile patch: cache write on success + background revalidate. */
   128→  const savePatch = async (patch: ProfilePatch): Promise<void> => {
   129→    try {
   130→      const dto = await profileApi.update(patch);
   131→      qc.setQueryData(["profile"], dto);
   132→      void qc.invalidateQueries({ queryKey: ["profile"] });
   133→      toast.success("Saved");
   134→      setEditing(null);
   135→    } catch (e) {
   136→      toast.error(e instanceof Error ? e.message : "Could not save");
   137→    }
   138→  };
   139→
   140→  if (!session || !settings) {
   141→    return (
   142→      <Screen topBar={<TopBar title="Profile" />}>
   143→        <ScrollBody contentClassName="flex flex-col gap-3" >
   144→          <div className="flex flex-col items-center gap-3 py-8" aria-busy="true" aria-label="Loading profile">
   145→            <Skeleton className="h-16 w-16 rounded-full" />
   146→            <Skeleton className="h-4 w-40 rounded-lg" />
   147→            <Skeleton className="h-3 w-56 rounded-lg" />
   148→          </div>
   149→          <Skeleton className="h-8 w-24 rounded-lg" />
   150→          {Array.from({ length: 6 }, (_, i) => (
   151→            <Skeleton key={i} className="h-14 w-full rounded-lg" />
   152→          ))}
   153→        </ScrollBody>
   154→      </Screen>
   155→    );
   156→  }
   157→
   158→  const initials =
   159→    (session.user.name ?? session.user.email)
   160→      .split(/[\s@.]/)
   161→      .filter(Boolean)
   162→      .slice(0, 2)
   163→      .map((p) => p[0]?.toUpperCase())
   164→      .join("") || "SF";
   165→

   167→    <Screen topBar={<TopBar title="Profile" />}>
   168→      <ScrollBody contentClassName="flex flex-col gap-4">

   170→        <header className="flex flex-col items-center gap-2 py-4 text-center">
   171→          <span
   172→            aria-hidden
   173→            className="flex h-16 w-16 items-center justify-center rounded-full bg-primary/10 text-xl font-bold text-primary"
   174→          >
   175→            {initials}
   176→          </span>
   177→          <div>
   178→            <p className="text-lg font-bold leading-none">{session.user.name ?? "Athlete"}</p>
   179→            <p className="mt-1 truncate text-xs text-muted-foreground">{session.user.email}</p>

   181→          <p className="text-sm text-muted-foreground">Ready to forge.</p>
   182→        </header>
   183→
   184→        <ProfileSection
   185→          profile={profile}
   186→          loading={isLoading}
   187→          imperial={imperial}
   188→          editing={editing}
   189→          setEditing={setEditing}
   190→          savePatch={savePatch}
   191→        />
   192→
   193→        <NotificationsSection />

   195→        <AccountSection />

   330→          onCancel={() => setEditing(null)}
   331→        />
   332→      ) : null}

   347→            if (goal !== v) void savePatch({ goal: v });
   348→          }}
   349→          onCancel={() => setEditing(null)}
   350→        />
   351→      ) : null}

   353→      <ProfileRow
   354→        label="Days per week"
   355→        value={profile.daysPerWeekTarget != null ? String(profile.daysPerWeekTarget) : null}

   357→        onToggle={() => toggle("days")}

   368→          ariaLabel="Training days per week"
   369→          stepLabel="Days"
   370→          onSave={(n) => savePatch({ daysPerWeekTarget: n })}
   371→          onCancel={() => setEditing(null)}
   372→        />

   424→/** Sign out + full local wipe (same behaviour as the settings screen). */
   425→function useSignOut() {
   426→  const setSession = useApp((s) => s.setSession);
   427→  const qc = useQueryClient();
   428→  return async () => {
   429→    try {
   430→      await authApi.logout();
   431→    } finally {
   432→      clearSwCaches();
   433→      wipeLocalData();
   434→      qc.clear();
   435→      setSession(null); // app-shell gate forces #/auth
   436→      toast.success("Signed out — local data cleared");
   437→    }
   438→  };
   439→}

   654→/** − n + stepper editor (age, days per week). */
   655→function StepperEditor({
   656→  title,
   657→  unitLabel,
   658→  min,
   659→  max,
   660→  initial,
   661→  formatValue,
   662→  ariaLabel,
   663→  stepLabel,
   664→  onSave,
   665→  onCancel,
   666→}: {
   667→  title: string;
   668→  unitLabel: string;
   669→  min: number;
   670→  max: number;
   671→  initial: number;
   672→  formatValue: (n: number) => string;
   673→  ariaLabel: string;
   674→  stepLabel: string;
   675→  onSave: (n: number) => Promise<void>;
   676→  onCancel: () => void;
   677→}) {
   678→  const [value, setValue] = useState(initial);
   679→  const [saving, setSaving] = useState(false);
   680→
   681→  const save = async () => {
   682→    setSaving(true);
   683→    try {
   684→      await onSave(value);
   685→    } finally {
   686→      setSaving(false);
   687→    }
   688→  };
   689→
   690→  return (
   691→    <EditorShell
   692→      title={`${title} · ${unitLabel}`}
   693→      footer={<SaveCancelButtons onSave={() => void save()} onCancel={onCancel} saving={saving} />}
   694→    >
   695→      <div className="flex items-center justify-between rounded-lg border bg-background px-2 py-1">
   696→        <span className="flex-none pl-1 text-xs text-muted-foreground">{stepLabel}</span>
   697→        <span className="flex items-center gap-1">
   698→          <Button
   699→            type="button"
   700→            variant="outline"
   701→            className="h-12 w-12 p-0"
   702→            disabled={value <= min || saving}
   703→            aria-label={`Decrease ${ariaLabel}`}
   704→            onClick={() => {
   705→              hapticTap();
   706→              setValue((v) => Math.max(min, v - 1));
   707→            }}
   708→          >
   709→            <Minus className="h-5 w-5" aria-hidden />
   710→          </Button>
   711→          <span
   712→            className="flex h-12 w-14 items-center justify-center text-base font-bold tabular-nums leading-none"
   713→            aria-live="polite"
   714→            aria-label={`${ariaLabel}: ${formatValue(value)}`}
   715→          >
   716→            {formatValue(value)}
   717→          </span>
   718→          <Button
   719→            type="button"
   720→            variant="outline"
   721→            className="h-12 w-12 p-0"
   722→            disabled={value >= max || saving}
   723→            aria-label={`Increase ${ariaLabel}`}
   724→            onClick={() => {
   725→              hapticTap();
   726→              setValue((v) => Math.min(max, v + 1));
   727→            }}
   728→          >
   729→            <Plus className="h-5 w-5" aria-hidden />
   730→          </Button>
   731→        </span>
   732→      </div>
   733→    </EditorShell>
   734→  );
   735→}

   737→/** Height/weight editor — number input in the display unit, stores metric. */
   738→function MeasureEditor({
   739→  title,
   740→  unitLabel,
   741→  placeholder,
   742→  initialMetric,
   743→  imperial,
   744→  toDisplay,
   745→  toMetric,
   746→  min,
   747→  max,
   748→  canonicalUnit,
   749→  onSave,
   750→  onCancel,
   751→}: {
   752→  title: string;
   753→  unitLabel: string;
   754→  placeholder: string;
   755→  initialMetric: number | null;
   756→  imperial: boolean;
   757→  toDisplay: (metric: number) => number;
   758→  toMetric: (display: number) => number;
   759→  min: number;
   760→  max: number;
   761→  canonicalUnit: string;
   762→  onSave: (metric: number) => Promise<void>;
   763→  onCancel: () => void;
   764→}) {
   765→  const [raw, setRaw] = useState(initialMetric != null ? String(roundTo1(toDisplay(initialMetric))) : "");
   766→  const [saving, setSaving] = useState(false);
   767→
   768→  const parsed = raw.trim() === "" ? null : Number(raw);
   769→  const metric = parsed == null || !Number.isFinite(parsed) ? null : toMetric(parsed);
   770→  const error =
   771→    parsed == null ? null
   772→    : !Number.isFinite(parsed) || parsed < min || parsed > max
   773→      ? `Enter ${min}–${max} ${unitLabel}`
   774→      : null;
   775→
   776→  const save = async () => {
   777→    if (metric == null || error) return;
   778→    setSaving(true);
   779→    try {
   780→      await onSave(roundTo1(metric));
   781→    } finally {
   782→      setSaving(false);
   783→    }
   784→  };
   785→
   786→  return (
   787→    <EditorShell
   788→      title={`${title} · ${unitLabel}`}
   789→      footer={
   790→        <SaveCancelButtons
   791→          onSave={() => void save()}
   792→          onCancel={onCancel}
   793→          saveDisabled={metric == null || error != null}
   794→          saving={saving}
   795→        />
   796→      }
   797→    >
   798→      <div className="flex flex-col gap-1.5">
   799→        <Input
   800→          type="number"
   801→          inputMode="decimal"
   802→          step="any"
   803→          min="0"
   804→          className={cn("h-12 rounded-lg text-base tabular-nums", error && "border-destructive")}
   805→          value={raw}
   806→          onChange={(e) => setRaw(e.target.value)}
   807→          placeholder={placeholder ?? ""}
   808→          aria-label={`${title} in ${unitLabel}`}
   809→          aria-invalid={error != null}
   810→        />
   811→        {error ? (
   812→          <p className="px-1 text-xs text-destructive" role="alert">
   813→            {error}
   814→          </p>
   815→        ) : imperial && metric != null ? (
   816→          <p className="px-1 text-xs tabular-nums text-muted-foreground">
   817→            ≈ {roundTo1(metric)} {canonicalUnit}
   818→          </p>
   819→        ) : (
   820→          <p className="px-1 text-xs text-muted-foreground">Stored as {canonicalUnit}.</p>
   821→        )}
   822→      </div>
   823→    </EditorShell>
   824→  );
   825→}
