    67→/**
    68→ * "Real logged data" = the user actually touched the workout:
    69→ * a completed set, a stamped completedAt, or a set with values whose
    70→ * updatedAt > createdAt (user edit — template prefills are untouched).
    71→ */
    72→export function workoutHasLoggedData(w: {
    73→  exercises: Array<{ sets: Array<{ isComplete: boolean; completedAt: Date | null; weight: number | null; reps: number | null; distance: number | null; timeSec: number | null; createdAt: Date; updatedAt: Date }> }>;
    74→}): boolean {
    75→  return w.exercises.some((e) =>
    76→    e.sets.some(
    77→      (s) =>
    78→        s.isComplete ||
    79→        s.completedAt != null ||
    80→        ((s.weight != null || s.reps != null || s.distance != null || s.timeSec != null) &&
    81→          s.updatedAt.getTime() > s.createdAt.getTime()),
    82→    ),
    83→  );
    84→}

    86→/** Workout logged from this routine/day with at least one set of real data. */
    87→async function findQualifyingWorkout(userId: string, routineId: string, dayId: string) {
    88→  const w = await db.workout.findFirst({
    89→    where: { userId, sourceRoutineId: routineId, sourceDayId: dayId },
    90→    include: { exercises: { include: { sets: true } } },
    91→    orderBy: { date: "desc" },
    92→  });
    93→  if (!w) return null;
    94→  const hasLoggedData = workoutHasLoggedData(w);
    95→  return {
    96→    id: w.id,
    97→    dateKey: dayKey(w.date),
    98→    finished: !!w.finishedAt,
    99→    hasLoggedData,
   100→    scheduledStart: w.scheduledStart,
   101→  };
   102→}

   111→/**
   112→ * Run the lazy cursor rules for a user. Returns the (possibly updated) active
   113→ * state + resolved today key. Safe to call on every relevant read.
   114→ */
   115→export async function applyProgramRules(userId: string) {
   116→  const settings = await getProgramSettings(userId);
   117→  const todayKey = localDateKey(settings.timezone, new Date());
   118→  const active = await loadActive(userId);
   119→  if (!active) return { todayKey, settings, active: null, days: [] as ProgramDay[] };
   120→
   121→  const days = active.routine.days;
   122→  let idx = clampCursor(active.cursorDayIndex, days.length);
   123→  let anchor =
   124→    active.lastAdvancedForDate ?? localDateKey(settings.timezone, active.startedAt);
   125→  let dirty = false;
   126→
   127→  const persist = async () => {
   128→    await db.activeRoutine.update({
   129→      where: { id: active.id },
   130→      data: {
   131→        cursorDayIndex: idx,
   132→        lastAdvancedAt: new Date(),
   133→        lastAdvancedForDate: anchor,
   134→      },
   135→    });
   136→  };
   137→
   138→  const guardMax = days.length * 2 + 8;
   139→  for (let guard = 0; guard < guardMax; guard++) {
   140→    const day = days[idx];
   141→    if (!day) break;
   142→
   143→    const workout = await findQualifyingWorkout(userId, active.routineId, day.id);
   144→    // schedule gate: logging a scheduled active-routine day moves the cursor
   145→    // only when scheduleMovesCursor is on
   146→    const qualifying =
   147→      workout && (!workout.scheduledStart || settings.scheduleMovesCursor)
   148→        ? workout
   149→        : null;
   150→
   151→    const advance = shouldAdvanceCursor({
   152→      day: toProgramDay(day),
   153→      todayKey,
   154→      anchorKey: anchor,
   155→      settings,
   156→      qualifyingWorkout: qualifying
   157→        ? {
   158→            hasLoggedData: qualifying.hasLoggedData,
   159→            dateKey: qualifying.dateKey,
   160→            finished: qualifying.finished,
   161→          }
   162→        : null,
   163→    });
   164→    if (!advance) break;
   165→
   166→    idx = advanceIndex(idx, days.length, 1);
   167→    anchor = addDaysKey(anchor, 1);
   168→    dirty = true;
   169→  }
   170→
   171→  if (dirty || idx !== active.cursorDayIndex) {
   172→    if (idx !== clampCursor(active.cursorDayIndex, days.length)) dirty = true;
   173→    await persist();
   174→  }
   175→
   176→  return {
   177→    todayKey,
   178→    settings,
   179→    active: { ...active, cursorDayIndex: idx },
   180→    days: days.map(toProgramDay),
   181→  };
   182→}

   184→/** Explicit cursor movement (skip / rest-done / advance / jump). */
   185→async function moveCursor(
   186→  userId: string,
   187→  mode: { kind: "advance"; n: number } | { kind: "jump"; dayIndex: number },
   188→) {
   189→  const settings = await getProgramSettings(userId);
   190→  const todayKey = localDateKey(settings.timezone, new Date());
   191→  const active = await loadActive(userId);
   192→  if (!active) throw badRequest("No program followed");
   193→  const len = active.routine.days.length;
   194→  if (len === 0) throw badRequest("Program has no days");
   195→
   196→  const nextIdx =
   197→    mode.kind === "advance"
   198→      ? advanceIndex(active.cursorDayIndex, len, mode.n)
   199→      : clampCursor(mode.dayIndex, len);
   200→
   201→  await db.activeRoutine.update({
   202→    where: { id: active.id },
   203→    data: {
   204→      cursorDayIndex: nextIdx,
   205→      lastAdvancedAt: new Date(),
   206→      lastAdvancedForDate: todayKey,
   207→    },
   208→  });
   209→  const day = active.routine.days[nextIdx];
   210→  return { dayIndex: nextIdx, day: toProgramDay(day) };
   211→}

   215→export async function followProgram(userId: string, routineId: string, input: { startDayIndex?: number } = {}) {
   216→  const r = await db.routine.findFirst({ where: { id: routineId, userId, deletedAt: null }, include: routineInclude });
   217→  if (!r) throw notFound("Program not found");
   218→  if (!isFollowable(r.kind ?? "ROUTINE", r.days)) {
   219→    throw badRequest("Only routines with at least one workout day can be followed");
   220→  }
   221→  const start = clampCursor(input.startDayIndex ?? 0, r.days.length);
   222→  const settings = await getProgramSettings(userId);
   223→  const todayKey = localDateKey(settings.timezone, new Date());
   224→  await db.$transaction(async (tx) => {
   225→    await tx.activeRoutine.deleteMany({ where: { userId } });
   226→    await tx.activeRoutine.create({
   227→      data: {
   228→        id: uuid7(),
   229→        userId,
   230→        routineId: r.id,
   231→        cursorDayIndex: start,
   232→        startedAt: new Date(),
   233→        lastAdvancedAt: null,
   234→        lastAdvancedForDate: todayKey,
   235→      },
   236→    });
   237→  });
   238→  return { routineId: r.id, dayIndex: start, day: toProgramDay(r.days[start]) };
   239→}

   274→/** Skip the current cursor day: advance + SKIPPED audit entry for today. */
   275→export async function skipCursorDay(userId: string) {
   276→  const settings = await getProgramSettings(userId);
   277→  const todayKey = localDateKey(settings.timezone, new Date());
   278→  const active = await loadActive(userId);
   279→  if (!active) throw badRequest("No program followed");
   280→  const len = active.routine.days.length;
   281→  if (len === 0) throw badRequest("Program has no days");
   282→  const skippedIdx = clampCursor(active.cursorDayIndex, len);
   283→  const skippedDay = active.routine.days[skippedIdx];
   284→  const nextIdx = advanceIndex(skippedIdx, len, 1);
   285→
   286→  await db.$transaction(async (tx) => {
   287→    await tx.activeRoutine.update({
   288→      where: { id: active.id },
   289→      data: {
   290→        cursorDayIndex: nextIdx,
   291→        lastAdvancedAt: new Date(),
   292→        lastAdvancedForDate: todayKey,
   293→      },
   294→    });
   295→    // audit: convert today's matching PLANNED entry, else create a SKIPPED one
   296→    const planned = await tx.scheduleEntry.findFirst({
   297→      where: {
   298→        userId,
   299→        date: toDayUtc(todayKey),
   300→        status: "PLANNED",
   301→        deletedAt: null,
   302→      },
   303→    });
   304→    if (planned && planned.routineId === active.routineId && planned.dayId === skippedDay.id) {
   305→      await tx.scheduleEntry.update({ where: { id: planned.id }, data: { status: "SKIPPED" } });
   306→    } else {
   307→      await tx.scheduleEntry.create({
   308→        data: {
   309→          id: uuid7(),
   310→          userId,
   311→          date: toDayUtc(todayKey),
   312→          sourceType: "ROUTINE_DAY",
   313→          routineId: active.routineId,
   314→          dayId: skippedDay.id,
   315→          status: "SKIPPED",
   316→        },
   317→      });
   318→    }
   319→  });
   320→  const nextDay = active.routine.days[nextIdx];
   321→  return { skipped: toProgramDay(skippedDay), dayIndex: nextIdx, day: toProgramDay(nextDay) };
   322→}

   324→/** "Mark rest done": advance from a REST day without waiting for midnight. */
   325→export async function markRestDone(userId: string) {
   326→  const active = await loadActive(userId);
   327→  if (!active) throw badRequest("No program followed");
   328→  const idx = clampCursor(active.cursorDayIndex, active.routine.days.length);
   329→  const day = active.routine.days[idx];
   330→  if ((day?.dayType ?? "WORKOUT") !== "REST") throw badRequest("Current day is not a rest day");
   331→  return moveCursor(userId, { kind: "advance", n: 1 });
   332→}

   336→/** Shared: append a routine day's exercises + predefined sets to a workout. */
   337→export async function appendDayToWorkout(
   338→  tx: Prisma.TransactionClient,
   339→  userId: string,
   340→  day: DayFull,
   341→  workoutId: string,
   342→  date: Date,
   343→): Promise<Set<string>> {
   344→  // recreate routine groups in the workout
   345→  const routineGroupIds = new Set(day.exercises.map((e) => e.groupId).filter(Boolean));
   346→  const groupMap = new Map<string, string>();
   347→  for (const gid of routineGroupIds) {
   348→    const rg = await tx.routineGroup.findFirst({ where: { id: gid!, userId } });
   349→    if (!rg) continue;
   350→    const existing = await tx.workoutGroup.findFirst({ where: { workoutId, name: rg.name } });
   351→    const g =
   352→      existing ??
   353→      (await tx.workoutGroup.create({
   354→        data: { id: uuid7(), userId, workoutId, name: rg.name, colour: rg.colour },
   355→      }));
   356→    groupMap.set(gid!, g.id);
   357→  }
   358→
   359→  const affected = new Set<string>();
   360→  let sortOrder = await tx.workoutExercise.count({ where: { workoutId } });
   361→
   362→  for (const re of day.exercises) {
   363→    affected.add(re.exerciseId);
   364→    let twe = await tx.workoutExercise.findFirst({ where: { workoutId, exerciseId: re.exerciseId } });
   365→    if (!twe) {
   366→      twe = await tx.workoutExercise.create({
   367→        data: {
   368→          id: uuid7(),
   369→          userId,
   370→          workoutId,
   371→          exerciseId: re.exerciseId,
   372→          sortOrder: sortOrder++,
   373→          groupId: re.groupId ? groupMap.get(re.groupId) ?? null : null,
   374→        },
   375→      });
   376→    } else if (re.groupId && !twe.groupId) {
   377→      await tx.workoutExercise.update({
   378→        where: { id: twe.id },
   379→        data: { groupId: groupMap.get(re.groupId) ?? null },
   380→      });
   381→    }
   382→
   383→    const predefined = re.sets.slice().sort((a, b) => a.sortOrder - b.sortOrder);
   384→    type SetValues = {
   385→      weight: number | null;
   386→      reps: number | null;
   387→      distance: number | null;
   388→      timeSec: number | null;
   389→      setType: string | null;
   390→      rpe: number | null;
   391→      tempo: string | null;
   392→      restPlannedSec: number | null;
   393→    };
   394→    const fromRow = (s: {
   395→      weight: number | null;
   396→      reps: number | null;
   397→      distance: number | null;
   398→      timeSec: number | null;
   399→      setType: string | null;
   400→      rpe: number | null;
   401→      tempo: string | null;
   402→      restPlannedSec: number | null;
   403→    }): SetValues => ({
   404→      weight: s.weight,
   405→      reps: s.reps,
   406→      distance: s.distance,
   407→      timeSec: s.timeSec,
   408→      setType: s.setType ?? null,
   409→      rpe: s.rpe ?? null,
   410→      tempo: s.tempo ?? null,
   411→      restPlannedSec: s.restPlannedSec ?? null,
   412→    });
   413→
   414→    const blank = (s: { weight: number | null; reps: number | null; distance: number | null; timeSec: number | null }) =>
   415→      s.weight == null && s.reps == null && s.distance == null && s.timeSec == null;
   416→    const isBlank = predefined.length > 0 && predefined.every(blank);
   417→    const anyBlank = predefined.some(blank);
   418→
   419→    // previous performance for this exercise (for copy-previous semantics)
   420→    const prev =
   421→      anyBlank || isBlank
   422→        ? await tx.workoutExercise.findFirst({
   423→            where: { userId, exerciseId: re.exerciseId, workout: { date: { lt: date } } },
   424→            include: { sets: { orderBy: { sortOrder: "asc" } } },
   425→            orderBy: { workout: { date: "desc" } },
   426→          })
   427→        : null;
   428→
   429→    let setsToAdd: Array<SetValues>;
   430→    if (isBlank) {
   431→      // all-blank template = "copy whatever I did last time" (set count follows previous)
   432→      setsToAdd = prev ? prev.sets.map(fromRow) : [];
   433→    } else {
   434→      // valued/mixed template: blank rows copy the previous set at the same
   435→      // index when one exists, otherwise they become empty placeholder rows
   436→      setsToAdd = predefined.map((s, i) => {
   437→        if (blank(s) && prev?.sets[i]) {
   438→          const p = fromRow(prev.sets[i]);
   439→          return { ...p, setType: s.setType ?? p.setType };
   440→        }
   441→        return fromRow(s);
   442→      });
   443→    }
   444→
   445→    let setOrder = await tx.trainingSet.count({ where: { workoutExerciseId: twe.id } });
   446→    for (const s of setsToAdd) {
   447→      await tx.trainingSet.create({
   448→        data: {
   449→          id: uuid7(),
   450→          userId,
   451→          workoutExerciseId: twe.id,
   452→          weight: s.weight,
   453→          reps: s.reps,
   454→          distance: s.distance,
   455→          timeSec: s.timeSec,
   456→          setType: s.setType ?? "NORMAL",
   457→          isWarmup: s.setType === "WARMUP",
   458→          rpe: s.rpe,
   459→          tempo: s.tempo,
   460→          restPlannedSec: s.restPlannedSec,
   461→          sortOrder: setOrder++,
   462→        },
   463→      });
   464→    }
   465→  }
   466→  return affected;
   467→}

   469→/**
   470→ * Start a program day / session: upsert the workout at `date`, stamp
   471→ * provenance, insert predefined sets (existing log-day logic), link/create the
   472→ * schedule DONE entry — all in ONE transaction.
   473→ */
   474→export async function startProgramDay(
   475→  userId: string,
   476→  routineId: string,
   477→  input: { dayId?: string; date?: string },
   478→) {
   479→  const r = await db.routine.findFirst({ where: { id: routineId, userId, deletedAt: null }, include: routineInclude });
   480→  if (!r) throw notFound("Program not found");
   481→  const kind = r.kind ?? "ROUTINE";
   482→
   483→  let day: DayFull | undefined;
   484→  if (input.dayId) {
   485→    day = r.days.find((d) => d.id === input.dayId);
   486→  } else if (kind === "SESSION") {
   487→    day = r.days.find((d) => (d.dayType ?? "WORKOUT") !== "REST");
   488→  } else {
   489→    const active = await loadActive(userId);
   490→    if (active && active.routineId === r.id) {
   491→      day = r.days[clampCursor(active.cursorDayIndex, r.days.length)];
   492→    }
   493→    day = day ?? r.days.find((d) => (d.dayType ?? "WORKOUT") !== "REST");
   494→  }
   495→  if (!day) throw badRequest("Program has no workout day to start");
   496→  if ((day.dayType ?? "WORKOUT") === "REST") throw badRequest("Cannot start a rest day");
   497→
   498→  const settings = await getProgramSettings(userId);
   499→  const todayKey = input.date ?? localDateKey(settings.timezone, new Date());
   500→  const date = toDayUtc(todayKey);
   501→
   502→  // was this day scheduled? (PLANNED entry matching routine+day+date)
   503→  const planned = await db.scheduleEntry.findFirst({
   504→    where: { userId, date, status: "PLANNED", deletedAt: null },
   505→  });
   506→  const wasScheduled =
   507→    !!planned && planned.routineId === r.id && (planned.dayId === day.id || kind === "SESSION");
   508→
   509→  const workout = await db.$transaction(async (tx) => {
   510→    const w = await tx.workout.upsert({
   511→      where: { userId_date: { userId, date } },
   512→      update: {},
   513→      create: { id: uuid7(), userId, date },
   514→    });
   515→    await tx.workout.update({
   516→      where: { id: w.id },
   517→      data: {
   518→        sourceType: kind === "SESSION" ? "SESSION" : "ROUTINE_DAY",
   519→        sourceRoutineId: r.id,
   520→        sourceDayId: day!.id,
   521→        scheduledStart: wasScheduled,
   522→        startAt: w.startAt ?? new Date(),
   523→      },
   524→    });
   525→    const affected = await appendDayToWorkout(tx, userId, day!, w.id, date);
   526→    for (const exerciseId of affected) await recomputePRs(tx, userId, exerciseId);
   527→
   528→    // schedule entry: PLANNED (matching) → DONE + link; else auto-create DONE
   529→    if (planned && wasScheduled) {
   530→      await tx.scheduleEntry.update({
   531→        where: { id: planned.id },
   532→        data: { status: "DONE", workoutId: w.id },
   533→      });
   534→    } else if (!planned) {
   535→      await tx.scheduleEntry.create({
   536→        data: {
   537→          id: uuid7(),
   538→          userId,
   539→          date,
   540→          sourceType: kind === "SESSION" ? "SESSION" : "ROUTINE_DAY",
   541→          routineId: r.id,
   542→          dayId: day!.id,
   543→          status: "DONE",
   544→          workoutId: w.id,
   545→        },
   546→      });
   547→    }
   548→    return tx.workout.findUnique({ where: { id: w.id }, include: workoutInclude });
   549→  });
   550→
   551→  return mapWorkout(workout!);
   552→}

   556→async function tryAdvanceForWorkout(userId: string, workout: Workout, opts: { forceFinishPath: boolean }) {
   557→  const settings = await getProgramSettings(userId);
   558→  const todayKey = localDateKey(settings.timezone, new Date());
   559→  const active = await loadActive(userId);
   560→  if (!active) return { advanced: false, nextDay: null };
   561→  if (workout.sourceRoutineId !== active.routineId) return { advanced: false, nextDay: null };
   562→  const days = active.routine.days;
   563→  const idx = clampCursor(active.cursorDayIndex, days.length);
   564→  const day = days[idx];
   565→  if (!day || day.id !== workout.sourceDayId) return { advanced: false, nextDay: null };
   566→  if (workout.scheduledStart && !settings.scheduleMovesCursor) {
   567→    return { advanced: false, nextDay: null };
   568→  }
   569→
   570→  const qualifying = await findQualifyingWorkout(userId, active.routineId, day.id);
   571→  if (!qualifying) return { advanced: false, nextDay: null };
   572→
   573→  const advance = opts.forceFinishPath
   574→    ? qualifying.hasLoggedData || workout.finishedAt != null
   575→    : shouldAdvanceCursor({
   576→        day: toProgramDay(day),
   577→        todayKey,
   578→        anchorKey: active.lastAdvancedForDate ?? localDateKey(settings.timezone, active.startedAt),
   579→        settings,
   580→        qualifyingWorkout: {
   581→          hasLoggedData: qualifying.hasLoggedData,
   582→          dateKey: qualifying.dateKey,
   583→          finished: qualifying.finished,
   584→        },
   585→      });
   586→  if (!advance) return { advanced: false, nextDay: null };
   587→
   588→  const nextIdx = advanceIndex(idx, days.length, 1);
   589→  await db.activeRoutine.update({
   590→    where: { id: active.id },
   591→    data: {
   592→      cursorDayIndex: nextIdx,
   593→      lastAdvancedAt: new Date(),
   594→      lastAdvancedForDate: todayKey,
   595→    },
   596→  });
   597→  const nextDay = days[nextIdx];
   598→  return { advanced: true, nextDay: toProgramDay(nextDay) };
   599→}

   655→function assertScheduleShape(r: RoutineFull, dayId?: string) {
   656→  const kind = r.kind ?? "ROUTINE";
   657→  if (kind === "SESSION") {
   658→    const day = r.days.find((d) => (d.dayType ?? "WORKOUT") !== "REST");
   659→    if (!day) throw badRequest("Session has no workout day");
   660→    if (!isValidSession(kind, r.days)) throw badRequest("Session must have exactly one workout day");
   661→    return day;
   662→  }
   663→  if (!dayId) throw badRequest("dayId is required for routines");
   664→  const day = r.days.find((d) => d.id === dayId);
   665→  if (!day) throw badRequest("Day not found in this routine");
   666→  if ((day.dayType ?? "WORKOUT") === "REST") throw badRequest("Cannot schedule a rest day");
   667→  return day;
   668→}

   721→export async function updateSchedule(
   722→  userId: string,
   723→  id: string,
   724→  input: { date?: string; status?: string; note?: string | null },
   725→) {
   726→  const entry = await db.scheduleEntry.findFirst({ where: { id, userId, deletedAt: null }, include: entryInclude });
   727→  if (!entry) throw notFound("Schedule entry not found");
   728→
   729→  const data: Prisma.ScheduleEntryUpdateInput = {};
   730→  if (input.date !== undefined) {
   731→    const date = toDayUtc(input.date);
   732→    if (!date.getTime() || Number.isNaN(date.getTime())) throw badRequest("Invalid date");
   733→    const conflictEntry = await findPlannedConflict(userId, date, id);
   734→    if (conflictEntry && entry.status === "PLANNED") {
   735→      throw conflict("A workout is already planned for this date", {
   736→        date: input.date,
   737→        entryId: conflictEntry.id,
   738→        routineName: conflictEntry.routine.name,
   739→      });
   740→    }
   741→    data.date = date;
   742→  }
   743→  if (input.note !== undefined) data.note = input.note;
   744→
   745→  if (input.status !== undefined) {
   746→    if (input.status === "PLANNED") {
   747→      // Reopen: only for date >= today
   748→      const settings = await getProgramSettings(userId);
   749→      const todayKey = localDateKey(settings.timezone, new Date());
   750→      const dateKey = dayKey(input.date !== undefined ? toDayUtc(input.date) : entry.date);
   751→      if (compareDateKeys(dateKey, todayKey) < 0) throw badRequest("Can only reopen entries from today onwards");
   752→      const conflictEntry = await findPlannedConflict(
   753→        userId,
   754→        input.date !== undefined ? toDayUtc(input.date) : entry.date,
   755→        id,
   756→      );
   757→      if (conflictEntry) throw conflict("A workout is already planned for this date");
   758→      data.status = "PLANNED";
   759→    } else if (input.status === "SKIPPED") {
   760→      data.status = "SKIPPED";
   761→    } else {
   762→      throw badRequest("Status can only be set to SKIPPED (or PLANNED to reopen)");
   763→    }
   764→  }
   765→
   766→  const updated = await db.scheduleEntry.update({ where: { id }, data, include: entryInclude });
   767→  return mapScheduleEntry(updated);
   768→}

   777→/** List entries in [from, to] with lazy status derivation + optional projection. */
   778→export async function listSchedule(userId: string, input: { from?: string; to?: string }) {
   779→  const settings = await getProgramSettings(userId);
   780→  const todayKey = localDateKey(settings.timezone, new Date());
   781→  await applyProgramRules(userId);
   782→
   783→  const to = input.to ?? addDaysKey(todayKey, 60);
   784→  const from = input.from ?? addDaysKey(todayKey, -60);
   785→  const rows = await db.scheduleEntry.findMany({
   786→    where: { userId, deletedAt: null, date: { gte: toDayUtc(from), lte: toDayUtc(to) } },
   787→    include: {
   788→      ...entryInclude,
   789→      workout: { include: { exercises: { include: { sets: true } } } },
   790→    },
   791→    orderBy: { date: "asc" },
   792→  });
   793→
   794→  const entries: Array<ReturnType<typeof mapScheduleEntry>> = [];
   795→  for (const e of rows) {
   796→    let status = e.status;
   797→    if (e.status === "PLANNED" && e.workoutId && e.workout) {
   798→      const hasData = workoutHasLoggedData(e.workout);
   799→      const derived = deriveEntryStatus({
   800→        storedStatus: "PLANNED",
   801→        dateKey: dayKey(e.date),
   802→        todayKey,
   803→        linkedWorkoutHasData: hasData,
   804→        linkedWorkoutFinished: !!e.workout.finishedAt,
   805→      });
   806→      if (derived !== "PLANNED") {
   807→        status = derived;
   808→        await db.scheduleEntry.update({ where: { id: e.id }, data: { status: derived } });
   809→      }
   810→    } else if (e.status === "PLANNED") {
   811→      const derived = deriveEntryStatus({
   812→        storedStatus: "PLANNED",
   813→        dateKey: dayKey(e.date),
   814→        todayKey,
   815→        linkedWorkoutHasData: false,
   816→        linkedWorkoutFinished: false,
   817→      });
   818→      if (derived !== "PLANNED") {
   819→        status = derived;
   820→        await db.scheduleEntry.update({ where: { id: e.id }, data: { status: derived } });
   821→      }
   822→    }
   823→    entries.push(mapScheduleEntry({ ...e, status }));
   824→  }
   825→
   826→  // projected ghost days (calendar)
   827→  let projected: Array<{ date: string; routineId: string; dayId: string; dayName: string; dayType: string }> = [];
   828→  if (settings.showProjectedDays) {
   829→    const active = await loadActive(userId);
   830→    if (active) {
   831→      const days = active.routine.days.map(toProgramDay);
   832→      const idx = clampCursor(active.cursorDayIndex, days.length);
   833→      const plannedDates = new Set(entries.filter((e) => e.status === "PLANNED").map((e) => e.date));
   834→      projected = projectCursor({ days, startIndex: idx, todayKey, settings, count: 28 })
   835→        .filter((p) => {
   836→          const inRange = compareDateKeys(p.dateKey, from) >= 0 && compareDateKeys(p.dateKey, to) <= 0;
   837→          return inRange && !plannedDates.has(p.dateKey) && p.day.dayType !== "REST";
   838→        })
   839→        .map((p) => ({
   840→          date: p.dateKey,
   841→          routineId: active.routineId,
   842→          dayId: p.day.id,
   843→          dayName: p.day.name,
   844→          dayType: p.day.dayType,
   845→        }));
   846→    }
   847→  }
   848→
   849→  return { entries, projected };
   850→}

   920→export async function listPrograms(userId: string, kind?: "ROUTINE" | "SESSION") {
   921→  await applyProgramRules(userId);

   923→    where: { userId, deletedAt: null, ...(kind ? { kind } : {}) },
   924→    include: routineInclude,
   925→    orderBy: { sortOrder: "asc" },
   926→  });
   927→  const active = await db.activeRoutine.findUnique({ where: { userId } });
   928→
   929→  const lastUsed = await db.workout.groupBy({
   930→    by: ["sourceRoutineId"],
   931→    where: { userId, sourceRoutineId: { not: null } },
   932→    _max: { date: true },
   933→  });
   934→  const lastUsedMap = new Map(lastUsed.map((r) => [r.sourceRoutineId!, r._max.date ?? null]));
   935→

   938→    const exerciseCount = workoutDays.reduce((acc, d) => acc + d.exercises.length, 0);
   939→    const last = lastUsedMap.get(r.id) ?? null;
   940→    return {
   941→      id: r.id,
   942→      name: r.name,
   943→      notes: r.notes ?? null,
   944→      kind: r.kind ?? "ROUTINE",
   945→      dayCount: r.days.length,
   946→      restCount: r.days.length - workoutDays.length,
   947→      exerciseCount,
   948→      lastUsedAt: last ? dayKey(last) : null,
   949→      isFollowed: active?.routineId === r.id,
   950→      cursor:
   951→        active?.routineId === r.id
   952→          ? {
   953→              dayIndex: clampCursor(active.cursorDayIndex, r.days.length),
   954→              dayCount: r.days.length,
   955→            }
   956→          : null,
   957→    };
   958→  });
   959→}

   963→export async function getDashboard(userId: string) {
   964→  const rules = await applyProgramRules(userId);
   965→  const { todayKey, settings } = rules;
   966→  const todayDate = toDayUtc(todayKey);
   967→
   968→  const [active, plannedToday, windowEntries, todayWorkoutRow] = await Promise.all([
   969→    loadActive(userId),
   970→    db.scheduleEntry.findFirst({
   971→      where: { userId, date: todayDate, deletedAt: null },
   972→      include: { ...entryInclude, workout: { include: { exercises: { include: { sets: true } } } } },
   973→      orderBy: [{ status: "asc" }, { createdAt: "desc" }], // PLANNED first
   974→    }),
   975→    db.scheduleEntry.findMany({
   976→      where: {
   977→        userId,
   978→        deletedAt: null,
   979→        date: { gte: addDaysUtc(todayDate, 1), lte: addDaysUtc(todayDate, 7) },
   980→      },
   981→      include: entryInclude,
   982→      orderBy: { date: "asc" },
   983→    }),
   984→    db.workout.findFirst({
   985→      where: { userId, date: todayDate },
   986→      include: { exercises: { include: { sets: true } } },
   987→    }),
   988→  ]);
   989→
   990→  // ---- today resolution ----
   991→  let today: {
   992→    date: string;
   993→    kind: "WORKOUT" | "REST" | "NONE";
   994→    scheduled: ReturnType<typeof mapScheduleEntry> | null;
   995→    routine: { id: string; name: string; kind: string } | null;
   996→    day: { id: string; name: string; dayType: string; index: number; count: number } | null;
   997→  };
   998→
   999→  const plannedEntry = plannedToday && plannedToday.status === "PLANNED" ? plannedToday : null;
  1000→
  1001→  if (plannedEntry) {
  1002→    const dayIdx = active && active.routineId === plannedEntry.routineId && plannedEntry.dayId
  1003→      ? active.routine.days.findIndex((d) => d.id === plannedEntry.dayId)
  1004→      : -1;
  1005→    today = {
  1006→      date: todayKey,
  1007→      kind: "WORKOUT",
  1008→      scheduled: mapScheduleEntry(plannedEntry),
  1009→      routine: { id: plannedEntry.routine.id, name: plannedEntry.routine.name, kind: plannedEntry.routine.kind ?? "ROUTINE" },
  1010→      day: plannedEntry.day
  1011→        ? {
  1012→            id: plannedEntry.day.id,
  1013→            name: plannedEntry.day.name,
  1014→            dayType: plannedEntry.day.dayType ?? "WORKOUT",
  1015→            index: dayIdx,
  1016→            count: active && active.routineId === plannedEntry.routineId ? active.routine.days.length : 0,
  1017→          }
  1018→        : null,
  1019→    };
  1020→  } else if (active) {
  1021→    const days = active.routine.days;
  1022→    const idx = clampCursor(active.cursorDayIndex, days.length);
  1023→    const day = days[idx];
  1024→    today = {
  1025→      date: todayKey,
  1026→      kind: (day?.dayType ?? "WORKOUT") === "REST" ? "REST" : "WORKOUT",
  1027→      scheduled: null,
  1028→      routine: { id: active.routine.id, name: active.routine.name, kind: active.routine.kind ?? "ROUTINE" },
  1029→      day: day
  1030→        ? { id: day.id, name: day.name, dayType: day.dayType ?? "WORKOUT", index: idx, count: days.length }
  1031→        : null,
  1032→    };
  1033→  } else {
  1034→    today = { date: todayKey, kind: "NONE", scheduled: null, routine: null, day: null };
  1035→  }
  1036→
  1037→  // ---- upcoming strip (7 days) ----
  1038→  type EntryRow = (typeof windowEntries)[number];
  1039→  const entryByDate = new Map<string, EntryRow[]>();
  1040→  for (const e of windowEntries) {
  1041→    const k = dayKey(e.date);
  1042→    entryByDate.set(k, [...(entryByDate.get(k) ?? []), e]);
  1043→  }
  1044→  const activeDays = active ? active.routine.days.map(toProgramDay) : [];
  1045→  const projection = active
  1046→    ? projectCursor({
  1047→        days: activeDays,
  1048→        startIndex: clampCursor(active.cursorDayIndex, activeDays.length),
  1049→        todayKey,
  1050→        settings,
  1051→        count: 7,
  1052→      })
  1053→    : [];
  1054→  const upcoming = projection.map((p) => {
  1055→    const entries = (entryByDate.get(p.dateKey) ?? []).filter((e) => e.status !== "DONE");
  1056→    const entry = entries.find((e) => e.status === "PLANNED") ?? entries[0] ?? null;
  1057→    if (entry) {
  1058→      return {
  1059→        date: p.dateKey,
  1060→        kind: "SCHEDULED" as const,
  1061→        label: entry.day?.name ?? entry.routine.name,
  1062→        entry: mapScheduleEntry(entry),
  1063→      };
  1064→    }
  1065→    return {
  1066→      date: p.dateKey,
  1067→      kind: p.day.dayType === "REST" ? ("REST" as const) : ("WORKOUT" as const),
  1068→      label: p.day.name,
  1069→      entry: null,
  1070→    };
  1071→  });
  1072→
  1073→  // ---- stats ----

  1085→          weekSets += 1;
  1086→          weekVolume += (s.weight ?? 0) * (s.reps ?? 0);
  1087→        }
  1088→      }
  1089→    }
  1090→  }
  1091→
  1092→  // streak: consecutive days (ending today or yesterday) with ≥1 set
  1093→  const daysWithData = new Set<string>();
  1094→  for (const w of weekWorkouts) {
  1095→    const has = w.exercises.some((e) => e.sets.length > 0);
  1096→    if (has) daysWithData.add(dayKey(w.date));
  1097→  }
  1098→  // widen the look-back for the streak
  1099→  const streakLookback = await db.workout.findMany({
  1100→    where: { userId, date: { gte: addDaysUtc(todayDate, -180), lte: addDaysUtc(todayDate, -7) } },
  1101→    include: { exercises: { include: { sets: { take: 1 } } } },
  1102→  });
  1103→  for (const w of streakLookback) {
  1104→    if (w.exercises.some((e) => e.sets.length > 0)) daysWithData.add(dayKey(w.date));
  1105→  }
  1106→  let streakDays = 0;
  1107→  let cursorDate = daysWithData.has(todayKey) ? todayKey : addDaysKey(todayKey, -1);
  1108→  while (daysWithData.has(cursorDate) && streakDays < 366) {
  1109→    streakDays += 1;
  1110→    cursorDate = addDaysKey(cursorDate, -1);

  1113→  // ---- quick sessions ----

  1147→  return {
