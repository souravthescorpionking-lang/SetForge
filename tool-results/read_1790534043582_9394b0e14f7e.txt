    99→    99→/** Map a CardSet patch (SetRow/ExerciseCard contract) onto the API's SetInput. */
   100→   100→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   101→   101→  const out: SetInput = {};
   102→   102→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   103→   103→  if ("reps" in patch) out.reps = patch.reps ?? null;
   104→   104→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   105→   105→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   106→   106→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   107→   107→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   108→   108→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   109→   109→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   110→   110→  if ("note" in patch) out.comment = patch.note ?? null;
   111→   111→  if ("done" in patch) out.isComplete = !!patch.done;
   112→   112→  return out;
   113→   113→}

   115→   115→/** §4.11 showMaxWeightBar — "Best 110×5" from the exercise's record table
   116→   116→ *  (max-weight entry; ties break to the higher rep count). */
   117→   117→function bestSetLine(records: RecordsDTO | undefined): string | null {
   118→   118→  if (!records) return null;
   119→   119→  const usable = records.actual.filter((r) => r.weight != null && r.weight > 0 && r.reps != null);
   120→   120→  const rows = usable.some((r) => !r.superseded) ? usable.filter((r) => !r.superseded) : usable;
   121→   121→  if (rows.length === 0) return null;
   122→   122→  let best = rows[0];
   123→   123→  for (const r of rows) {
   124→   124→    if (r.weight! > best.weight! || (r.weight === best.weight && r.reps! > best.reps!)) best = r;
   125→   125→  }
   126→   126→  return `Best ${trimNum(best.weight!)}×${best.reps!}`;
   127→   127→}

   869→   869→                    {showTipRow ? <TrainerTipRow tip={we.exercise.trainerTip!} /> : null}
   870→   870→                  </>
   871→   871→                ) : undefined;
   872→   872→              return (
   873→   873→                <div
   874→   874→                  key={we.id}
   875→   875→                  ref={(el) => {
   876→   876→                    if (el) cardRefs.current.set(we.id, el);
   877→   877→                    else cardRefs.current.delete(we.id);
   878→   878→                  }}
   879→   879→                  className="relative flex-none"
   880→   880→                >
   881→   881→                  <ExerciseCard
   882→   882→                    mode="edit"
   883→   883→                    exercise={toCardExercise(we)}
   884→   884→                    sets={cardSets(we)}
   885→   885→                    collapsed={collapsedIds.has(we.id)}
   886→   886→                    visibleColumns={visibleColumns}
   887→   887→                    groupColour={group?.colour}
   888→   888→                    groupName={group?.name}
   889→   889→                    groupCode={groupCodes.codes.get(we.id)}
   890→   890→                    metaExtra={bestByWeId.get(we.id)}
   891→   891→                    underHeader={underHeader}
   892→   892→                    className={isCurrentGuidedCard ? "ring-2 ring-primary/40" : undefined}
   893→   893→                    onAction={handleCardAction(we)}
   894→   894→                  />
   895→   895→                  {notesWeId === we.id && (
   896→   896→                    <ExerciseNotesPopover
   897→   897→                      exercise={we.exercise}
   898→   898→                      open
   899→   899→                      onClose={() => setNotesWeId(null)}
   900→   900→                    />
   901→   901→                  )}
   902→   902→                  {groupWeId === we.id && (
   903→   903→                    <ExerciseGroupPopover
   904→   904→                      workout={workout}
   905→   905→                      we={we}
   906→   906→                      open
   907→   907→                      onClose={() => setGroupWeId(null)}
   908→   908→                    />
   909→   909→                  )}
   910→   910→                </div>
   911→   911→              );
   912→   912→            })}
   913→   913→
   914→   914→            <SummaryRow workout={workout} unit={defaultUnitFor(settings)} />
   915→   915→
   916→   916→            {/* bottom breathing spacer — bars are flex siblings, nothing to clear */}
   917→   917→            <div className="h-4 flex-none" aria-hidden />
   918→   918→
   919→   919→            <ConfirmRemoveExercise
   920→   920→              exerciseName={removeWe?.exercise.name ?? ""}
   921→   921→              open={removeWeId != null}
   922→   922→              onClose={() => setRemoveWeId(null)}
   923→   923→              onConfirm={async () => {
   924→   924→                const id = removeWeId;
   925→   925→                setRemoveWeId(null);
   926→   926→                if (id) await removeExercise(id);
   927→   927→              }}
   928→   928→            />
   929→   929→
   930→   930→            {/* §4.11 finishBehaviour=ASK — Save/Discard dialogs (mounted only
   931→   931→                while the ASK stage is open; ALWAYS_SAVE never reaches it) */}
   932→   932→            {settings?.finishBehaviour === "ASK" ? (
   933→   933→              <FinishAskFlow
   934→   934→                workout={workout}
   935→   935→                open={finishAskOpen}
   936→   936→                onClose={() => setFinishAskOpen(false)}
   937→   937→                onSave={() => void finishWorkout()}
   938→   938→                dateKey={dateKey}
   939→   939→                onDiscarded={() => {
   940→   940→                  // a discarded day can't keep a rest countdown running
   941→   941→                  rest.skip();
   942→   942→                  setFinishAskOpen(false);
   943→   943→                }}
   944→   944→              />
   945→   945→            ) : null}
   946→   946→
   947→   947→            {sessionFromWorkout.dialog}
   948→   948→          </>
   949→   949→        )}
   950→   950→      </ScrollBody>
   951→   951→    </Screen>
   952→   952→  );
   953→   953→}
