   142→function ExerciseNotesPopover({
   143→  exerciseName,
   144→  notes,
   145→  open,
   146→  onClose,
   147→}: {
   148→  exerciseName: string;
   149→  notes: string | null;
   150→  open: boolean;
   151→  onClose: () => void;
   152→}) {
   153→  return (
   154→    <Popover open={open} onOpenChange={(o) => !o && onClose()}>
   155→      <PopoverTrigger asChild>
   156→        <span className="absolute bottom-2 left-2 h-1 w-1" aria-hidden />
   157→      </PopoverTrigger>
   158→      <PopoverContent
   159→        align="start"
   160→        className="w-72"
   161→        onOpenAutoFocus={(e) => e.preventDefault()}
   162→        // The ⋮ menu closes by returning focus to its trigger; without this
   163→        // the popover would be instantly dismissed (same Radix gotcha as the
   164→        // Today card popovers). Pointer-outside + Escape still close.
   165→        onFocusOutside={(e) => e.preventDefault()}
   166→      >
   167→        <p className="pb-1 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">
   168→          {exerciseName} — notes
   169→        </p>
   170→        <p className="text-sm leading-relaxed text-foreground">
   171→          {notes && notes.trim().length > 0 ? notes : "No notes on this exercise."}
   172→        </p>
   173→      </PopoverContent>
   174→    </Popover>
   175→  );
   176→}
