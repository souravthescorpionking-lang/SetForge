    98→/** Map a CardSet patch (SetRow/ExerciseCard contract) onto the API's SetInput. */
    99→function cardPatchToSetInput(patch: Partial<CardSet>): SetInput {
   100→  const out: SetInput = {};
   101→  if ("weightKg" in patch) out.weight = patch.weightKg ?? null;
   102→  if ("reps" in patch) out.reps = patch.reps ?? null;
   103→  if ("distanceM" in patch) out.distance = patch.distanceM ?? null;
   104→  if ("timeSec" in patch) out.timeSec = patch.timeSec ?? null;
   105→  if ("rpe" in patch) out.rpe = patch.rpe ?? null;
   106→  if ("tempo" in patch) out.tempo = patch.tempo ?? null;
   107→  if ("restPlannedSec" in patch) out.restPlannedSec = patch.restPlannedSec ?? null;
   108→  if ("setType" in patch) out.setType = patch.setType ?? "NORMAL";
   109→  if ("note" in patch) out.comment = patch.note ?? null;
   110→  if ("done" in patch) out.isComplete = !!patch.done;
   111→  return out;
   112→}

   114→/** §4.11 showMaxWeightBar — "Best 110×5" from the exercise's record table
   115→ *  (max-weight entry; ties break to the higher rep count). */
   116→function bestSetLine(records: RecordsDTO | undefined): string | null {
   117→  if (!records) return null;
   118→  const usable = records.actual.filter((r) => r.weight != null && r.weight > 0 && r.reps != null);
   119→  const rows = usable.some((r) => !r.superseded) ? usable.filter((r) => !r.superseded) : usable;
   120→  if (rows.length === 0) return null;
   121→  let best = rows[0];
   122→  for (const r of rows) {
   123→    if (r.weight! > best.weight! || (r.weight === best.weight && r.reps! > best.reps!)) best = r;
   124→  }
   125→  return `Best ${trimNum(best.weight!)}×${best.reps!}`;
   126→}

   489→   *  (offline-queued or failed sets keep the pointer where it is; the
   490→   *  skip-effect advances it once the cache catches up). */
   491→  const logGuidedSet = async () => {
   492→    const p = pointerOrder[guidedIdx];
   493→    if (!p) return;
   494→    const we = exercises.find((w) => w.id === p.exerciseId);
   495→    const set = sortedSetsByWe.get(p.exerciseId)?.[p.setIndex];
   496→    if (!we || !set || set.isComplete) return;
   497→    const logged = await toggleSetDone(we, set);
   498→    if (logged) advanceGuidedPast(guidedIdx + 1);
   499→  };
   500→
   501→  // card element registry — the guided pointer scrolls the current card into view
   502→  const cardRefs = useRef<Map<string, HTMLDivElement>>(new Map());
   503→  useEffect(() => {
   504→    if (!guidedActive) return;
   505→    const p = pointerOrder[guidedIdx];
   506→    if (!p) return;
   507→    const el = cardRefs.current.get(p.exerciseId);
   508→    if (!el) return;
   509→    const reduced =
   510→      typeof window.matchMedia === "function" &&
   511→      window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   512→    el.scrollIntoView({ block: "nearest", behavior: reduced ? "auto" : "smooth" });
   513→  }, [guidedActive, guidedIdx, pointerOrder]);
   514→
   515→  // ---------- §4.11 derived card data ----------
   516→
   517→  /** §4.10 group codes for today's cards (A1/A2… on superset members). */
   518→  const groupCodes = useMemo(() => computeGroupCodes(exercises), [exercises]);
   519→
   520→  const guidedCardWeId = guidedIdx < pointerOrder.length ? pointerOrder[guidedIdx].exerciseId : null;
   521→  const guidedAllDone = guidedActive && guidedIdx >= pointerOrder.length;
   522→  const guidedLabel = (() => {
   523→    if (!guidedActive) return null;
   524→    if (guidedAllDone) return "All sets done";
   525→    const p = pointerOrder[guidedIdx];
   526→    const code = groupCodes.codes.get(p.exerciseId);
   527→    const total = sortedSetsByWe.get(p.exerciseId)?.length ?? 0;
   528→    return code ? `${code} · set ${p.setIndex + 1}/${total}` : `set ${p.setIndex + 1}/${total}`;
   529→  })();
   530→
   531→  /** §4.11 sets progress — non-warmup performed sets / total sets. */
   532→  const setsProgress = useMemo(() => {
   533→    if (!workout) return null;
   534→    let total = 0;
   535→    let completed = 0;
   536→    for (const we of workout.exercises) {
   537→      for (const s of we.sets) {
   538→        total += 1;
   539→        if (s.isComplete && !s.isWarmup && (s.setType ?? "NORMAL") !== "WARMUP") completed += 1;
   540→      }
   541→    }
   542→    return total > 0 ? { completed, total } : null;
   543→  }, [workout]);
   544→
   545→  /** §4.11 showMaxWeightBar — "Best W×R" per exercise from its records. */
   546→  const showBestLine = !!settings?.showMaxWeightBar;
   547→  const recordsQueries = useQueries({
   548→    queries: exercises.map((we) => ({
   549→      queryKey: qk.exerciseRecords(we.exerciseId),
   550→      queryFn: () => exercisesApi.records(we.exerciseId),
   551→      enabled: showBestLine,
   552→      staleTime: 60_000,
   553→    })),
   554→  });
   555→  const bestByWeId = useMemo(() => {
   556→    const m = new Map<string, string>();
   557→    if (!showBestLine) return m;
   558→    exercises.forEach((we, i) => {
   559→      const line = bestSetLine(recordsQueries[i]?.data);
   560→      if (line) m.set(we.id, line);
   561→    });
   562→    return m;
   563→  }, [showBestLine, exercises, recordsQueries]);
   564→
   565→  /** Any card has a trainer tip → the MetaRow "Tip" chip renders. */
   566→  const anyTips = exercises.some((we) => !!we.exercise.trainerTip);
   567→
   568→  // ---------- Part 5: finish / undo-finish ----------
   569→  const finishWorkout = async () => {
   570→    if (!workout || !online) {
   571→      if (!online) toast.info("Finishing needs a connection");
   572→      return;
   573→    }
   574→    try {
   575→      const res = await workoutsApi.finish(workout.id);
   576→      invalidate.workout();
   577→      invalidate.dashboard();
   578→      invalidate.schedule();
   579→      invalidate.programs();
   580→      // "Day X up next" — index comes from the routine days (fallback: name)
   581→      let nextLabel = "Done";
   582→      if (res.nextDay) {
   583→        const routine = routinesQuery.data?.routines.find(
   584→          (r) => r.id === workout.sourceRoutineId,
   585→        );
   586→        const idx = routine?.days.findIndex((d) => d.id === res.nextDay?.id) ?? -1;
   587→        nextLabel = idx >= 0 ? `Day ${idx + 1} up next` : `${res.nextDay.name} up next`;
   588→      }
   589→      toast.success(`Workout finished · ${nextLabel}`, {
   590→        duration: 10_000,
   591→        action: {
   592→          label: "Undo",
   593→          onClick: () => void undoFinish(workout.id),
   594→        },
   595→      });
   596→    } catch (e) {
   597→      toast.error(e instanceof Error ? e.message : "Could not finish workout");
   598→    }
   599→  };
   600→
   601→  const undoFinish = async (workoutId: string) => {
   602→    try {
   603→      await workoutsApi.undoFinish(workoutId);
   604→      invalidate.workout();
   605→      invalidate.dashboard();
   606→      invalidate.schedule();
   607→      invalidate.programs();
   608→      toast.success("Finish undone — keep logging");
   609→    } catch (e) {
   610→      if (e instanceof ApiError && e.status === 404) {
   611→        toast.info("Too late to undo (past the 10s window)");
   612→      } else {
   613→        toast.error(e instanceof Error ? e.message : "Could not undo");
   614→      }
   615→    }
   616→  };
   617→
   618→  // §4.11 finishBehaviour — ALWAYS_SAVE (default) finishes directly; ASK opens
   619→  // the Save/Discard dialogs (finish-ask.tsx).
   620→  const handleFinishPressed = () => {
   621→    if (settings?.finishBehaviour === "ASK") setFinishAskOpen(true);
   622→    else void finishWorkout();
   623→  };
   624→
   625→  // ---------- card mapping ----------
   626→  const visibleColumns: CardVisibleColumns = {
   627→    setType: settings?.showSetType ?? true,
   628→    rpe: settings?.showRpe ?? true,
   629→    tempo: settings?.showTempo ?? true,
   630→    rest: settings?.showRest ?? true,
   631→  };
   632→
   633→  const toCardExercise = (we: WorkoutExerciseDTO): CardExercise => ({
   634→    id: we.id,
   635→    name: we.exercise.name,
   636→    categoryLabel: we.exercise.category?.name ?? "Exercise",
   637→    categoryColour: we.exercise.category?.colour ?? "#71717a",
   638→    modality: we.exercise.type,
   639→    unit: exerciseUnit(we.exercise, settings),
   640→    weightIncrement: we.exercise.weightIncrement ?? settings?.defaultWeightIncrement ?? 2.5,
   641→  });
   642→
   643→  const cardSets = (we: WorkoutExerciseDTO): CardSet[] =>
   644→    [...we.sets].sort((a, b) => a.sortOrder - b.sortOrder).map((s, i) => toCardSet(s, i + 1));
   645→
   646→  const handleCardAction =
   647→    (we: WorkoutExerciseDTO) =>
   648→    (action: CardAction): void => {
   649→      switch (action.type) {
   650→        case "toggle-collapse":
   651→          setCollapsedIds((prev) => {
   652→            const next = new Set(prev);
   653→            if (next.has(we.id)) next.delete(we.id);
   654→            else next.add(we.id);
   655→            return next;
   656→          });
   657→          break;
   658→        case "add-set":
   659→          void addSetToCard(we);
   660→          break;
   661→        case "update-set":
   662→          void patchSet(we, action.setId, cardPatchToSetInput(action.patch));
   663→          break;
   664→        case "toggle-done": {
   665→          const set = we.sets.find((s) => s.id === action.setId);
   666→          if (set) void toggleSetDone(we, set);
   667→          break;
   668→        }
   669→        case "copy-last":
   670→          void copyLast(we, action.setId);
   671→          break;
   672→        case "notes":
   673→          setNotesWeId(we.id);
   674→          break;
   675→        case "rest-timer":
   676→          rest.start(we.exercise.restSec ?? undefined, action.setId ?? null);
   677→          break;
   678→        case "move-up":
   679→          void moveExercise(we, -1);
   680→          break;
   681→        case "move-down":
   682→          void moveExercise(we, 1);
   683→          break;
   684→        case "add-to-group":
   685→          setGroupWeId(we.id);
   686→          break;
   687→        case "replace":
   688→          // p3-4: the picker opens in replace mode with the day context
   689→          navigate(`/exercises?replace=${we.id}&date=${dateKey}`);
   690→          break;
   691→        case "remove":
   692→          setRemoveWeId(we.id);
   693→          break;
   694→        case "select":
   695→          // multi-select omitted in the p3-3 Today rebuild (see report)
   696→          toast.info("Multi-select arrives with the next build");
   697→          break;
   698→        case "open":
   699→          // p3-4 "Focus view" — carry the day context into the training screen
   700→          navigate(`/today/${we.id}?date=${dateKey}`);
   701→          break;
   702→        case "toggle-select":
   703→          break; // preview-mode only; edit rows never emit it
   704→      }
   705→    };
   706→
   707→  const restActive = rest.remainingSec != null;
   708→  // §4.11 RING display — in-flow block while resting (RestBar stays too)
   709→  const ringRestActive = restActive && settings?.restDisplay === "RING";
   710→
   711→  // ---------- render ----------
   712→  return (
   713→    <Screen
   714→      topBar={
   715→        <TopBar
   716→          title="SetForge"
   717→          actions={
   718→            <>
   719→              <Button
   720→                type="button"
   721→                variant="ghost"
   722→                size="icon"
   723→                className="h-11 w-11 flex-none"
   724→                aria-label="Calendar"
   725→                tour={{ id: "today.calendar", label: "Calendar", help: "Open the calendar to jump to another day.", order: 10 }}
   726→                onClick={() => navigate("/calendar")}
   727→              >
   728→                <CalendarDays className="h-5 w-5" aria-hidden />
   729→              </Button>
   730→              <DropdownMenu>
   731→                <DropdownMenuTrigger asChild>
   732→                  <Button
   733→                    type="button"
   734→                    variant="ghost"
   735→                    size="icon"
   736→                    className="h-11 w-11 flex-none"
   737→                    aria-label="More actions"
   738→                    tour={{ id: "today.menu", label: "Day menu", help: "Save as session, arrange, history, tools or settings.", order: 20 }}
   739→                  >
   740→                    <MoreVertical className="h-5 w-5" aria-hidden />
   741→                  </Button>
   742→                </DropdownMenuTrigger>
   743→                <DropdownMenuContent align="end" className="w-52">
   744→                  {workout && workout.exercises.length > 0 ? (
   745→                    <>
   746→                      <DropdownMenuItem
   747→                        onClick={() => sessionFromWorkout.openFor(workout.id, dateKey)}
   748→                      >
   749→                        <Dumbbell className="h-4 w-4" aria-hidden /> Save as session
   750→                      </DropdownMenuItem>
   751→                      {/* §4.10c — today's arrange screen (6-d) */}
   752→                      <DropdownMenuItem
   753→                        onClick={() =>
   754→                          navigate(
   755→                            dateKey === todayKey() ? "/today/arrange" : `/today/arrange?date=${dateKey}`,
   756→                          )
   757→                        }
   758→                      >
   759→                        <ArrowDownUp className="h-4 w-4" aria-hidden /> Arrange exercises
   760→                      </DropdownMenuItem>
   761→                    </>
   762→                  ) : null}
   763→                  <DropdownMenuItem onClick={() => navigate("/history")}>
   764→                    <History className="h-4 w-4" aria-hidden /> History
   765→                  </DropdownMenuItem>
   766→                  <DropdownMenuItem onClick={() => navigate("/exercises")}>
   767→                    <Dumbbell className="h-4 w-4" aria-hidden /> Exercises
   768→                  </DropdownMenuItem>
   769→                  <DropdownMenuItem onClick={() => navigate("/tools")}>
   770→                    <Wrench className="h-4 w-4" aria-hidden /> Tools
   771→                  </DropdownMenuItem>
   772→                  <DropdownMenuItem onClick={() => navigate("/settings")}>
   773→                    <Settings className="h-4 w-4" aria-hidden /> Settings
   774→                  </DropdownMenuItem>
   775→                </DropdownMenuContent>
   776→              </DropdownMenu>
   777→              <TopBarHelp />
   778→            </>
   779→          }
   780→        />
   781→      }
   782→      subBar={
   783→        <DateStrip
   784→          dateKey={dateKey}
   785→          onChange={goTo}
   786→          workoutExists={!!workout}
   787→          progress={settings?.showSetsProgressBar !== false ? setsProgress : null}
   788→        />
   789→      }
   790→      bottomBar={
   791→        <BottomBar>
   792→          {restActive ? (
   793→            <RestBar
   794→              display={formatDuration(rest.remainingSec ?? 0)}
   795→              onAdjust={rest.adjust}
   796→              onSkip={rest.skip}
   797→            />
   798→          ) : workout?.finishedAt ? (
   799→            // finished — single disabled confirmation row
   800→            <FinishedBar />
   801→          ) : guidedActive ? (
   802→            // §4.11 guided mode — Prev / Log set / Next replaces Add/Finish
   803→            <GuidedBar
   804→              onPrev={() => moveGuided(-1)}
   805→              onNext={() => moveGuided(1)}
   806→              onLog={() => void logGuidedSet()}
   807→              onFinish={handleFinishPressed}
   808→              canLog={guidedIdx < pointerOrder.length && !isPointerComplete(guidedIdx)}
   809→              allDone={guidedAllDone}
   810→              atStart={guidedIdx <= 0}
   811→            />
   812→          ) : workout && workout.exercises.length > 0 ? (
   813→            // logging in progress — two equal actions
   814→            <FinishBar
   815→              onAdd={() => navigate(`/exercises?date=${dateKey}`)}
   816→              onFinish={handleFinishPressed}
   817→            />
   818→          ) : (
   819→            <AddExerciseBar onAdd={() => navigate(`/exercises?date=${dateKey}`)} />
   820→          )}
   821→        </BottomBar>
   822→      }
   823→    >
   824→      <ScrollBody>
   825→        {isLoading ? (
   826→          <DaySkeleton />
   827→        ) : !workout ? (
   828→          <TodayEmpty
   829→            hasPrevious={hasPrevious}
   830→            starting={creating}
   831→            onStartNew={() => void startNewWorkout()}
   832→            onCopyPrevious={() => void copyPreviousWorkout()}
   833→          />
   834→        ) : (
   835→          <>
   836→            {/* §4.11 RING rest display — FIRST child, pushes content down,
   837→                removed when rest ends (no overlay; RestBar stays in the BottomBar) */}
   838→            {ringRestActive ? (
   839→              <RestRingBlock
   840→                remainingSec={rest.remainingSec ?? 0}
   841→                totalSec={rest.totalSec}
   842→                onAdjust={rest.adjust}
   843→                onSkip={rest.skip}
   844→              />
   845→            ) : null}
   846→
   847→            <MetaRow
   848→              workout={workout}
   849→              source={source}
   850→              restRemainingSec={rest.remainingSec}
   851→              onToggleTimer={() => void toggleTimer()}
   852→              setsProgress={settings?.showSetsProgressBar !== false ? setsProgress : null}
   853→              guidedPointer={guidedLabel}
   854→              tipToggle={
   855→                anyTips ? { open: tipsOpen, onToggle: () => setTipsOpen((v) => !v) } : null
   856→              }
   857→            />
   858→
   859→            {exercises.map((we) => {
   860→              const group = we.groupId ? groupById.get(we.groupId) ?? null : null;
   861→              const isCurrentGuidedCard = guidedCardWeId === we.id;
   862→              // §4.11 guided video panel — only the CURRENT guided card, only
   863→              // when a video exists (MediaBlock itself collapses to 0px otherwise)
   864→              const showGuidedVideo =
   865→                isCurrentGuidedCard && !!settings?.showVideoPanel && !!we.exercise.videoUrl;
   866→              const showTipRow = tipsOpen && !!we.exercise.trainerTip;
   867→              const underHeader =
   868→                showGuidedVideo || showTipRow ? (
   869→                  <>
   870→                    {showGuidedVideo ? (
   871→                      // Part 7 — exerciseCard.mediaBlock anchor (the block itself
   872→                      // is consumer-supplied via the underHeader slot).
   873→                      <div
   874→                        {...tourAttrs({ id: "exerciseCard.mediaBlock", label: "Form video", help: "Guided demo video for the current exercise.", order: 140, when: ["guided"] })}
   875→                      >
   876→                        <GuidedVideoPanel weId={we.id} videoUrl={we.exercise.videoUrl} height={180} />
   877→                      </div>
   878→                    ) : null}
   879→                    {showTipRow ? (
   880→                      // Part 7 — exerciseCard.tipRow anchor (same slot contract).
   881→                      <div
   882→                        {...tourAttrs({ id: "exerciseCard.tipRow", label: "Trainer tip", help: "Coaching cue for this exercise; tap to expand.", order: 150 })}
   883→                      >
   884→                        <TrainerTipRow tip={we.exercise.trainerTip!} />
   885→                      </div>
   886→                    ) : null}
   887→                  </>
   888→                ) : undefined;
   889→              return (
   890→                <div
   891→                  key={we.id}
   892→                  ref={(el) => {
   893→                    if (el) cardRefs.current.set(we.id, el);
   894→                    else cardRefs.current.delete(we.id);
   895→                  }}
   896→                  className="relative flex-none"
   897→                >
   898→                  <SoloCard
   899→                    mode="edit"
   900→                    exercise={toCardExercise(we)}
   901→                    sets={cardSets(we)}
   902→                    collapsed={collapsedIds.has(we.id)}
   903→                    visibleColumns={visibleColumns}
   904→                    groupColour={group?.colour}
   905→                    groupName={group?.name}
   906→                    groupCode={groupCodes.codes.get(we.id)}
   907→                    metaExtra={bestByWeId.get(we.id)}
   908→                    underHeader={underHeader}
   909→                    className={isCurrentGuidedCard ? "ring-2 ring-primary/40" : undefined}
   910→                    onAction={handleCardAction(we)}
   911→                  />
   912→                  {notesWeId === we.id && (
   913→                    <ExerciseNotesPopover
   914→                      exercise={we.exercise}
   915→                      open
   916→                      onClose={() => setNotesWeId(null)}
   917→                    />
   918→                  )}
   919→                  {groupWeId === we.id && (
   920→                    <ExerciseGroupPopover
   921→                      workout={workout}
   922→                      we={we}
   923→                      open
   924→                      onClose={() => setGroupWeId(null)}
   925→                    />
   926→                  )}
   927→                </div>
   928→              );
   929→            })}
   930→
   931→            {/* Part 7 — today.summary anchor (SummaryRow is a display-only row;
   932→                the wrapper carries the declaration without touching it). */}
   933→            <div
   934→              {...tourAttrs({ id: "today.summary", label: "Day summary", help: "Sets, volume and PRs logged for the day.", order: 70, when: ["populated"] })}
   935→            >
   936→              <SummaryRow workout={workout} unit={defaultUnitFor(settings)} />
   937→            </div>
   938→
   939→            {/* bottom breathing spacer — bars are flex siblings, nothing to clear */}
   940→            <div className="h-4 flex-none" aria-hidden />
   941→
   942→            <ConfirmRemoveExercise
   943→              exerciseName={removeWe?.exercise.name ?? ""}
   944→              open={removeWeId != null}
   945→              onClose={() => setRemoveWeId(null)}
   946→              onConfirm={async () => {
   947→                const id = removeWeId;
   948→                setRemoveWeId(null);
   949→                if (id) await removeExercise(id);
   950→              }}
   951→            />
   952→
   953→            {/* §4.11 finishBehaviour=ASK — Save/Discard dialogs (mounted only
   954→                while the ASK stage is open; ALWAYS_SAVE never reaches it) */}
   955→            {settings?.finishBehaviour === "ASK" ? (
   956→              <FinishAskFlow
   957→                workout={workout}
   958→                open={finishAskOpen}
   959→                onClose={() => setFinishAskOpen(false)}
   960→                onSave={() => void finishWorkout()}
   961→                dateKey={dateKey}
   962→                onDiscarded={() => {
   963→                  // a discarded day can't keep a rest countdown running
   964→                  rest.skip();
   965→                  setFinishAskOpen(false);
   966→                }}
   967→              />
   968→            ) : null}
   969→
   970→            {sessionFromWorkout.dialog}
   971→          </>
   972→        )}
   973→      </ScrollBody>
   974→    </Screen>
   975→  );
   976→}
   977→
   978→function DaySkeleton() {
   979→  return (
   980→    <div className="flex flex-col gap-3" aria-busy="true" aria-label="Loading day">
   981→      <div className="h-12 animate-pulse rounded-lg bg-muted/60" />
   982→      {[0, 1, 2].map((i) => (
   983→        <div key={i} className="h-14 animate-pulse rounded-lg bg-muted/40" />
   984→      ))}
   985→    </div>
   986→  );
   987→}
   988→
